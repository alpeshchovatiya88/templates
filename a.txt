import plotly.express as px

book_dist = df["BOOKYEAR"].value_counts().sort_index().reset_index()
book_dist.columns = ["BOOKYEAR", "Count"]

book_dist["Percentage"] = (
    book_dist["Count"] / book_dist["Count"].sum() * 100
)

book_dist["Label"] = (
    book_dist["Count"].map("{:,}".format)
    + " (" +
    book_dist["Percentage"].map("{:.1f}%".format)
    + ")"
)

fig = px.line(
    book_dist,
    x="BOOKYEAR",
    y="Count",
    markers=True,
    text="Label",
    title="Data Distribution by BOOKYEAR"
)

fig.update_traces(
    textposition="top center"
)

fig.update_layout(
    xaxis_title="BOOKYEAR",
    yaxis_title="Number of Records",
    hovermode="x unified"
)

fig.show()
