bad = df[df["target"] == 1].copy()

# Create flags
bad["COLLECTION"] = (bad["CRA_COLLECTION"] == "Y").astype(int)
bad["BANKRUPTCY"] = (bad["CRA_BANKRUPTCY_FORECLOSURE_IND"] == "Y").astype(int)
bad["JUDGEMENT"] = (bad["CRA_JUDGEMENT"] == "Y").astype(int)
bad["PUBLIC_DEROG"] = (bad["CRD_PUBLIC_DEROG_TU"] > 0).astype(int)

flag_cols = ["COLLECTION", "BANKRUPTCY", "JUDGEMENT", "PUBLIC_DEROG"]

# Count adverse indicators
bad["ADVERSE_COUNT"] = bad[flag_cols].sum(axis=1)

# Keep only overlaps of 2, 3, and 4
overlap = (
    bad[bad["ADVERSE_COUNT"].isin([2, 3, 4])]
    .groupby(flag_cols)
    .size()
    .reset_index(name="Row_Count")
)

# Create readable combination names
overlap["Combination"] = overlap.apply(
    lambda x: " + ".join([
        col for col in flag_cols if x[col] == 1
    ]),
    axis=1
)

# Show results
overlap = overlap[["Combination", "Row_Count"]].sort_values(
    "Row_Count", ascending=False
)

print(overlap)
