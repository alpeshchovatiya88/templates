# Threshold for Precision, Recall and FPR
threshold = 0.50


def evaluate(df, name, probability_col):

    # Exclude missing target rows from evaluation
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty:
        print(f"{name}: No observed targets.")
        return None

    y = data[target_col].astype(int).to_numpy()

    probabilities = data[
        probability_col
    ].to_numpy(dtype=float)

    # --------------------------------------------
    # Number of observations and bad events
    # --------------------------------------------

    n_observed = len(data)
    n_bad_events = int(y.sum())

    # --------------------------------------------
    # Classification at selected threshold
    # --------------------------------------------

    y_pred = (
        probabilities >= threshold
    ).astype(int)

    tn, fp, fn, tp = confusion_matrix(
        y,
        y_pred,
        labels=[0, 1]
    ).ravel()

    # --------------------------------------------
    # Precision
    # --------------------------------------------

    precision = precision_score(
        y,
        y_pred,
        zero_division=0
    )

    # --------------------------------------------
    # Recall / Sensitivity / TPR
    # --------------------------------------------

    recall = recall_score(
        y,
        y_pred,
        zero_division=0
    )

    # --------------------------------------------
    # False Positive Rate
    # --------------------------------------------

    fpr = (
        fp / (fp + tn)
        if (fp + tn) > 0
        else np.nan
    )

    # --------------------------------------------
    # AUC and KS
    # --------------------------------------------

    if len(np.unique(y)) == 2:

        auc = roc_auc_score(
            y,
            probabilities
        )

        fpr_curve, tpr_curve, _ = roc_curve(
            y,
            probabilities
        )

        ks = np.max(
            tpr_curve - fpr_curve
        )

    else:
        auc = np.nan
        ks = np.nan

    # --------------------------------------------
    # Return performance metrics
    # --------------------------------------------

    return {
        "Dataset": name,
        "N_Observed": n_observed,
        "N_Bad_Events": n_bad_events,
        "Actual_Bad_Rate": y.mean(),
        "Mean_PD": probabilities.mean(),
        "AUC": auc,
        "KS": ks,
        "Precision": precision,
        "Recall": recall,
        "FPR": fpr,
        "Brier_Score": brier_score_loss(
            y,
            probabilities
        ),
        "Log_Loss": log_loss(
            y,
            probabilities,
            labels=[0, 1]
        )
    }

# 10. PLOT CALIBRATION CURVES

# These plots use exactly the same probabilities
# as the two performance tables.

def plot_calibration(df, name):

    # Use observed targets only
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty or data[target_col].nunique() < 2:
        print(f"{name}: Insufficient target data.")
        return

    y = data[target_col].astype(int)

    plt.figure(figsize=(8, 6))

    for col, label in [
        ("PD_ORIGINAL", "Original XGBoost"),
        ("PD_CALIBRATED", "Calibrated XGBoost")
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            data[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
    )

    plt.xlabel("Mean Predicted Probability")
    plt.ylabel("Observed Bad Rate")
    plt.title(name + " Calibration Curve")
    plt.legend()
    plt.grid(True)
    plt.show()


plot_calibration(df_train, "TRAIN")
