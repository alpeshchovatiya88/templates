def extract_hit_score(
    df,
    score_cols,
    output_col
):

    hit_cols = [
        f"{c}_HIT"
        for c in score_cols
    ]

    available_score_cols = [
        c for c in score_cols
        if c in df.columns
    ]

    available_hit_cols = [
        c for c in hit_cols
        if c in df.columns
    ]

    if (
        not available_score_cols
        or not available_hit_cols
    ):
        return pd.DataFrame(
            columns=[
                "BORROWER_SID",
                output_col
            ]
        )

    mask = (
        df[available_hit_cols]
        .eq("hit")
        .any(axis=1)
    )

    out = df.loc[
        mask,
        (
            ["BORROWER_SID"]
            + available_score_cols
            + available_hit_cols
        )
    ].copy()

    out[output_col] = np.nan

    for score_col in available_score_cols:

        hit_col = f"{score_col}_HIT"

        if hit_col not in out.columns:
            continue

        m = out[hit_col].eq("hit")

        out.loc[
            m,
            output_col
        ] = out.loc[
            m,
            score_col
        ]

    out = out[
        ["BORROWER_SID", output_col]
    ].copy()

    out[output_col] = pd.to_numeric(
        out[output_col],
        errors="coerce"
    )

    return out
