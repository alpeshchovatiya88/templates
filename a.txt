from preprocess import build_policy_query, build_borrower_query
from util import load_data_from_snowflake

dt_model = "202605"
region = "ca-central-1"

# Policy
policy = load_data_from_snowflake(
    build_policy_query(dt_model=dt_model),
    region=region
)
policy.to_csv("raw_policy.csv", index=False)

# Borrower
borrower = load_data_from_snowflake(
    build_borrower_query(dt_model=dt_model),
    region=region
)
borrower.to_csv("raw_borrower.csv", index=False)

print("Policy:", policy.shape)
print("Borrower:", borrower.shape)
print("Done")
