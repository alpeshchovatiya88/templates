def plot_numeric_distributions(data, numeric_cols, bins=50):

    for col in numeric_cols:

        plt.figure(figsize=(10, 5))

        sns.histplot(
            data=data,
            x=col,
            bins=bins,
            kde=True,
            stat="density",
            element="step"
        )

        plt.title(f"Distribution of {col}")
        plt.xlabel(col)
        plt.ylabel("Density")
        plt.tight_layout()
        plt.show()
