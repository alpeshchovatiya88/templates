
# Build summary
summary = (
    df[df["CRA_COLLECTION"].fillna("N").eq("Y")]
      .groupby("CRD_SCORE_bin")
      .agg(
          Total=("target", "count"),
          Total_Bad=("target", "sum"),
          Bad_Pct=("target", lambda x: round(x.mean() * 100, 2))
      )
      .reset_index()
)

# ---------------- Plot ----------------

x = np.arange(len(summary))

fig, ax1 = plt.subplots(figsize=(10, 6))

# Bars
bars = ax1.bar(
    x,
    summary["Total"],
    width=0.6
)

ax1.set_xlabel("CRD_SCORE_bin", fontweight="bold")
ax1.set_ylabel("Total Count", fontweight="bold")

ax1.set_xticks(x)
ax1.set_xticklabels(
    summary["CRD_SCORE_bin"].astype(int),
    fontweight="bold"
)

ax1.grid(axis="y", alpha=0.3)

# Bar labels (Total_Bad)
for i, bar in enumerate(bars):

    height = bar.get_height()

    total_bad = summary["Total_Bad"].iloc[i]

    ax1.text(
        bar.get_x() + bar.get_width()/2,
        height,
        f"{int(height):,}\n(Bad={int(total_bad):,})",
        ha="center",
        va="bottom",
        fontweight="bold",
        fontsize=9
    )

# Line for Bad %
ax2 = ax1.twinx()

ax2.plot(
    x,
    summary["Bad_Pct"],
    marker="o",
    linewidth=2
)

ax2.set_ylabel("Bad %", fontweight="bold")

for i, pct in enumerate(summary["Bad_Pct"]):

    ax2.text(
        x[i],
        pct,
        f"{pct:.1f}%",
        ha="center",
        va="bottom",
        fontweight="bold"
    )

plt.title(
    "CRA_COLLECTION = Y by CRD_SCORE_bin",
    fontsize=16,
    fontweight="bold"
)

plt.tight_layout()

plt.show()
