    def _save_feature_importance(self) -> None:
        model = self.best_model_.named_steps["model"]

        importance_values = model.get_feature_importance()

        if len(importance_values) != len(self.feature_names):
            raise ValueError(
                "Feature importance count does not match feature names."
            )

        rows = []
        for feature, value in zip(self.feature_names, importance_values):
            rows.append({
                "feature": str(feature),
                "importance": float(value)
            })

        fi = pd.DataFrame(rows, columns=["feature", "importance"])

        if not fi.empty:
            fi = fi.sort_values(
                "importance", ascending=False
            ).reset_index(drop=True)

            total_importance = fi["importance"].sum()

            if total_importance > 0:
                fi["importance_pct"] = (
                    fi["importance"] / total_importance
                )
            else:
                fi["importance_pct"] = 0.0

            fi["feature_rank"] = np.arange(1, len(fi) + 1)

        self.feature_importance_ = fi

        path = os.path.join(
            self.results_dir,
            "feature_importance_catboost.csv"
        )
        fi.to_csv(path, index=False)
