import pandas as pd
import numpy as np
import plotly.graph_objects as go
from plotly.subplots import make_subplots


def create_swapset_dashboard(
    df,
    filename="swapset_dashboard.html"
):

    # ==========================================
    # 1. PREPARE DATA
    # ==========================================

    df = df.copy()

    for source, target in [
        ("rr_model_decile_xgb", "decile_xgb_num"),
        ("rr_model_decile_cat", "decile_cat_num")
    ]:

        if target not in df.columns:

            df[target] = (
                df[source]
                .astype("string")
                .str.extract(r"(\d+)", expand=False)
                .astype("Int64")
            )

        else:

            df[target] = pd.to_numeric(
                df[target],
                errors="coerce"
            ).astype("Int64")

    df = df.dropna(
        subset=[
            "decile_xgb_num",
            "decile_cat_num"
        ]
    )

    # Keep only valid deciles
    df = df[
        df["decile_xgb_num"].between(1, 20)
        & df["decile_cat_num"].between(1, 20)
    ]

    deciles = list(range(1, 21))

    regions = sorted(
        df["ECON_REGION"].dropna().unique(),
        key=str
    )

    # ==========================================
    # 2. CREATE SWAPSET
    # ==========================================

    def make_swapset(data):

        swap = pd.crosstab(
            data["decile_xgb_num"],
            data["decile_cat_num"]
        )

        swap = swap.reindex(
            index=deciles,
            columns=deciles,
            fill_value=0
        )

        return swap

    # ==========================================
    # 3. FORMAT TABLE WITH COUNT + %
    # ==========================================

    def format_swapset(swap):

        row_totals = swap.sum(axis=1)
        col_totals = swap.sum(axis=0)
        grand_total = int(swap.values.sum())

        display = []
        colors = []

        # First column: XGB deciles
        display.append(
            [str(i) for i in deciles] + ["All"]
        )

        colors.append(
            ["#E6EAF0"] * 21
        )

        for col in deciles:

            col_values = []
            col_colors = []

            for row in deciles:

                count = int(swap.loc[row, col])
                total = int(row_totals.loc[row])

                pct = (
                    count / total * 100
                    if total > 0 else 0
                )

                col_values.append(
                    f"{count:,} ({pct:.2f}%)"
                )

                if count == 0:
                    color = "#FFFFFF"

                elif row == col:
                    color = "#D5F0DB"

                elif abs(row - col) <= 2:
                    color = "#FFE4B5"

                else:
                    color = "#F8C4C4"

                col_colors.append(color)

            # Column total
            count = int(col_totals.loc[col])

            pct = (
                count / grand_total * 100
                if grand_total > 0 else 0
            )

            col_values.append(
                f"{count:,} ({pct:.2f}%)"
            )

            col_colors.append("#E6EAF0")

            display.append(col_values)
            colors.append(col_colors)

        # Total column
        total_values = [
            f"{int(row_totals.loc[row]):,} (100.00%)"
            if row_totals.loc[row] > 0
            else "0 (0.00%)"
            for row in deciles
        ]

        total_values.append(
            f"{grand_total:,} (100.00%)"
            if grand_total > 0
            else "0 (0.00%)"
        )

        display.append(total_values)

        colors.append(
            ["#E6EAF0"] * 21
        )

        return display, colors

    # ==========================================
    # 4. BUILD FIGURE WITH TABLE + HEATMAP
    # ==========================================

    fig = make_subplots(
        rows=2,
        cols=1,
        specs=[
            [{"type": "table"}],
            [{"type": "heatmap"}]
        ],
        row_heights=[0.55, 0.45],
        vertical_spacing=0.08,
        subplot_titles=[
            "Swapset Table: Count (Row %)",
            "XGBoost vs CatBoost Decile Movement Heatmap"
        ]
    )

    datasets = [("ALL", df)] + [
        (
            str(region),
            df[df["ECON_REGION"] == region]
        )
        for region in regions
    ]

    # ==========================================
    # 5. ADD TABLE AND HEATMAP PER REGION
    # ==========================================

    for idx, (region_name, data) in enumerate(datasets):

        swap = make_swapset(data)

        table_values, cell_colors = format_swapset(
            swap
        )

        is_visible = (idx == 0)

        # TABLE
        fig.add_trace(
            go.Table(

                header=dict(
                    values=(
                        ["XGB / CAT"]
                        + [str(i) for i in deciles]
                        + ["All"]
                    ),
                    fill_color="#4472C4",
                    font=dict(
                        color="white",
                        size=12
                    ),
                    align="center",
                    height=35
                ),

                cells=dict(
                    values=table_values,
                    fill_color=cell_colors,
                    font=dict(
                        color="#202020",
                        size=10
                    ),
                    align="center",
                    height=28
                ),

                columnwidth=[75] + [115] * 21,

                visible=is_visible,

                name=region_name
            ),
            row=1,
            col=1
        )

        # HEATMAP
        fig.add_trace(
            go.Heatmap(

                z=swap.values,

                x=deciles,
                y=deciles,

                colorscale="Blues",

                text=swap.values,

                texttemplate="%{text:,}",

                textfont=dict(size=9),

                hovertemplate=(
                    "XGBoost Decile: %{y}<br>"
                    "CatBoost Decile: %{x}<br>"
                    "Count: %{z:,}"
                    "<extra></extra>"
                ),

                colorbar=dict(
                    title="Count",
                    len=0.35,
                    y=0.20
                ),

                showscale=is_visible,

                visible=is_visible,

                name=region_name
            ),
            row=2,
            col=1
        )

    # ==========================================
    # 6. CREATE DROPDOWN
    # ==========================================

    buttons = []

    n_regions = len(datasets)

    for i, (region_name, _) in enumerate(datasets):

        visibility = [False] * (2 * n_regions)

        visibility[2 * i] = True
        visibility[2 * i + 1] = True

        scale_visibility = [False] * (2 * n_regions)
        scale_visibility[2 * i + 1] = True

        buttons.append(
            dict(
                label=region_name,
                method="update",
                args=[
                    {
                        "visible": visibility,
                        "showscale": scale_visibility
                    },
                    {
                        "title.text": (
                            "XGBoost vs CatBoost "
                            f"Swapset Dashboard - {region_name}"
                        )
                    }
                ]
            )
        )

    # ==========================================
    # 7. LAYOUT
    # ==========================================

    fig.update_layout(

        title=dict(
            text=(
                "XGBoost vs CatBoost "
                "Swapset Dashboard - ALL"
            ),
            x=0.5,
            xanchor="center",
            y=0.98
        ),

        height=1600,
        width=1600,

        margin=dict(
            l=80,
            r=80,
            t=180,
            b=80
        ),

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,
                x=0.01,
                xanchor="left",
                y=1.08,
                yanchor="top"
            )
        ],

        annotations=[
            dict(
                text=(
                    "Green: Same Decile | "
                    "Orange: 1-2 Decile Movement | "
                    "Red: 3+ Decile Movement | "
                    "White: Zero | "
                    "Percentage: XGB Row Total"
                ),
                x=0.5,
                y=1.025,
                xref="paper",
                yref="paper",
                showarrow=False,
                font=dict(size=12)
            )
        ]
    )

    # ==========================================
    # 8. HEATMAP AXES
    # ==========================================

    fig.update_xaxes(
        title_text="CatBoost Decile",
        tickmode="array",
        tickvals=deciles,
        row=2,
        col=1
    )

    fig.update_yaxes(
        title_text="XGBoost Decile",
        tickmode="array",
        tickvals=deciles,
        autorange="reversed",
        row=2,
        col=1
    )

    # ==========================================
    # 9. SAVE AND DISPLAY
    # ==========================================

    fig.write_html(
        filename,
        include_plotlyjs=True
    )

    fig.show()

    return fig


# ==============================================
# RUN
# ==============================================

fig = create_swapset_dashboard(
    df,
    filename="swapset_dashboard.html"
)
