importance_values = model.get_feature_importance()

if len(importance_values) != len(self.feature_names):
    raise ValueError(
        "Feature importance count does not match "
        "encoded feature count: "
        f"{len(importance_values)} != {len(self.feature_names)}"
    )

rows = []

for feature, value in zip(
    self.feature_names,
    importance_values
):
    rows.append({
        "feature": feature,
        "importance": float(value)
    })
