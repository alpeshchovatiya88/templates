
import pandas as pd

import matplotlib.pyplot as plt

df = df.copy()
df["BOOKYEAR"] = pd.to_numeric(df["BOOKYEAR"])

periods = {
    "Training (2009–2015)": (2009, 2015, "blue"),
    "Test (2021–2023)": (2021, 2023, "red"),
    "Evaluation (2025–2026)": (2025, 2026, "orange")
}

def plot_overlay(df, column, reference_value=None):
    fig, ax = plt.subplots(figsize=(12, 6))

    for label, (start, end, color) in periods.items():
        subset = df[
            df["BOOKYEAR"].between(start, end)
        ].copy()

        # Average values for each year
        yearly = (
            subset.groupby("BOOKYEAR")[column]
            .mean()
            .sort_index()
        )

        if yearly.empty:
            continue

        # Relative year starting from zero
        x = yearly.index.to_numpy() - start

        ax.plot(
            x, yearly.values,
            marker="o",
            linewidth=2,
            color=color,
            label=label
        )

    if reference_value is not None:
        ax.axhline(
            y=reference_value,
            color="green",
            linewidth=2,
            linestyle="--",
            label="4-year reference"
        )

    ax.set_xlabel("Years from period start")
    ax.set_ylabel(column)
    ax.set_title(f"{column} - Period Comparison")
    ax.legend()
    ax.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

plot_overlay(df, "HPI_+4")
plot_overlay(df, "UE_+4")
