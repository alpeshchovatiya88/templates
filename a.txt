# Totals per BOOKYEAR
totals = (
    df.groupby("BOOKYEAR")
      .agg(
          Total_N=("target", "count"),
          Total_Bad=("target", "sum"),
          Bad_Rate=("target", lambda x: round(x.mean() * 100, 2))
      )
)

# Create display string
formatted["Overall"] = (
    "n=" + totals["Total_N"].astype(str)
    + " (bad=" + totals["Total_Bad"].astype(str)
    + ", " + totals["Bad_Rate"].astype(str) + "%)"
)

formatted
