import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ---------------------------------------------------------
# Clean keys
# ---------------------------------------------------------

df_new = df_new.copy()
df_2016_2025 = df_2016_2025.copy()

df_new["ECON_REGION"] = df_new["ECON_REGION"].astype(str).str.strip()
df_2016_2025["ECON_REGION"] = (
    df_2016_2025["ECON_REGION"].astype(str).str.strip()
)

df_new["EFFECT_SID"] = pd.to_numeric(
    df_new["EFFECT_SID"], errors="coerce"
).astype("Int64")

df_2016_2025["EFFECT_SID"] = pd.to_numeric(
    df_2016_2025["EFFECT_SID"], errors="coerce"
).astype("Int64")

df_new["HPI"] = pd.to_numeric(df_new["HPI"], errors="coerce")
df_2016_2025["HPI"] = pd.to_numeric(
    df_2016_2025["HPI"], errors="coerce"
)


# ---------------------------------------------------------
# Build HPI history
# ---------------------------------------------------------

hpi_old = df_new[
    ["ECON_REGION", "EFFECT_SID", "HPI"]
].copy()

hpi_new = df_2016_2025[
    ["ECON_REGION", "EFFECT_SID", "HPI"]
].copy()

hpi_all = pd.concat(
    [hpi_old, hpi_new],
    ignore_index=True
)

hpi_all = hpi_all.dropna(
    subset=["ECON_REGION", "EFFECT_SID", "HPI"]
)

hpi_all = hpi_all.drop_duplicates(
    subset=["ECON_REGION", "EFFECT_SID"],
    keep="first"
)


# ---------------------------------------------------------
# Event dates/months
# ---------------------------------------------------------

df_new["DELQ90_IND_DATE"] = pd.to_datetime(
    df_new["DELQ90_IND_DATE"],
    errors="coerce"
)

df_new["CLAIM_IND_DATE"] = pd.to_datetime(
    df_new["CLAIM_IND_DATE"],
    errors="coerce"
)

df_new["DELQ_MONTH"] = (
    df_new["DELQ90_IND_DATE"]
    .dt.strftime("%Y%m")
    .astype("Int64")
)

df_new["CLAIM_MONTH"] = (
    df_new["CLAIM_IND_DATE"]
    .dt.strftime("%Y%m")
    .astype("Int64")
)


# ---------------------------------------------------------
# HPI lookup
# ---------------------------------------------------------

hpi_map = hpi_all.set_index(
    ["ECON_REGION", "EFFECT_SID"]
)["HPI"]

df_new["HPI_AT_DELQ"] = [
    hpi_map.get((region, month), np.nan)
    if pd.notna(month) else np.nan
    for region, month in zip(
        df_new["ECON_REGION"],
        df_new["DELQ_MONTH"]
    )
]

df_new["HPI_AT_CLAIM"] = [
    hpi_map.get((region, month), np.nan)
    if pd.notna(month) else np.nan
    for region, month in zip(
        df_new["ECON_REGION"],
        df_new["CLAIM_MONTH"]
    )
]


# ---------------------------------------------------------
# HPI ratios
# ---------------------------------------------------------

df_new["HPI_RATIO_DELQ"] = (
    df_new["HPI_AT_DELQ"] / df_new["HPI"]
)

df_new["HPI_RATIO_CLAIM"] = (
    df_new["HPI_AT_CLAIM"] / df_new["HPI"]
)

# Claim takes priority when both indicators = 1
df_new.loc[
    df_new["CLAIM_IND"] == 1.0,
    "HPI_RATIO_DELQ"
] = np.nan


# ---------------------------------------------------------
# Year buckets
# ---------------------------------------------------------

df_new["DELQ_YEAR"] = np.ceil(
    df_new["TIME_TO_DELQ_YR"]
)

df_new["CLAIM_YEAR"] = np.ceil(
    df_new["TIME_TO_CLAIM_YR"]
)


# ---------------------------------------------------------
# Average HPI ratio by year
# ---------------------------------------------------------

years = np.arange(1, 18)

delq_ratio = (
    df_new.loc[
        (df_new["DELQ90_IND"] == 1.0) &
        (df_new["CLAIM_IND"] != 1.0) &
        (df_new["DELQ_YEAR"].between(1, 17))
    ]
    .groupby("DELQ_YEAR")["HPI_RATIO_DELQ"]
    .mean()
    .reindex(years)
)

claim_ratio = (
    df_new.loc[
        (df_new["CLAIM_IND"] == 1.0) &
        (df_new["CLAIM_YEAR"].between(1, 17))
    ]
    .groupby("CLAIM_YEAR")["HPI_RATIO_CLAIM"]
    .mean()
    .reindex(years)
)


# ---------------------------------------------------------
# GRAPH ONLY
# ---------------------------------------------------------

plt.figure(figsize=(12, 6))

plt.plot(
    years,
    delq_ratio.values,
    marker="o",
    label="Delinquency"
)

plt.plot(
    years,
    claim_ratio.values,
    marker="o",
    label="Claim"
)

plt.axhline(
    y=1.0,
    linestyle="--",
    alpha=0.5
)

plt.xticks(years)
plt.xlim(1, 17)

plt.xlabel("Years from Effective Date")
plt.ylabel("Average HPI at Event / HPI at Origination")

plt.title("HPI Ratio at Delinquency and Claim")

plt.legend()
plt.grid(alpha=0.3)

plt.tight_layout()
plt.show()
