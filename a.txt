      .groupby("BOOKYEAR", as_index=False)
      .agg(
          n=("STATUS", "size"),

          Actual_n_Delqs=(
              "category",
              lambda x: x.isin(["Delq Only"]).sum()
          ),

          Actual_n_Claims=(
              "category",
              lambda x: x.isin(["Delq Claim"]).sum()
          ),

          Actual_n_Delq_and_Claims=(
              "category",
              lambda x: x.isin(["Delq Only", "Delq Claim"]).sum()
          ),

          Predicted_n_Delq_and_Claims=(
              "oof_pred_proba",
              "sum"
          ),

          SMI3_LOSS=(
              "SMI3_BT_LOSS",
              "sum"
          ),

          ACTUAL_LOSS=(
              "TOTAL_INCURRED_LOSS",
              "sum"
          ),

          POD_PREMIUM_WRITTEN=(
              "POD_PREMIUM_WRITTEN",
              "sum"
          )
      )
)

# Compute frequency metrics
LR_bt["n_diff"] = (
    LR_bt["Predicted_n_Delq_and_Claims"]
    - LR_bt["Actual_n_Delq_and_Claims"]
)

LR_bt["per_diff"] = (
    100
    * (
        LR_bt["Predicted_n_Delq_and_Claims"]
        - LR_bt["Actual_n_Delq_and_Claims"]
    )
    / LR_bt["Actual_n_Delq_and_Claims"]
)

LR_bt["Delq_Proportion"] = (
    LR_bt["Actual_n_Delqs"]
    / LR_bt["Actual_n_Delq_and_Claims"]
)

# Compute LR
LR_bt["SMI3_LR"] = (
    LR_bt["SMI3_LOSS"]
    / LR_bt["POD_PREMIUM_WRITTEN"]
)

LR_bt["ACTUAL_LR"] = (
    LR_bt["ACTUAL_LOSS"]
    / LR_bt["POD_PREMIUM_WRITTEN"]
)

# Adjusted LR
LR_bt["SMI3_ADJ_LR"] = (
    LR_bt["SMI3_LR"]
    * LR_bt["Delq_Proportion"]
)

# Differences
LR_bt["LR_diff"] = (
    LR_bt["SMI3_LR"]
    - LR_bt["ACTUAL_LR"]
)

LR_bt["ADJ_LR_diff"] = (
    LR_bt["SMI3_ADJ_LR"]
    - LR_bt["ACTUAL_LR"]
)
