
import pandas as pd
import numpy as np

from openpyxl import Workbook
from openpyxl.styles import (
    Font, PatternFill, Alignment, Border, Side
)

# =====================================================
# 1. EXPERIMENTS - EDIT THIS DICTIONARY
# =====================================================

dfs = {
    "HPI_+4": df_hpi_4,
    "HPI_+3": df_hpi_3,
    "HPI_+5": df_hpi_5,
}

# Optional descriptions
descriptions = {
    # "HPI_+4": "Description",
    # "HPI_+3": "Description",
}

DECILE_COL = "rr_model_decile"
LOSS_COL = "SMI3_LOSS"
LR_COL = "SMI3_LR"

LR_SCALE = 1  # Set to 100 if LR is stored as decimal fractions

OUTPUT_FILE = "seed_experiments.xlsx"

# =====================================================
# 2. WORKBOOK - SINGLE SHEET
# =====================================================

wb = Workbook()
ws = wb.active
ws.title = "Seed Experiments"

yellow = PatternFill("solid", fgColor="FFFF00")
light_blue = PatternFill("solid", fgColor="DDEBF7")

thin = Side(style="thin", color="B7B7B7")
table_border = Border(
    left=thin,
    right=thin,
    top=thin,
    bottom=thin
)

# IMPORTANT:
# No column_dimensions.width changes.
# No row_dimensions.height changes.

# =====================================================
# 3. HELPER FOR MIN/MAX
# =====================================================

def get_range(series):
    values = pd.to_numeric(series, errors="coerce")
    values = values.replace([np.inf, -np.inf], np.nan)
    values = values.dropna()

    if values.empty:
        return None, None

    return float(values.min()), float(values.max())


# =====================================================
# 4. BUILD EXPERIMENT TABLES
# =====================================================

row = 1

for experiment, df in dfs.items():

    temp = df.copy()

    required = [DECILE_COL, LOSS_COL, LR_COL]
    missing = [c for c in required if c not in temp.columns]

    if missing:
        raise ValueError(
            f"{experiment}: Missing columns {missing}"
        )

    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL]
        .astype(str)
        .str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    temp[LOSS_COL] = pd.to_numeric(
        temp[LOSS_COL], errors="coerce"
    )

    temp[LR_COL] = (
        pd.to_numeric(temp[LR_COL], errors="coerce")
        * LR_SCALE
    )

    decile1 = temp[temp["_decile"] == 1]

    overall_loss_min, overall_loss_max = get_range(
        temp[LOSS_COL]
    )
    overall_lr_min, overall_lr_max = get_range(
        temp[LR_COL]
    )

    decile_loss_min, decile_loss_max = get_range(
        decile1[LOSS_COL]
    )
    decile_lr_min, decile_lr_max = get_range(
        decile1[LR_COL]
    )

    # -----------------------------------------
    # Row 1: Yellow experiment name
    # -----------------------------------------

    name_cell = ws.cell(row, 1, str(experiment))
    name_cell.fill = yellow
    name_cell.font = Font(bold=True, size=12)
    name_cell.alignment = Alignment(vertical="center")

    # Merged headers
    ws.merge_cells(
        start_row=row, start_column=2,
        end_row=row, end_column=3
    )
    ws.merge_cells(
        start_row=row, start_column=4,
        end_row=row, end_column=5
    )

    for col, text in [
        (2, "Loss range"),
        (4, "Loss ratio range")
    ]:
        cell = ws.cell(row, col, text)
        cell.font = Font(bold=True, size=12)
        cell.alignment = Alignment(
            horizontal="center",
            vertical="center"
        )

    # -----------------------------------------
    # Row 2: Blue description and min/max
    # -----------------------------------------

    desc = ws.cell(
        row + 1,
        1,
        descriptions.get(experiment, "")
    )
    desc.fill = light_blue
    desc.alignment = Alignment(
        vertical="center",
        wrap_text=False
    )

    for col, text in enumerate(
        ["min", "max", "min", "max"], start=2
    ):
        cell = ws.cell(row + 1, col, text)
        cell.font = Font(bold=True, size=12)
        cell.alignment = Alignment(
            horizontal="center",
            vertical="center"
        )

    # -----------------------------------------
    # Rows 3-4: Values
    # -----------------------------------------

    data_rows = [
        [
            "Overall",
            overall_loss_min,
            overall_loss_max,
            overall_lr_min,
            overall_lr_max
        ],
        [
            "Decile 1",
            decile_loss_min,
            decile_loss_max,
            decile_lr_min,
            decile_lr_max
        ]
    ]

    for r, values in enumerate(
        data_rows, start=row + 2
    ):
        for col, value in enumerate(values, start=1):

            cell = ws.cell(r, col, value)

            if col in (2, 3):
                cell.number_format = '$#,##0'
            elif col in (4, 5):
                cell.number_format = '0.00'

            cell.alignment = Alignment(
                horizontal="left" if col == 1 else "right",
                vertical="center"
            )

    # -----------------------------------------
    # Borders
    # -----------------------------------------

    for r in range(row, row + 4):
        for col in range(1, 6):
            ws.cell(r, col).border = table_border

    # Two blank rows between experiments
    row += 6

# =====================================================
# 5. SAVE
# =====================================================

wb.save(OUTPUT_FILE)

print(f"Saved: {OUTPUT_FILE}")
