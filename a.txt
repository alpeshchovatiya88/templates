policy_clip = policy_product

clip_rows = []

for col, (lo, hi) in FEATURE_RANGES.items():

    if col not in policy_clip.columns:
        print("Missing column, skipped:", col)
        continue

    x = pd.to_numeric(
        policy_clip[col],
        errors="coerce"
    )

    valid = x.notna()
    below = valid & (x < lo)
    above = valid & (x > hi)

    clip_rows.append({
        "column": col,
        "lower_bound": lo,
        "upper_bound": hi,
        "below": below.sum(),
        "above": above.sum(),
        "total_clipped": below.sum() + above.sum(),
    })

    policy_clip.loc[below, col] = lo
    policy_clip.loc[above, col] = hi

clip_summary = pd.DataFrame(clip_rows)

clip_summary.sort_values(
    "total_clipped",
    ascending=False
)
