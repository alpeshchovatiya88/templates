limport pandas as pd
import numpy as np

# Read files
dev = pd.read_csv("dev.csv")
val = pd.read_csv("val.csv")

# Select required rows
dev = dev.iloc[9:16].copy()
val = val.iloc[14:21].copy()

# Remove spaces from column names
dev.columns = dev.columns.str.strip()
val.columns = val.columns.str.strip()

# Columns to compare
metrics = [
    "Total Application",
    "Actual Loss",
    "Predicted Loss",
    "Actual LR",
    "Predicted LR",
    "Actual N of Delq",
    "Claims"
]

columns = ["Bookyear"] + metrics

# Keep only required columns
dev = dev[columns]
val = val[columns]

# Merge using Bookyear
df = pd.merge(
    dev,
    val,
    on="Bookyear",
    how="outer",
    suffixes=("_DEV", "_VAL"),
    indicator=True,
    validate="one_to_one"
)

# Compare each metric
for col in metrics:

    dev_value = pd.to_numeric(
        df[f"{col}_DEV"], errors="coerce"
    )

    val_value = pd.to_numeric(
        df[f"{col}_VAL"], errors="coerce"
    )

    df[f"{col}_Status"] = np.where(
        np.isclose(
            dev_value,
            val_value,
            atol=0.0001,
            rtol=0,
            equal_nan=False
        ),
        "PASS",
        "FAIL"
    )

# Overall status
status_cols = [f"{col}_Status" for col in metrics]

df["Overall_Status"] = np.where(
    (df[status_cols] == "PASS").all(axis=1)
    & (df["_merge"] == "both"),
    "PASS",
    "FAIL"
)

# Final result
print(df[["Bookyear"] + status_cols + ["Overall_Status"]])

# Export
df.to_excel("Reproducibility_Report.xlsx", index=False)
