import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns


def plot_grouped_bad_rate(
    df,
    category_col,
    target_col="target",
    year_col="BOOKYEAR",
    palette="turbo",
    figsize=(18,8),
    save_image=True
):

    # -------------------------
    # Overall
    # -------------------------

    overall = (
        df.groupby(category_col)[target_col]
          .agg(
              total_count="count",
              bad_count="sum",
              bad_rate="mean"
          )
          .reset_index()
    )

    overall[year_col] = "Overall"


    # -------------------------
    # Yearly
    # -------------------------

    agg_year = (
        df.groupby([year_col, category_col])[target_col]
          .agg(
              total_count="count",
              bad_count="sum",
              bad_rate="mean"
          )
          .reset_index()
    )


    # -------------------------
    # Combine
    # -------------------------

    agg_df = pd.concat(
        [overall, agg_year],
        ignore_index=True
    )


    # -------------------------
    # Order
    # -------------------------

    year_order = (
        ["Overall"]
        + sorted(
            df[year_col]
            .dropna()
            .unique()
            .tolist()
        )
    )


    # -------------------------
    # Plot
    # -------------------------

    plt.figure(figsize=figsize)

    sns.set_theme(style="whitegrid")

    ax = sns.barplot(
        data=agg_df,
        x=year_col,
        y="bad_rate",
        hue=category_col,
        order=year_order,
        palette=palette
    )


    # -------------------------
    # Labels
    # -------------------------

    for container in ax.containers:

        labels = []

        subset = (
            agg_df[
                agg_df[category_col]
                == container.get_label()
            ]
            .set_index(year_col)
            .reindex(year_order)
            .reset_index()
        )

        for _, row in subset.iterrows():

            total = int(row["total_count"])

            bad = int(row["bad_count"])

            rate = row["bad_rate"]

            labels.append(
                f"T:{total}\nB:{bad}\n({rate:.1%})"
            )

        ax.bar_label(
            container,
            labels=labels,
            padding=4,
            fontsize=7,
            fontweight="bold"
        )


    # -------------------------
    # Formatting
    # -------------------------

    clean_name = (
        category_col
        .replace("_", " ")
        .title()
    )

    plt.title(
        f"Bad Rate by {clean_name} and BOOKYEAR",
        fontsize=16,
        fontweight="bold"
    )

    plt.xlabel("BOOKYEAR")

    plt.ylabel("Bad Rate (%)")

    ax.yaxis.set_major_formatter(
        plt.FuncFormatter(
            lambda y, _: f"{y:.0%}"
        )
    )

    plt.xticks(rotation=45)

    plt.legend(
        title=clean_name,
        bbox_to_anchor=(1.02,1),
        loc="upper left"
    )

    plt.tight_layout()


    # -------------------------
    # Save image
    # -------------------------

    if save_image:

        filename = (
            f"{category_col.lower()}_overall_bookyear.png"
        )

        plt.savefig(
            filename,
            dpi=300,
            bbox_inches="tight"
        )

        print(f"Saved: {filename}")


    plt.show()
