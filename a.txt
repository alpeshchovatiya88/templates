def numeric_summary(df, features, sample_name):

    x = df[features]

    result = pd.DataFrame({
        "N": x.count(),
        "Missing_%": x.isna().mean() * 100,
        "Mean": x.mean(),
        "Std": x.std(),
        "Min": x.min(),
        "P01": x.quantile(0.01),
        "P05": x.quantile(0.05),
        "P25": x.quantile(0.25),
        "Median": x.median(),
        "P75": x.quantile(0.75),
        "P95": x.quantile(0.95),
        "P99": x.quantile(0.99),
        "Max": x.max(),
    })

    result["Dataset"] = sample_name

    return result.reset_index(
        names="Feature"
    )

def category_compare(feature):

    train = (
        train_df[feature]
        .fillna("MISSING")
        .value_counts(normalize=True)
        .mul(100)
        .rename("Train_%")
    )

    val = (
        validation_df[feature]
        .fillna("MISSING")
        .value_counts(normalize=True)
        .mul(100)
        .rename("Validation_%")
    )

    test = (
        test_df[feature]
        .fillna("MISSING")
        .value_counts(normalize=True)
        .mul(100)
        .rename("Test_%")
    )

    result = pd.concat(
        [train, val, test],
        axis=1
    ).fillna(0)

    result["Val_minus_Train_pp"] = (
        result["Validation_%"]
        - result["Train_%"]
    )

    result["Test_minus_Train_pp"] = (
        result["Test_%"]
        - result["Train_%"]
    )

    return result.sort_values(
        "Test_%",
        ascending=False
    )
