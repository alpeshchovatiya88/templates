
import boto3
import json
import pandas as pd

# SageMaker endpoint
ENDPOINT_NAME = "YOUR_ENDPOINT_NAME"

runtime = boto3.client(
    "sagemaker-runtime",
    region_name="ca-central-1"
)

# Load existing applications
with open("applications.json", "r") as f:
    applications = json.load(f)

if isinstance(applications, dict):
    applications = [applications]

results = []

# Test each application
for app in applications:
    app_id = app.get("APPLICATION_NUMBER")

    try:
        response = runtime.invoke_endpoint(
            EndpointName=ENDPOINT_NAME,
            ContentType="application/json",
            Body=json.dumps(app)
        )

        prediction = json.loads(
            response["Body"].read().decode("utf-8")
        )

        results.append({
            "APPLICATION_NUMBER": app_id,
            "STATUS": "PASS",
            "CLAIM_60TH": prediction.get("CLAIM_60TH"),
            "LOSS_60TH": prediction.get("LOSS_60TH"),
            "CLAIM_95TH": prediction.get("CLAIM_95TH"),
            "LOSS_95TH": prediction.get("LOSS_95TH"),
            "ERROR": ""
        })

        print(f"PASS: {app_id}")

    except Exception as e:
        results.append({
            "APPLICATION_NUMBER": app_id,
            "STATUS": "FAIL",
            "ERROR": str(e)
        })

        print(f"FAIL: {app_id} - {e}")

# Save test results
df = pd.DataFrame(results)
df.to_csv("test_results.csv", index=False)

print("\n", df)
