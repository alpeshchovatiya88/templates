import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import math

# --------------------------------------------------
# SETTINGS
# --------------------------------------------------
year_col = "BOOKYEAR"
category_col = "PRD_LOAN_PURPOSE"
bad_col = "BAD"     # change if your target has another name

# Explicit years
years = list(range(2009, 2026))

# --------------------------------------------------
# SUMMARY FUNCTION
# --------------------------------------------------
def get_summary(data, label):

    out = (
        data.groupby(category_col, dropna=False)
            .agg(
                T=(bad_col, "size"),
                B=(bad_col, "sum")
            )
            .reset_index()
    )

    out["Bad_Rate"] = (out["B"] / out["T"]) * 100
    out["BOOKYEAR_LABEL"] = str(label)

    return out

# 2009 - 2025
for yr in years:
    temp = df[df[year_col] == yr]

    if len(temp) > 0:
        summary_list.append(get_summary(temp, yr))

summary = pd.concat(summary_list, ignore_index=True)


# --------------------------------------------------
# KEEP SAME CATEGORY ORDER AND COLORS
# --------------------------------------------------
categories = sorted(
    df[category_col].dropna().unique()
)

# Different color for each PRD_LOAN_PURPOSE
colors = sns.color_palette(
    "Set2",
    n_colors=len(categories)
)

color_dict = dict(zip(categories, colors))


# --------------------------------------------------
# PLOT
# --------------------------------------------------
plot_order = ["overall"] + [str(y) for y in years]

ncols = 3
nrows = math.ceil(len(plot_order) / ncols)

fig, axes = plt.subplots(
    nrows=nrows,
    ncols=ncols,
    figsize=(18, 5 * nrows)
)

axes = np.array(axes).flatten()


for i, yr in enumerate(plot_order):

    ax = axes[i]

    temp = summary[
        summary["BOOKYEAR_LABEL"] == yr
    ].copy()

    # If no observations for a particular year
    if temp.empty:
        ax.set_title(f"BOOKYEAR: {yr}")
        ax.text(
            0.5, 0.5,
            "No data",
            ha="center",
            va="center",
            transform=ax.transAxes
        )
        ax.axis("off")
        continue

    # Reindex so categories always appear
    # in the SAME ORDER
    temp = (
        temp.set_index(category_col)
            .reindex(categories)
            .reset_index()
    )

    # Plot manually so each category
    # always gets the same color
    x = np.arange(len(categories))

    bars = ax.bar(
        x,
        temp["Bad_Rate"].fillna(0),
        color=[color_dict[c] for c in categories],
        width=0.7
    )

    # X axis
    ax.set_xticks(x)
    ax.set_xticklabels(
        categories,
        rotation=0,
        fontsize=9
    )

    # Titles
    ax.set_title(
        f"BOOKYEAR: {yr}",
        fontsize=13,
        fontweight="bold",
        pad=10
    )

    ax.set_xlabel("PRD_LOAN_PURPOSE")
    ax.set_ylabel("Bad Rate (%)")

    # Grid only on Y axis
    ax.grid(
        axis="y",
        linestyle="--",
        alpha=0.3
    )

    ax.set_axisbelow(True)

    # --------------------------------------------------
    # LABELS ABOVE BARS
    # --------------------------------------------------
    max_rate = temp["Bad_Rate"].max(skipna=True)

    if pd.isna(max_rate):
        max_rate = 1

    for bar, (_, row) in zip(bars, temp.iterrows()):

        if pd.isna(row["T"]):
            continue

        height = row["Bad_Rate"]

        label = (
            f"T: {int(row['T']):,}\n"
            f"B: {int(row['B']):,}\n"
            f"{height:.1f}%"
        )

        ax.text(
            bar.get_x() + bar.get_width()/2,
            height + max_rate * 0.03,
            label,
            ha="center",
            va="bottom",
            fontsize=8
        )

    # Give labels enough space
    ax.set_ylim(
        0,
        max_rate * 1.30
    )


# --------------------------------------------------
# REMOVE UNUSED SUBPLOTS
# --------------------------------------------------
for j in range(len(plot_order), len(axes)):
    fig.delaxes(axes[j])


# --------------------------------------------------
# MAIN TITLE
# --------------------------------------------------
fig.suptitle(
    "PRD_LOAN_PURPOSE Bad Rate by BOOKYEAR",
    fontsize=18,
    fontweight="bold",
    y=1.002
)

plt.tight_layout()
plt.show()
