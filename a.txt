import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from pathlib import Path
from matplotlib.backends.backend_pdf import PdfPages
from matplotlib.ticker import FuncFormatter


def export_seed_pdf_matplotlib(
    dfs,
    filename="Random_Seed_Sensitivity_Loss_Comparison.pdf",
    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    metrics=None,
    baseline="RS42",
    claims_share=0.62,
    colors=None,
    decimals=2,
):
    """
    Create chart-only multi-page PDF.

    Pages included:
      - All Ventiles / Overall
      - Each Ventile
      - Each ECON_REGION
      - Each POD_MARKET_CODE
      - Each PRK_TENURE

    For both:
      - SMI3_LOSS
      - SMI3_LR

    Each page contains 4 bars:
      RS42, RS07, RS101, RS151

    Inside each bar:
      Min, Max, Avg, P10, P50, P90
      Pred Delq + Claims
      Pred Claims
      Pred Delqs
      Delta vs RS42

    No table.
    No Plotly/Kaleido required.
    """

    # --------------------------------------------------------
    # Validation / defaults
    # --------------------------------------------------------

    if not dfs:
        raise ValueError("dfs cannot be empty")

    if baseline not in dfs:
        raise ValueError(
            f"Baseline '{baseline}' is not in dfs"
        )

    if not 0 <= claims_share <= 1:
        raise ValueError(
            "claims_share must be between 0 and 1"
        )

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    seeds = list(dfs.keys())
    metric_names = list(metrics.keys())

    required_cols = list(dict.fromkeys([
        ventile_col,
        region_col,
        market_col,
        tenure_col,
        probability_col,
        *metric_names
    ]))

    # --------------------------------------------------------
    # Prepare data once
    # --------------------------------------------------------

    prepared = {}

    all_ventiles = set()
    all_regions = set()
    all_markets = set()
    all_tenures = set()

    for seed, source_df in dfs.items():

        missing = [
            c for c in required_cols
            if c not in source_df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed}: missing columns {missing}"
            )

        df = source_df[required_cols].copy()

        df["_Ventile"] = pd.to_numeric(
            df[ventile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        df["_Region"] = (
            df[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Market"] = (
            df[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Tenure"] = (
            df[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Probability"] = (
            pd.to_numeric(
                df[probability_col],
                errors="coerce"
            )
            .replace([np.inf, -np.inf], np.nan)
        )

        for metric in metric_names:
            df[metric] = (
                pd.to_numeric(
                    df[metric],
                    errors="coerce"
                )
                .replace([np.inf, -np.inf], np.nan)
            )

        all_ventiles.update(
            df["_Ventile"]
            .dropna()
            .astype(int)
            .tolist()
        )

        all_regions.update(
            df["_Region"].tolist()
        )

        all_markets.update(
            df["_Market"].tolist()
        )

        all_tenures.update(
            df["_Tenure"].tolist()
        )

        prepared[str(seed)] = df

    all_ventiles = sorted(all_ventiles)
    all_regions = sorted(all_regions)
    all_markets = sorted(all_markets)
    all_tenures = sorted(all_tenures)

    # --------------------------------------------------------
    # Helpers
    # --------------------------------------------------------

    def fmt_number(value):

        if value is None or not np.isfinite(value):
            return "N/A"

        return f"{value:,.{decimals}f}"


    def display_metric(value, metric):

        if value is None or not np.isfinite(value):
            return "N/A"

        cfg = metrics[metric]

        if cfg.get("is_ratio", False):

            scale = cfg.get(
                "ratio_scale",
                1
            )

            display_value = (
                value / scale * 100
            )

            return (
                f"{display_value:,.{decimals}f}%"
            )

        return f"{value:,.{decimals}f}"


    def display_delta(value, metric):

        if value is None or not np.isfinite(value):
            return "N/A"

        sign = "+" if value > 0 else ""

        cfg = metrics[metric]

        if cfg.get("is_ratio", False):

            scale = cfg.get(
                "ratio_scale",
                1
            )

            display_value = (
                value / scale * 100
            )

            return (
                f"{sign}"
                f"{display_value:,.{decimals}f} pp"
            )

        return (
            f"{sign}{value:,.{decimals}f}"
        )


    def display_delta_pct(value):

        if value is None or not np.isfinite(value):
            return "N/A"

        sign = "+" if value > 0 else ""

        return (
            f"{sign}{value:,.{decimals}f}%"
        )

    # --------------------------------------------------------
    # Calculate stats for one filtered page
    # --------------------------------------------------------

    def calculate_stats(
        metric,
        filter_type=None,
        filter_value=None
    ):

        cfg = metrics[metric]

        result = {}

        for seed in seeds:

            df = prepared[str(seed)]

            if filter_type == "ventile":

                subset = df[
                    df["_Ventile"] == filter_value
                ]

            elif filter_type == "region":

                subset = df[
                    df["_Region"] == str(filter_value)
                ]

            elif filter_type == "market":

                subset = df[
                    df["_Market"] == str(filter_value)
                ]

            elif filter_type == "tenure":

                subset = df[
                    df["_Tenure"] == str(filter_value)
                ]

            else:

                subset = df

            values = (
                subset[metric]
                .dropna()
                .to_numpy(dtype=float)
            )

            if len(values) > 0:

                total_sum = float(
                    np.sum(values)
                )

                avg = float(
                    np.mean(values)
                )

                aggregate = (
                    total_sum
                    if cfg["agg"] == "sum"
                    else avg
                )

                minimum = float(
                    np.min(values)
                )

                maximum = float(
                    np.max(values)
                )

                p10 = float(
                    np.quantile(values, 0.10)
                )

                p50 = float(
                    np.quantile(values, 0.50)
                )

                p90 = float(
                    np.quantile(values, 0.90)
                )

            else:

                aggregate = np.nan
                minimum = np.nan
                maximum = np.nan
                avg = np.nan
                p10 = np.nan
                p50 = np.nan
                p90 = np.nan

            probability = (
                subset["_Probability"]
                .dropna()
            )

            if len(probability) > 0:

                predicted_total = float(
                    probability.sum()
                )

                predicted_claims = (
                    predicted_total
                    * claims_share
                )

                predicted_delqs = (
                    predicted_total
                    * (1 - claims_share)
                )

            else:

                predicted_total = np.nan
                predicted_claims = np.nan
                predicted_delqs = np.nan

            result[str(seed)] = {
                "aggregate": aggregate,
                "min": minimum,
                "max": maximum,
                "avg": avg,
                "p10": p10,
                "p50": p50,
                "p90": p90,
                "predictedTotal": predicted_total,
                "predictedClaims": predicted_claims,
                "predictedDelqs": predicted_delqs
            }

        base_value = (
            result[str(baseline)]["aggregate"]
        )

        for seed in seeds:

            seed = str(seed)

            value = result[seed]["aggregate"]

            if (
                np.isfinite(value)
                and np.isfinite(base_value)
            ):

                delta = value - base_value

            else:

                delta = np.nan

            if (
                np.isfinite(delta)
                and np.isfinite(base_value)
                and base_value != 0
            ):

                delta_pct = (
                    delta
                    / base_value
                    * 100
                )

            else:

                delta_pct = np.nan

            result[seed]["delta"] = delta
            result[seed]["deltaPct"] = delta_pct

        return result

    # --------------------------------------------------------
    # Draw one page
    # --------------------------------------------------------

    def draw_page(
        pdf,
        metric,
        subtitle,
        filter_type=None,
        filter_value=None
    ):

        cfg = metrics[metric]

        stats = calculate_stats(
            metric=metric,
            filter_type=filter_type,
            filter_value=filter_value
        )

        values = np.array([
            stats[str(seed)]["aggregate"]
            for seed in seeds
        ], dtype=float)

        x = np.arange(
            len(seeds)
        )

        fig, ax = plt.subplots(
            figsize=(16, 9)
        )

        bars = ax.bar(
            x,
            values,
            width=0.75,
            color=[
                colors.get(
                    str(seed),
                    "#64748B"
                )
                for seed in seeds
            ]
        )

        # ----------------------------------------------------
        # Titles
        # ----------------------------------------------------

        fig.suptitle(
            f"{metric} — Random Seed Sensitivity",
            fontsize=22,
            fontweight="bold",
            y=0.96
        )

        ax.set_title(
            subtitle,
            fontsize=13,
            pad=18,
            color="#666666"
        )

        # ----------------------------------------------------
        # X-axis
        # ----------------------------------------------------

        ax.set_xticks(x)

        ax.set_xticklabels(
            [str(seed) for seed in seeds],
            fontsize=11
        )

        ax.set_xlabel(
            "Experiment",
            fontsize=11,
            labelpad=14
        )

        # ----------------------------------------------------
        # Y-axis
        # ----------------------------------------------------

        ax.set_ylabel(
            metric,
            fontsize=11
        )

        if cfg.get(
            "is_ratio",
            False
        ):

            scale = cfg.get(
                "ratio_scale",
                1
            )

            ax.yaxis.set_major_formatter(
                FuncFormatter(
                    lambda y, pos:
                    f"{y / scale * 100:.0f}%"
                )
            )

        ax.grid(
            axis="y",
            alpha=0.20
        )

        ax.set_axisbelow(True)

        ax.spines["top"].set_visible(False)
        ax.spines["right"].set_visible(False)
        ax.spines["left"].set_visible(False)

        # ----------------------------------------------------
        # Y range / headroom
        # ----------------------------------------------------

        valid_values = values[
            np.isfinite(values)
        ]

        if len(valid_values) > 0:

            minimum = min(
                0,
                float(valid_values.min())
            )

            maximum = max(
                0,
                float(valid_values.max())
            )

            span = max(
                maximum - minimum,
                1e-9
            )

            ax.set_ylim(
                minimum - span * 0.08,
                maximum + span * 0.25
            )

        # ----------------------------------------------------
        # Labels inside each bar
        # ----------------------------------------------------

        for i, bar in enumerate(bars):

            seed = str(seeds[i])

            r = stats[seed]

            aggregate = r["aggregate"]

            if not np.isfinite(aggregate):
                continue

            # Top aggregate label
            ax.text(
                bar.get_x()
                + bar.get_width() / 2,
                aggregate,
                display_metric(
                    aggregate,
                    metric
                ),
                ha="center",
                va="bottom"
                    if aggregate >= 0
                    else "top",
                fontsize=14,
                fontweight="bold"
            )

            if seed == str(baseline):

                delta_text = "Baseline"

            else:

                delta_text = display_delta(
                    r["delta"],
                    metric
                )

            lines = [
                f"Min: {display_metric(r['min'], metric)}",
                f"Max: {display_metric(r['max'], metric)}",
                f"Avg: {display_metric(r['avg'], metric)}",
                f"P10: {display_metric(r['p10'], metric)}",
                f"P50: {display_metric(r['p50'], metric)}",
                f"P90: {display_metric(r['p90'], metric)}",
                "",
                "Pred Delq + Claims: "
                + fmt_number(
                    r["predictedTotal"]
                ),
                "Pred Claims: "
                + fmt_number(
                    r["predictedClaims"]
                ),
                "Pred Delqs: "
                + fmt_number(
                    r["predictedDelqs"]
                ),
                "",
                f"Δ: {delta_text}"
            ]

            if not cfg.get(
                "is_ratio",
                False
            ):

                delta_pct_text = (
                    "Baseline"
                    if seed == str(baseline)
                    else display_delta_pct(
                        r["deltaPct"]
                    )
                )

                lines.append(
                    f"Δ%: {delta_pct_text}"
                )

            inside_text = "\n".join(
                lines
            )

            # Put text around middle of the bar
            text_y = aggregate * 0.50

            ax.text(
                bar.get_x()
                + bar.get_width() / 2,
                text_y,
                inside_text,
                ha="center",
                va="center",
                fontsize=9.5,
                fontweight="bold",
                color="white",
                linespacing=1.25
            )

        # ----------------------------------------------------
        # Footer
        # ----------------------------------------------------

        fig.text(
            0.02,
            0.025,
            "Data - October 2025 - May 2026",
            fontsize=9,
            color="#666666"
        )

        fig.text(
            0.98,
            0.025,
            "Note - Ventile means 5% increments in percentile.",
            fontsize=9,
            color="#666666",
            ha="right"
        )

        fig.tight_layout(
            rect=[0.03, 0.06, 0.98, 0.92]
        )

        pdf.savefig(
            fig,
            bbox_inches="tight"
        )

        plt.close(fig)

    # --------------------------------------------------------
    # Create all pages
    # --------------------------------------------------------

    output = Path(
        filename
    ).resolve()

    page_count = (
        len(metric_names)
        * (
            1
            + len(all_ventiles)
            + len(all_regions)
            + len(all_markets)
            + len(all_tenures)
        )
    )

    print(
        f"Creating {page_count} PDF pages..."
    )

    current_page = 0

    with PdfPages(output) as pdf:

        for metric in metric_names:

            # Overall
            current_page += 1

            print(
                f"Page {current_page}/{page_count}: "
                f"{metric} - All Ventiles"
            )

            draw_page(
                pdf,
                metric,
                "All Ventiles"
            )

            # Ventiles
            for ventile in all_ventiles:

                current_page += 1

                print(
                    f"Page {current_page}/{page_count}: "
                    f"{metric} - Ventile {ventile}"
                )

                draw_page(
                    pdf,
                    metric,
                    f"Ventile {ventile}",
                    "ventile",
                    ventile
                )

            # Regions
            for region in all_regions:

                current_page += 1

                print(
                    f"Page {current_page}/{page_count}: "
                    f"{metric} - ECON_REGION: {region}"
                )

                draw_page(
                    pdf,
                    metric,
                    f"ECON_REGION: {region}",
                    "region",
                    region
                )

            # Markets
            for market in all_markets:

                current_page += 1

                print(
                    f"Page {current_page}/{page_count}: "
                    f"{metric} - POD_MARKET_CODE: {market}"
                )

                draw_page(
                    pdf,
                    metric,
                    f"POD_MARKET_CODE: {market}",
                    "market",
                    market
                )

            # Tenure
            for tenure in all_tenures:

                current_page += 1

                print(
                    f"Page {current_page}/{page_count}: "
                    f"{metric} - PRK_TENURE: {tenure}"
                )

                draw_page(
                    pdf,
                    metric,
                    f"PRK_TENURE: {tenure}",
                    "tenure",
                    tenure
                )

    print()
    print(
        f"PDF report created: {output}"
    )

    print(
        f"File size: "
        f"{output.stat().st_size / (1024 ** 2):.2f} MB"
    )

    return output
