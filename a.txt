bad = df[df["target"] == 1].copy()

bad["COLLECTION_FLAG"] = (bad["CRA_COLLECTION"] == "Y").astype(int)
bad["BANKRUPTCY_FLAG"] = (bad["CRA_BANKRUPTCY_FORECLOSURE_IND"] == "Y").astype(int)
bad["JUDGEMENT_FLAG"] = (bad["CRA_JUDGEMENT"] == "Y").astype(int)
bad["DEROG_FLAG"] = (bad["CRD_PUBLIC_DEROG_TU"] > 0).astype(int)

flag_cols = [
    "COLLECTION_FLAG",
    "BANKRUPTCY_FLAG",
    "JUDGEMENT_FLAG",
    "DEROG_FLAG"
]

bad["ADVERSE_COUNT"] = bad[flag_cols].sum(axis=1)

overlap = (
    bad[bad["ADVERSE_COUNT"].isin([2, 3, 4])]
    .groupby(flag_cols + ["ADVERSE_COUNT"])
    .size()
    .reset_index(name="Row_Count")
    .sort_values(["ADVERSE_COUNT", "Row_Count"], ascending=[True, False])
)

overlap




def combo_name(row):
    names = []
    if row["COLLECTION_FLAG"] == 1:
        names.append("Collection")
    if row["BANKRUPTCY_FLAG"] == 1:
        names.append("Bankruptcy/Foreclosure")
    if row["JUDGEMENT_FLAG"] == 1:
        names.append("Judgement")
    if row["DEROG_FLAG"] == 1:
        names.append("Public Derog")
    return " + ".join(names)

overlap["Combination"] = overlap.apply(combo_name, axis=1)

overlap[["Combination", "ADVERSE_COUNT", "Row_Count"]]
