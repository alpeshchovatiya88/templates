import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import math

# ---------------------------------------------------
# SETTINGS
# ---------------------------------------------------
year_col = "BOOKYEAR"
category_col = "PRD_LOAN_PURPOSE"

# If you have a target/bad flag, put the column name here.
# Example: bad_col = "BAD_FLAG"
bad_col = "BAD"

# ---------------------------------------------------
# FUNCTION TO CREATE SUMMARY
# ---------------------------------------------------
def get_summary(data, label):
    out = (
        data.groupby(category_col, dropna=False)
            .agg(
                T=(bad_col, "size"),     # Total
                B=(bad_col, "sum")       # Bad count
            )
            .reset_index()
    )

    out["Bad_Rate"] = out["B"] / out["T"] * 100
    out["BOOKYEAR_LABEL"] = label

    return out

# PLOT
# ---------------------------------------------------
plot_order = ["overall"] + [str(x) for x in years]

n_plots = len(plot_order)
ncols = 4
nrows = math.ceil(n_plots / ncols)

fig, axes = plt.subplots(
    nrows=nrows,
    ncols=ncols,
    figsize=(18, 4.5 * nrows)
)

axes = np.array(axes).flatten()

for i, yr in enumerate(plot_order):

    ax = axes[i]

    temp = summary[
        summary["BOOKYEAR_LABEL"] == yr
    ].copy()

    # Sort categories if desired
    temp = temp.sort_values(category_col)

    sns.barplot(
        data=temp,
        x=category_col,
        y="Bad_Rate",
        ax=ax
    )

    ax.set_title(f"BOOKYEAR: {yr}", fontweight="bold")
    ax.set_xlabel("PRD_LOAN_PURPOSE")
    ax.set_ylabel("Bad Rate (%)")

    # Add T / B / Bad rate above each bar
    for j, row in temp.reset_index(drop=True).iterrows():

        label = (
            f"T: {int(row['T'])}\n"
            f"B: {int(row['B'])}\n"
            f"{row['Bad_Rate']:.1f}%"
        )

        ax.text(
            j,
            row["Bad_Rate"] + 0.15,
            label,
            ha="center",
            va="bottom",
            fontsize=8
        )

    ax.tick_params(axis="x", rotation=45)

    # Give some room for labels above bars
    max_rate = temp["Bad_Rate"].max()

    if pd.notna(max_rate):
        ax.set_ylim(0, max_rate * 1.30)


# Remove empty panels
for j in range(n_plots, len(axes)):
    fig.delaxes(axes[j])


plt.tight_layout()
plt.show()
