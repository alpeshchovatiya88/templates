
import pandas as pd
import numpy as np
import plotly.graph_objects as go
import ipywidgets as widgets
from IPython.display import display

def plot_seed_sensitivity(
    dfs,
    decile_col,
    value_col,
    region_col="ECON_REGION",
    agg="sum",
    baseline="RS42",
    decimals=2,
    title=None
):
    if baseline not in dfs:
        raise ValueError(f"{baseline} must be included in dfs")

    seeds = list(dfs)
    palette = ["#2563EB", "#F97316", "#10B981", "#8B5CF6"]
    colors = {
        seed: palette[i % len(palette)]
        for i, seed in enumerate(seeds)
    }

    def fmt(v):
        if pd.isna(v):
            return "N/A"
        if abs(v) >= 1e6:
            return f"{v / 1e6:,.2f}M"
        if abs(v) >= 1e3:
            return f"{v / 1e3:,.2f}K"
        return f"{v:,.{decimals}f}"

    def signed(v):
        return f"{'+' if v > 0 else ''}{fmt(v)}" if pd.notna(v) else "N/A"

    # Prepare data
    frames = []

    for seed, df in dfs.items():
        temp = df[[decile_col, value_col, region_col]].copy()

        temp["Seed"] = seed
        temp["Decile"] = pd.to_numeric(
            temp[decile_col].astype("string").str.extract(
                r"(\d+)", expand=False
            ),
            errors="coerce"
        )
        temp[value_col] = pd.to_numeric(
            temp[value_col], errors="coerce"
        )
        frames.append(temp)

    data = pd.concat(frames, ignore_index=True)
    data = data.dropna(subset=["Decile"])
    data["Decile"] = data["Decile"].astype(int)

    deciles = sorted(data["Decile"].unique())
    regions = sorted(data[region_col].dropna().unique(), key=str)

    decile_filter = widgets.Dropdown(
        options=[("All Deciles", None)] +
                [(f"Decile {d}", d) for d in deciles],
        value=None,
        description="Decile:"
    )

    region_filter = widgets.Dropdown(
        options=[("All Regions", None)] +
                [(str(r), r) for r in regions],
        value=None,
        description="ECON_REGION:",
        style={"description_width": "initial"}
    )

    output = widgets.Output()

    def update_chart(change=None):
        with output:
            output.clear_output(wait=True)

            subset = data

            if region_filter.value is not None:
                subset = subset[
                    subset[region_col] == region_filter.value
                ]

            if decile_filter.value is not None:
                subset = subset[
                    subset["Decile"] == decile_filter.value
                ]

            stats = (
                subset.groupby(["Seed", "Decile"])[value_col]
                .agg(
                    total=agg,
                    min="min",
                    max="max",
                    avg="mean",
                    p10=lambda x: x.quantile(0.10),
                    p20=lambda x: x.quantile(0.20),
                    p50="median"
                )
                .reset_index()
            )

            # Compare each seed against RS42 within the same decile
            base = (
                stats[stats["Seed"] == baseline]
                [["Decile", "total"]]
                .rename(columns={"total": "baseline_total"})
            )

            stats = stats.merge(base, on="Decile", how="left")

            stats["difference"] = (
                stats["total"] - stats["baseline_total"]
            )

            stats["pct_difference"] = np.where(
                stats["baseline_total"].notna() &
                (stats["baseline_total"] != 0),
                stats["difference"] / stats["baseline_total"] * 100,
                np.nan
            )

            stats.loc[
                stats["Seed"] == baseline,
                ["difference", "pct_difference"]
            ] = 0

            fig = go.Figure()

            selected_deciles = (
                [decile_filter.value]
                if decile_filter.value is not None
                else deciles
            )

            single_decile = decile_filter.value is not None

            for seed in seeds:
                s = (
                    stats[stats["Seed"] == seed]
                    .set_index("Decile")
                    .reindex(selected_deciles)
                )

                inside = []
                hover = []
                totals = []

                for d, r in s.iterrows():
                    if pd.isna(r["total"]):
                        inside.append("")
                        hover.append("No data")
                        totals.append("")
                        continue

                    diff_text = (
                        "Baseline"
                        if seed == baseline
                        else signed(r["difference"])
                    )

                    pct_text = (
                        "Baseline"
                        if seed == baseline
                        else (
                            f"{r['pct_difference']:+.2f}%"
                            if pd.notna(r["pct_difference"])
                            else "N/A"
                        )
                    )

                    details = (
                        f"<b>Min: {fmt(r['min'])}</b><br>"
                        f"<b>Max: {fmt(r['max'])}</b><br>"
                        f"<b>Avg: {fmt(r['avg'])}</b><br>"
                        f"<b>P10: {fmt(r['p10'])}</b><br>"
                        f"<b>P20: {fmt(r['p20'])}</b><br>"
                        f"<b>P50: {fmt(r['p50'])}</b><br>"
                        f"────────────<br>"
                        f"<b>Δ: {diff_text}</b><br>"
                        f"<b>Δ%: {pct_text}</b>"
                    )

                    inside.append(details)
                    hover.append(
                        f"<b>{seed} | Decile {d}</b><br>"
                        f"Total: {r['total']:,.{decimals}f}<br>"
                        + details
                    )
                    totals.append(f"<b>{fmt(r['total'])}</b>")

                # Permanent statistics inside each bar when one decile
                # is selected; full details remain on hover otherwise.
                fig.add_trace(go.Bar(
                    x=[str(d) for d in selected_deciles],
                    y=s["total"],
                    name=seed,
                    marker_color=colors[seed],
                    text=inside if single_decile else totals,
                    textposition="inside" if single_decile else "outside",
                    insidetextanchor="middle",
                    textfont=dict(
                        size=12 if single_decile else 13,
                        color="white" if single_decile else "black",
                        family="Arial"
                    ),
                    cliponaxis=False,
                    customdata=np.array(hover, dtype=object),
                    hovertemplate="%{customdata}<extra></extra>"
                ))

            fig.update_layout(
                title=title or f"{value_col} by Decile Across Random Seeds",
                template="plotly_white",
                barmode="group",
                height=720,
                bargap=0.20,
                bargroupgap=0.04,
                legend=dict(orientation="h", y=1.10),
                margin=dict(t=100, b=70, l=80, r=40),
                uniformtext=dict(minsize=10, mode="show")
            )

            fig.update_xaxes(
                title="Decile",
                type="category",
                categoryorder="array",
                categoryarray=[str(d) for d in deciles]
            )
            fig.update_yaxes(
                title=value_col,
                tickformat="~s",
                zeroline=True
            )

            fig.show()

            # Exact numerical comparison below the chart
            table = stats[
                ["Decile", "Seed", "total", "difference",
                 "pct_difference", "min", "max",
                 "avg", "p10", "p20", "p50"]
            ].copy()

            table = table.sort_values(
                ["Decile", "Seed"],
                key=lambda s: (
                    s.map({seed: i for i, seed in enumerate(seeds)})
                    if s.name == "Seed" else s
                )
            )

            display(
                table.style.format({
                    "total": f"{{:,.{decimals}f}}",
                    "difference": f"{{:+,.{decimals}f}}",
                    "pct_difference": "{:+.2f}%",
                    "min": f"{{:,.{decimals}f}}",
                    "max": f"{{:,.{decimals}f}}",
                    "avg": f"{{:,.{decimals}f}}",
                    "p10": f"{{:,.{decimals}f}}",
                    "p20": f"{{:,.{decimals}f}}",
                    "p50": f"{{:,.{decimals}f}}"
                }, na_rep="N/A")
            )

    decile_filter.observe(update_chart, names="value")
    region_filter.observe(update_chart, names="value")

    display(widgets.HBox([decile_filter, region_filter]), output)
    update_chart()
