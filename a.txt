
bad = df[df["target"] == 1].copy()

results = []

total_collection = (bad["CRA_COLLECTION"] == "Y").sum()

features = [
    "CRA_BANKRUPTCY_FORECLOSURE_IND",
    "CRA_JUDGEMENT",
    "CRD_PUBLIC_DEROG_TU"
]

# Individual rows
for col in features:

    if col == "CRD_PUBLIC_DEROG_TU":
        feature_flag = bad[col] > 0
    else:
        feature_flag = bad[col] == "Y"

    collection_flag = bad["CRA_COLLECTION"] == "Y"

    total_feature = feature_flag.sum()

    overlap = (collection_flag & feature_flag).sum()

    overlap_pct = round(overlap / total_feature * 100, 1)

    results.append({
        "Feature": col,
        "Total_Bad_Count": total_feature,
        "Overlap_With_CRA_COLLECTION": overlap,
        "Overlap_%": overlap_pct
    })


# Combined row
combined_flag = (
    (bad["CRA_BANKRUPTCY_FORECLOSURE_IND"] == "Y")
    | (bad["CRA_JUDGEMENT"] == "Y")
    | (bad["CRD_PUBLIC_DEROG_TU"] > 0)
)

combined_total = combined_flag.sum()

combined_overlap = (
    (bad["CRA_COLLECTION"] == "Y")
    & combined_flag
).sum()

combined_pct = round(
    combined_overlap / combined_total * 100,
    1
)

results.append({
    "Feature": "ALL_3_COMBINED",
    "Total_Bad_Count": combined_total,
    "Overlap_With_CRA_COLLECTION": combined_overlap,
    "Overlap_%": combined_pct
})

results = pd.DataFrame(results)

print("Total bad CRA_COLLECTION =", total_collection)
