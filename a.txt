import pandas as pd
import plotly.graph_objects as go
from plotly.subplots import make_subplots

def plot_by_decile(
    dfs,
    decile_col,
    value_col,
    agg="sum",
    title=None,
    decimals=2
):
    # Prepare summary
    summary = pd.DataFrame({
        name: df.assign(
            _decile=pd.to_numeric(
                df[decile_col].astype(str).str.extract(r"(\d+)", expand=False),
                errors="coerce"
            )
        ).groupby("_decile")[value_col].agg(agg)
        for name, df in dfs.items()
    }).sort_index()

    summary.index = summary.index.astype(int)
    summary = summary.reindex(range(1, 21))

    colors = ["#2563EB", "#F97316", "#10B981", "#8B5CF6"]

    fig = make_subplots(
        rows=2, cols=1,
        specs=[[{"type": "xy"}], [{"type": "table"}]],
        row_heights=[0.72, 0.28],
        vertical_spacing=0.10
    )

    # Grouped bars
    for i, name in enumerate(summary.columns):
        values = summary[name]

        fig.add_trace(
            go.Bar(
                x=summary.index.astype(str),
                y=values,
                name=name,
                marker_color=colors[i % len(colors)],
                text=[
                    f"{v/1e6:.1f}M" if abs(v) >= 1e6
                    else f"{v/1e3:.1f}K" if abs(v) >= 1e3
                    else f"{v:,.{decimals}f}"
                    if pd.notna(v) else ""
                    for v in values
                ],
                textposition="outside",
                textfont=dict(size=10),
                cliponaxis=False,
                customdata=values,
                hovertemplate=(
                    "Decile: %{x}<br>"
                    f"Seed: {name}<br>"
                    "Value: %{y:,.2f}<extra></extra>"
                )
            ),
            row=1, col=1
        )

    # Full-value comparison table
    table_values = [
        summary.index.tolist(),
        *[
            [
                f"{v:,.{decimals}f}" if pd.notna(v) else "—"
                for v in summary[name]
            ]
            for name in summary.columns
        ]
    ]

    fig.add_trace(
        go.Table(
            header=dict(
                values=["Decile"] + list(summary.columns),
                fill_color="#1E3A5F",
                font=dict(color="white", size=12),
                align="center",
                height=30
            ),
            cells=dict(
                values=table_values,
                fill_color="#F3F6FA",
                align="center",
                font=dict(size=11),
                height=25
            )
        ),
        row=2, col=1
    )

    # Dropdown: show all or zoom into one decile
    buttons = [
        dict(
            label="All Deciles",
            method="relayout",
            args=[{
                "xaxis.range": [-0.5, 19.5],
                "yaxis.autorange": True
            }]
        )
    ]

    for d in summary.index:
        row_values = summary.loc[d].dropna()
        ymin = min(0, row_values.min()) if len(row_values) else 0
        ymax = max(0, row_values.max()) if len(row_values) else 1
        padding = max((ymax - ymin) * 0.20, 1)

        buttons.append(
            dict(
                label=f"Decile {d}",
                method="relayout",
                args=[{
                    "xaxis.range": [d - 1.5, d - 0.5],
                    "yaxis.range": [ymin - padding, ymax + padding]
                }]
            )
        )

    fig.update_layout(
        title=dict(
            text=title or f"{value_col} by Decile — Random Seed Sensitivity",
            x=0.5
        ),
        template="plotly_white",
        barmode="group",
        height=850,
        width=1500,
        bargap=0.25,
        bargroupgap=0.05,
        legend=dict(
            orientation="h",
            y=1.10,
            x=0.5,
            xanchor="center"
        ),
        updatemenus=[dict(
            buttons=buttons,
            direction="down",
            x=1,
            xanchor="right",
            y=1.17,
            yanchor="top"
        )],
        margin=dict(t=140, b=40, l=70, r=40),
        uniformtext=dict(minsize=9, mode="hide")
    )

    fig.update_xaxes(
        title_text="Decile",
        type="category",
        categoryorder="array",
        categoryarray=[str(i) for i in range(1, 21)],
        row=1, col=1
    )

    fig.update_yaxes(
        title_text=value_col,
        tickformat="~s",
        zeroline=True,
        row=1, col=1
    )

    fig.show()
    return summary
