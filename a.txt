
import pandas as pd
import matplotlib.pyplot as plt

def plot_indicator(df, indicator, reference_date=202212):

    indicator = indicator.upper()

    col = {
        "HPI": "HPI_+4",
        "UE": "UE_+4"
    }[indicator]

    data = df.copy()
    data["BOOKYEAR"] = pd.to_numeric(data["BOOKYEAR"])

    # Get reference value at December 2022
    reference_value = data.loc[
        data["BOOKYEAR"] == reference_date, col
    ].mean()

    if pd.isna(reference_value):
        raise ValueError(
            f"No reference data found for {reference_date}"
        )

    windows = {
        "Train": (200901, 201512, "blue"),
        "Test": (202101, 202312, "red"),
        "Validation": (202510, 202606, "orange")
    }

    fig, ax = plt.subplots(figsize=(12, 5))

    for name, (start, end, color) in windows.items():

        subset = data.loc[
            data["BOOKYEAR"].between(start, end)
        ]

        series = (
            subset.groupby("BOOKYEAR")[col]
            .mean()
            .sort_index()
        )

        if series.empty:
            continue

        dates = pd.to_datetime(
            series.index.astype(str), format="%Y%m"
        )

        ax.plot(
            dates,
            series.values,
            color=color,
            linewidth=2,
            label=name
        )

    ax.axhline(
        reference_value,
        color="green",
        linestyle="--",
        linewidth=2,
        label=f"{indicator} reference ({reference_date})"
    )

    ax.set_title(f"{indicator} +4: Train, Test & Validation")
    ax.set_xlabel("BOOKYEAR")
    ax.set_ylabel(col)
    ax.legend()
    ax.grid(alpha=0.3)

    plt.tight_layout()
    plt.show()

==========

plot_indicator(df, "HPI")
