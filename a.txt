import plotly.express as px

def plot_numeric_distributions(data, numeric_cols, bins=50):

    for col in numeric_cols:

        fig = px.histogram(
            data,
            x=col,
            nbins=bins,
            histnorm="probability density",
            marginal="box",
            title=f"Distribution of {col}"
        )

        fig.update_layout(
            xaxis_title=col,
            yaxis_title="Density",
            hovermode="x"
        )

        fig.show()
