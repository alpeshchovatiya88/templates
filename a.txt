def psi_yoy_quarter(
    df,
    period_col,
    numeric_cols,
    categorical_cols,
    start_year,
    end_year
):

    results = []

    for year in range(start_year + 1, end_year + 1):

        for q in [1, 2, 3, 4]:

            previous_period = f"{year-1}_Q{q}"
            current_period = f"{year}_Q{q}"

            expected = df[df[period_col] == previous_period]
            actual = df[df[period_col] == current_period]

            if expected.empty or actual.empty:
                print(
                    f"Skipping {previous_period} vs {current_period}"
                )
                continue

            temp = run_psi_for_model(
                expected=expected,
                actual=actual,
                model_name="funded_model",
                numeric_cols=numeric_cols,
                categorical_cols=categorical_cols
            )

            temp["period"] = current_period

            results.append(temp)

    result = pd.concat(results, ignore_index=True)

    psi_wide = result.pivot(
        index="feature",
        columns="period",
        values="psi"
    ).reset_index()

    return result, psi_wide
