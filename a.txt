
import pandas as pd
import numpy as np
import plotly.graph_objects as go
import ipywidgets as widgets
from IPython.display import display

def plot_seed_sensitivity(
    dfs,
    decile_col,
    value_col,
    region_col="ECON_REGION",
    agg="sum",
    baseline="RS42",
    decimals=2,
    title=None,
    colors=None,
    inside_font_size=16,
    total_font_size=17
):
    if baseline not in dfs:
        raise ValueError(f"{baseline} not found in dfs")

    seeds = list(dfs)

    if colors is None:
        colors = {
            "RS42": "#264E86",
            "RS07": "#D97736",
            "RS101": "#21867A",
            "RS151": "#7856A6"
        }

    fallback = ["#264E86", "#D97736", "#21867A", "#7856A6"]

    def fmt(v):
        if pd.isna(v):
            return "N/A"
        if abs(v) >= 1e6:
            return f"{v / 1e6:,.2f}M"
        if abs(v) >= 1e3:
            return f"{v / 1e3:,.2f}K"
        return f"{v:,.{decimals}f}"

    def signed(v):
        if pd.isna(v):
            return "N/A"
        return ("+" if v > 0 else "") + fmt(v)

    # Prepare scored data
    frames = []

    for seed, df in dfs.items():
        temp = df[[decile_col, value_col, region_col]].copy()

        temp["Seed"] = seed

        temp["Decile"] = pd.to_numeric(
            temp[decile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        temp[value_col] = pd.to_numeric(
            temp[value_col],
            errors="coerce"
        )

        frames.append(temp)

    data = pd.concat(frames, ignore_index=True)
    data = data.dropna(subset=["Decile"])
    data["Decile"] = data["Decile"].astype(int)

    deciles = sorted(data["Decile"].unique())
    regions = sorted(
        data[region_col].dropna().unique(),
        key=str
    )

    # Filters
    decile_filter = widgets.Dropdown(
        options=[("All Deciles", None)] +
                [(f"Decile {d}", d) for d in deciles],
        value=None,
        description="Decile:"
    )

    region_filter = widgets.Dropdown(
        options=[("All Regions", None)] +
                [(str(r), r) for r in regions],
        value=None,
        description="ECON_REGION:",
        style={"description_width": "initial"}
    )

    output = widgets.Output()

    def update_chart(change=None):
        with output:
            output.clear_output(wait=True)

            subset = data.copy()

            # ECON_REGION filter
            if region_filter.value is not None:
                subset = subset[
                    subset[region_col] == region_filter.value
                ]

            # Decile filter
            all_deciles = decile_filter.value is None

            if all_deciles:
                # Combine all deciles into one summary group
                subset["Plot_Decile"] = 0
                group_label = "All Deciles"
            else:
                subset = subset[
                    subset["Decile"] == decile_filter.value
                ].copy()

                subset["Plot_Decile"] = decile_filter.value
                group_label = f"Decile {decile_filter.value}"

            # Statistics for the selected population
            stats = (
                subset.groupby("Seed")[value_col]
                .agg(
                    total=agg,
                    min="min",
                    max="max",
                    avg="mean",
                    p10=lambda x: x.quantile(0.10),
                    p20=lambda x: x.quantile(0.20),
                    p50="median",
                    count="count"
                )
                .reindex(seeds)
            )

            baseline_total = stats.loc[baseline, "total"]

            stats["difference"] = (
                stats["total"] - baseline_total
            )

            stats["pct_difference"] = (
                stats["difference"] / baseline_total * 100
                if pd.notna(baseline_total) and baseline_total != 0
                else np.nan
            )

            stats.loc[baseline, ["difference", "pct_difference"]] = 0

            # Explicit numeric x positions prevent overlapping SUM labels
            x_positions = np.arange(len(seeds))

            fig = go.Figure()

            for i, seed in enumerate(seeds):
                r = stats.loc[seed]

                if pd.isna(r["total"]):
                    continue

                diff = (
                    "Baseline"
                    if seed == baseline
                    else signed(r["difference"])
                )

                pct = (
                    "Baseline"
                    if seed == baseline
                    else (
                        f"{r['pct_difference']:+.2f}%"
                        if pd.notna(r["pct_difference"])
                        else "N/A"
                    )
                )

                inside_text = (
                    f"<b>Min: {fmt(r['min'])}</b><br>"
                    f"<b>Max: {fmt(r['max'])}</b><br>"
                    f"<b>Avg: {fmt(r['avg'])}</b><br>"
                    f"<b>P10: {fmt(r['p10'])}</b><br>"
                    f"<b>P20: {fmt(r['p20'])}</b><br>"
                    f"<b>P50: {fmt(r['p50'])}</b><br>"
                    f"────────────<br>"
                    f"<b>Δ: {diff}</b><br>"
                    f"<b>Δ%: {pct}</b>"
                )

                hover_text = (
                    f"<b>{seed} | {group_label}</b><br>"
                    f"<b>{agg.upper()}: "
                    f"{r['total']:,.{decimals}f}</b><br>"
                    f"Count: {int(r['count']):,}<br>"
                    + inside_text
                )

                # Main bar
                fig.add_trace(go.Bar(
                    x=[i],
                    y=[r["total"]],
                    width=0.72,
                    name=seed,
                    marker=dict(
                        color=colors.get(
                            seed,
                            fallback[i % len(fallback)]
                        ),
                        line=dict(color="white", width=1.5)
                    ),
                    text=[inside_text],
                    textposition="inside",
                    insidetextanchor="middle",
                    textfont=dict(
                        size=inside_font_size,
                        color="white",
                        family="Arial"
                    ),
                    customdata=[hover_text],
                    hovertemplate="%{customdata}<extra></extra>",
                    showlegend=False
                ))

                # SUM above the correct bar
                fig.add_annotation(
                    x=i,
                    y=float(r["total"]),
                    xref="x",
                    yref="y",
                    text=f"<b>{agg.upper()}: {fmt(r['total'])}</b>",
                    showarrow=False,
                    yshift=18,
                    font=dict(
                        size=total_font_size,
                        color="#172033",
                        family="Arial Black"
                    )
                )

            # Give labels extra space above the tallest bar
            values = stats["total"].dropna()
            if not values.empty:
                ymin = min(0, values.min())
                ymax = max(0, values.max())
                padding = max((ymax - ymin) * 0.18, 0.01)
                y_range = [ymin - padding * 0.15, ymax + padding]
            else:
                y_range = [0, 1]

            fig.update_layout(
                title=dict(
                    text=(
                        title or
                        f"{value_col} — Random Seed Sensitivity"
                    ) + f"<br><sup>{group_label} | "
                        f"{region_filter.value if region_filter.value is not None else 'All Regions'}</sup>",
                    x=0.5,
                    font=dict(size=22, color="#172033")
                ),
                template="plotly_white",
                height=800,
                plot_bgcolor="white",
                paper_bgcolor="white",
                font=dict(
                    family="Arial",
                    size=15,
                    color="#172033"
                ),
                margin=dict(t=150, b=90, l=100, r=60),
                bargap=0.25,
                uniformtext=dict(minsize=12, mode="show")
            )

            fig.update_xaxes(
                title="Random Seed",
                tickmode="array",
                tickvals=x_positions,
                ticktext=seeds,
                range=[-0.6, len(seeds) - 0.4],
                tickfont=dict(size=16),
                showgrid=False
            )

            fig.update_yaxes(
                title=value_col,
                tickformat="~s",
                range=y_range,
                tickfont=dict(size=14),
                gridcolor="#E7ECF2",
                zeroline=True
            )

            fig.show()

            # Detailed comparison table
            table = stats.reset_index().rename(
                columns={"Seed": "Random Seed"}
            )

            table.insert(0, "Selection", group_label)

            display(
                table.style.format({
                    "total": f"{{:,.{decimals}f}}",
                    "difference": f"{{:+,.{decimals}f}}",
                    "pct_difference": "{:+.2f}%",
                    "min": f"{{:,.{decimals}f}}",
                    "max": f"{{:,.{decimals}f}}",
                    "avg": f"{{:,.{decimals}f}}",
                    "p10": f"{{:,.{decimals}f}}",
                    "p20": f"{{:,.{decimals}f}}",
                    "p50": f"{{:,.{decimals}f}}",
                    "count": "{:,.0f}"
                }, na_rep="N/A")
            )

    decile_filter.observe(update_chart, names="value")
    region_filter.observe(update_chart, names="value")

    display(
        widgets.HBox([decile_filter, region_filter]),
        output
    )

    update_chart()
