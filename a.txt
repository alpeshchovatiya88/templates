def export_interactive_seed_report(
    dfs,
    filename="Random_Seed_Sensitivity_Report.html",
    decile_col="rr_model_decile",
    region_col="ECON_REGION",
    metrics=None,
    baseline="RS42",
    colors=None,
    decimals=2,
    inside_font_size=15,
    total_font_size=18,
):
    """
    Export a fully standalone interactive HTML report.

    Filters:
        - Metric
        - Decile
        - ECON_REGION

    The HTML works without Python, Jupyter, or internet.

    metrics example:
        {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    ratio_scale:
        1   -> 0.125 represents 12.5%
        100 -> 12.5 represents 12.5%
    """

    if baseline not in dfs:
        raise ValueError(
            f"Baseline {baseline!r} is not in dfs."
        )

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    seeds = list(dfs)

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#D97736",
            "RS101": "#147D74",
            "RS151": "#7952B3"
        }

    # ----------------------------
    # Prepare input observations
    # ----------------------------

    required = [decile_col, region_col] + list(metrics)
    frames = []

    for seed, df in dfs.items():
        missing = [
            col for col in required
            if col not in df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed} is missing columns: {missing}"
            )

        t = df[required].copy()

        t["Seed"] = seed

        t["Decile"] = pd.to_numeric(
            t[decile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        t["Region"] = (
            t[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        for metric in metrics:
            t[metric] = pd.to_numeric(
                t[metric],
                errors="coerce"
            )

        frames.append(
            t[["Seed", "Decile", "Region"] + list(metrics)]
        )

    data = pd.concat(frames, ignore_index=True)

    data = data.dropna(subset=["Decile"]).copy()
    data["Decile"] = data["Decile"].astype(int)

    deciles = sorted(data["Decile"].unique().tolist())
    regions = sorted(data["Region"].unique().tolist())

    # --------------------------------------------------
    # Precompute every metric/decile/region combination.
    #
    # Browser filtering will use these statistics.
    # No Python calculations are required after export.
    # --------------------------------------------------

    records = {}

    for metric, config in metrics.items():

        aggregation = config.get("agg", "sum")

        if aggregation not in ("sum", "mean"):
            raise ValueError(
                f"Only 'sum' and 'mean' are supported: {metric}"
            )

        for decile in [None] + deciles:

            for region in [None] + regions:

                subset = data

                if decile is not None:
                    subset = subset[
                        subset["Decile"] == decile
                    ]

                if region is not None:
                    subset = subset[
                        subset["Region"] == region
                    ]

                grouped = (
                    subset.groupby("Seed")[metric]
                    .agg(
                        total=aggregation,
                        min="min",
                        max="max",
                        avg="mean",
                        p10=lambda x: x.quantile(.10),
                        p20=lambda x: x.quantile(.20),
                        p50="median",
                        count="count"
                    )
                    .reindex(seeds)
                )

                baseline_total = grouped.loc[
                    baseline, "total"
                ]

                rows = []

                for seed in seeds:
                    r = grouped.loc[seed]

                    total = r["total"]

                    if (
                        pd.notna(total)
                        and pd.notna(baseline_total)
                    ):
                        difference = total - baseline_total
                    else:
                        difference = np.nan

                    if (
                        pd.notna(difference)
                        and baseline_total != 0
                    ):
                        pct_difference = (
                            difference / baseline_total * 100
                        )
                    else:
                        pct_difference = np.nan

                    if seed == baseline and pd.notna(total):
                        difference = 0.0
                        pct_difference = 0.0

                    def clean(v):
                        if pd.isna(v):
                            return None
                        return float(v)

                    rows.append({
                        "seed": seed,
                        "total": clean(total),
                        "min": clean(r["min"]),
                        "max": clean(r["max"]),
                        "avg": clean(r["avg"]),
                        "p10": clean(r["p10"]),
                        "p20": clean(r["p20"]),
                        "p50": clean(r["p50"]),
                        "count": int(r["count"]),
                        "difference": clean(difference),
                        "pct_difference": clean(pct_difference)
                    })

                key = json.dumps(
                    [metric, decile, region],
                    separators=(",", ":")
                )

                records[key] = rows

    # ----------------------------
    # Embed data and Plotly JS
    # ----------------------------

    report_data = {
        "seeds": seeds,
        "baseline": baseline,
        "deciles": deciles,
        "regions": regions,
        "metrics": metrics,
        "colors": colors,
        "decimals": decimals,
        "insideFontSize": inside_font_size,
        "totalFontSize": total_font_size,
        "records": records
    }

    # Escape '<' to keep embedded JSON safe in HTML.
    report_json = json.dumps(
        report_data,
        ensure_ascii=False,
        allow_nan=False
    ).replace("<", "\\u003c")

    const report = JSON.parse(

    function addOption(select, value, label) {
        const option = document.createElement("option");
        option.value = value;
        option.textContent = label;
        select.appendChild(option);
    }

    Object.keys(report.metrics).forEach(metric => {
        addOption(metricFilter, metric, metric);
    });

    addOption(decileFilter, "", "All Deciles");

    report.deciles.forEach(d => {
        addOption(decileFilter, String(d), "Decile " + d);
    });

    addOption(regionFilter, "", "All Regions");

    report.regions.forEach(r => {
        addOption(regionFilter, r, r);
    });

    function formatNumber(v, metric, compact=false) {
        if (v === null || v === undefined || !Number.isFinite(v)) {
            return "N/A";
        }

        const config = report.metrics[metric];

        if (config.is_ratio) {
            const scale = config.ratio_scale || 1;
            return (v / scale * 100).toLocaleString(
                "en-US",
                {
                    minimumFractionDigits: report.decimals,
                    maximumFractionDigits: report.decimals
                }
            ) + "%";
        }

        if (compact && Math.abs(v) >= 1000000) {
            return (v / 1000000).toFixed(2) + "M";
        }

        if (compact && Math.abs(v) >= 1000) {
            return (v / 1000).toFixed(2) + "K";
        }

        return v.toLocaleString(
            "en-US",
            {
                minimumFractionDigits: report.decimals,
                maximumFractionDigits: report.decimals
            }
        );
    }

    function signedNumber(v, metric) {
        if (v === null || !Number.isFinite(v)) {
            return "N/A";
        }

        return (v > 0 ? "+" : "") + formatNumber(v, metric, true);
    }

    function signedPercent(v) {
        if (v === null || !Number.isFinite(v)) {
            return "N/A";
        }

        return (v > 0 ? "+" : "") + v.toFixed(2) + "%";
    }

    function escapeHtml(value) {
        return String(value)
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;");
    }

    function update() {

        const metric = metricFilter.value;

        const decile = decileFilter.value === ""
            ? null
            : Number(decileFilter.value);

        const region = regionFilter.value === ""
            ? null
            : regionFilter.value;

        const key = JSON.stringify([metric, decile, region]);
        const rows = report.records[key] || [];

        const config = report.metrics[metric];

        const aggregateLabel = config.agg.toUpperCase();

        const decileLabel = decile === null
            ? "All Deciles"
            : "Decile " + decile;

        const regionLabel = region === null
            ? "All Regions"
            : region;

        const traces = [];
        const annotations = [];

        rows.forEach((r, i) => {

            if (r.total === null) {
                return;
            }

            const deltaText = r.seed === report.baseline

                : signedNumber(r.difference, metric);

            const pctText = r.seed === report.baseline
                ? "Baseline"
                : signedPercent(r.pct_difference);

            const insideText = [
                "<b>Min: " + formatNumber(r.min, metric, true) + "</b>",

                "Count: " + r.count,
                insideText
            ].join("<br>");

            traces.push({

                width: 0.72,
                name: r.seed,

        let maxValue = Math.max(0, ...values);

            minValue - span * 0.08,

        ];
