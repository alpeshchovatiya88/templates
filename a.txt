METRICS = {
    "SMI3_LOSS": "sum",
    "SMI3_LR": "mean"

# ============================================================
# 1. PREPARE DATA — SHARED BY HTML AND PDF
# ============================================================

def prepare_seed_data(
    dfs,
    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability"
):
    segments = {
        "Ventile": ventile_col,
        "ECON_REGION": region_col,
        "POD_MARKET_CODE": market_col,
        "PRK_TENURE": tenure_col
    }

    required = list(dict.fromkeys(
        list(segments.values())
        + list(METRICS)
        + [probability_col]
    ))

    prepared = []

    for seed, df in dfs.items():

        missing = [c for c in required if c not in df.columns]

        if missing:
            raise ValueError(f"{seed}: missing columns {missing}")

        t = df[required].copy()
        t["Seed"] = seed

        t["Ventile"] = (
            pd.to_numeric(
                t[ventile_col]
                .astype("string")
                .str.extract(r"(\d+)", expand=False),
                errors="coerce"
            )
            .astype("Int64")
            .astype("string")
            .fillna("(Missing)")
        )

        for label, source in list(segments.items())[1:]:
            t[label] = (
                t[source]
                .astype("string")
                .fillna("(Missing)")
            )

        for metric in [*METRICS, probability_col]:
            t[metric] = (
                pd.to_numeric(t[metric], errors="coerce")
                .replace([np.inf, -np.inf], np.nan)
            )

        prepared.append(
            t[
                [
                    "Seed",
                    *segments.keys(),
                    *METRICS,
                    probability_col
                ]
            ]
        )

    return pd.concat(prepared, ignore_index=True)


# ============================================================
# 2. SHARED STATISTICS FOR PDF
# ============================================================

def summarize_seed_data(
    data,
    metric,
    segment=None,
    baseline="RS42",
    probability_col="prediction_probability",
    claims_share=0.62,
    lr_scale=1
):
    keys = ["Seed"] if segment is None else [segment, "Seed"]

    grouped = data.groupby(
        keys,
        dropna=False,
        sort=False
    )

    result = grouped.agg(
        N=(metric, "count"),
        Aggregate=(metric, METRICS[metric]),
        Min=(metric, "min"),
        Max=(metric, "max"),
        Avg=(metric, "mean"),
        P10=(metric, lambda x: x.quantile(0.10)),
        P50=(metric, lambda x: x.quantile(0.50)),
        P90=(metric, lambda x: x.quantile(0.90)),
        Predicted=(
            probability_col,
            lambda x: x.sum(min_count=1)
        )
    ).reset_index()

    result["Claims"] = result["Predicted"] * claims_share
    result["Delqs"] = result["Predicted"] * (1 - claims_share)

    if metric == "SMI3_LR":
        for col in [
            "Aggregate", "Min", "Max",
            "Avg", "P10", "P50", "P90"
        ]:
            result[col] *= lr_scale

    if segment is None:

        baseline_rows = result.loc[
            result["Seed"] == baseline,
            "Aggregate"
        ]

        result["Baseline"] = (
            baseline_rows.iloc[0]
            if not baseline_rows.empty
            else np.nan
        )

    else:

        baseline_values = (
            result.loc[
                result["Seed"] == baseline,
                [segment, "Aggregate"]
            ]
            .rename(columns={"Aggregate": "Baseline"})
        )

        result = result.merge(
            baseline_values,
            on=segment,
            how="left",
            validate="many_to_one"
        )

    result["Delta"] = (
        result["Aggregate"] - result["Baseline"]
    )

    result["DeltaPct"] = np.where(
        result["Baseline"].notna()
        & result["Baseline"].ne(0),
        result["Delta"] / result["Baseline"] * 100,
        np.nan
    )

    if segment == "Ventile":

        result["_sort"] = pd.to_numeric(
            result["Ventile"],
            errors="coerce"
        )

        result = (
            result.sort_values(
                ["_sort", "Seed"],
                na_position="last"
            )
            .drop(columns="_sort")
        )

    elif segment is not None:
        result = result.sort_values([segment, "Seed"])

    else:
        result = result.sort_values("Seed")

    return result

def export_seed_pdf(
    data,
    seeds,
    filename="Random_Seed_Sensitivity_Detailed.pdf",
    baseline="RS42",
    probability_col="prediction_probability",
    claims_share=0.62,
    lr_scale=1,
    decimals=2
):

    page_size = landscape(A3)

    doc = SimpleDocTemplate(
        str(Path(filename).resolve()),
        pagesize=page_size,
        leftMargin=30,
        rightMargin=30,
        topMargin=38,
        bottomMargin=38
    )

    styles = getSampleStyleSheet()

    styles.add(
        ParagraphStyle(
            name="SeedReportTitle",
            parent=styles["Title"],
            fontSize=22,
            leading=27,
            textColor=colors.HexColor("#17365D"),
            spaceAfter=14
        )
    )

    styles.add(
        ParagraphStyle(
            name="SeedSectionTitle",
            parent=styles["Heading2"],
            fontSize=15,
            spaceBefore=16,
            spaceAfter=10,
            textColor=colors.HexColor("#17365D")
        )
    )

    styles.add(
        ParagraphStyle(
            name="SeedTableCell",
            fontName="Helvetica",
            fontSize=6.5,
            leading=8
        )
    )

    story = []

    # --------------------------------------------------------
    # PDF chart
    # --------------------------------------------------------

    def add_chart(frame, segment, metric):

        group_col = segment if segment else "Portfolio"

        view = frame.copy()

        if segment is None:
            view[group_col] = "All Records"

        pivot = view.pivot(
            index=group_col,
            columns="Seed",
            values="Aggregate"
        )

        if pivot.empty:
            return

        if segment == "Ventile":
            pivot = pivot.reindex(
                sorted(
                    pivot.index,
                    key=lambda x: (
                        not str(x).isdigit(),
                        int(x) if str(x).isdigit() else 999999
                    )
                )
            )

        fig, ax = plt.subplots(figsize=(15, 4.3))

        x = np.arange(len(pivot))
        width = 0.8 / len(seeds)

        for i, seed in enumerate(seeds):

            if seed not in pivot.columns:
                continue

            ax.bar(
                x - 0.4 + (i + 0.5) * width,
                pivot[seed],
                width,
                label=seed,
                color=SEED_COLORS.get(seed, "#64748B")
            )

        ax.set_xticks(x)

        ax.set_xticklabels(
            pivot.index.astype(str),
            rotation=90 if len(pivot) > 12 else 0,
            fontsize=7
        )

        ax.set_ylabel(
            "Total Loss"
            if metric == "SMI3_LOSS"
            else "Average LR (%)"
        )

        ax.legend(ncol=4, fontsize=8)
        ax.grid(axis="y", alpha=0.2)

        fig.tight_layout()

        buffer = BytesIO()

        fig.savefig(
            buffer,
            format="png",
            dpi=130,
            bbox_inches="tight"
        )

        plt.close(fig)
        buffer.seek(0)

        story.append(
            Image(
                buffer,
                width=950,
                height=272
            )
        )

    # --------------------------------------------------------
    # PDF table
    # --------------------------------------------------------

    def add_table(frame, segment, metric):

        first_cols = [segment] if segment else []

        columns = first_cols + [
            "Seed",
            "N",
            "Aggregate",
            "Delta"
        ]

        if metric == "SMI3_LOSS":
            columns.append("DeltaPct")

        columns += [
            "Min",
            "Max",
            "Avg",
            "P10",
            "P50",
            "P90",
            "Predicted",
            "Claims",
            "Delqs"
        ]

        headings = {
            "Aggregate": (
                "SUM"
                if metric == "SMI3_LOSS"
                else "MEAN (%)"
            ),
            "Delta": (
                "Delta"
                if metric == "SMI3_LOSS"
                else "Delta (pp)"
            ),
            "DeltaPct": "Delta (%)",
            "Predicted": "Pred Total",
            "Claims": "Pred Claims",
            "Delqs": "Pred Delqs"
        }

        def fmt(value, col):

            if col == segment or col == "Seed":
                return str(value)

            if pd.isna(value):
                return "-"

            if col == "N":
                return f"{int(value):,}"

            return f"{value:,.{decimals}f}"

        values = [[
            Paragraph(
                escape(headings.get(col, col)),
                styles["SeedTableCell"]
            )
            for col in columns
        ]]

        for _, row in frame.iterrows():

            values.append([
                Paragraph(
                    escape(fmt(row[col], col)),
                    styles["SeedTableCell"]
                )
                for col in columns
            ])

        available_width = page_size[0] - 60

        weights = [
            2.0 if col == segment
            else 1.1 if col == "Seed"
            else 1.0
            for col in columns
        ]

        widths = [
            available_width * weight / sum(weights)
            for weight in weights
        ]

        table = Table(
            values,
            colWidths=widths,
            repeatRows=1,
            hAlign="LEFT"
        )

        table.setStyle(
            TableStyle([
                (
                    "BACKGROUND",
                    (0, 0), (-1, 0),
                    colors.HexColor("#17365D")
                ),
                (
                    "TEXTCOLOR",
                    (0, 0), (-1, 0),
                    colors.white
                ),
                (
                    "ROWBACKGROUNDS",
                    (0, 1), (-1, -1),
                    [
                        colors.white,
                        colors.HexColor("#F2F5F9")
                    ]
                ),
                (
                    "VALIGN",

def export_seed_reports(
    dfs,
    html_filename="Random_Seed_Sensitivity_Interactive.html",
    pdf_filename="Random_Seed_Sensitivity_Detailed.pdf",
    baseline="RS42",
    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    claims_share=0.62,
    lr_scale=1,
    decimals=2
):

    if baseline not in dfs:
        raise ValueError(f"Baseline {baseline} not found")

    if not 0 <= claims_share <= 1:
        raise ValueError("claims_share must be between 0 and 1")

    data = prepare_seed_data(
        dfs=dfs,
        ventile_col=ventile_col,
        region_col=region_col,
        market_col=market_col,
        tenure_col=tenure_col,
        probability_col=probability_col
    )

    shared = dict(
        data=data,
        seeds=list(dfs),
        baseline=baseline,
        probability_col=probability_col,
        claims_share=claims_share,
        lr_scale=lr_scale,
        decimals=decimals
    )

    html_path = export_seed_html(
        filename=html_filename,
        **shared
    )

    pdf_path = export_seed_pdf(
        filename=pdf_filename,
        **shared
    )

    print("\nBoth reports generated successfully.")

    return html_path, pdf_path
