import plotly.express as px

target_dist = df["target"].value_counts().sort_index().reset_index()
target_dist.columns = ["Target", "Count"]

target_dist["Percentage"] = (
    target_dist["Count"] / target_dist["Count"].sum() * 100
)

target_dist["Label"] = (
    target_dist["Count"].map("{:,}".format)
    + " (" +
    target_dist["Percentage"].map("{:.2f}%".format)
    + ")"
)

fig = px.bar(
    target_dist,
    x="Target",
    y="Count",
    text="Label",
    title="Target Distribution"
)

fig.update_traces(
    textposition="outside"
)

fig.update_layout(
    xaxis_title="Target",
    yaxis_title="Number of Records"
)

fig.show()
