import os
import sys

sys.path.insert(
    0,
    os.path.join(os.path.dirname(__file__), "pipelines", "smi3")
)

from pipelines.smi3.preprocess import (
    build_policy_query,
    build_borrower_query,
)

# Use the SAME dt_model as your pipeline
DT_MODEL = "202606"

REGION = "ca-central-1"

OUTPUT_DIR = "raw_data"

os.makedirs(OUTPUT_DIR, exist_ok=True)


print("Loading POLICY from Snowflake...")

policy = load_data_from_snowflake(
    build_policy_query(dt_model=DT_MODEL),
    region=REGION,
)

    os.path.join(OUTPUT_DIR, "policy_raw.csv"),
    index=False,
)

print("Policy shape:", policy.shape)

borrower = load_data_from_snowflake(
    build_borrower_query(dt_model=DT_MODEL),
    region=REGION,

borrower.to_csv(
    os.path.join(OUTPUT_DIR, "borrower_raw.csv"),
    index=False,
)
