def validate_columns(df, required_cols, name):
    missing = [c for c in required_cols if c not in df.columns]

    if missing:
        raise KeyError(
            f"{name} missing required columns: {missing}"
        )


def normalize_columns(df):
    df = df.copy()
    df.columns = df.columns.str.strip()
    return df

def clean_policy(policy):

    policy = normalize_columns(policy)

    validate_columns(
        policy,
        [
            "APPLICATION_NUMBER",
            "RECEIVED_DAY_SID",
            "EFFECT_DAY_SID",
            "LOB_CODE",
            "LRUPLUS_FLAG",
        ],
        "policy",
    )

    rows_before = len(policy)

    policy = policy[
        policy["APPLICATION_NUMBER"].notna()
    ].copy()

    policy["APPLICATION_NUMBER"] = (
        policy["APPLICATION_NUMBER"].astype("int64")
    )

    policy["RECEIVED_DAY_SID"] = pd.to_numeric(
        policy["RECEIVED_DAY_SID"],
        errors="coerce"
    )

    policy["EFFECT_DAY_SID"] = pd.to_numeric(
        policy["EFFECT_DAY_SID"],
        errors="coerce"
    )

    effect_invalid = (
        policy["EFFECT_DAY_SID"].isna()
        | policy["EFFECT_DAY_SID"].le(0)
    )

    received_invalid = (
        policy["RECEIVED_DAY_SID"].isna()
        | policy["RECEIVED_DAY_SID"].le(0)
    )

    policy.loc[
        effect_invalid & ~received_invalid,
        "EFFECT_DAY_SID"
    ] = policy.loc[
        effect_invalid & ~received_invalid,
        "RECEIVED_DAY_SID"
    ]

    policy["EFFECT_SID"] = (
        policy["EFFECT_DAY_SID"] // 100
    ).astype("Int64")

    policy["BOOKYEAR"] = (
        policy["EFFECT_SID"] // 100
    ).astype("Int64")

    dup_count = policy.duplicated(
        "APPLICATION_NUMBER"
    ).sum()

    print(
        "BOOKYEARS:\n",
        policy["BOOKYEAR"]
        .value_counts(dropna=False)
    )

    print(
        "Policy rows before APPLICATION_NUMBER cleanup:",
        f"{rows_before:,}"
    )

    print(
        "Policy duplicate APPLICATION_NUMBER count:",
        f"{dup_count:,}"
    )

    policy = policy.drop_duplicates(
        subset="APPLICATION_NUMBER",
        keep="last"
    )

    print("Policy shape after cleanup:", policy.shape)

    return policy

def prepare_etd_claim_file(etd_claim_file):

    etd_claim_file = normalize_columns(
        etd_claim_file
    )

    validate_columns(
        etd_claim_file,
        [
            "APPLICATION_NUMBER",
            "PMP_BEST_DATE",
            "DELQ90_IND",
            "CLAIM_IND",
            "TOTAL_INCURED_LOSS",
        ],
        "etd_claim_file",
    )

    etd_claim_file["APPLICATION_NUMBER"] = (
        etd_claim_file["APPLICATION_NUMBER"]
        .astype("int64")
    )

    pmp_date = pd.to_datetime(
        etd_claim_file["PMP_BEST_DATE"],
        errors="coerce"
    )

    etd_claim_file["PMP_BEST_DATE"] = (
        pmp_date.dt.strftime("%Y%m%d")
        .astype("Int64")
    )

    etd_claim_file = etd_claim_file[
        pmp_date.dt.year.ne(2020)
    ].copy()

    dup_rows = etd_claim_file.duplicated(
        "APPLICATION_NUMBER"
    ).sum()

    dup_apps = (
        etd_claim_file.loc[
            etd_claim_file.duplicated(
                "APPLICATION_NUMBER"
            ),
            "APPLICATION_NUMBER",
        ]
        .nunique()
    )

    print(
        "ETD duplicate episodes:",
        f"{dup_rows:,}",
        "rows across",
        f"{dup_apps:,}",
        "applications"
    )

    etd_claim_file = (
        etd_claim_file
        .groupby(
            "APPLICATION_NUMBER",
            as_index=False
        )
        .agg(
            DELQ90_IND=("DELQ90_IND", "max"),
            CLAIM_IND=("CLAIM_IND", "max"),
            TOTAL_INCURED_LOSS=(
                "TOTAL_INCURED_LOSS",
                "sum"
            ),
        )
    )

    print(
        "ETD shape after application-level aggregation:",
        etd_claim_file.shape
    )

    return etd_claim_file
