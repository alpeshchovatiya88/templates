
import pandas as pd
import numpy as np

from pathlib import Path
from datetime import datetime
from sklearn.metrics import (
    roc_auc_score,
    brier_score_loss,
    log_loss
)


def check_reproducibility(
    developer_file,
    validator_file,
    dataset_name,
    key_columns,
    target_col="Target",
    probability_col="Predicted_Probability",
    loss_col="Loss",
    loss_ratio_col="Loss_Ratio",
    probability_tolerance=1e-8,
    metric_tolerance=1e-8,
    loss_tolerance=1e-6,
    loss_ratio_tolerance=1e-8,
    total_loss_tolerance=1e-4,
    output_folder="reproducibility_reports"
):
    """
    Compare Developer and Validator results for binary models.

    Required:
        Record ID(s), Target, Predicted Probability

    Optional:
        Loss, Loss Ratio

    Dataset metrics:
        AUC, Brier Score, Log Loss

    Returns:
        Dictionary of comparison results and report paths.
    """

    # -------------------------------------------------------
    # 1. Validate configuration
    # -------------------------------------------------------

    if isinstance(key_columns, str):
        key_columns = [key_columns]
    else:
        key_columns = list(key_columns)

    if not key_columns:
        raise ValueError("At least one key column is required.")

    required = key_columns + [target_col, probability_col]

    if len(set(required)) != len(required):
        raise ValueError("Required column names must be distinct.")

    for tolerance in [
        probability_tolerance,
        metric_tolerance,
        loss_tolerance,
        loss_ratio_tolerance,
        total_loss_tolerance
    ]:
        if not np.isfinite(tolerance) or tolerance < 0:
            raise ValueError("Invalid tolerance.")

    # -------------------------------------------------------
    # 2. Read CSV files
    # -------------------------------------------------------

    key_dtypes = {col: "string" for col in key_columns}

    dev = pd.read_csv(
        developer_file,
        dtype=key_dtypes,
        low_memory=False
    )

    val = pd.read_csv(
        validator_file,
        dtype=key_dtypes,
        low_memory=False
    )

    # -------------------------------------------------------
    # 3. Validate data
    # -------------------------------------------------------

    for name, df in [("Developer", dev), ("Validator", val)]:

        if not df.columns.is_unique:
            raise ValueError(f"{name}: duplicate column names.")

        missing_cols = [c for c in required if c not in df]

        if missing_cols:
            raise ValueError(
                f"{name}: missing columns {missing_cols}"
            )

        if df[key_columns].isna().any().any():
            raise ValueError(f"{name}: missing record IDs.")

        if df.duplicated(subset=key_columns).any():
            raise ValueError(f"{name}: duplicate record IDs.")

        for col in [target_col, probability_col]:

            df[col] = pd.to_numeric(df[col], errors="raise")

            if df[col].isna().any():
                raise ValueError(
                    f"{name}: missing values in {col}."
                )

            if not np.isfinite(df[col]).all():
                raise ValueError(
                    f"{name}: non-finite values in {col}."
                )

        if not df[target_col].isin([0, 1]).all():
            raise ValueError(
                f"{name}: Target must contain only 0 and 1."
            )

        if not df[probability_col].between(0, 1).all():
            raise ValueError(
                f"{name}: probabilities must be in [0, 1]."
            )

    # -------------------------------------------------------
    # 4. Identify optional metrics
    # -------------------------------------------------------

    optional_metrics = {}

    for metric, col, tolerance in [
        ("Loss", loss_col, loss_tolerance),
        ("Loss_Ratio", loss_ratio_col, loss_ratio_tolerance)
    ]:

        if col is None:
            continue

        in_dev = col in dev.columns
        in_val = col in val.columns

        if in_dev != in_val:
            raise ValueError(
                f"{col} exists in only one file."
            )

        if in_dev:

            for name, df in [
                ("Developer", dev),
                ("Validator", val)
            ]:
                df[col] = pd.to_numeric(
                    df[col],
                    errors="raise"
                )

                if np.isinf(df[col]).any():
                    raise ValueError(
                        f"{name}: infinite values in {col}."
                    )

            optional_metrics[metric] = (col, tolerance)

    # -------------------------------------------------------
    # 5. Align records
    # -------------------------------------------------------

    columns = (
        key_columns
        + [target_col, probability_col]
        + [col for col, _ in optional_metrics.values()]
    )

    columns = list(dict.fromkeys(columns))

    merged = dev[columns].merge(
        val[columns],
        on=key_columns,
        how="outer",
        suffixes=("_DEV", "_VAL"),
        indicator=True,
        validate="one_to_one"
    )

    missing_records = merged.loc[
        merged["_merge"] != "both",
        key_columns + ["_merge"]
    ].copy()

    missing_records["Issue"] = (
        missing_records["_merge"]
        .astype(str)
        .replace({
            "left_only": "Missing in Validator",
            "right_only": "Extra in Validator"
        })
    )

    missing_records = missing_records[
        key_columns + ["Issue"]
    ].reset_index(drop=True)

    matched = merged.loc[
        merged["_merge"] == "both"
    ].copy()

    # -------------------------------------------------------
    # 6. Record-level comparisons
    # -------------------------------------------------------

    record_metrics = {
        "Target": (target_col, 0.0),
        "Predicted_Probability": (
            probability_col,
            probability_tolerance
        )
    }

    record_metrics.update(optional_metrics)

    summary_rows = []
    mismatch_frames = []

    for metric, (col, tolerance) in record_metrics.items():

        dev_values = matched[f"{col}_DEV"]
        val_values = matched[f"{col}_VAL"]

        diff = (dev_values - val_values).abs()

        passed = (
            dev_values.notna()
            & val_values.notna()
            & (diff <= tolerance)
        )

        failed = ~passed
        mismatch_count = int(failed.sum())

        summary_rows.append({
            "Dataset": dataset_name,
            "Metric": metric,
            "Records_Compared": len(matched),
            "Mismatch_Count": mismatch_count,
            "Mismatch_Percentage": (
                100 * mismatch_count / len(matched)
                if len(matched) else np.nan
            ),
            "Max_Absolute_Difference": diff.max(),
            "Tolerance": tolerance,
            "Status": (
                "PASS"
                if len(matched) > 0 and mismatch_count == 0
                else "FAIL"
            )
        })

        if mismatch_count:

            detail = matched.loc[
                failed, key_columns
            ].copy()

            detail["Dataset"] = dataset_name
            detail["Metric"] = metric
            detail["Developer_Value"] = dev_values.loc[failed]
            detail["Validator_Value"] = val_values.loc[failed]
            detail["Absolute_Difference"] = diff.loc[failed]
            detail["Tolerance"] = tolerance

            mismatch_frames.append(detail)

    record_summary = pd.DataFrame(summary_rows)

    if mismatch_frames:
        mismatches = pd.concat(
            mismatch_frames,
            ignore_index=True
        )
    else:
        mismatches = pd.DataFrame(
            columns=key_columns + [
                "Dataset", "Metric",
                "Developer_Value", "Validator_Value",
                "Absolute_Difference", "Tolerance"
            ]
        )

    # -------------------------------------------------------
    # 7. Dataset-level metrics
    # -------------------------------------------------------

    def calculate_metrics(df):

        y = df[target_col].to_numpy(dtype=int)
        p = df[probability_col].to_numpy(dtype=float)

        if len(np.unique(y)) != 2:
            raise ValueError(
                f"{dataset_name}: AUC requires both classes."
            )

        return {
            "AUC": roc_auc_score(y, p),
            "Brier_Score": brier_score_loss(y, p),
            "Log_Loss": log_loss(y, p, labels=[0, 1])
        }

    dev_metrics = calculate_metrics(dev)
    val_metrics = calculate_metrics(val)

    performance_rows = []

    for metric in dev_metrics:

        dev_value = dev_metrics[metric]
        val_value = val_metrics[metric]

        diff = abs(dev_value - val_value)

        performance_rows.append({
            "Dataset": dataset_name,
            "Metric": metric,
            "Developer_Value": dev_value,
            "Validator_Value": val_value,
            "Absolute_Difference": diff,
            "Tolerance": metric_tolerance,
            "Status": (
                "PASS" if diff <= metric_tolerance else "FAIL"
            )
        })

    performance_summary = pd.DataFrame(performance_rows)

    # -------------------------------------------------------
    # 8. Aggregate total loss
    # -------------------------------------------------------

    aggregate_rows = []

    if "Loss" in optional_metrics:

        dev_total = dev[loss_col].sum(min_count=1)
        val_total = val[loss_col].sum(min_count=1)

        diff = abs(dev_total - val_total)

        valid = (
            dev[loss_col].notna().all()
            and val[loss_col].notna().all()
            and np.isfinite(dev_total)
            and np.isfinite(val_total)
        )

        aggregate_rows.append({
            "Dataset": dataset_name,
            "Metric": "Total_Loss",
            "Developer_Value": dev_total,
            "Validator_Value": val_total,
            "Absolute_Difference": diff,
            "Tolerance": total_loss_tolerance,
            "Status": (
                "PASS"
                if valid and diff <= total_loss_tolerance
                else "FAIL"
            )
        })

    aggregate_summary = pd.DataFrame(
        aggregate_rows,
        columns=[
            "Dataset", "Metric",
            "Developer_Value", "Validator_Value",
            "Absolute_Difference", "Tolerance", "Status"
        ]
    )

    # -------------------------------------------------------
    # 9. Overall PASS / FAIL
    # -------------------------------------------------------

    overall_pass = (
        len(matched) > 0
        and missing_records.empty
        and record_summary["Status"].eq("PASS").all()
        and performance_summary["Status"].eq("PASS").all()
        and (
            aggregate_summary.empty
            or aggregate_summary["Status"].eq("PASS").all()
        )
    )

    overall_status = "PASS" if overall_pass else "FAIL"

    overall_summary = pd.DataFrame([{
        "Dataset": dataset_name,
        "Developer_Rows": len(dev),
        "Validator_Rows": len(val),
        "Matched_Rows": len(matched),
        "Missing_Extra_Rows": len(missing_records),
        "Record_Mismatches": len(mismatches),
        "Overall_Status": overall_status,
        "Execution_Time": datetime.now().isoformat(
            timespec="seconds"
        )
    }])

    # -------------------------------------------------------
    # 10. Export report
    # -------------------------------------------------------

    output_dir = Path(output_folder)
    output_dir.mkdir(parents=True, exist_ok=True)

    safe_name = "".join(
        c if c.isalnum() or c in "-_" else "_"
        for c in dataset_name
    )

    if not safe_name:
        raise ValueError("Invalid dataset name.")

    report_path = (
        output_dir / f"{safe_name}_Reproducibility.xlsx"
    )

    mismatch_path = (
        output_dir / f"{safe_name}_Mismatches.csv"
    )

    missing_path = (
        output_dir / f"{safe_name}_Missing_Records.csv"
    )

    mismatches.to_csv(mismatch_path, index=False)
    missing_records.to_csv(missing_path, index=False)

    excel_limit = 1_048_575

    with pd.ExcelWriter(
        report_path,
        engine="openpyxl"
    ) as writer:

        overall_summary.to_excel(
            writer,
            sheet_name="Executive Summary",
            index=False
        )

        record_summary.to_excel(
            writer,
            sheet_name="Record Checks",
            index=False
        )

        performance_summary.to_excel(
            writer,
            sheet_name="Performance Metrics",
            index=False
        )

        aggregate_summary.to_excel(
            writer,
            sheet_name="Aggregate Loss",
            index=False
        )

        mismatches.head(excel_limit).to_excel(
            writer,
            sheet_name="Mismatches",
            index=False
        )

        missing_records.head(excel_limit).to_excel(
            writer,
            sheet_name="Missing Records",
            index=False
        )

    # -------------------------------------------------------
    # 11. Display and return results
    # -------------------------------------------------------

    print(f"\n{'=' * 60}")
    print(f"REPRODUCIBILITY: {dataset_name}")
    print(f"{'=' * 60}")

    print("\nRecord-Level Checks:")
    print(record_summary.to_string(index=False))

    print("\nDataset-Level Performance:")
    print(performance_summary.to_string(index=False))

    if not aggregate_summary.empty:
        print("\nAggregate Loss:")
        print(aggregate_summary.to_string(index=False))

    print(f"\nOverall Status: {overall_status}")
    print(f"Report: {report_path}")

    return {
        "overall_status": overall_status,
        "overall_summary": overall_summary,
        "record_summary": record_summary,
        "performance_summary": performance_summary,
        "aggregate_summary": aggregate_summary,
        "mismatches": mismatches,
        "missing_records": missing_records,
        "report_path": str(report_path)
    }
