# Reliability

def psi(expected, actual, bins=10):
    expected = expected[np.isfinite(expected)]
    actual = actual[np.isfinite(actual)]
    cuts = np.unique(np.quantile(expected, np.linspace(0, 1, bins + 1)))
    cuts[0], cuts[-1] = -np.inf, np.inf
    e = np.clip(np.histogram(expected, bins=cuts)[0] / len(expected), 1e-6, None)
    a = np.clip(np.histogram(actual, bins=cuts)[0] / len(actual), 1e-6, None)
    return float(np.sum((a - e) * np.log(a / e)))

print("PSI score train->test", psi(y_score(train), y_score(test)))
print("PSI score train->val", psi(y_score(train), y_score(val)))
print("PSI target train->val", psi(y_true(train), y_true(val)))

def bootstrap_auc(df, n=400):
    yt, ys, _ = finite(df)
    stats = []
    for _ in range(n):

def perf(yt, ys):

def slice_metric(df, col, min_n=200):
    rows = []
    for key, g in df.groupby(col, dropna=False):
        if len(g) < min_n:
            continue
        yt, ys, _ = finite(g)
        if len(np.unique(yt)) < 2:
            continue
        stat = roc_auc_score(yt, ys) if TASK == "classification" else mean_absolute_error(yt, ys)
        rows.append({"slice": col, "value": key, "n": len(yt), "metric": stat, "bad_rate": yt.mean()})
    return pd.DataFrame(rows)

# Weakness regions

MIN_N, MAE_LIFT, AUC_DROP = 300, 1.25, 0.05
SLICE_COLS = [c for c in ["PROVINCE", "PRK_PROPERTY_TYPE", "PRK_OCCUPANCY", "PRD_LOAN_PURPOSE", "PRD_RATE_TYPE", "PRK_CONS_TYPE", "ECON_REGION"] if c in val.columns]

def weakness(df):
    yt, ys, _ = finite(df)
    book_mae = np.mean(np.abs(yt - ys))
    book_auc = roc_auc_score(yt, ys)
    rows = []
    for col in SLICE_COLS:
        for key, g in df.groupby(col, dropna=False):
            if len(g) < MIN_N:
                continue
            gyt, gys, _ = finite(g)
            if len(np.unique(gyt)) < 2:
                continue
            mae = np.mean(np.abs(gyt - gys))
            auc = roc_auc_score(gyt, gys)
            if mae > book_mae * MAE_LIFT or auc < book_auc - AUC_DROP:
                rows.append({"feature": col, "value": key, "n": len(gyt), "mae": mae, "book_mae": book_mae, "auc": auc, "book_auc": book_auc, "bad_rate": gyt.mean(), "avg_score": gys.mean()})
    return pd.DataFrame(rows).sort_values("mae", ascending=False)

weak_val = weakness(val)
weak_test = weakness(test)

def auc(df):
    yt, ys, _ = finite(df)
    return roc_auc_score(yt, ys)

def brier(df):
    yt, ys, _ = finite(df)
