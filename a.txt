def adjust_probability(probabilities, factor=34.05):

    probabilities = pd.to_numeric(
        probabilities,
        errors="coerce"
    )

    return (
        probabilities
        / (
            probabilities
            + factor * (1 - probabilities)
        )
    )

        0,

    # ORIGINAL LOSS - FOR VERIFICATION ONLY
    # -------------------------------------------------

    df["SMI3_LOSS_original_check"] = (
        lgd
        * original_pd
        * loss_multiplier

    # -------------------------------------------------
    # ORIGINAL LOSS RATIO - FOR VERIFICATION ONLY
    # -------------------------------------------------

    df["SMI3_LR_original_check"] = (
        df["SMI3_LOSS_original_check"]
        / safe_premium
    ) * scale

    # -------------------------------------------------
    # CALIBRATED LOSS
    # -------------------------------------------------

    df["SMI3_LOSS_calibrated"] = (
        lgd
        * calibrated_pd
        * loss_multiplier

    # -------------------------------------------------
    # CALIBRATED LOSS RATIO
    # -------------------------------------------------

    df["SMI3_LR_calibrated"] = (
        df["SMI3_LOSS_calibrated"]

# 8B. CREATE ORIGINAL AND CALIBRATED D1-D20

# Recreated original decile:
# decile_original
#
# Calibrated decile:
# rr_model_decile_calibrated
#
# Existing production decile:
# rr_model_decile
#
# Ranking convention assumed here:
# D1 = highest risk
# D20 = lowest risk
#
# Groups are approximately equal-sized within
# each dataset.
#
# Compare decile_original with rr_model_decile
# before relying on swapset movements.


def create_20_deciles(
    scores,
    highest_risk_first=True
):

    scores = pd.to_numeric(
        scores,
        errors="coerce"
    )

    result = pd.Series(
        pd.NA,
        index=scores.index,
        dtype="string"
    )

    valid = scores.notna()

    n_valid = int(valid.sum())

    if n_valid == 0:
        return result

    # Descending score: highest risk first.
    # method="first" resolves ties by row order.

    ranks = scores.loc[valid].rank(
        method="first",
        ascending=not highest_risk_first
    )

    n_groups = min(20, n_valid)

    group_numbers = (
        pd.qcut(
            ranks,
            q=n_groups,
            labels=False
        ) + 1
    )

    result.loc[valid] = (
        "D"
        + group_numbers.astype(int).astype(str)
    )

    return result

# NORMALIZE EXISTING D1-D20 VALUES
# ------------------------------------------------------------

def normalize_deciles(series):

    values = (
        series.astype("string")
        .str.strip()
        .str.upper()
        .str.replace(
            r"^D",
            "",
            regex=True
        )
    )

    numbers = pd.to_numeric(
        values,
        errors="coerce"
    ).astype("Int64")

    return numbers

# 9. EVALUATE PERFORMANCE
# ============================================================

def evaluate(df, name, probability_col):

    data = df[
        df[target_col].isin([0, 1])
        & df[probability_col].notna()
    ].copy()

    if data.empty:
        print(
            f"{name}: No valid observed outcomes."
        )
        return None

    y = data[target_col].astype(int).to_numpy()

    probabilities = (
        data[probability_col]
        .to_numpy(dtype=float)
    )

    n_observed = len(data)
    n_bad_events = int(y.sum())

    # Binary predictions
    y_pred = (
        probabilities >= threshold
    ).astype(int)

    # Confusion matrix
    tn, fp, fn, tp = confusion_matrix(
        y,
        y_pred,
        labels=[0, 1]
    ).ravel()

    # Precision
    precision = precision_score(
        y,
        y_pred,
        zero_division=0
    )

    # Recall
    recall = recall_score(
        y,
        y_pred,
        zero_division=0
    )

    # False Positive Rate
    fpr = (
        fp / (fp + tn)
        if fp + tn > 0
        else np.nan
    )

    # AUC and KS
    if np.unique(y).size == 2:

        auc = roc_auc_score(
            y,
            probabilities
        )

        fpr_curve, tpr_curve, _ = roc_curve(
            y,
            probabilities
        )

        ks = np.max(
            tpr_curve - fpr_curve
        )

    else:

        auc = np.nan
        ks = np.nan

    return {
        "Dataset": name,
        "N_Observed": n_observed,
        "N_Bad_Events": n_bad_events,
        "Actual_Bad_Rate": y.mean(),
        "Mean_PD": probabilities.mean(),
        "AUC": auc,
        "KS": ks,
        "Precision": precision,
        "Recall": recall,
        "FPR": fpr,
        "Brier_Score": brier_score_loss(
            y,
            probabilities
        ),
        "Log_Loss": log_loss(
            y,
            probabilities,
            labels=[0, 1]
        )
    }

def plot_calibration(df, name):

    data = df[
        df[target_col].isin([0, 1])
        & df["PD_BASELINE"].notna()
        & df["PD_CALIBRATED"].notna()
    ].copy()

    if data.empty or data[target_col].nunique() < 2:

        print(
            f"{name}: Insufficient target data."
        )
        return

    y = data[target_col].astype(int)

    plt.figure(figsize=(8, 6))

    for col, label in [
        (
            "PD_BASELINE",
            "Original XGBoost"
        ),
        (
            "PD_CALIBRATED",
            "Isotonic Calibrated XGBoost"
        )
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            data[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
