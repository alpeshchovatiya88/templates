import numpy as np
import pandas as pd


def make_shap_importance(
    df: pd.DataFrame,
    feature_cols: list,
    subgroup_cols: list = None,
    min_rows: int = 200,
    max_levels: int = 12,
    combine_filters: bool = False,
    output_path: str = None,
) -> pd.DataFrame:

    subgroup_cols = list(subgroup_cols or [])
    feature_cols = list(feature_cols)

    # Validate input
    required_cols = feature_cols + subgroup_cols
    missing = [c for c in required_cols if c not in df.columns]

    if missing:
        raise ValueError(f"Missing columns: {missing}")

    if not feature_cols:
        raise ValueError("feature_cols cannot be empty")

    if df.empty:
        raise ValueError("Input DataFrame is empty")

    results = []

    # --------------------------------------------------
    # Calculate SHAP importance
    # --------------------------------------------------
    def importance_table(data, subgroup_label):

        sv = data[feature_cols].to_numpy(dtype=float)

        mean_signed = np.nanmean(sv, axis=0)
        mean_abs = np.nanmean(np.abs(sv), axis=0)

        imp = pd.DataFrame({
            "Predictor Name": feature_cols,
            "SHAP value": mean_signed,
            "Abs SHAP value": mean_abs
        })

        imp = (
            imp.sort_values(
                "Abs SHAP value",
                ascending=False
            )
            .reset_index(drop=True)
        )

        imp["Predictor Rank"] = np.arange(1, len(imp) + 1)
        imp["Sub-Group"] = subgroup_label

        imp["Direction"] = np.where(
            imp["SHAP value"] > 0,
            "increases",
            np.where(
                imp["SHAP value"] < 0,
                "lowers",
                "both"
            )
        )

        return imp[
            [
                "Sub-Group",
                "Direction",
                "Predictor Rank",
                "Predictor Name",
                "SHAP value",
                "Abs SHAP value"
            ]
        ]

    # --------------------------------------------------
    # Overall SHAP importance
    # --------------------------------------------------
    results.append(
        importance_table(df, "All")
    )

    # --------------------------------------------------
    # Independent subgroup filtering
    # --------------------------------------------------
    for col in subgroup_cols:

        if col.lower() == "all":
            continue

        values = df[col].astype("string")

        levels = (
            values.value_counts(dropna=False)
            .index.tolist()[:max_levels]
        )

        for level in levels:

            mask = values.eq(str(level)).fillna(False)
            subset = df.loc[mask]

            if len(subset) < min_rows:
                continue

            label = f"{col}={level}"

            results.append(
                importance_table(subset, label)
            )

    # --------------------------------------------------
    # Combined filtering
    # --------------------------------------------------
    if combine_filters and len(subgroup_cols) == 2:

        col1, col2 = subgroup_cols

        levels1 = (
            df[col1].astype("string")
            .value_counts(dropna=False)
            .index.tolist()[:max_levels]
        )

        levels2 = (
            df[col2].astype("string")
            .value_counts(dropna=False)
            .index.tolist()[:max_levels]
        )

        for level1 in levels1:
            for level2 in levels2:

                mask1 = (
                    df[col1].astype("string")
                    .eq(str(level1))
                    .fillna(False)
                )

                mask2 = (
                    df[col2].astype("string")
                    .eq(str(level2))
                    .fillna(False)
                )

                subset = df.loc[mask1 & mask2]

                if len(subset) < min_rows:
                    continue

                label = (
                    f"{col1}={level1} | "
                    f"{col2}={level2}"
                )

                results.append(
                    importance_table(subset, label)
                )

    final = pd.concat(
        results,
        ignore_index=True
    )

    if output_path:
        final.to_csv(output_path, index=False)

    return final
