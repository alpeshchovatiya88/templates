# ---------------------------------------------------------
# PRK_PROPERTY_AGE missing-value treatment
# Exact logic from current policy_processor.py
# ---------------------------------------------------------

# 1. Identify rows where property age is missing
missing_property_age = policy_df["PRK_PROPERTY_AGE"].isna()

# 2. If AVM_YEAR_BUILT exists, calculate property age
policy_df.loc[
    missing_property_age
    & policy_df["AVM_YEAR_BUILT"].notna(),
    "PRK_PROPERTY_AGE"
] = (
    policy_df["BOOKYEAR"]
    - policy_df["AVM_YEAR_BUILT"]
)

# 3. For new construction, always force property age = 1
policy_df.loc[
    policy_df["PRK_CONS_TYPE"].eq("N"),
    "PRK_PROPERTY_AGE"
] = 1
