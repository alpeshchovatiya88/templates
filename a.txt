import pandas as pd
import numpy as np
import plotly.graph_objects as go
from plotly.subplots import make_subplots

# =====================================================
# 1. CHANGE ONLY THESE TWO COLUMN NAMES
# =====================================================

base_col = "decile_rs42"
compare_col = "decile_rs07"

base_name = "RS42"
compare_name = "RS07"

# =====================================================
# 2. PREPARE DATA
# =====================================================

data = df.copy()

for col in [base_col, compare_col]:

    # Extract numeric decile, including strings like "Decile 5"
    data[col] = pd.to_numeric(
        data[col]
        .astype("string")
        .str.extract(r"(\d+)", expand=False),
        errors="coerce"
    )

data = data.dropna(subset=[base_col, compare_col])

data = data[
    data[base_col].between(1, 20)
    & data[compare_col].between(1, 20)
].copy()

data[base_col] = data[base_col].astype(int)
data[compare_col] = data[compare_col].astype(int)

if data.empty:
    raise ValueError("No valid paired decile observations found.")

# =====================================================
# 3. CALCULATE DECILE MOVEMENT
# =====================================================

data["Movement"] = data[compare_col] - data[base_col]
data["Abs_Movement"] = data["Movement"].abs()

data["Direction"] = np.select(
    [
        data["Movement"] > 0,
        data["Movement"] < 0
    ],
    [
        "Higher Decile",
        "Lower Decile"
    ],
    default="No Movement"
)

data["Movement_Group"] = np.select(
    [
        data["Abs_Movement"] == 0,
        data["Abs_Movement"] == 1,
        data["Abs_Movement"] == 2,
        data["Abs_Movement"].between(3, 5),
        data["Abs_Movement"] > 5
    ],
    [
        "No Movement",
        "1 Decile",
        "2 Deciles",
        "3-5 Deciles",
        "More Than 5"
    ],
    default="Other"
)

total_files = len(data)

# =====================================================
# 4. MOVEMENT SUMMARY
# =====================================================

movement_summary = pd.crosstab(
    data["Abs_Movement"],
    data["Direction"]
)

movement_summary = movement_summary.reindex(
    index=range(20),
    columns=[
        "Lower Decile",
        "No Movement",
        "Higher Decile"
    ],
    fill_value=0
)

movement_summary["Total Files"] = (
    movement_summary[
        ["Lower Decile", "No Movement", "Higher Decile"]
    ].sum(axis=1)
)

movement_summary["Percentage"] = (
    movement_summary["Total Files"] / total_files * 100
).round(2)

movement_summary.index.name = "Deciles Moved"

# =====================================================
# 5. MOVEMENT GROUP SUMMARY
# =====================================================

movement_groups = pd.crosstab(
    data["Movement_Group"],
    data["Direction"]
)

movement_groups = movement_groups.reindex(
    index=[
        "No Movement",
        "1 Decile",
        "2 Deciles",
        "3-5 Deciles",
        "More Than 5"
    ],
    columns=[
        "Lower Decile",
        "No Movement",
        "Higher Decile"
    ],
    fill_value=0
)

movement_groups["Total Files"] = movement_groups.sum(axis=1)

movement_groups["Percentage"] = (
    movement_groups["Total Files"] / total_files * 100
).round(2)

# =====================================================
# 6. STABILITY METRICS
# =====================================================

metrics = pd.DataFrame({
    "Metric": [
        "Total Files",
        "No Movement %",
        "Within 1 Decile %",
        "Within 2 Deciles %",
        "Within 3 Deciles %",
        "Moved More Than 5 %",
        "Higher Decile %",
        "Lower Decile %",
        "Average Absolute Movement",
        "Median Absolute Movement",
        "Maximum Movement",
        "Spearman Correlation"
    ],
    "Value": [
        total_files,
        (data["Abs_Movement"] == 0).mean() * 100,
        (data["Abs_Movement"] <= 1).mean() * 100,
        (data["Abs_Movement"] <= 2).mean() * 100,
        (data["Abs_Movement"] <= 3).mean() * 100,
        (data["Abs_Movement"] > 5).mean() * 100,
        (data["Movement"] > 0).mean() * 100,
        (data["Movement"] < 0).mean() * 100,
        data["Abs_Movement"].mean(),
        data["Abs_Movement"].median(),
        data["Abs_Movement"].max(),
        data[base_col].corr(
            data[compare_col], method="spearman"
        )
    ]
})

metrics["Value"] = metrics["Value"].round(4)

# =====================================================
# 7. DECILE-LEVEL SUMMARY
# =====================================================

decile_summary = data.groupby(base_col).agg(
    Total_Files=("Movement", "size"),
    Same_Decile=("Movement", lambda x: (x == 0).sum()),
    Higher_Decile=("Movement", lambda x: (x > 0).sum()),
    Lower_Decile=("Movement", lambda x: (x < 0).sum()),
    Within_1=("Abs_Movement", lambda x: (x <= 1).sum()),
    Within_2=("Abs_Movement", lambda x: (x <= 2).sum()),
    More_Than_5=("Abs_Movement", lambda x: (x > 5).sum()),
    Avg_Abs_Movement=("Abs_Movement", "mean")
)

for col in [
    "Same_Decile",
    "Higher_Decile",
    "Lower_Decile",
    "Within_1",
    "Within_2",
    "More_Than_5"
]:
    decile_summary[col + "_Pct"] = (
        decile_summary[col]
        / decile_summary["Total_Files"] * 100
    ).round(2)

decile_summary = decile_summary.round(2)

# =====================================================
# 8. CREATE ORIGINAL SWAPSET
# =====================================================

deciles = list(range(1, 21))

swap = pd.crosstab(
    data[base_col],
    data[compare_col]
).reindex(
    index=deciles,
    columns=deciles,
    fill_value=0
)

swap["All"] = swap.sum(axis=1)
swap.loc["All"] = swap.sum(axis=0)

formatted = swap.copy().astype(str)
cell_colors = []

grand_total = swap.loc["All", "All"]

for col in swap.columns:

    column_colors = []

    for row in swap.index:

        count = int(swap.loc[row, col])

        if row == "All":
            denominator = grand_total
        else:
            denominator = swap.loc[row, "All"]

        pct = (
            count / denominator * 100
            if denominator > 0 else 0
        )

        formatted.loc[row, col] = (
            f"{count:,}<br>({pct:.2f}%)"
        )

        if row == "All" or col == "All":
            color = "#E6EAF0"

        elif count == 0:
            color = "#FFFFFF"

        elif row == col:
            color = "#D5F0DB"

        elif abs(int(row) - int(col)) <= 2:
            color = "#FFE4B5"

        else:
            color = "#F8C4C4"

        column_colors.append(color)

    cell_colors.append(column_colors)

cell_colors = [
    ["#E6EAF0"] * len(swap.index)
] + cell_colors

# =====================================================
# 9. CREATE INTERACTIVE DASHBOARD
# =====================================================

fig = make_subplots(
    rows=3,
    cols=1,
    specs=[
        [{"type": "table"}],
        [{"type": "xy"}],
        [{"type": "xy"}]
    ],
    row_heights=[0.55, 0.23, 0.22],
    vertical_spacing=0.08,
    subplot_titles=[
        "Decile Swapset: Count and Row Percentage",
        "Number of Files by Deciles Moved",
        "Movement Direction by Decile Distance"
    ]
)

# Original colored swapset
fig.add_trace(
    go.Table(
        header=dict(
            values=(
                [f"{base_name} ↓ / {compare_name} →"]
                + [str(c) for c in swap.columns]
            ),
            fill_color="#4472C4",
            font=dict(color="white", size=12),
            align="center",
            height=45
        ),
        cells=dict(
            values=[
                [str(x) for x in formatted.index]
            ] + [
                formatted[col].tolist()
                for col in formatted.columns
            ],
            fill_color=cell_colors,
            font=dict(color="#202020", size=10),
            align="center",
            height=40
        ),
        columnwidth=[120] + [115] * 21
    ),
    row=1,
    col=1
)

# Number of files by movement
fig.add_trace(
    go.Bar(
        x=movement_summary.index,
        y=movement_summary["Total Files"],
        name="Total Files",
        marker_color="#4472C4"
    ),
    row=2,
    col=1
)

# Higher and lower movement
fig.add_trace(
    go.Bar(
        x=movement_summary.index,
        y=movement_summary["Higher Decile"],
        name="Higher Decile",
        marker_color="#E6A23C"
    ),
    row=3,
    col=1
)

fig.add_trace(
    go.Bar(
        x=movement_summary.index,
        y=movement_summary["Lower Decile"],
        name="Lower Decile",
        marker_color="#D9534F"
    ),
    row=3,
    col=1
)

fig.update_layout(
    title=dict(
        text=f"Decile Stability Analysis: {base_name} vs {compare_name}",
        x=0.5,
        xanchor="center"
    ),
    height=1900,
    width=1600,
    margin=dict(l=70, r=40, t=150, b=70),
    barmode="group",
    legend=dict(
        orientation="h",
        y=-0.04
    )
)

fig.update_xaxes(
    title_text="Number of Deciles Moved",
    dtick=1,
    row=2,
    col=1
)

fig.update_xaxes(
    title_text="Number of Deciles Moved",
    dtick=1,
    row=3,
    col=1
)

fig.update_yaxes(
    title_text="Number of Files",
    row=2,
    col=1
)

fig.update_yaxes(
    title_text="Number of Files",
    row=3,
    col=1
)

# 10. DISPLAY ALL RESULTS
# =====================================================

print("\n" + "=" * 60)
print(f"DECILE STABILITY: {base_name} VS {compare_name}")
print("=" * 60)

print("\n1. OVERALL STABILITY METRICS")
display(metrics)

print("\n2. EXACT NUMBER OF DECILES MOVED")
display(movement_summary)

print("\n3. MOVEMENT GROUP SUMMARY")
display(movement_groups)

print("\n4. DECILE-LEVEL ANALYSIS")
display(decile_summary)

print("\n5. INTERACTIVE DASHBOARD")
fig.show()
