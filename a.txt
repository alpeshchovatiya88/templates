years = np.arange(1, 18)

delq_count = (
    df_new.loc[
        (df_new["DELQ90_IND"] == 1.0) &
        (df_new["CLAIM_IND"] != 1.0) &
        (df_new["DELQ_YEAR"].between(1, 17)),
        "DELQ_YEAR"
    ]
    .value_counts()
    .reindex(years, fill_value=0)
    .sort_index()
)

claim_count = (
    df_new.loc[
        (df_new["CLAIM_IND"] == 1.0) &
        (df_new["CLAIM_YEAR"].between(1, 17)),
        "CLAIM_YEAR"
    ]
    .value_counts()
    .reindex(years, fill_value=0)
    .sort_index()
)

count_table = pd.DataFrame({
    "YEAR": years,
    "DELQ_COUNT": delq_count.values,
    "CLAIM_COUNT": claim_count.values
})

print(count_table)
