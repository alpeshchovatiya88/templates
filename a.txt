# Totals by BOOKYEAR + CRD_SCORE_bin
row_totals = (
    df.groupby(["BOOKYEAR", "CRD_SCORE_bin"])
      .agg(
          Total_N=("target", "count"),
          Total_Bad=("target", "sum"),
          Bad_Rate=("target", lambda x: round(x.mean() * 100, 2))
      )
)

# Add a formatted total column
row_totals["Overall"] = (
    "n=" + row_totals["Total_N"].astype(str)
    + " (bad=" + row_totals["Total_Bad"].astype(str)
    + ", " + row_totals["Bad_Rate"].astype(str) + "%)"
)
