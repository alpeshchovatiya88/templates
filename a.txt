def export_interactive_seed_report(
    dfs,
    filename="Random_Seed_Sensitivity_Report.html",
    decile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    metrics=None,
    baseline="RS42",
    colors=None,
    decimals=2,
    inside_font_size=16,
    total_font_size=18,
):

    if baseline not in dfs:
        raise ValueError(f"Baseline {baseline} not found.")

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    seeds = list(dfs)

    required = list(dict.fromkeys([
        decile_col,
        region_col,
        market_col,
        tenure_col,
        *metrics.keys()
    ]))

    # -------------------------------------------------
    # 1. Prepare all seed datasets
    # -------------------------------------------------

    frames = []

    for seed, df in dfs.items():

        missing = [c for c in required if c not in df.columns]

        if missing:
            raise ValueError(
                f"{seed} missing columns: {missing}"
            )

        t = df[required].copy()

        t["Seed"] = seed

        # Supports 1, 2, ... and "Decile 1", "Decile 2", ...
        t["Decile"] = pd.to_numeric(
            t[decile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        # Convert categorical filters to strings
        t["Region"] = (
            t[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        t["Market"] = (
            t[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        t["Tenure"] = (
            t[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        for metric in metrics:

            t[metric] = (
                pd.to_numeric(t[metric], errors="coerce")
                .replace([np.inf, -np.inf], np.nan)
            )

        frames.append(
            t[
                [
                    "Seed",
                    "Decile",
                    "Region",
                    "Market",
                    "Tenure",
                    *metrics.keys()
                ]
            ]
        )

    data = pd.concat(frames, ignore_index=True)

    deciles = sorted(
        data["Decile"]
        .dropna()
        .astype(int)
        .unique()
        .tolist()
    )

    regions = sorted(data["Region"].unique().tolist())
    markets = sorted(data["Market"].unique().tolist())
    tenures = sorted(data["Tenure"].unique().tolist())

    # -------------------------------------------------
    # 2. Precalculate filter combinations
    # -------------------------------------------------

    # To avoid generating an enormous Cartesian product
    # of Decile x Region x Market x Tenure, aggregate by
    # actual observed combinations, then calculate each
    # selected combination on demand in the browser.
    #
    # For correct percentiles, individual metric values
    # are included in the HTML data payload.
    #
    # This produces a fully offline, interactive report.

    records = []

    for metric in metrics:

        subset = data[
            [
                "Seed",
                "Decile",
                "Region",
                "Market",
                "Tenure",
                metric
            ]
        ].copy()

        subset = subset.rename(columns={metric: "Value"})

        # Drop missing metric values only.
        subset = subset.dropna(subset=["Value"])

        # Compact column-based records for JavaScript.
        for row in subset.itertuples(index=False, name=None):

            seed, decile, region, market, tenure, value = row

            records.append([
                seed,
                int(decile) if pd.notna(decile) else None,
                region,
                market,
                tenure,
                metric,
                float(value)
            ])

        "seeds": seeds,
        "baseline": baseline,
        "deciles": deciles,
        "regions": regions,
        "markets": markets,
        "tenures": tenures,
        "metrics": metrics,
        "colors": colors,
        "decimals": decimals,
        "insideFontSize": inside_font_size,
        "totalFontSize": total_font_size,
        "records": records
    }

    report_json = json.dumps(
        report_data,
        ensure_ascii=False,
        allow_nan=False,
        separators=(",", ":")
    ).replace("<", "\\u003c")

function formatNumber(v, metric, compact=false) {

    if (v === null || !Number.isFinite(v)) {
        return "N/A";
    }

    const config = report.metrics[metric];

    if (config.is_ratio) {

        const scale = config.ratio_scale || 1;

        return (v / 1000000).toFixed(2) + "M";
    }

    if (compact && Math.abs(v) >= 1000) {
        return (v / 1000).toFixed(2) + "K";
    }

    return v.toLocaleString(
        "en-US",
        {
            minimumFractionDigits: report.decimals,
            maximumFractionDigits: report.decimals
        }
    );
}

function signedNumber(v, metric) {

    if (v === null || !Number.isFinite(v)) {
        return "N/A";
    }

    return (
        v > 0 ? "+" : ""

}

function formatDelta(v, metric) {

        return "N/A";

    if (config.is_ratio) {

            (pp > 0 ? "+" : "") +

}

function formatDeltaPercent(v) {

    if (v === null || !Number.isFinite(v)) {
        return "N/A";
    }

    return (
        (v > 0 ? "+" : "") +
        v.toFixed(2) +
        "%"
    );
}

function escapeHtml(value) {

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
}

function quantile(sortedValues, p) {

    if (!sortedValues.length) {
        return null;
    }

    if (sortedValues.length === 1) {
        return sortedValues[0];
    }

    const position = (sortedValues.length - 1) * p;
    const lower = Math.floor(position);
    const upper = Math.ceil(position);
    const fraction = position - lower;

    return (
        sortedValues[lower] * (1 - fraction) +
        sortedValues[upper] * fraction
    );
}

function calculateStats(values, agg) {

    if (!values.length) {
        return {
            total: null,
            min: null,
            max: null,
            avg: null,
            p10: null,
            p50: null,
            p90: null,
            count: 0
        };
    }

    const sorted = [...values].sort((a, b) => a - b);

    const sum = values.reduce((a, b) => a + b, 0);
    const avg = sum / values.length;

    return {
        total: agg === "sum" ? sum : avg,
        min: sorted[0],
        max: sorted[sorted.length - 1],
        avg: avg,
        p10: quantile(sorted, 0.10),
        p50: quantile(sorted, 0.50),
        p90: quantile(sorted, 0.90),
        count: values.length
    };
}
