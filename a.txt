def make_year_time_cv(df_train, year_col="BOOKYEAR"):
    splits = [
        ((2009, 2012), (2013, 2013)),
        ((2009, 2013), (2014, 2014)),
        ((2009, 2014), (2015, 2015)),
        ((2009, 2015), (2016, 2016)),
        ((2009, 2016), (2017, 2017)),
    ]

    cv = []

    for train_years, valid_years in splits:
        train_idx = np.where(df_train[year_col].between(*train_years))[0]
        valid_idx = np.where(df_train[year_col].between(*valid_years))[0]
        cv.append((train_idx, valid_idx))

    return cv
