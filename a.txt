LR_summary = (
    df
    .group_by("train_test")
    .agg(
        SMI3_LOSS = pl.col("SMI3_LOSS").sum(),
        ACTUAL_LOSS = pl.col("ACTUAL_LOSS").sum(),
        POD_PREMIUM_WRITTEN = pl.col("POD_PREMIUM_WRITTEN").sum()
    )
    .with_columns(
        SMI3_LR = pl.col("SMI3_LOSS") / pl.col("POD_PREMIUM_WRITTEN"),
        ACTUAL_LR = pl.col("ACTUAL_LOSS") / pl.col("POD_PREMIUM_WRITTEN")
    )
    .with_columns(
        LR_diff = pl.col("SMI3_LR") - pl.col("ACTUAL_LR")
    )
)

LR_summary
