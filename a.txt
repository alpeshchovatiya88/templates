    df_val["SMI3_LOSS_calibrated"]

# ------------------------------------------------------------
# 8A.2 D1-D20 FUNCTION
# ------------------------------------------------------------

# Assumed ranking:
# D1 = highest loss ratio
# D20 = lowest loss ratio
#
# Approximately equal-sized groups.
# Ties broken by row order.
#
# Confirm this matches the existing
# rr_model_decile methodology.

def create_20_deciles(scores):

    scores = pd.to_numeric(
        scores,
        errors="coerce"
    )

    result = pd.Series(
        pd.NA,
        index=scores.index,
        dtype="string"
    )

    valid = scores.notna()

    if valid.any():

        ranks = scores.loc[valid].rank(
            method="first",
            ascending=False
        )

        n_groups = min(
            20,
            int(valid.sum())
        )

        groups = (
            pd.qcut(
                ranks,
                q=n_groups,
                labels=False
            ) + 1
        )

        result.loc[valid] = (
            "D" + groups.astype(str)
        )

    return result

# 9. EVALUATE CALIBRATION PERFORMANCE

# Threshold used for Precision, Recall, FPR
threshold = 0.50


def evaluate(df, name, probability_col):

    # Exclude missing-target rows
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty:
        print(f"{name}: No observed targets.")
        return None

    y = data[target_col].astype(int)

    probabilities = data[probability_col]

    n_observed = len(data)
    n_bad_events = int(y.sum())

    # Predicted class
    y_pred = (
        probabilities >= threshold
    ).astype(int)

    # Confusion matrix
    tn, fp, fn, tp = confusion_matrix(
        y,
        y_pred,
        labels=[0, 1]
    ).ravel()

    # Precision
    precision = precision_score(
        y,
        y_pred,
        zero_division=0
    )

    # Recall
    recall = recall_score(
        y,
        y_pred,
        zero_division=0
    )

    # False Positive Rate
    fpr = (
        fp / (fp + tn)
        if (fp + tn) > 0
        else np.nan
    )

    # AUC and KS require both classes
    if y.nunique() == 2:

        auc = roc_auc_score(
            y,
            probabilities
        )

        fpr_curve, tpr_curve, _ = roc_curve(
            y,
            probabilities
        )

        ks = np.max(
            tpr_curve - fpr_curve
        )

    else:
        auc = np.nan
        ks = np.nan

    return {
        "Dataset": name,
        "N_Observed": n_observed,
        "N_Bad_Events": n_bad_events,
        "Actual_Bad_Rate": y.mean(),
        "Mean_PD": probabilities.mean(),
        "AUC": auc,
        "KS": ks,
        "Precision": precision,
        "Recall": recall,
        "FPR": fpr,
        "Brier_Score": brier_score_loss(
            y,
            probabilities
        ),
        "Log_Loss": log_loss(
            y,
            probabilities,
            labels=[0, 1]
        )
    }

# 10. PLOT CALIBRATION CURVES

def plot_calibration(df, name):

    # Original observed-target filtering
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty or data[target_col].nunique() < 2:
        print(f"{name}: Insufficient target data.")
        return

    y = data[target_col].astype(int)

    plt.figure(figsize=(8, 6))

    for col, label in [
        ("PD_ORIGINAL", "Original XGBoost"),
        ("PD_CALIBRATED", "Calibrated XGBoost")
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            data[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
    )

    plt.xlabel("Mean Predicted Probability")
    plt.ylabel("Observed Bad Rate")
    plt.title(name + " Calibration Curve")
    plt.legend()
    plt.grid(True)
    plt.show()


plot_calibration(df_train, "TRAIN")
