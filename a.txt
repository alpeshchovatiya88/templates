import math
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.ticker as mtick

# Make sure results already has:
# CRD_SCORE_bin, Feature, Total_Bad_Count, Overlap_With_CRA_COLLECTION, Overlap_%

plot_data = results.copy()

bins = sorted(plot_data["CRD_SCORE_bin"].dropna().unique())

ncols = 4
nrows = math.ceil(len(bins) / ncols)

fig, axes = plt.subplots(nrows, ncols, figsize=(22, 5 * nrows))
axes = axes.flatten()

for i, score_bin in enumerate(bins):

    ax = axes[i]

    plot_df = plot_data[plot_data["CRD_SCORE_bin"] == score_bin].copy()

    x = np.arange(len(plot_df))
    width = 0.35

    bars1 = ax.bar(
        x - width/2,
        plot_df["Total_Bad_Count"],
        width=width,
        label="Total Bad Count"
    )

    bars2 = ax.bar(
        x + width/2,
        plot_df["Overlap_With_CRA_COLLECTION"],
        width=width,
        label="Overlap with CRA_COLLECTION"
    )

    for bar in bars1:
        height = bar.get_height()
        ax.text(
            bar.get_x() + bar.get_width()/2,
            height,
            f"{int(height)}",
            ha="center",
            va="bottom",
            fontsize=8
        )

    for j, bar in enumerate(bars2):
        height = bar.get_height()
        pct = plot_df["Overlap_%"].iloc[j]

        ax.text(
            bar.get_x() + bar.get_width()/2,
            height,
            f"{int(height)}\n({pct}%)",
            ha="center",
            va="bottom",
            fontsize=8
        )

    ax.set_title(f"CRD_SCORE_bin: {score_bin}", fontweight="bold")
    ax.set_xticks(x)
    ax.set_xticklabels(plot_df["Feature"], rotation=30, ha="right")
    ax.set_ylabel("Bad Count")
    ax.grid(axis="y", alpha=0.3)

# Hide empty subplots
for j in range(i + 1, len(axes)):
    axes[j].axis("off")

handles, labels = axes[0].get_legend_handles_labels()
fig.legend(handles, labels, loc="upper center", ncol=2)

plt.suptitle("Overlap Analysis with CRA_COLLECTION by CRD_SCORE_bin", fontsize=16, fontweight="bold", y=1.02)
plt.tight_layout()
plt.show()
