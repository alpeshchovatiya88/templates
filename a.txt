import matplotlib.pyplot as plt

df["target"].value_counts().sort_index().plot(kind="bar")

plt.title("Target Distribution")
plt.xlabel("Target")
plt.ylabel("Count")
plt.xticks(rotation=0)
plt.show()

import matplotlib.pyplot as plt

# ECON_REGION distribution
df["ECON_REGION"].value_counts().plot(
    kind="bar",
    figsize=(10, 5)
)

plt.title("Data Distribution by ECON_REGION")
plt.xlabel("ECON_REGION")
plt.ylabel("Number of Records")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
