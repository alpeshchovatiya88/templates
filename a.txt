import pandas as pd
import numpy as np

from openpyxl.styles import Font, PatternFill, Alignment
from openpyxl.utils import get_column_letter


def reproducibility_report(
    df1,
    df2,
    dataset="Train",
    tolerance=1e-8,
    output_file="Reproducibility_Assessment.xlsx"
):

    # ==========================================
    # 1. CONFIGURATION
    # ==========================================

    id_col = "APPLICATION_NUMBER"

    metrics = {
        "endpoint_probability": "Endpoint_Probability",
        "endpoint_SMI3_LOSS": "Endpoint_Loss"
    }

    required_cols = [id_col] + list(metrics.keys())

    for name, df in [
        ("Developer", df1),
        ("Validator", df2)
    ]:
        missing_cols = [
            col for col in required_cols
            if col not in df.columns
        ]

        if missing_cols:
            raise ValueError(
                f"{name} missing columns: {missing_cols}"
            )

        if df[id_col].isna().any():
            raise ValueError(
                f"{name} has missing APPLICATION_NUMBER"
            )

        if df[id_col].duplicated().any():
            raise ValueError(
                f"{name} has duplicate APPLICATION_NUMBER"
            )

    dev_df = df1[required_cols].copy()
    val_df = df2[required_cols].copy()

    for col in metrics:
        dev_df[col] = pd.to_numeric(
            dev_df[col], errors="raise"
        )
        val_df[col] = pd.to_numeric(
            val_df[col], errors="raise"
        )

    # ==========================================
    # 2. MERGE ON APPLICATION_NUMBER
    # ==========================================

    merged = dev_df.merge(
        val_df,
        on=id_col,
        how="outer",
        suffixes=("_Developer", "_Validator"),
        indicator=True,
        validate="one_to_one"
    )

    id_match = merged["_merge"].eq("both").all()

    # ==========================================
    # 3. RECORD-LEVEL CHECKS
    # ==========================================

    record_results = []
    mismatch_masks = {}

    for col, metric_name in metrics.items():

        dev = merged[f"{col}_Developer"]
        val = merged[f"{col}_Validator"]

        matched_ids = merged["_merge"].eq("both")

        both_missing = (
            matched_ids
            & dev.isna()
            & val.isna()
        )

        one_sided_missing = (
            ~matched_ids
            | (
                matched_ids
                & dev.isna().ne(val.isna())
            )
        )

        both_present = (
            matched_ids
            & dev.notna()
            & val.notna()
        )

        abs_diff = (dev - val).abs()

        mismatch = (
            both_present
            & ~pd.Series(
                np.isclose(
                    dev,
                    val,
                    atol=tolerance,
                    rtol=0,
                    equal_nan=False
                ),
                index=merged.index
            )
        )

        mismatch_masks[col] = (
            one_sided_missing | mismatch
        )

        mismatch_count = int(mismatch.sum())
        records_compared = int(matched_ids.sum())

        mismatch_percentage = (
            mismatch_count / records_compared * 100
            if records_compared else 0
        )

        max_abs_diff = (
            abs_diff[both_present].max()
            if both_present.any() else 0
        )

        status = (
            "PASS"
            if mismatch_count == 0
            and not one_sided_missing.any()
            else "FAIL"
        )

        record_results.append({
            "Dataset": dataset,
            "Metric": metric_name,
            "Records_Compared": records_compared,
            "Both_Missing": int(both_missing.sum()),
            "One_Sided_Missing": int(one_sided_missing.sum()),
            "Mismatch_Count": mismatch_count,
            "Mismatch_Percentage": mismatch_percentage,
            "Max_Absolute_Difference": max_abs_diff,
            "Status": status
        })

    record_report = pd.DataFrame(record_results)

    # ==========================================
    # 4. AGGREGATE CHECKS
    # ==========================================

    aggregate_data = [
        (
            "Total_Rows",
            len(dev_df),
            len(val_df)
        ),
        (
            "Total_Loss",
            dev_df["endpoint_SMI3_LOSS"].sum(),
            val_df["endpoint_SMI3_LOSS"].sum()
        ),
        (
            "Total_Probability",
            dev_df["endpoint_probability"].sum(),
            val_df["endpoint_probability"].sum()
        ),
        (
            "Unique_Application_IDs",
            dev_df[id_col].nunique(),
            val_df[id_col].nunique()
        )
    ]

    aggregate_results = []

    for metric, dev_value, val_value in aggregate_data:

        abs_difference = abs(dev_value - val_value)

        if metric in [
            "Total_Rows",
            "Unique_Application_IDs"
        ]:
            passed = dev_value == val_value
            check_tolerance = 0
        else:
            passed = bool(
                np.isclose(
                    dev_value,
                    val_value,
                    atol=tolerance,
                    rtol=0
                )
            )
            check_tolerance = tolerance

        aggregate_results.append({
            "Dataset": dataset,
            "Metric": metric,
            "Developer_Value": dev_value,
            "Validator_Value": val_value,
            "Absolute_Difference": abs_difference,
            "Tolerance": check_tolerance,
            "Status": "PASS" if passed else "FAIL"
        })

    # Explicit ID comparison
    id_difference = len(
        set(dev_df[id_col]).symmetric_difference(
            set(val_df[id_col])
        )
    )

    aggregate_results.append({
        "Dataset": dataset,
        "Metric": "Application_ID_Match",
        "Developer_Value": len(dev_df),
        "Validator_Value": len(val_df),
        "Absolute_Difference": id_difference,
        "Tolerance": 0,
        "Status": "PASS" if id_match else "FAIL"
    })

    aggregate_report = pd.DataFrame(aggregate_results)

    # ==========================================
    # 5. MISSING VALUE SUMMARY
    # ==========================================

    missing_results = []

    for col, metric_name in metrics.items():

        dev_missing = int(dev_df[col].isna().sum())
        val_missing = int(val_df[col].isna().sum())

        matched = merged["_merge"].eq("both")

        both_missing_count = int(
            (
                matched
                & merged[f"{col}_Developer"].isna()
                & merged[f"{col}_Validator"].isna()
            ).sum()
        )

        one_sided_count = int(
            (
                ~matched
                | (
                    matched
                    & merged[f"{col}_Developer"].isna().ne(
                        merged[f"{col}_Validator"].isna()
                    )
                )
            ).sum()
        )

        missing_results.append({
            "Dataset": dataset,
            "Metric": metric_name,
            "Developer_Missing": dev_missing,
            "Validator_Missing": val_missing,
            "Both_Missing": both_missing_count,
            "One_Sided_Missing": one_sided_count,
            "Status": (
                "PASS"
                if one_sided_count == 0
                else "FAIL"
            )
        })

    missing_report = pd.DataFrame(missing_results)

    # ==========================================
    # 6. MISMATCH DETAILS
    # ==========================================

    mismatch_mask = merged["_merge"].ne("both")

    for col in metrics:
        mismatch_mask = (
            mismatch_mask | mismatch_masks[col]
        )

    mismatch_details = merged.loc[
        mismatch_mask
    ].copy()

    for col in metrics:
        mismatch_details[
            f"{col}_Absolute_Difference"
        ] = (
            mismatch_details[f"{col}_Developer"]
            - mismatch_details[f"{col}_Validator"]
        ).abs()

    # ==========================================
    # 7. OVERALL REPRODUCIBILITY STATUS
    # ==========================================

    all_statuses = pd.concat([
        record_report["Status"],
        aggregate_report["Status"],
        missing_report["Status"]
    ])

    overall_status = (
        "PASS"
        if all_statuses.eq("PASS").all()
        else "FAIL"
    )

    # ==========================================
    # 8. SUMMARY REPORT
    # ==========================================

    summary_results = []

    for _, row in aggregate_report.iterrows():
        summary_results.append({
            "Dataset": dataset,
            "Check": row["Metric"],
            "Developer_Value": row["Developer_Value"],
            "Validator_Value": row["Validator_Value"],
            "Status": row["Status"]
        })

    for _, row in record_report.iterrows():
        summary_results.append({
            "Dataset": dataset,
            "Check": row["Metric"],
            "Developer_Value": row["Records_Compared"],
            "Validator_Value": row["Mismatch_Count"],
            "Status": row["Status"]
        })

    summary_results.append({
        "Dataset": dataset,
        "Check": "OVERALL REPRODUCIBILITY",
        "Developer_Value": None,
        "Validator_Value": None,
        "Status": overall_status
    })

    summary_report = pd.DataFrame(summary_results)

    # ==========================================
    # 9. SAVE ALL REPORTS TO EXCEL
    # ==========================================

    with pd.ExcelWriter(
        output_file,
        engine="openpyxl"
    ) as writer:

        summary_report.to_excel(
            writer,
            sheet_name="Summary",
            index=False
        )

        record_report.to_excel(
            writer,
            sheet_name="Record_Level_Checks",
            index=False
        )

        aggregate_report.to_excel(
            writer,
            sheet_name="Aggregate_Checks",
            index=False
        )

        missing_report.to_excel(
            writer,
            sheet_name="Missing_Value_Summary",
            index=False
        )

        mismatch_details.to_excel(
            writer,
            sheet_name="Mismatch_Details",
            index=False
        )

        # ======================================
        # 10. EXCEL FORMATTING
        # ======================================

        workbook = writer.book

        header_fill = PatternFill(
            fill_type="solid",
            fgColor="17365D"
        )

        pass_fill = PatternFill(
            fill_type="solid",
            fgColor="C6EFCE"
        )

        fail_fill = PatternFill(
            fill_type="solid",
            fgColor="FFC7CE"
        )

        for sheet in workbook.worksheets:

            sheet.freeze_panes = "A2"
            sheet.auto_filter.ref = sheet.dimensions

            # Header style
            for cell in sheet[1]:

                cell.fill = header_fill
                cell.font = Font(
                    bold=True,
                    color="FFFFFF"
                )

                cell.alignment = Alignment(
                    horizontal="center"
                )

            # Auto-adjust column width
            for column in sheet.columns:

                column_letter = get_column_letter(
                    column[0].column
                )

                max_length = max(
                    len(str(cell.value))
                    if cell.value is not None else 0
                    for cell in column
                )

                sheet.column_dimensions[
                    column_letter
                ].width = min(max_length + 3, 45)

            # PASS / FAIL formatting
            for row in sheet.iter_rows(min_row=2):

                for cell in row:

                    if cell.value == "PASS":

                        cell.fill = pass_fill
                        cell.font = Font(
                            bold=True,
                            color="006100"
                        )

                    elif cell.value == "FAIL":

                        cell.fill = fail_fill
                        cell.font = Font(
                            bold=True,
                            color="9C0006"
                        )

    # ==========================================
    # 11. DISPLAY RESULTS
    # ==========================================

    print("=" * 80)
    print(f"REPRODUCIBILITY ASSESSMENT: {dataset}")
    print("=" * 80)

    print("\nRECORD-LEVEL CHECKS")
    print(record_report.to_string(index=False))

    print("\nAGGREGATE CHECKS")
    print(aggregate_report.to_string(index=False))

    print("\nMISSING VALUE SUMMARY")
    print(missing_report.to_string(index=False))

    print("\n" + "=" * 80)
    print(f"OVERALL REPRODUCIBILITY STATUS: {overall_status}")
    print("=" * 80)

    print(f"\nExcel report saved: {output_file}")

    return (
        summary_report,
        record_report,
        aggregate_report,
        missing_report,
        mismatch_details
    )


# ==========================================
# 12. RUN THE REPORT
# ==========================================

results = reproducibility_report(
    df1=df1,
    df2=df2,
    dataset="Train",
    tolerance=1e-8,
    output_file="Train_Reproducibility_Report.xlsx"
)
