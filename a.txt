# 1. PSI INTERPRETATION
# ============================================================

def psi_interpretation(psi):
    """
    Interpret PSI value.
    """

    if pd.isna(psi):
        return "Not calculated"
    elif psi < 0.10:
        return "Stable"
    elif psi < 0.25:
        return "Moderate shift"
    else:
        return "Significant shift"


# ============================================================
# 2. NUMERIC PSI
# ============================================================

def calculate_psi_numeric(
    expected,
    actual,
    feature,
    bins=10,
    eps=0.0001
):
    """
    Calculate PSI for one numeric feature.

    Bin limits are created from the expected population.
    """

    expected_data = (
        pd.to_numeric(expected[feature], errors="coerce")
        .replace([np.inf, -np.inf], np.nan)
        .dropna()
    )

    actual_data = (
        pd.to_numeric(actual[feature], errors="coerce")
        .replace([np.inf, -np.inf], np.nan)
        .dropna()
    )

    if expected_data.empty or actual_data.empty:
        return np.nan

    if expected_data.nunique() < 2:
        return np.nan

    try:
        # Create quantile bins from expected population
        _, bin_edges = pd.qcut(
            expected_data,
            q=bins,
            retbins=True,
            duplicates="drop"
        )

        bin_edges = np.unique(bin_edges)

        if len(bin_edges) < 2:
            return np.nan

        # Include values outside the expected range
        bin_edges[0] = -np.inf
        bin_edges[-1] = np.inf

        expected_bins = pd.cut(
            expected_data,
            bins=bin_edges,
            include_lowest=True
        )

        actual_bins = pd.cut(
            actual_data,
            bins=bin_edges,
            include_lowest=True
        )

        expected_pct = expected_bins.value_counts(
            normalize=True,
            sort=False
        )

        actual_pct = actual_bins.value_counts(
            normalize=True,
            sort=False
        )

        all_bins = expected_pct.index.union(actual_pct.index)

        expected_pct = expected_pct.reindex(
            all_bins,
            fill_value=0
        )

        actual_pct = actual_pct.reindex(
            all_bins,
            fill_value=0
        )

        expected_pct = expected_pct.clip(lower=eps)
        actual_pct = actual_pct.clip(lower=eps)

        psi_component = (
            (actual_pct - expected_pct)
            * np.log(actual_pct / expected_pct)
        )

        return float(psi_component.sum())

    except (ValueError, TypeError, KeyError):
        return np.nan


# ============================================================
# 3. CATEGORICAL PSI
# ============================================================

def calculate_psi_categorical(
    expected,
    actual,
    feature,
    eps=0.0001
):
    """
    Calculate PSI for one categorical feature.
    Missing values are treated as a separate category.
    """

    expected_data = (
        expected[feature]
        .astype("object")
        .where(expected[feature].notna(), "MISSING")
        .astype(str)
    )

    actual_data = (
        actual[feature]
        .astype("object")
        .where(actual[feature].notna(), "MISSING")
        .astype(str)
    )

    if expected_data.empty or actual_data.empty:
        return np.nan

    levels = sorted(
        set(expected_data.unique()).union(
            set(actual_data.unique())
        )
    )

    expected_pct = (
        expected_data
        .value_counts(normalize=True)
        .reindex(levels, fill_value=0)
        .clip(lower=eps)
    )

    actual_pct = (
        actual_data
        .value_counts(normalize=True)
        .reindex(levels, fill_value=0)
        .clip(lower=eps)
    )

    psi_component = (
        (actual_pct - expected_pct)
        * np.log(actual_pct / expected_pct)
    )

    return float(psi_component.sum())


# ============================================================
# 4. RUN PSI FOR ONE PERIOD COMPARISON
# ============================================================

def run_psi_for_model(
    expected,
    actual,
    model_name,
    numeric_cols=None,
    categorical_cols=None,
    bins=10
):
    """
    Calculate PSI for all supplied features.
    """

    numeric_cols = numeric_cols or []
    categorical_cols = categorical_cols or []

    results = []

    for col in numeric_cols:

        if col not in expected.columns or col not in actual.columns:
            print(f"Skipping missing numeric column: {col}")
            continue

        psi_value = calculate_psi_numeric(
            expected=expected,
            actual=actual,
            feature=col,
            bins=bins
        )

        results.append({
            "model": model_name,
            "feature": col,
            "type": "numeric",
            "psi": psi_value,
            "interpretation": psi_interpretation(psi_value)
        })

    for col in categorical_cols:

        if col not in expected.columns or col not in actual.columns:
            print(f"Skipping missing categorical column: {col}")
            continue

        psi_value = calculate_psi_categorical(
            expected=expected,
            actual=actual,
            feature=col
        )

        results.append({
            "model": model_name,
            "feature": col,
            "type": "categorical",
            "psi": psi_value,
            "interpretation": psi_interpretation(psi_value)
        })

    return pd.DataFrame(results)

# 5. CREATE YEAR / QUARTER PERIOD COLUMN
# ============================================================

def create_psi_period(
    df,
    date_col=None,
    year_col=None,
    quarter_col=None,
    annual_years=None,
    output_col="psi_period"
):
    """
    Create a PSI period column.

    Example required output:
        2021
        2022_Q1
        2022_Q2
        2022_Q3
        2022_Q4

    You can provide either:

    1. date_col

    OR

    2. year_col and quarter_col

    annual_years:
        Years that should be treated as full-year periods.
        Example: [2021]
    """

    data = df.copy()

    annual_years = annual_years or []

    if date_col is not None:

        if date_col not in data.columns:
            raise KeyError(
                f"Date column '{date_col}' is not present."
            )

        data[date_col] = pd.to_datetime(
            data[date_col],
            errors="coerce"
        )

        data["_psi_year"] = data[date_col].dt.year
        data["_psi_quarter"] = data[date_col].dt.quarter

    else:

        if year_col is None or quarter_col is None:
            raise ValueError(
                "Provide date_col, or both year_col and quarter_col."
            )

        if year_col not in data.columns:
            raise KeyError(
                f"Year column '{year_col}' is not present."
            )

        if quarter_col not in data.columns:
            raise KeyError(
                f"Quarter column '{quarter_col}' is not present."
            )

        data["_psi_year"] = pd.to_numeric(
            data[year_col],
            errors="coerce"
        )

        quarter_text = (
            data[quarter_col]
            .astype(str)
            .str.upper()
            .str.strip()
            .str.replace("QUARTER", "", regex=False)
            .str.replace("Q", "", regex=False)
        )

        data["_psi_quarter"] = pd.to_numeric(
            quarter_text,
            errors="coerce"
        )

    data[output_col] = np.where(
        data["_psi_year"].isin(annual_years),
        data["_psi_year"].astype("Int64").astype(str),
        (
            data["_psi_year"].astype("Int64").astype(str)
            + "_Q"
            + data["_psi_quarter"].astype("Int64").astype(str)
        )
    )

    invalid_mask = (
        data["_psi_year"].isna()
        | (
            ~data["_psi_year"].isin(annual_years)
            & ~data["_psi_quarter"].isin([1, 2, 3, 4])
        )
    )

    data.loc[invalid_mask, output_col] = np.nan

    return data

# 6. PREVIOUS-PERIOD PSI
# ============================================================

def calculate_previous_period_psi(
    df,
    period_col,
    periods,
    numeric_cols=None,
    categorical_cols=None,
    model_name="model",
    bins=10,
    multiply_by_100=True,
    include_first_period=True
):
    """
    Calculate PSI where each period is compared with the
    immediately previous period.

    For periods:
        2021, 2022_Q1, 2022_Q2, 2022_Q3, 2022_Q4

    Comparisons:
        2022_Q1 vs 2021
        2022_Q2 vs 2022_Q1
        2022_Q3 vs 2022_Q2
        2022_Q4 vs 2022_Q3

    Returns
    -------
    detailed_df:
        Long-format PSI results.

    wide_df:
        Features as rows and periods as columns.
    """

    if period_col not in df.columns:
        raise KeyError(
            f"Period column '{period_col}' is not present."
        )

    if len(periods) < 2:
        raise ValueError(
            "At least two periods are required."
        )

    numeric_cols = numeric_cols or []
    categorical_cols = categorical_cols or []

    data = df[df[period_col].isin(periods)].copy()

    all_results = []

    for i in range(1, len(periods)):

        previous_period = periods[i - 1]
        current_period = periods[i]

        expected = data[
            data[period_col] == previous_period
        ].copy()

        actual = data[
            data[period_col] == current_period
        ].copy()

        if expected.empty or actual.empty:
            print(
                f"Skipping {previous_period} vs {current_period}: "
                "data is missing for one or both periods."
            )
            continue

        period_result = run_psi_for_model(
            expected=expected,
            actual=actual,
            model_name=model_name,
            numeric_cols=numeric_cols,
            categorical_cols=categorical_cols,
            bins=bins
        )

        if period_result.empty:
            continue

        period_result["previous_period"] = previous_period
        period_result["period"] = current_period
        period_result["comparison"] = (
            f"{previous_period}_vs_{current_period}"
        )

        if multiply_by_100:
            period_result["psi_pct"] = (
                period_result["psi"] * 100
            )
        else:
            period_result["psi_pct"] = period_result["psi"]

        all_results.append(period_result)

    if not all_results:
        return pd.DataFrame(), pd.DataFrame()

    detailed_df = pd.concat(
        all_results,
        ignore_index=True
    )

    wide_df = detailed_df.pivot_table(
        index=["feature", "type"],
        columns="period",
        values="psi_pct",
        aggfunc="first"
    ).reset_index()

    wide_df.columns.name = None

    # Add the first period as a blank reference column
    if include_first_period:
        first_period = periods[0]

        if first_period not in wide_df.columns:
            wide_df.insert(
                loc=2,
                column=first_period,
                value=np.nan
            )

    # Force columns into the requested order
    ordered_period_columns = [
        period
        for period in periods
        if period in wide_df.columns
    ]

    wide_df = wide_df[
        ["feature", "type"] + ordered_period_columns
    ]

    return detailed_df, wide_df
