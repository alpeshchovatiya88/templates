target_years = [year for year in range(2009, 2026) if year != 2020]

df = df[df['BOOKYEAR'].isin(target_years)]

summary = (
    df.groupby(['CRD_SCORE_bin', 'BOOKYEAR'])
      .agg(
          Total_Count=('target', 'count'),
          Bad_Count=('target', 'sum'),
          Bad_Rate=('target', 'mean')
      )
      .reset_index()
)

summary['Bad_Rate'] = (summary['Bad_Rate'] * 100).round(2)

summary.to_csv('badrate_summary.csv', index=False)
