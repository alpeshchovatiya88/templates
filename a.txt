import matplotlib.pyplot as plt

# Optional: filter train/test separately
plot_df = (
    LR_bt_by
    .sort_values("BOOKYEAR")
)

plt.figure(figsize=(10, 6))

plt.plot(
    plot_df["BOOKYEAR"],
    plot_df["ACTUAL_LR"],
    marker="o",
    label="Actual LR"
)

plt.plot(
    plot_df["BOOKYEAR"],
    plot_df["SMI3_LR"],
    marker="o",
    label="SMI3 LR"
)

plt.xlabel("Book Year")
plt.ylabel("Loss Ratio")
plt.title("Actual LR vs SMI3 LR by Book Year")
plt.grid(True, alpha=0.3)
plt.legend()

plt.tight_layout()
plt.show()
