
# ============================================================
# 9. EVALUATE CALIBRATION PERFORMANCE
#    MEAN PD USING ALL ROWS
# ============================================================

from sklearn.metrics import (
    roc_auc_score,
    roc_curve,
    confusion_matrix,
    precision_score,
    recall_score,
    brier_score_loss,
    log_loss
)

threshold = 0.50


def evaluate(df, name, probability_col):

    # --------------------------------------------
    # ALL ROWS: N_Total and Mean PD
    # --------------------------------------------

    all_probabilities = pd.to_numeric(
        df[probability_col],
        errors="coerce"
    )

    if all_probabilities.isna().any():
        raise ValueError(
            f"{name}: Missing probabilities "
            f"in {probability_col}."
        )

    n_total = len(df)

    mean_pd = all_probabilities.mean()


    # --------------------------------------------
    # OBSERVED TARGETS: PERFORMANCE METRICS
    # --------------------------------------------

    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty:
        print(f"{name}: No observed targets.")
        return None

    y = data[target_col].astype(int)

    probabilities = pd.to_numeric(
        data[probability_col],
        errors="coerce"
    )

    n_observed = len(data)

    n_bad_events = int(y.sum())

    actual_bad_rate = y.mean()


    # --------------------------------------------
    # CLASSIFICATION AT THRESHOLD 0.50
    # --------------------------------------------

    y_pred = (
        probabilities >= threshold
    ).astype(int)

    tn, fp, fn, tp = confusion_matrix(
        y,
        y_pred,
        labels=[0, 1]
    ).ravel()

    precision = precision_score(
        y,
        y_pred,
        zero_division=0
    )

    recall = recall_score(
        y,
        y_pred,
        zero_division=0
    )

    fpr = (
        fp / (fp + tn)
        if (fp + tn) > 0
        else np.nan
    )


    # --------------------------------------------
    # AUC AND KS
    # --------------------------------------------

    if y.nunique() == 2:

        auc = roc_auc_score(
            y,
            probabilities
        )

        fpr_curve, tpr_curve, _ = roc_curve(
            y,
            probabilities
        )

        ks = np.max(
            tpr_curve - fpr_curve
        )

    else:
        auc = np.nan
        ks = np.nan


    # --------------------------------------------
    # RETURN METRICS
    # --------------------------------------------

    return {
        "Dataset": name,
        "N_Total": n_total,
        "N_Observed": n_observed,
        "N_Bad_Events": n_bad_events,
        "Actual_Bad_Rate": actual_bad_rate,
        "Mean_PD": mean_pd,
        "AUC": auc,
        "KS": ks,
        "Precision": precision,
        "Recall": recall,
        "FPR": fpr,
        "Brier_Score": brier_score_loss(
            y,
            probabilities
        ),
        "Log_Loss": log_loss(
            y,
            probabilities,
            labels=[0, 1]
        )
    }
