
import pandas as pd
import numpy as np
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side

# =====================================================
# 1. CHANGE ONLY THIS DICTIONARY
# =====================================================

dfs = {
    "RS42": df_rs42,
    "RS532": df_rs532,
    "RS3458": df_rs3458,
    "RS499": df_rs499,
    "RS345": df_rs345,
    "RS1999": df_rs1999,
    "RS2324": df_rs2324,
}

DECILE_COL = "rr_model_decile"
LR_SCALE = 1  # Use 100 if SMI3_LR contains decimals such as 0.85
OUTPUT_FILE = "seed_experiments.xlsx"

# Optional: descriptions under experiment names
descriptions = {
    # "RS42": "Experiment description",
    # "RS532": "Another description",
}

# =====================================================
# 2. EXCEL FORMATTING
# =====================================================

wb = Workbook()
ws = wb.active
ws.title = "Seed Experiments"

yellow = PatternFill("solid", fgColor="FFFF00")
blue = PatternFill("solid", fgColor="DDEBF7")

thin = Side(style="thin", color="B7B7B7")
border = Border(left=thin, right=thin, top=thin, bottom=thin)

ws.column_dimensions["A"].width = 65
ws.column_dimensions["B"].width = 23
ws.column_dimensions["C"].width = 23
ws.column_dimensions["D"].width = 19
ws.column_dimensions["E"].width = 19

# =====================================================
# 3. AUTOMATICALLY CREATE A SECTION FOR EACH EXPERIMENT
# =====================================================

row = 1

for experiment, df in dfs.items():

    temp = df.copy()

    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL].astype(str).str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    temp["SMI3_LOSS"] = pd.to_numeric(
        temp["SMI3_LOSS"], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan)

    temp["SMI3_LR"] = pd.to_numeric(
        temp["SMI3_LR"], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan) * LR_SCALE

    decile1 = temp[temp["_decile"] == 1]

    # Header row
    ws.cell(row, 1, experiment)
    ws.cell(row, 1).fill = yellow
    ws.cell(row, 1).font = Font(size=15, bold=True)

    ws.merge_cells(
        start_row=row, start_column=2,
        end_row=row, end_column=3
    )
    ws.merge_cells(
        start_row=row, start_column=4,
        end_row=row, end_column=5
    )

    ws.cell(row, 2, "Loss range")
    ws.cell(row, 4, "Loss ratio range")

    for col in [2, 4]:
        cell = ws.cell(row, col)
        cell.font = Font(size=14, bold=True)
        cell.alignment = Alignment(horizontal="center")

    ws.row_dimensions[row].height = 30

    # Description and min/max headings
    ws.cell(row + 1, 1, descriptions.get(experiment, ""))
    ws.cell(row + 1, 1).fill = blue
    ws.cell(row + 1, 1).alignment = Alignment(
        wrap_text=True,
        vertical="center"
    )

    for col, text in zip(
        [2, 3, 4, 5],
        ["min", "max", "min", "max"]
    ):
        cell = ws.cell(row + 1, col, text)
        cell.font = Font(size=13, bold=True)
        cell.alignment = Alignment(horizontal="center")

    ws.row_dimensions[row + 1].height = 35

    # Overall and Decile 1
    data = [
        (
            "Overall",
            temp["SMI3_LOSS"].min(),
            temp["SMI3_LOSS"].max(),
            temp["SMI3_LR"].min(),
            temp["SMI3_LR"].max()
        ),
        (
            "Decile 1",
            decile1["SMI3_LOSS"].min(),
            decile1["SMI3_LOSS"].max(),
            decile1["SMI3_LR"].min(),
            decile1["SMI3_LR"].max()
        )
    ]

    for r, values in enumerate(data, start=row + 2):

        for col, value in enumerate(values, start=1):

            if col > 1 and pd.isna(value):
                value = None
            elif isinstance(value, (np.integer, np.floating)):
                value = value.item()

            cell = ws.cell(r, col, value)

            cell.alignment = Alignment(
                horizontal="left" if col == 1 else "right",
                vertical="center"
            )

            if col in [2, 3]:
                cell.number_format = '$#,##0'
            elif col in [4, 5]:
                cell.number_format = '0.00'

        ws.row_dimensions[r].height = 28

    # Table borders
    for r in range(row, row + 4):
        for col in range(1, 6):
            ws.cell(r, col).border = border

    # Two blank rows before next experiment
    row += 6

# =====================================================
# 4. SAVE EXCEL - ONE SHEET ONLY
# =====================================================

wb.save(OUTPUT_FILE)

print(f"Saved: {OUTPUT_FILE}")
