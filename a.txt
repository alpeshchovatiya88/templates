)

    plot_df.groupby("BOOKYEAR")["count"].transform("sum")
    * 100
)

fig = px.bar(
    plot_df,
    x="BOOKYEAR",
    y="pct",
    color="PRD_LOAN_PURPOSE",
    text="pct",
    hover_data=["count"],
    title="PRD_LOAN_PURPOSE Distribution by BOOKYEAR"
)

    texttemplate="%{text:.1f}%",
    textposition="inside"
)

fig.update_layout(
    yaxis_title="Percentage (%)",
    xaxis_title="BOOKYEAR",
    barmode="stack"
)

fig.show()
