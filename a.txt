
"""Standalone offline seed-sensitivity dashboard with Python-only statistics.

Dependencies: pip install pandas numpy plotly
"""
from __future__ import annotations

import itertools
import json
from pathlib import Path
from string import Template

import numpy as np
import pandas as pd
from plotly.offline import get_plotlyjs


def export_interactive_seed_report(
    dfs: dict,
    output_html: str = "seed_sensitivity_python_precomputed.html",
    *,
    ventile_col: str = "VENTILE",
    region_col: str = "REGION",
    market_col: str = "MARKET",
    tenure_col: str = "TENURE",
    probability_col: str = "prediction_probability",
    metrics: dict | None = None,
    baseline_seed: str = "RS42",
    claims_share: float = 0.62,
    max_combinations: int = 100_000,
) -> Path:
    """Build a standalone report; all numeric calculations run in Python.

    Filter values are exact string matches. Missing categorical values are shown
    as '(Missing)'. A blank ventile selection means ALL ventiles.

    The number of precomputed combinations is
      (2 ** n_ventiles) * (n_regions+1) * (n_markets+1) * (n_tenures+1).
    One additional choice per categorical filter represents 'All'.

    Metrics configuration, for example:
      {"SMI3_LOSS": {"agg": "sum", "label": "SMI3 Loss", "scale": 1},
       "SMI3_LR": {"agg": "mean", "label": "SMI3 LR", "scale": 100,
                    "unit": "%", "delta_unit": "pp"}}
    """
    if not dfs:
        raise ValueError("dfs must contain at least one seed DataFrame")
    if baseline_seed not in dfs:
        raise ValueError(f"Baseline seed {baseline_seed!r} is missing")
    if not 0 <= claims_share <= 1:
        raise ValueError("claims_share must be between 0 and 1")
    if metrics is None:
        metrics = {
            "SMI3_LOSS": {"agg": "sum", "label": "SMI3 Loss", "scale": 1},
            "SMI3_LR": {"agg": "mean", "label": "SMI3 LR", "scale": 100,
                         "unit": "%", "delta_unit": "pp"},
        }
    for metric, cfg in metrics.items():
        if cfg.get("agg") not in ("sum", "mean"):
            raise ValueError(f"{metric}: agg must be 'sum' or 'mean'")
        if not np.isfinite(float(cfg.get("scale", 1))):
            raise ValueError(f"{metric}: scale must be finite")

    required = [ventile_col, region_col, market_col, tenure_col,
                probability_col, *metrics]
    frames = []
    for seed, frame in dfs.items():
        missing = [col for col in required if col not in frame.columns]
        if missing:
            raise ValueError(f"{seed}: missing columns {missing}")
        f = frame[required].copy()
        f["_seed"] = str(seed)
        # Accept numeric ventiles or text such as 'Ventile 03'.
        f["_ventile"] = pd.to_numeric(
            f[ventile_col].astype(str).str.extract(r"(-?\d+)")[0],
            errors="coerce",
        )
        f = f.dropna(subset=["_ventile"])
        f["_ventile"] = f["_ventile"].astype(int)
        for col, key in [(region_col, "_region"),
                         (market_col, "_market"),
                         (tenure_col, "_tenure")]:
            f[key] = f[col].astype("string").fillna("(Missing)").astype(str)
        for col in [probability_col, *metrics]:
            f[col] = pd.to_numeric(f[col], errors="coerce")
            f[col] = f[col].replace([np.inf, -np.inf], np.nan)
        frames.append(f)
    data = pd.concat(frames, ignore_index=True)
    seeds = [str(s) for s in dfs]
    ventiles = sorted(int(v) for v in data["_ventile"].unique())
    regions = sorted(data["_region"].unique().tolist())
    markets = sorted(data["_market"].unique().tolist())
    tenures = sorted(data["_tenure"].unique().tolist())

    combination_count = ((2 ** len(ventiles)) * (len(regions) + 1)
                         * (len(markets) + 1) * (len(tenures) + 1))
    if combination_count > max_combinations:
        raise ValueError(
            f"Precomputation requires {combination_count:,} filter combinations, "
            f"exceeding max_combinations={max_combinations:,}. "
            "Reduce the number of ventiles/categories, restrict selections, "
            "or use a Python backend for unrestricted filtering."
        )

    def clean(value):
        if value is None or pd.isna(value):
            return None
        return float(value)

    def summarize(frame, metric, cfg):
        values = frame[metric].dropna()
        probs = frame[probability_col].dropna()
        scale = float(cfg.get("scale", 1))
        predicted = clean(probs.sum()) if len(probs) else None
        if len(values):
            aggregate = values.sum() if cfg["agg"] == "sum" else values.mean()
            stats = {
                "count": int(len(values)),
                "aggregate": clean(aggregate * scale),
                "mean": clean(values.mean() * scale),
                "min": clean(values.min() * scale),
                "max": clean(values.max() * scale),
                "p10": clean(values.quantile(.10) * scale),
                "p50": clean(values.quantile(.50) * scale),
                "p90": clean(values.quantile(.90) * scale),
            }
        else:
            stats = {"count": 0, "aggregate": None, "mean": None,
                     "min": None, "max": None, "p10": None, "p50": None,
                     "p90": None}
        stats.update({
            "probability_count": int(len(probs)),
            "predicted_total": predicted,
            "predicted_claims": predicted * claims_share if predicted is not None else None,
            "predicted_delqs": predicted * (1 - claims_share) if predicted is not None else None,
        })
        return stats

    # Cache filtered slices per category tuple, then apply each ventile subset.
    records = {}
    category_choices = itertools.product(
        [""] + regions, [""] + markets, [""] + tenures
    )
    for region, market, tenure in category_choices:
        category_frame = data
        for col, choice in [("_region", region), ("_market", market),
                            ("_tenure", tenure)]:
            if choice:
                category_frame = category_frame[category_frame[col] == choice]
        for mask in range(2 ** len(ventiles)):
            selected = [v for i, v in enumerate(ventiles) if mask & (1 << i)]
            # Both zero selected and all selected are interpreted as ALL.
            subset = (category_frame if not selected or len(selected) == len(ventiles)
                      else category_frame[category_frame["_ventile"].isin(selected)])
            result = {}
            for metric, cfg in metrics.items():
                by_seed = {}
                for seed in seeds:
                    seed_frame = subset[subset["_seed"] == seed]
                    by_seed[seed] = summarize(seed_frame, metric, cfg)
                baseline = by_seed[baseline_seed]["aggregate"]
                for seed in seeds:
                    stat = by_seed[seed]
                    current = stat["aggregate"]
                    stat["delta"] = (current - baseline
                                     if current is not None and baseline is not None
                                     else None)
                    stat["delta_pct"] = (
                        (current - baseline) / baseline * 100
                        if current is not None and baseline is not None and baseline != 0
                        else None
                    )
                result[metric] = by_seed
            # mask=0 and mask=all have equivalent results; retain both keys.
            records[f"{mask}|{region}|{market}|{tenure}"] = result

    payload = {
        "seeds": seeds,
        "baseline": baseline_seed,
        "ventiles": ventiles,
        "regions": regions,
        "markets": markets,
        "tenures": tenures,
        "metrics": metrics,
        "records": records,
    }
    # Avoid literal closing script tags in any user-provided categorical text.
    json_data = json.dumps(payload, separators=(",", ":"), ensure_ascii=False,
                           allow_nan=False).replace("<", "\\u003c")

    html = Template(r'''<!DOCTYPE html>

h1{font-size:27px;margin:0 0 7px}.sub{color:#637186;margin:0 0 22px}
.controls{display:flex;gap:14px;flex-wrap:wrap;align-items:end;background:white;padding:18px;border-radius:13px;box-shadow:0 2px 12px #1e2b3910}

select,button{border:1px solid #cbd4e1;border-radius:7px;padding:9px 11px;background:white;font:inherit;color:inherit}
button{cursor:pointer}.ventiles{display:flex;flex-wrap:wrap;gap:9px;max-width:100%;margin-top:8px}
.ventiles label{font-weight:400;background:#f3f6fb;padding:7px;border-radius:6px;white-space:nowrap}

.table-scroll{overflow:auto}.negative{color:#ba3131}.positive{color:#137b53}
</style>
<script>$PLOTLY_JS</script>
</head>
