
import json
from pathlib import Path

import numpy as np
import pandas as pd
from plotly.offline import get_plotlyjs


def export_interactive_seed_report(
    dfs,
    filename="Random_Seed_Sensitivity_Report.html",
    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    metrics=None,
    baseline="RS42",
    claims_share=0.62,
    colors=None,
    decimals=2,
    inside_font_size=13,
    chart_height=950
):
    """
    Interactive offline random seed sensitivity report.

    Metric dropdown:
        SMI3_LOSS
        SMI3_LR

    Filters:
        Metric, Ventile, ECON_REGION,
        POD_MARKET_CODE, PRK_TENURE

    Inside every bar:
        Min, Max, Avg, P10, P50, P90
        Predicted Delq + Claims
        Predicted Claims
        Predicted Delqs
        Difference versus baseline

    Predicted counts are NOT dropdown metrics.
    """

    if baseline not in dfs:
        raise ValueError(f"Baseline {baseline} not found")

    if not 0 <= claims_share <= 1:
        raise ValueError("claims_share must be between 0 and 1")

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    for metric, config in metrics.items():
        if config["agg"] not in ("sum", "mean"):
            raise ValueError(
                f"{metric}: agg must be 'sum' or 'mean'"
            )

    # --------------------------------------------------------
    # Prepare observations
    #
    # One row per observation and metric.
    # Prediction probability is embedded separately,
    # so it does not appear in the metric dropdown.
    # --------------------------------------------------------

    records = []
    ventile_values = set()
    region_values = set()
    market_values = set()
    tenure_values = set()

    for seed, source_df in dfs.items():

        required = list(dict.fromkeys([
            ventile_col,
            region_col,
            market_col,
            tenure_col,
            probability_col,
            *metrics.keys()
        ]))

        missing = [
            c for c in required
            if c not in source_df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed} missing columns: {missing}"
            )

        df = source_df[required].copy()

        ventiles = pd.to_numeric(
            df[ventile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        regions = (
            df[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        markets = (
            df[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        tenures = (
            df[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        ventile_values.update(
            ventiles.dropna().astype(int).tolist()
        )
        region_values.update(regions.tolist())
        market_values.update(markets.tolist())
        tenure_values.update(tenures.tolist())

        # [seed, ventile, region, market, tenure,
        #  metric, value, probability]

        probability = (
            pd.to_numeric(
                df[probability_col],
                errors="coerce"
            )
            .replace([np.inf, -np.inf], np.nan)
        )

        for metric in metrics:

            values = (
                pd.to_numeric(df[metric], errors="coerce")
                .replace([np.inf, -np.inf], np.nan)
            )

            # Include a row if metric OR probability exists.
            # The JS code independently handles missing values.

            valid = values.notna() | probability.notna()

            for i in np.flatnonzero(valid.to_numpy()):

                v = values.iloc[i]
                p = probability.iloc[i]
                ventile = ventiles.iloc[i]

                records.append([
                    str(seed),
                    int(ventile)
                        if pd.notna(ventile) else None,
                    str(regions.iloc[i]),
                    str(markets.iloc[i]),
                    str(tenures.iloc[i]),
                    metric,
                    float(v) if pd.notna(v) else None,
                    float(p) if pd.notna(p) else None
                ])

    report = {
        "seeds": list(dfs.keys()),
        "baseline": baseline,
        "metrics": metrics,
        "colors": colors,
        "ventiles": sorted(ventile_values),
        "regions": sorted(region_values),
        "markets": sorted(market_values),
        "tenures": sorted(tenure_values),
        "records": records,
        "claimsShare": claims_share,
        "decimals": decimals,
        "fontSize": inside_font_size,
        "chartHeight": chart_height
    }

    report_json = json.dumps(
        report,
        ensure_ascii=False,
        allow_nan=False,
        separators=(",", ":")
    ).replace("<", "\\u003c")

    html = r"""
<!DOCTYPE html>
<html lang="en">

<head>
<meta charset="UTF-8">
<meta name="viewport"
      content="width=device-width, initial-scale=1">

<title>Random Seed Sensitivity Analysis</title>

<style>

* { box-sizing: border-box; }

body {
    margin: 0;
    font-family: Arial, sans-serif;
    background: #f4f6fa;
    color: #172033;
}

.container {
    max-width: 1700px;
    margin: auto;
    padding: 24px;
}

h1 { margin-bottom: 6px; }

.subtitle {
    color: #64748b;
    margin-bottom: 22px;
}

.filters {
    display: flex;
    flex-wrap: wrap;
    gap: 15px;
    padding: 20px;
    background: white;
    border-radius: 12px;
    margin-bottom: 20px;
    align-items: end;
}

.filter {
    display: flex;
    flex-direction: column;
    gap: 7px;
}

label {
    font-weight: bold;
    font-size: 13px;
}

select {
    min-width: 170px;
    max-width: 250px;
    padding: 10px;
    border: 1px solid #cbd5e1;
    border-radius: 6px;
    background: white;
}

button {
    padding: 11px 18px;
    background: #17365D;
    color: white;
    border: 0;
    border-radius: 6px;
    cursor: pointer;
}

.card {
    background: white;
    padding: 18px;
    border-radius: 12px;
    margin-bottom: 20px;
}

#chart {
    width: 100%;
    min-height: 950px;
}

.table-wrap {
    overflow-x: auto;
}

table {
    width: 100%;
    border-collapse: collapse;
    white-space: nowrap;
    font-size: 13px;
}

    background: #17365D;
    color: white;
    text-align: right;
    padding: 12px;
}

td {
    padding: 12px;
    text-align: right;
    border-bottom: 1px solid #e2e8f0;
}

th:first-child,
td:first-child {
    text-align: left;
}

tbody tr:nth-child(even) {
    background: #f8fafc;
}

.note {
    color: #64748b;
    font-size: 13px;
    margin-top: 15px;
}

</style>

<script>
__PLOTLY_JS__
</script>

</head>

<body>

<div class="container">

<h1>Random Seed Sensitivity Analysis</h1>

<div class="subtitle">
Interactive comparison against baseline
</div>

<div class="filters">

<div class="filter">
<label>Metric</label>
<select id="metric"></select>
</div>

<div class="filter">
<label>Ventile</label>
<select id="ventile"></select>
</div>

<div class="filter">
<label>ECON_REGION</label>
<select id="region"></select>
</div>

<div class="filter">
<label>POD_MARKET_CODE</label>
<select id="market"></select>
</div>

<div class="filter">
<label>PRK_TENURE</label>
<select id="tenure"></select>
</div>

<button id="reset">Reset Filters</button>

</div>

<div class="card">
<div id="chart"></div>
</div>

<div class="card">

<h2>Detailed Statistics</h2>

<div class="table-wrap">
<table>

<thead>
<tr>
<th>Seed</th>
<th>Count</th>
<th>Aggregate</th>
<th id="deltaHead">Δ</th>
<th id="pctHead">Δ%</th>
<th>Min</th>
<th>Max</th>
<th>Avg</th>
<th>P10</th>
<th>P50</th>
<th>P90</th>
<th>Predicted Delq + Claims</th>
<th>Predicted Claims</th>
<th>Predicted Delqs</th>
</tr>
</thead>

<tbody id="statsBody"></tbody>

</table>
</div>

<div class="note" id="note"></div>

</div>

<script id="reportData" type="application/json">
__REPORT_JSON__
</script>

<script>

(function () {

"use strict";

const report = JSON.parse(
    document.getElementById("reportData").textContent
);

const $ = id => document.getElementById(id);

function addOption(select, value, label) {
    const option = document.createElement("option");
    option.value = value;
    option.textContent = label;
    select.appendChild(option);
}

// Metric dropdown contains ONLY configured metrics.

Object.keys(report.metrics).forEach(metric => {
    addOption($("metric"), metric, metric);
});

addOption($("ventile"), "", "All Ventiles");

report.ventiles.forEach(v => {
    addOption(
        $("ventile"),
        String(v),
        "Ventile " + v
    );
});

addOption($("region"), "", "All Regions");
report.regions.forEach(x =>
    addOption($("region"), x, x)
);

addOption($("market"), "", "All Markets");
report.markets.forEach(x =>
    addOption($("market"), x, x)
);

addOption($("tenure"), "", "All Tenures");
report.tenures.forEach(x =>
    addOption($("tenure"), x, x)
);

function number(value, digits=report.decimals) {
    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    return value.toLocaleString("en-US", {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
    });
}

function metricNumber(value, metric) {
    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    const cfg = report.metrics[metric];

    if (cfg.is_ratio) {
        return number(
            value / (cfg.ratio_scale || 1) * 100
        ) + "%";
    }

    return number(value);
}

function deltaNumber(value, metric) {
    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    const cfg = report.metrics[metric];
    const sign = value > 0 ? "+" : "";

    if (cfg.is_ratio) {
        return sign + number(
            value / (cfg.ratio_scale || 1) * 100
        ) + " pp";
    }

    return sign + number(value);
}

function percentNumber(value) {
    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    return (
        (value > 0 ? "+" : "") +
        number(value) +
        "%"
    );
}

function quantile(sorted, p) {
    if (!sorted.length) return null;

    if (sorted.length === 1) {
        return sorted[0];
    }

    const position = (sorted.length - 1) * p;
    const low = Math.floor(position);
    const high = Math.ceil(position);
    const weight = position - low;

    return (
        sorted[low] * (1 - weight) +
        sorted[high] * weight
    );
}

function calculateStats(values, aggregation) {

    if (!values.length) {
        return {
            count: 0,
            total: null,
            min: null,
            max: null,
            avg: null,
            p10: null,
            p50: null,
            p90: null
        };
    }

    const sorted = [...values].sort((a, b) => a - b);
    const sum = values.reduce((a, b) => a + b, 0);
    const avg = sum / values.length;

    return {
        count: values.length,
        total: aggregation === "sum" ? sum : avg,
        min: sorted[0],
        max: sorted[sorted.length - 1],
        avg: avg,
        p10: quantile(sorted, 0.10),
        p50: quantile(sorted, 0.50),
        p90: quantile(sorted, 0.90)
    };
}

function update() {

    const metric = $("metric").value;
    const cfg = report.metrics[metric];

    const ventile = $("ventile").value === ""
        ? null
        : Number($("ventile").value);

    const region = $("region").value;
    const market = $("market").value;
    const tenure = $("tenure").value;

    // Select only the active metric.
    // This prevents predicted counts being duplicated.

    const filtered = report.records.filter(r =>
        r[5] === metric &&
        (ventile === null || r[1] === ventile) &&
        (region === "" || r[2] === region) &&
        (market === "" || r[3] === market) &&
        (tenure === "" || r[4] === tenure)
    );

    const grouped = {};

    report.seeds.forEach(seed => {
        grouped[seed] = {
            values: [],
            probabilitySum: 0,
            probabilityCount: 0
        };
    });

    filtered.forEach(r => {

        const group = grouped[r[0]];

        if (r[6] !== null) {
            group.values.push(r[6]);
        }

        if (r[7] !== null) {
            group.probabilitySum += r[7];
            group.probabilityCount += 1;
        }
    });

    const result = {};

    report.seeds.forEach(seed => {

        const group = grouped[seed];

        const s = calculateStats(
            group.values,
            cfg.agg
        );

        const p = group.probabilityCount
            ? group.probabilitySum
            : null;

        s.predictedTotal = p;

        s.predictedClaims = p === null
            ? null
            : p * report.claimsShare;

        s.predictedDelqs = p === null
            ? null
            : p * (1 - report.claimsShare);

        result[seed] = s;
    });

    const baselineTotal = result[report.baseline].total;

    const rows = report.seeds.map(seed => {

        const s = result[seed];

        const delta = (
            s.total !== null &&
            baselineTotal !== null
        )
            ? s.total - baselineTotal
            : null;

        const deltaPct = (
            delta !== null &&
            baselineTotal !== 0
        )
            ? delta / baselineTotal * 100
            : null;

        return {
            seed,
            ...s,
            delta,
            deltaPct
        };
    });

    // --------------------------------------------------------
    // BAR CHART
    // --------------------------------------------------------

    const traces = [];
    const annotations = [];

    rows.forEach((r, i) => {

        if (r.total === null) return;

        const deltaText = r.seed === report.baseline
            ? "Baseline"
            : deltaNumber(r.delta, metric);

        const deltaPctText = r.seed === report.baseline
            ? "Baseline"
            : percentNumber(r.deltaPct);

        const lines = [
            "<b>Min: " + metricNumber(r.min, metric) + "</b>",
            "<b>Max: " + metricNumber(r.max, metric) + "</b>",
            "<b>Avg: " + metricNumber(r.avg, metric) + "</b>",
            "<b>P10: " + metricNumber(r.p10, metric) + "</b>",
            "<b>P50: " + metricNumber(r.p50, metric) + "</b>",
            "<b>P90: " + metricNumber(r.p90, metric) + "</b>",
            "────────────",
            "<b>Pred Delq + Claims: " +
                number(r.predictedTotal) + "</b>",
            "<b>Pred Claims: " +
                number(r.predictedClaims) + "</b>",
            "<b>Pred Delqs: " +
                number(r.predictedDelqs) + "</b>",
            "────────────",
            "<b>Δ: " + deltaText + "</b>"
        ];

        if (!cfg.is_ratio) {
            lines.push(
                "<b>Δ%: " + deltaPctText + "</b>"
            );
        }

        const insideText = lines.join("<br>");

        traces.push({
            type: "bar",
            x: [i],
            y: [r.total],
            width: 0.78,
            marker: {
                color: report.colors[r.seed] || "#64748b"
            },
            text: [insideText],
            textposition: "inside",
            insidetextanchor: "middle",
            constraintext: "none",
            cliponaxis: false,
            textfont: {
                size: report.fontSize,
                color: "white",
                family: "Arial"
            },
            hovertemplate:
                "<b>" + r.seed + "</b><br>" +
                "Aggregate: " +
                metricNumber(r.total, metric) +
                "<br>" + insideText +
                "<extra></extra>",
            showlegend: false
        });

        annotations.push({
            x: i,
            y: r.total,
            text: "<b>" +
                metricNumber(r.total, metric) +
                "</b>",
            showarrow: false,
            yshift: r.total >= 0 ? 20 : -20,
            font: {
                size: 17,
                color: "#172033"
            }
        });
    });

    const totals = rows
        .map(r => r.total)
        .filter(v => v !== null && Number.isFinite(v));

    const minValue = Math.min(0, ...totals);
    const maxValue = Math.max(0, ...totals);
    const span = Math.max(maxValue - minValue, 1e-9);

    const layout = {
        title: {
            text:
                "<b>" + metric +
                " — Random Seed Sensitivity</b>" +
                "<br><sup>" +
                (
                    ventile === null
                    ? "All Ventiles"
                    : "Ventile " + ventile
                ) +
                "</sup>",
            x: 0.5,
            font: {size: 22}
        },

        height: report.chartHeight,

        template: "plotly_white",

        margin: {
            t: 125,
            b: 90,
            l: 100,
            r: 60
        },

        xaxis: {
            tickmode: "array",
            tickvals: report.seeds.map((_, i) => i),
            ticktext: report.seeds,
            range: [
                -0.6,
                report.seeds.length - 0.4
            ],
            showgrid: false,
            title: "Experiment"
        },

        yaxis: {
            title: metric,
            range: [
                minValue - span * 0.08,
                maxValue + span * 0.25
            ],
            ticksuffix:
                cfg.is_ratio &&
                (cfg.ratio_scale || 1) === 100
                ? "%"
                : ""
        },

        annotations,

        uniformtext: {
            minsize: 9,
            mode: "show"
        },

        paper_bgcolor: "white",
        plot_bgcolor: "white"
    };

    Plotly.react(
        "chart",
        traces,
        layout,
        {
            responsive: true,
            displaylogo: false
        }
    );

    // --------------------------------------------------------
    // TABLE
    // --------------------------------------------------------

    $("deltaHead").textContent =
        cfg.is_ratio ? "Δ (pp)" : "Δ";

    $("pctHead").style.display =
        cfg.is_ratio ? "none" : "";

    $("statsBody").innerHTML = "";

    rows.forEach(r => {

        const tr = document.createElement("tr");

        const fields = [
            r.seed,
            number(r.count, 0),
            metricNumber(r.total, metric),
            r.seed === report.baseline
                ? "Baseline"
                : deltaNumber(r.delta, metric),
            r.seed === report.baseline
                ? "Baseline"
                : percentNumber(r.deltaPct),
            metricNumber(r.min, metric),
            metricNumber(r.max, metric),
            metricNumber(r.avg, metric),
            metricNumber(r.p10, metric),
            metricNumber(r.p50, metric),
            metricNumber(r.p90, metric),
            number(r.predictedTotal),
            number(r.predictedClaims),
            number(r.predictedDelqs)
        ];

        fields.forEach((value, index) => {

            const td = document.createElement("td");
            td.textContent = value;

            if (index === 4 && cfg.is_ratio) {
                td.style.display = "none";
            }

            tr.appendChild(td);
        });

        $("statsBody").appendChild(tr);
    });

    $("note").textContent = cfg.is_ratio
        ? "SMI3_LR differences are shown in percentage points."
        : "Δ = experiment minus baseline; Δ% = Δ / baseline × 100.";
}

["metric", "ventile", "region", "market", "tenure"]
    .forEach(id => {
        $(id).addEventListener("change", update);
    });

$("reset").addEventListener("click", () => {
    $("metric").selectedIndex = 0;
    $("ventile").value = "";
    $("region").value = "";
    $("market").value = "";
    $("tenure").value = "";
    update();
});

update();

})();
</script>

</body>
</html>
"""

    final_html = (
        html.replace("__PLOTLY_JS__", get_plotlyjs())
            .replace("__REPORT_JSON__", report_json)
    )

    output = Path(filename).resolve()
    output.write_text(final_html, encoding="utf-8")

    print(f"Report created: {output}")
    print(
        f"File size: "
        f"{output.stat().st_size / 1024**2:.2f} MB"
    )

    return output
