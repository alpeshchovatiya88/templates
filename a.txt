
import pandas as pd
import plotly.graph_objects as go


def plot_by_decile(
    dfs,
    decile_col,
    value_col,
    agg="sum",
    title=None,
    decimals=2,
    label_font_size=15,
    height=700
):

    # --------------------------------------------------
    # 1. Aggregate metric by decile for each seed
    # --------------------------------------------------

    summary = pd.DataFrame({
        name: (
            df.assign(
                _decile=pd.to_numeric(
                    df[decile_col]
                    .astype(str)
                    .str.extract(r"(\d+)", expand=False),
                    errors="coerce"
                )
            )
            .groupby("_decile")[value_col]
            .agg(agg)
        )
        for name, df in dfs.items()
    }).sort_index()

    # --------------------------------------------------
    # 2. Create grouped bar chart
    # --------------------------------------------------

    fig = go.Figure()

    for name in summary.columns:

        values = summary[name]

        fig.add_trace(
            go.Bar(
                x=summary.index,
                y=values,
                name=name,

                # Bold formatted values above bars
                text=[
                    f"{v:,.{decimals}f}"
                    if pd.notna(v) else ""
                    for v in values
                ],

                textposition="outside",

                textfont=dict(
                    family="Arial Black",
                    size=label_font_size,
                    color="black"
                ),

                cliponaxis=False,

                hovertemplate=(
                    "Decile: %{x}<br>"
                    f"Seed: {name}<br>"
                    "Value: %{y:,.2f}"
                    "<extra></extra>"
                )
            )
        )

    # --------------------------------------------------
    # 3. Decile dropdown
    # --------------------------------------------------

    buttons = [
        dict(
            label="All Deciles",
            method="relayout",
            args=[
                {"xaxis.range": [0.5, 20.5]}
            ]
        )
    ]

    for d in summary.index:

        buttons.append(
            dict(
                label=f"Decile {int(d)}",
                method="relayout",
                args=[
                    {
                        "xaxis.range": [
                            float(d) - 0.5,
                            float(d) + 0.5
                        ]
                    }
                ]
            )
        )

    # --------------------------------------------------
    # 4. Set axis limits to leave space for labels
    # --------------------------------------------------

    max_value = summary.max().max()
    min_value = summary.min().min()

    if pd.isna(max_value):
        max_value = 0

    if pd.isna(min_value):
        min_value = 0

    max_value = float(max_value)
    min_value = float(min_value)

    # Extra 25% headroom above tallest positive bar
    # and below lowest negative bar.

    upper = (
        max_value * 1.25
        if max_value > 0
        else max(1, abs(min_value) * 0.10)
    )

    lower = (
        min_value * 1.25
        if min_value < 0
        else 0
    )

    # --------------------------------------------------
    # 5. Chart formatting
    # --------------------------------------------------

    fig.update_layout(

        title=dict(
            text=(
                title
                or f"{value_col} by Decile Across Random Seeds"
            ),
            x=0.5,
            xanchor="center",
            font=dict(
                size=22,
                family="Arial Black"
            )
        ),

        xaxis_title="Decile",
        yaxis_title=value_col,

        barmode="group",

        height=height,

        margin=dict(
            t=160,
            b=90,
            l=90,
            r=60
        ),

        font=dict(
            family="Arial",
            size=13
        ),

        legend=dict(
            orientation="h",
            x=0.5,
            xanchor="center",
            y=1.02,
            yanchor="bottom"
        ),

        uniformtext=dict(
            minsize=10,
            mode="show"
        ),

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,

                x=1,
                xanchor="right",

                y=1.24,
                yanchor="top",

                bgcolor="white",
                bordercolor="#CBD5E1",
                borderwidth=1,

                font=dict(
                    size=13
                )
            )
        ],

        plot_bgcolor="white",
        paper_bgcolor="white",

        bargap=0.20,
        bargroupgap=0.05
    )

    # --------------------------------------------------
    # 6. X-axis formatting
    # --------------------------------------------------

    fig.update_xaxes(
        tickmode="array",
        tickvals=summary.index.tolist(),
        type="linear",
        range=[0.5, 20.5],

        showgrid=False,

        tickfont=dict(
            size=12
        )
    )

    # --------------------------------------------------
    # 7. Y-axis formatting
    # --------------------------------------------------

    fig.update_yaxes(
        range=[lower, upper],
        automargin=True,
        gridcolor="#E5E7EB",
        zeroline=True,
        zerolinecolor="#94A3B8"
    )

    # --------------------------------------------------
    # 8. Display chart
    # --------------------------------------------------

    fig.show()

    return summary
