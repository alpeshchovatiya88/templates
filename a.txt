
import math
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.ticker import FuncFormatter

# =====================================================
# 1. DATAFRAMES
# =====================================================

dfs = {
    "RS42": df_rs42,
    "RS532": df_rs532,
    "RS3458": df_rs3458,
    "RS499": df_rs499,
    "RS345": df_rs345,
    "RS1999": df_rs1999,
    "RS2324": df_rs2324,
}

SEEDS = list(dfs.keys())

# =====================================================
# 2. IMPROVED COLOR PALETTE
# =====================================================

colors = {
    "RS42":   "#355C7D",  # Steel blue
    "RS532":  "#6C9A8B",  # Sage teal
    "RS3458": "#8172B3",  # Muted violet
    "RS499":  "#DD9A59",  # Soft orange
    "RS345":  "#C76D7E",  # Dusty rose
    "RS1999": "#7A9E65",  # Olive green
    "RS2324": "#5B8DB8",  # Cornflower blue
}

# =====================================================
# 3. CONFIGURATION
# =====================================================

DECILE_COL = "rr_model_decile"

# Change to 1 if SMI3_LR is already stored as %
LR_SCALE = 100

N_COLS = 4

# Font sizes
VALUE_LABEL_SIZE = 14
SEED_LABEL_SIZE = 12
VENTILE_TITLE_SIZE = 17
MAIN_TITLE_SIZE = 24

# Figure settings
SUBPLOT_WIDTH = 8
SUBPLOT_HEIGHT = 5.5
DPI = 300

output_dir = Path("seed_sensitivity_plots")
output_dir.mkdir(parents=True, exist_ok=True)

# =====================================================
# 4. PREPARE DATA
# =====================================================

prepared = {}

for seed, df in dfs.items():

    temp = df.copy()

    # Extract numeric decile/ventile
    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL]
        .astype(str)
        .str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    for metric in ["SMI3_LOSS", "SMI3_LR"]:

        temp[metric] = pd.to_numeric(
            temp[metric],
            errors="coerce"
        ).replace(
            [np.inf, -np.inf],
            np.nan
        )

    temp = temp.dropna(subset=["_decile"])
    temp["_decile"] = temp["_decile"].astype(int)

    prepared[seed] = temp

# Automatically detect available groups
deciles = sorted(
    set().union(
        *(set(df["_decile"].unique()) for df in prepared.values())
    )
)

if not deciles:
    raise ValueError(
        "No valid deciles found in rr_model_decile."
    )

N_ROWS = math.ceil(len(deciles) / N_COLS)

# =====================================================
# 5. PLOTTING FUNCTION
# =====================================================

def plot_metric(metric, aggregation, scale=1):

    fig, axes = plt.subplots(
        N_ROWS,
        N_COLS,
        figsize=(
            N_COLS * SUBPLOT_WIDTH,
            N_ROWS * SUBPLOT_HEIGHT
        ),
        squeeze=False,
        facecolor="white"
    )

    axes = axes.flatten()

    # -----------------------------------------
    # Aggregate values for each seed
    # -----------------------------------------

    aggregated = {}

    for seed, df in prepared.items():

        grouped = df.groupby("_decile")[metric]

        if aggregation == "sum":
            values = grouped.sum(min_count=1)
        elif aggregation == "mean":
            values = grouped.mean()
        else:
            raise ValueError(
                "aggregation must be 'sum' or 'mean'"
            )

        aggregated[seed] = (
            values.reindex(deciles) * scale
        )

    # -----------------------------------------
    # Create one subplot per decile
    # -----------------------------------------

    for i, decile in enumerate(deciles):

        ax = axes[i]

        values = [
            aggregated[seed].loc[decile]
            for seed in SEEDS
        ]

        x = np.arange(len(SEEDS))

        # -------------------------------------
        # Bars with improved colors
        # -------------------------------------

        bars = ax.bar(
            x,
            values,
            color=[colors[seed] for seed in SEEDS],
            width=0.70,
            edgecolor="white",
            linewidth=1.2,
            zorder=3
        )

        # -------------------------------------
        # LARGE BOLD VALUE LABELS
        # -------------------------------------

        for bar, value in zip(bars, values):

            if pd.isna(value):
                continue

            if metric == "SMI3_LOSS":
                label = f"{value / 1_000_000:.2f}M"
            else:
                label = f"{value:.2f}%"

            ax.annotate(
                label,
                xy=(
                    bar.get_x() + bar.get_width() / 2,
                    bar.get_height()
                ),
                xytext=(0, 7),
                textcoords="offset points",
                ha="center",
                va="bottom",
                fontsize=VALUE_LABEL_SIZE,
                fontweight="bold",
                color="#263238",
                clip_on=False
            )

        # -------------------------------------
        # BOLD VENTILE TITLE
        # -------------------------------------

        ax.set_title(
            f"VENTILE: {decile}",
            fontsize=VENTILE_TITLE_SIZE,
            fontweight="bold",
            color="#1F2937",
            pad=20
        )

        # -------------------------------------
        # BOLD SEED LABELS
        # -------------------------------------

        ax.set_xticks(x)

        ax.set_xticklabels(
            SEEDS,
            rotation=45,
            ha="right",
            fontsize=SEED_LABEL_SIZE,
            fontweight="bold",
            color="#374151"
        )

        # -------------------------------------
        # Y-AXIS FORMATTING
        # -------------------------------------

        ax.tick_params(
            axis="y",
            labelsize=11,
            colors="#4B5563"
        )

        for tick in ax.get_yticklabels():
            tick.set_fontweight("bold")

        if metric == "SMI3_LOSS":

            ax.yaxis.set_major_formatter(
                FuncFormatter(
                    lambda value, pos:
                    f"{value / 1_000_000:.1f}M"
                )
            )

        else:

            ax.yaxis.set_major_formatter(
                FuncFormatter(
                    lambda value, pos:
                    f"{value:.0f}%"
                )
            )

        # -------------------------------------
        # SOFT GRIDLINES
        # -------------------------------------

        ax.grid(
            axis="y",
            color="#E5E7EB",
            linestyle="-",
            linewidth=0.8,
            zorder=0
        )

        ax.set_axisbelow(True)

        # -------------------------------------
        # ADD HEADROOM ABOVE LABELS
        # -------------------------------------

        valid_values = [
            float(v)
            for v in values
            if pd.notna(v)
        ]

        if valid_values and min(valid_values) >= 0:

            ymax = max(valid_values)

            ax.set_ylim(
                0,
                ymax * 1.35 if ymax > 0 else 1
            )

        # -------------------------------------
        # CLEAN SUBPLOT STYLE
        # -------------------------------------

        ax.set_facecolor("white")

        ax.spines["top"].set_visible(False)
        ax.spines["right"].set_visible(False)

        ax.spines["left"].set_color("#D1D5DB")
        ax.spines["bottom"].set_color("#D1D5DB")

        ax.tick_params(
            axis="x",
            length=0,
            pad=7
        )

    # Hide unused subplot positions
    for j in range(len(deciles), len(axes)):
        axes[j].axis("off")

    # -----------------------------------------
    # MAIN TITLE
    # -----------------------------------------

    fig.suptitle(
        f"Random Seed Sensitivity - {metric} by Ventile",
        fontsize=MAIN_TITLE_SIZE,
        fontweight="bold",
        color="#111827",
        y=0.995
    )

    # Spacing between subplots
    fig.tight_layout(
        rect=[0.01, 0.01, 0.99, 0.97],
        h_pad=4.0,
        w_pad=2.5
    )

    # -----------------------------------------
    # SAVE PNG
    # -----------------------------------------

    png_path = (
        output_dir /
        f"{metric}_by_decile.png"
    )

    fig.savefig(
        png_path,
        dpi=DPI,
        bbox_inches="tight",
        facecolor="white"
    )

    print(f"Saved: {png_path.resolve()}")

    plt.show()

    return fig


# =====================================================
# 6. SMI3_LOSS: SUM BY DECILE
# =====================================================

fig_loss = plot_metric(
    metric="SMI3_LOSS",
    aggregation="sum"
)


# =====================================================
# 7. SMI3_LR: MEAN BY DECILE
# =====================================================

fig_lr = plot_metric(
    metric="SMI3_LR",
    aggregation="mean",
    scale=LR_SCALE
)
