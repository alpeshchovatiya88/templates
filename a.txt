import pandas as pd
import plotly.graph_objects as go


def create_swapset_html(
    df,
    filename="swapset_by_region.html"
):

    # ---------------------------------------
    # 1. PREPARE DATA
    # ---------------------------------------

    df = df.copy()

    # Extract numeric deciles if necessary
    for source, target in [
        ("rr_model_decile_xgb", "decile_xgb_num"),
        ("rr_model_decile_cat", "decile_cat_num")
    ]:
        if target not in df.columns:
            df[target] = (
                df[source]
                .astype("string")
                .str.extract(r"(\d+)", expand=False)
                .astype("Int64")
            )

    # Remove rows with missing deciles
    df = df.dropna(
        subset=["decile_xgb_num", "decile_cat_num"]
    )

    # Get economic regions
    regions = sorted(
        df["ECON_REGION"].dropna().unique(),
        key=str
    )

    # Decile range
    deciles = list(range(1, 21))

    # ---------------------------------------
    # 2. FUNCTION TO CREATE SWAPSET
    # ---------------------------------------

    def make_swapset(data):

        swap = pd.crosstab(
            data["decile_xgb_num"],
            data["decile_cat_num"]
        )

        # Ensure all 20 deciles are present
        swap = swap.reindex(
            index=deciles,
            columns=deciles,
            fill_value=0
        )

        # Add row totals
        swap["All"] = swap.sum(axis=1)

        # Add column totals
        swap.loc["All"] = swap.sum(axis=0)

        return swap

    # ---------------------------------------
    # 3. FORMAT COUNTS, % AND COLORS
    # ---------------------------------------

    def format_swapset(swap):

        formatted = swap.copy().astype(str)

        cell_colors = []

        grand_total = swap.loc["All", "All"]

        # Iterate column by column for Plotly
        for col in swap.columns:

            column_colors = []

            for row in swap.index:

                count = swap.loc[row, col]

                # Calculate row percentage
                if row == "All":

                    pct = (
                        count / grand_total * 100
                        if grand_total > 0 else 0
                    )

                else:

                    row_total = swap.loc[row, "All"]

                    pct = (
                        count / row_total * 100
                        if row_total > 0 else 0
                    )

                # Format count + percentage
                formatted.loc[row, col] = (
                    f"{count:,} ({pct:.2f}%)"
                )

                # Determine cell color
                if row == "All" or col == "All":

                    color = "#E6EAF0"  # Gray

                elif count == 0:

                    color = "#FFFFFF"  # White

                elif row == col:

                    color = "#D5F0DB"  # Green

                elif abs(int(row) - int(col)) <= 2:

                    color = "#FFE4B5"  # Orange

                else:

                    color = "#F8C4C4"  # Red

                column_colors.append(color)

            cell_colors.append(column_colors)

        # Index column colors
        index_colors = [
            "#E6EAF0"
        ] * len(swap.index)

        cell_colors = [index_colors] + cell_colors

        return formatted, cell_colors

    # ---------------------------------------
    # 4. FUNCTION TO ADD PLOTLY TABLE
    # ---------------------------------------

    fig = go.Figure()

    def add_swapset_table(data, visible=False):

        swap = make_swapset(data)

        swap_display, cell_colors = format_swapset(
            swap
        )

        # Table values
        table_values = [
            swap_display.index.tolist()
        ] + [
            swap_display[col].tolist()
            for col in swap_display.columns
        ]

        # Header labels
        headers = (
            ["XGB"]
            + [str(col) for col in swap_display.columns]
        )

        fig.add_trace(
            go.Table(

                header=dict(
                    values=headers,
                    fill_color="#4472C4",
                    font=dict(
                        color="white",
                        size=12
                    ),
                    align="center",
                    height=35
                ),

                cells=dict(
                    values=table_values,
                    fill_color=cell_colors,
                    font=dict(
                        color="#202020",
                        size=11
                    ),
                    align="center",
                    height=30
                ),

                columnwidth=[65] + [110] * 21,

                visible=visible
            )
        )

    # ---------------------------------------
    # 5. ALL REGIONS
    # ---------------------------------------

    add_swapset_table(
        df,
        visible=True
    )

    # ---------------------------------------
    # 6. EACH ECONOMIC REGION
    # ---------------------------------------

    for region in regions:

        temp = df[
            df["ECON_REGION"] == region
        ]

        add_swapset_table(
            temp,
            visible=False
        )

    # ---------------------------------------
    # 7. CREATE DROPDOWN
    # ---------------------------------------

    buttons = []

    # ALL regions button
    buttons.append(
        dict(
            label="ALL",
            method="update",
            args=[
                {
                    "visible":
                    [True] + [False] * len(regions)
                },
                {
                    "title.text":
                    "Swapset: XGBoost Model vs CatBoost Model - ALL"
                }
            ]
        )
    )

    # Individual region buttons
    for i, region in enumerate(regions):

        visible = [False] * (
            len(regions) + 1
        )

        visible[i + 1] = True

        buttons.append(
            dict(
                label=str(region),
                method="update",
                args=[
                    {
                        "visible": visible
                    },
                    {
                        "title.text":
                        f"Swapset: XGBoost Model vs CatBoost Model - {region}"
                    }
                ]
            )
        )

    # ---------------------------------------
    # 8. LAYOUT
    # ---------------------------------------

    fig.update_layout(

        title=dict(
            text=(
                "Swapset: XGBoost Model "
                "vs CatBoost Model - ALL"
            ),
            x=0.5,
            xanchor="center",
            y=0.96,
            yanchor="top"
        ),

        height=950,

        margin=dict(
            l=20,
            r=20,
            t=150,
            b=40
        ),

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,
                x=0.01,
                xanchor="left",
                y=1.08,
                yanchor="top"
            )
        ],

        annotations=[
            dict(
                text=(
                    "Green: Same decile | "
                    "Orange: 1-2 deciles movement | "
                    "Red: 3+ deciles movement | "
                    "White: Zero observations | "
                    "Percentage: XGB row total"
                ),
                x=0.5,
                y=1.02,
                xref="paper",
                yref="paper",
                showarrow=False,
                font=dict(size=12)
            )
        ]
    )

    # ---------------------------------------
    # 9. SAVE INTERACTIVE HTML
    # ---------------------------------------

    fig.write_html(
        filename,
        include_plotlyjs=True
    )

    # ---------------------------------------
    # 10. DISPLAY
    # ---------------------------------------

    fig.show()

    return fig


# ===========================================
# RUN THE FUNCTION
# ===========================================

fig = create_swapset_html(
    df,
    filename="swapset_by_region.html"
)
