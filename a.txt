# Keep only bad accounts
bad = df[df["target"] == 1].copy()

results = []

# Total bad CRA_COLLECTION count
total_collection = (bad["CRA_COLLECTION"] == "Y").sum()

features = [
    "CRA_BANKRUPTCY_FORECLOSURE_IND",
    "CRA_JUDGEMENT",
    "CRD_PUBLIC_DEROG_TU"
]

for col in features:

    # Total bad count of each feature
    if col == "CRD_PUBLIC_DEROG_TU":

        total_feature = (bad[col] > 0).sum()

        overlap = (
            (bad["CRA_COLLECTION"] == "Y")
            & (bad[col] > 0)
        ).sum()

    else:

        total_feature = (bad[col] == "Y").sum()

        overlap = (
            (bad["CRA_COLLECTION"] == "Y")
            & (bad[col] == "Y")
        ).sum()

    overlap_pct = round(
        overlap / total_collection * 100,
        1
    )

    results.append({
        "Feature": col,
        "Total_Bad_Count": total_feature,
        "Overlap_With_CRA_COLLECTION": overlap,
        "Overlap_%": overlap_pct
    })

results = pd.DataFrame(results)

print("Total bad CRA_COLLECTION =", total_collection)
