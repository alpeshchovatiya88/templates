
# ============================================================
# VALIDATION ONLY: ADJUSTED PD, LOSS, LR AND D1-D20
# ============================================================

import numpy as np
import pandas as pd

odds_factor = 34.05

# 1. Adjust original and calibrated PD
def adjust_pd(p):
    p = pd.to_numeric(p, errors="coerce")
    return p / (p + odds_factor * (1 - p))


df_val["PD_ORIGINAL_adjusted"] = adjust_pd(
    df_val["PD_ORIGINAL"]
)

df_val["PD_CALIBRATED_adjusted"] = adjust_pd(
    df_val["PD_CALIBRATED"]
)


# 2. Calibrated loss and loss ratio
lgd = pd.to_numeric(df_val["lgd"], errors="coerce")

premium = pd.to_numeric(
    df_val["premium"], errors="coerce"
).replace(0, np.nan)

df_val["SMI3_LOSS_calibrated"] = (
    lgd * df_val["PD_CALIBRATED_adjusted"] * 0.62
)

df_val["SMI3_L-R_calibrated"] = (
    df_val["SMI3_LOSS_calibrated"] / premium
) * 100


# 3. D1-D20 decile function
# D1 = highest loss ratio; D20 = lowest

def create_20_deciles(scores):
    scores = pd.to_numeric(scores, errors="coerce")

    result = pd.Series(
        pd.NA, index=scores.index, dtype="string"
    )

    valid = scores.notna()

    if valid.any():
        ranks = scores.loc[valid].rank(
            method="first", ascending=False
        )

        groups = pd.qcut(
            ranks,
            q=min(20, int(valid.sum())),
            labels=False
        ) + 1

        result.loc[valid] = (
            "D" + groups.astype(str)
        )

    return result


# 4. Original adjusted loss ratio for ranking
original_lr_adjusted = (
    lgd * df_val["PD_ORIGINAL_adjusted"] * 0.62
    / premium
) * 100

df_val["decile_original"] = create_20_deciles(
    original_lr_adjusted
)

df_val["decile_calibrated"] = create_20_deciles(
    df_val["SMI3_L-R_calibrated"]
)


# 5. Compare with existing production deciles
display_cols = [
    "PD_ORIGINAL_adjusted",
    "PD_CALIBRATED_adjusted",
    "SMI3_LOSS_calibrated",
    "SMI3_L-R_calibrated",
    "decile_original",
    "decile_calibrated"
]

if "rr_model_decile" in df_val.columns:
    display_cols.insert(4, "rr_model_decile")

display(df_val[display_cols].head(20))


# 6. Save validation file
df_val.to_csv(
    "df_val_calibrated.csv",
    index=False
)

print("Validation calibrated columns saved.")
