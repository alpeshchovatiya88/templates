import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import math


def plot_feature_by_bookyear(
    df,
    feature,
    bad_col="BAD",
    year_col="BOOKYEAR",
    start_year=2009,
    end_year=2025,
    ncols=3,
    figsize_per_plot=(6, 5),
    include_missing=False
):
    """
    Plot bad rate by BOOKYEAR for any categorical feature.

    Bar height = Bad Rate %
    Bar width  = Total observations
    Bar color  = Feature category

    Parameters
    ----------
    df : pandas DataFrame
    feature : str
        Feature/column to plot.
    bad_col : str
        Binary target column where 1 = Bad, 0 = Good.
    year_col : str
        Year column.
    start_year : int
    end_year : int
    ncols : int
        Number of subplot columns.
    figsize_per_plot : tuple
        Width and height per subplot.
    include_missing : bool
        Whether to include missing feature values as "Missing".

    Returns
    -------
    summary : pandas DataFrame
        Table used to create the plots.
    """

    # ---------------------------------------------
    # Copy required columns
    # ---------------------------------------------
    data = df[
        [year_col, feature, bad_col]
    ].copy()

    # Optional missing-value category
    if include_missing:
        data[feature] = data[feature].fillna("Missing")
    else:
        data = data[data[feature].notna()]

    # ---------------------------------------------
    # Years
    # ---------------------------------------------
    years = list(
        range(start_year, end_year + 1)
    )

    # ---------------------------------------------
    # Summary function
    # ---------------------------------------------
    def create_summary(temp_data, label):

        result = (
            temp_data
            .groupby(feature, dropna=False)
            .agg(
                T=(bad_col, "size"),
                B=(bad_col, "sum")
            )
            .reset_index()
        )

        result["Bad_Rate"] = (
            result["B"] / result["T"]
        ) * 100

        result["BOOKYEAR_LABEL"] = str(label)

        return result

    # ---------------------------------------------
    # Overall summary
    # ---------------------------------------------
    summary_list = [
        create_summary(data, "overall")
    ]

    # ---------------------------------------------
    # Year summaries
    # ---------------------------------------------
    for year in years:

        temp_year = data[
            data[year_col] == year
        ]

        if not temp_year.empty:
            summary_list.append(
                create_summary(
                    temp_year,
                    year
                )
            )

    summary = pd.concat(
        summary_list,
        ignore_index=True
    )

    # ---------------------------------------------
    # Feature categories
    # ---------------------------------------------
    categories = sorted(
        data[feature]
        .dropna()
        .unique(),
        key=lambda x: str(x)
    )

    # ---------------------------------------------
    # Consistent colors
    # ---------------------------------------------
    colors = sns.color_palette(
        "Set2",
        n_colors=len(categories)
    )

    color_dict = dict(
        zip(categories, colors)
    )

    # ---------------------------------------------
    # Plot order
    # ---------------------------------------------
    plot_order = (
        ["overall"] +
        [str(y) for y in years]
    )

    nplots = len(plot_order)

    nrows = math.ceil(
        nplots / ncols
    )

    fig_width = (
        figsize_per_plot[0] * ncols
    )

    fig_height = (
        figsize_per_plot[1] * nrows
    )

    fig, axes = plt.subplots(
        nrows=nrows,
        ncols=ncols,
        figsize=(fig_width, fig_height)
    )

    axes = np.array(
        axes
    ).reshape(-1)

    # ---------------------------------------------
    # Create each subplot
    # ---------------------------------------------
    for i, year_label in enumerate(plot_order):

        ax = axes[i]

        temp = summary[
            summary["BOOKYEAR_LABEL"]
            == year_label
        ].copy()

        # No data
        if temp.empty:

            ax.set_title(
                f"{year_col}: {year_label}",
                fontweight="bold"
            )

            ax.text(
                0.5,
                0.5,
                "No data",
                ha="center",
                va="center",
                transform=ax.transAxes
            )

            ax.axis("off")

            continue

        # -----------------------------------------
        # Same categories/order in every plot
        # -----------------------------------------
        temp = (
            temp
            .set_index(feature)
            .reindex(categories)
            .reset_index()
        )

        x = np.arange(
            len(categories)
        )

        # -----------------------------------------
        # Bar width based on total
        # -----------------------------------------
        totals = temp["T"].fillna(0)

        if totals.max() > 0:

            # sqrt scaling stops small groups
            # becoming invisible
            widths = (
                0.25
                +
                0.65
                * np.sqrt(
                    totals / totals.max()
                )
            )

        else:

            widths = np.repeat(
                0.7,
                len(categories)
            )

        # -----------------------------------------
        # Plot bars
        # -----------------------------------------
        bars = ax.bar(
            x,
            temp["Bad_Rate"].fillna(0),
            width=widths,
            color=[
                color_dict[c]
                for c in categories
            ]
        )

        # -----------------------------------------
        # Labels / title
        # -----------------------------------------
        ax.set_title(
            f"{year_col}: {year_label}",
            fontsize=13,
            fontweight="bold",
            pad=12
        )

        ax.set_xlabel(feature)

        ax.set_ylabel(
            "Bad Rate (%)"
        )

        ax.set_xticks(x)

        ax.set_xticklabels(
            categories,
            rotation=45,
            ha="right",
            fontsize=9
        )

        # Grid
        ax.grid(
            axis="y",
            linestyle="--",
            alpha=0.3
        )

        ax.set_axisbelow(True)

        # -----------------------------------------
        # Y range
        # -----------------------------------------
        max_rate = temp[
            "Bad_Rate"
        ].max(skipna=True)

        if (
            pd.isna(max_rate)
            or max_rate == 0
        ):
            max_rate = 1

        ax.set_ylim(
            0,
            max_rate * 1.35
        )

        # -----------------------------------------
        # T / B / Bad Rate labels
        # -----------------------------------------
        for bar, (_, row) in zip(
            bars,
            temp.iterrows()
        ):

            if pd.isna(row["T"]):
                continue

            total = int(row["T"])
            bad = int(row["B"])
            rate = row["Bad_Rate"]

            text = (
                f"T: {total:,}\n"
                f"B: {bad:,}\n"
                f"{rate:.1f}%"
            )

            ax.text(
                bar.get_x()
                + bar.get_width() / 2,
                rate + max_rate * 0.025,
                text,
                ha="center",
                va="bottom",
                fontsize=8
            )

    # ---------------------------------------------
    # Remove unused panels
    # ---------------------------------------------
    for j in range(
        nplots,
        len(axes)
    ):
        fig.delaxes(
            axes[j]
        )

    # ---------------------------------------------
    # Overall title
    # ---------------------------------------------
    fig.suptitle(
        f"{feature} Bad Rate by {year_col}\n"
        "Bar Height = Bad Rate | Bar Width = Total",
        fontsize=18,
        fontweight="bold",
        y=1.002
    )

    plt.tight_layout(
        h_pad=3,
        w_pad=2
    )

    plt.show()

    return summary
