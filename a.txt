# SECTION 3 — LOAD, CHECK AND AUDIT DATA
from pathlib import Path
from itertools import combinations
import numpy as np
import pandas as pd

frames, evaluation, data_audit, overlap = {}, {}, [], []

for name, path in FILES.items():
    df = pd.read_csv(
        Path(path), dtype={ID_COLUMN: "string"}, low_memory=False
    )
    raw_rows = len(df)

    if POPULATION_FILTERS.get(name):
        df = df.query(POPULATION_FILTERS[name], engine="python").copy()
    df = df.reset_index(drop=True)

    required = [ID_COLUMN, SCORE_COLUMNS[name]]
    required += [GROUP_COLUMN] if GROUP_COLUMN else []
    required += [TARGET] if MATURE[name] else []
    absent = [c for c in required if c not in df.columns]
    if absent:
        raise ValueError(f"{name}: missing columns {absent}")
    if df.empty:
        raise ValueError(f"{name}: no eligible rows")

    for column in [ID_COLUMN] + ([GROUP_COLUMN] if GROUP_COLUMN else []):
        values = df[column].astype("string")
        if (values.isna() | values.str.strip().eq("")).any():
            raise ValueError(f"{name}: missing/blank {column}")
    if df[ID_COLUMN].duplicated().any():
        raise ValueError(f"{name}: duplicate application IDs")

    df["_p"] = pd.to_numeric(
        df[SCORE_COLUMNS[name]], errors="coerce"
    ).to_numpy(dtype=float, na_value=np.nan)
    if not (np.isfinite(df["_p"]) & df["_p"].between(0, 1)).all():
        raise ValueError(f"{name}: invalid probabilities")

    df = df.drop(columns="_y", errors="ignore")
    labeled_rows = missing_targets = 0

    if MATURE[name]:
        text = df[TARGET].astype("string").str.strip()
        missing = text.isna() | text.eq("").fillna(False)
        y = pd.to_numeric(text, errors="coerce")
        valid = y.isin([0, 1])

        if ((~missing) & (~valid)).any():
            raise ValueError(f"{name}: unexpected target codes")
        if missing.any() and name != "test":
            raise ValueError(f"{name}: missing mature outcomes")
        if not valid.any():
            raise ValueError(f"{name}: no labeled outcomes")

        df["_y"] = y.to_numpy(dtype=float, na_value=np.nan)
        evaluation[name] = df.loc[valid].copy()
        evaluation[name]["_y"] = evaluation[name]["_y"].astype(int)
        labeled_rows = int(valid.sum())
        missing_targets = int(missing.sum())

        if missing_targets:
            record_status(
                f"{name}_target_completeness", "REVIEW",
                "Exploratory labeled-subset evaluation; "
                "maturity and missing-outcome cause unconfirmed."
            )

    frames[name] = df
    data_audit.append({
        "cohort": name, "raw_rows": raw_rows,
        "eligible_rows": len(df), "mature": MATURE[name],
        "evaluated_rows": labeled_rows,
        "missing_targets": missing_targets if MATURE[name] else np.nan,
    })

show_table(pd.DataFrame(data_audit), "01_population_audit")
show_table(pd.DataFrame({
    name: df.drop(columns=["_p", "_y"], errors="ignore").isna().mean()
    for name, df in frames.items()
}), "01_missingness")

for first, second in combinations(frames, 2):
    for column in dict.fromkeys(
        [ID_COLUMN] + ([GROUP_COLUMN] if GROUP_COLUMN else [])
    ):
        shared = len(
            set(frames[first][column].astype("string"))
            & set(frames[second][column].astype("string"))
        )
        overlap.append({
            "first": first, "second": second,
            "identifier": column, "shared_identifiers": shared,
        })

show_table(pd.DataFrame(overlap), "01_overlap")
if any(row["shared_identifiers"] for row in overlap):
    record_status("population_overlap", "REVIEW", "Shared IDs detected.")

record_status("data_controls", "COMPLETED", "Loading checks completed.")
