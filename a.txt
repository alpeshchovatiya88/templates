def adjust_probability(probability, factor=34.05):

    p = pd.to_numeric(
        probability,
        errors="coerce"
    )

    return p / (
        p + factor * (1 - p)
    )

# 8B. CREATE ORIGINAL AND CALIBRATED D1-D20
# ============================================================

# decile_original:
# Recreated original ranking for verification.
#
# rr_model_decile_calibrated:
# New calibrated ranking for swapset.
#
# Keep existing rr_model_decile unchanged.
#
# Assumed method:
# 20 approximately equal-count groups,
# D1 = highest score.
#
# Exact match to original ranking is not
# guaranteed until the production ranking
# logic has been verified.


def create_20_deciles(scores):

    numeric = pd.to_numeric(
        scores,
        errors="coerce"
    )

    output = pd.Series(
        pd.NA,
        index=numeric.index,
        dtype="string"
    )

    valid = numeric.notna()

    if not valid.any():
        return output

    # High risk = D1
    ranks = numeric.loc[valid].rank(
        method="first",
        ascending=False
    )

    n_groups = min(20, int(valid.sum()))

    group_numbers = (
        pd.qcut(
            ranks,
            q=n_groups,
            labels=False
        ) + 1
    )

    output.loc[valid] = (
        "D"
        + group_numbers.astype(int).astype(str)
    )

    return output

# COMPARE RECREATED DECILE AGAINST EXISTING
# ------------------------------------------------------------

def normalize_decile(series):

    values = (
        series.astype("string")
        .str.strip()
        .str.upper()
        .str.replace(
            r"^D",
            "",
            regex=True
        )
    )

    return pd.to_numeric(
        values,
        errors="coerce"
    ).astype("Int64")

        continue

    existing = normalize_decile(

    )

    recreated = normalize_decile(
        df["decile_original"]
    )

    valid = existing.notna() & recreated.notna()

    if valid.any():

        matches = (

# Classification threshold for Precision,
# Recall and FPR = 0.50

def evaluate(df, name, probability_col):

    # Exclude missing targets and scores
    data = df[
        df[target_col].isin([0, 1])
        & df[probability_col].notna()
    ].copy()

    if data.empty:
        print(f"{name}: No observed targets.")
        return None

    y = data[target_col].astype(int).to_numpy()

    probability = data[
        probability_col
    ].to_numpy(dtype=float)

    y_pred = (
        probability >= threshold
    ).astype(int)

    tn, fp, fn, tp = confusion_matrix(
        y,
        y_pred,
        labels=[0, 1]
    ).ravel()

    precision = precision_score(
        y,
        y_pred,
        zero_division=0
    )

    recall = recall_score(
        y,
        y_pred,
        zero_division=0
    )

    fpr = (
        fp / (fp + tn)
        if (fp + tn) > 0
        else np.nan
    )

    if len(np.unique(y)) == 2:

        auc = roc_auc_score(
            y,
            probability
        )

        fpr_curve, tpr_curve, _ = roc_curve(
            y,
            probability
        )

        ks = np.max(
            tpr_curve - fpr_curve
        )

    else:

        auc = np.nan
        ks = np.nan

    return {
        "Dataset": name,
        "N_Observed": len(data),
        "N_Bad_Events": int(y.sum()),
        "Actual_Bad_Rate": y.mean(),
        "Mean_PD": probability.mean(),
        "AUC": auc,
        "KS": ks,
        "Precision": precision,
        "Recall": recall,
        "FPR": fpr,
        "Brier_Score": brier_score_loss(
            y,
            probability
        ),
        "Log_Loss": log_loss(
            y,
            probability,
            labels=[0, 1]
        )
    }

# Same plotting method as your original code.
# Only the probability columns have changed.

def plot_calibration(df, name):

    data = df[
        df[target_col].isin([0, 1])
        & df["PD_BASELINE"].notna()
        & df["PD_CALIBRATED"].notna()
    ].copy()

    if data.empty or data[target_col].nunique() < 2:
        print(f"{name}: Insufficient target data.")
        return

    y = data[target_col].astype(int)

    plt.figure(figsize=(8, 6))

    for col, label in [
        ("PD_BASELINE", "Original XGBoost"),
        ("PD_CALIBRATED", "Calibrated XGBoost")
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            data[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
    )

    plt.xlabel("Mean Predicted Probability")
    plt.ylabel("Observed Bad Rate")
    plt.title(name + " Calibration Curve")
    plt.legend()
    plt.grid(True)
    plt.show()


plot_calibration(df_train, "TRAIN")

plot_calibration(df_val, "VALIDATION")
