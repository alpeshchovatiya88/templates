import numpy as np
import pandas as pd
from sklearn.metrics import (
    roc_auc_score, average_precision_score, brier_score_loss, log_loss,
    accuracy_score, precision_score, recall_score, f1_score,
    mean_absolute_error, mean_squared_error, r2_score,
)

TRAIN_PATH = "df_scored_train.csv"
TEST_PATH = "df_scored_test.csv"
VAL_PATH = "df_scored_val.csv"

TARGET_COL = "TARGET"
SCORE_COL = "SCORE"
TASK = "classification"          # or "regression"
THRESHOLD = 0.5
ID_COLS = ["APPLICATION_NUMBER", "UAID", "UUAID", "Date"]
PROTECTED_COLS = ["PROVINCE"]
DATE_COL = "Date"

train = pd.read_csv(TRAIN_PATH)
test = pd.read_csv(TEST_PATH)
val = pd.read_csv(VAL_PATH)
frames = {
    "train_2009_2015": train,
    "test_2021_2023": test,
    "val_202510_202605": val,
}

def y_true(df):
    return df[TARGET_COL].astype(float).to_numpy()

def y_score(df):
    return df[SCORE_COL].astype(float).to_numpy()

def y_pred(df):
    if TASK == "classification":
        return (y_score(df) >= THRESHOLD).astype(int)
    return y_score(df)

def feature_cols(df):
    drop = {TARGET_COL, SCORE_COL, *ID_COLS}
    return [c for c in df.columns if c not in drop and pd.api.types.is_numeric_dtype(df[c])]

def finite(df):
    m = np.isfinite(y_true(df)) & np.isfinite(y_score(df))
    return y_true(df)[m], y_score(df)[m], y_pred(df)[m]

def ks_stat(y, s):
    order = np.argsort(s)
    y = y[order]
    n1, n0 = y.sum(), len(y) - y.sum()
    if n1 == 0 or n0 == 0:
        return np.nan
    cdf1 = np.cumsum(y) / n1
    cdf0 = np.cumsum(1 - y) / n0
    return float(np.max(np.abs(cdf1 - cdf0)))

def metric(df):
    yt, ys, _ = finite(df)
    if TASK == "classification":
        return {"AUC": roc_auc_score(yt, ys), "Brier": brier_score_loss(yt, np.clip(ys, 1e-6, 1 - 1e-6))}
    return {"MAE": mean_absolute_error(yt, ys)}

def reliability_table(df, n_bins=10):
    yt, ys, _ = finite(df)
    tab = pd.DataFrame({"y": yt, "s": ys, "bin": pd.qcut(ys, n_bins, duplicates="drop")})
    tab = tab.groupby("bin", observed=True).agg(n=("y", "size"), mean_score=("s", "mean"), observed_rate=("y", "mean"))
    tab["gap"] = tab["mean_score"] - tab["observed_rate"]
    return tab

def ece(df):
    tab = reliability_table(df)
    return float(np.average(np.abs(tab["gap"]), weights=tab["n"]))

for name, df in frames.items():
    print(name, "ECE", round(ece(df), 4))
    display(reliability_table(df))

fig, ax = plt.subplots(figsize=(6, 6))

    frac, pred = calibration_curve(yt, ys, n_bins=10, strategy="quantile")
    ax.plot(pred, frac, marker="o", label=name)
ax.plot([0, 1], [0, 1], "--", color="gray")
ax.set_xlabel("Mean score")
ax.set_ylabel("Observed rate")
ax.legend()
plt.show()
