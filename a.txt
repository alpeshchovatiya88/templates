import pandas as pd
import numpy as np

def analyze_decile_movement(
    df,
    base_col,
    compare_col
):

    temp = df[[base_col, compare_col]].copy()

    temp[base_col] = pd.to_numeric(
        temp[base_col], errors="coerce"
    )

    temp[compare_col] = pd.to_numeric(
        temp[compare_col], errors="coerce"
    )

    temp = temp.dropna()

    # Positive = higher numbered decile
    # Negative = lower numbered decile
    temp["movement"] = (
        temp[compare_col] - temp[base_col]
    ).astype(int)

    temp["abs_movement"] = (
        temp["movement"].abs()
    )

    temp["direction"] = np.select(
        [
            temp["movement"] > 0,
            temp["movement"] < 0
        ],
        [
            "Higher Decile",
            "Lower Decile"
        ],
        default="No Movement"
    )

    # Number of files by movement
    summary = (
        temp.groupby(
            ["direction", "abs_movement"]
        )
        .size()
        .reset_index(name="file_count")
    )

    summary["percentage"] = (
        summary["file_count"]
        / len(temp) * 100
    ).round(2)

    return temp, summary
