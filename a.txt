
def check_reproducibility(
    developer_df,
    validator_df,
    dataset_name,
    key_columns,
    target_col="Target",
    probability_col="Predicted_Probability",
    loss_col="Loss",
    loss_ratio_col="Loss_Ratio",
    probability_tolerance=1e-8,
    metric_tolerance=1e-8,
    loss_tolerance=1e-6,
    loss_ratio_tolerance=1e-8,
    total_loss_tolerance=1e-4,
    output_folder=None
):
    """
    Compare Developer and Validator DataFrames.

    Required columns:
        Key(s), Target, Predicted_Probability

    Optional columns:
        Loss, Loss_Ratio

    Dataset metrics:
        AUC, Brier Score, Log Loss

    If output_folder is provided, save Excel and CSV reports.
    Otherwise, return results without writing files.
    """

    # --------------------------------------------------
    # 1. Configuration validation
    # --------------------------------------------------

    if isinstance(key_columns, str):
        key_columns = [key_columns]
    else:
        key_columns = list(key_columns)

    if not key_columns:
        raise ValueError("Provide at least one key column.")

    if not isinstance(developer_df, pd.DataFrame):
        raise TypeError("developer_df must be a DataFrame.")

    if not isinstance(validator_df, pd.DataFrame):
        raise TypeError("validator_df must be a DataFrame.")

    required = key_columns + [target_col, probability_col]

    if len(set(required)) != len(required):
        raise ValueError(
            "Key, target and probability columns must be distinct."
        )

    tolerances = [
        probability_tolerance,
        metric_tolerance,
        loss_tolerance,
        loss_ratio_tolerance,
        total_loss_tolerance
    ]

    if any(
        not np.isfinite(tol) or tol < 0
        for tol in tolerances
    ):
        raise ValueError(
            "Tolerances must be finite and nonnegative."
        )

    # Copies avoid changing original DataFrames.
    dev = developer_df.copy()
    val = validator_df.copy()

    # --------------------------------------------------
    # 2. Validate required columns and data
    # --------------------------------------------------

    for name, df in [
        ("Developer", dev),
        ("Validator", val)
    ]:

        if not df.columns.is_unique:
            raise ValueError(
                f"{name}: duplicate column names."
            )

        missing_cols = [
            col for col in required
            if col not in df.columns
        ]

        if missing_cols:
            raise ValueError(
                f"{name}: missing columns {missing_cols}"
            )

        if df[key_columns].isna().any().any():
            raise ValueError(
                f"{name}: missing record identifiers."
            )

        if df.duplicated(subset=key_columns).any():
            raise ValueError(
                f"{name}: duplicate record identifiers."
            )

        for col in [target_col, probability_col]:

            df[col] = pd.to_numeric(
                df[col],
                errors="raise"
            )

            if df[col].isna().any():
                raise ValueError(
                    f"{name}: missing values in {col}."
                )

            if not np.isfinite(df[col]).all():
                raise ValueError(
                    f"{name}: non-finite values in {col}."
                )

        if not df[target_col].isin([0, 1]).all():
            raise ValueError(
                f"{name}: Target must contain only 0 and 1."
            )

        if not df[probability_col].between(0, 1).all():
            raise ValueError(
                f"{name}: probabilities must be between 0 and 1."
            )

    # --------------------------------------------------
    # 3. Optional Loss and Loss Ratio columns
    # --------------------------------------------------

    optional_metrics = {}

    for metric, col, tolerance in [
        ("Loss", loss_col, loss_tolerance),
        ("Loss_Ratio", loss_ratio_col, loss_ratio_tolerance)
    ]:

        if col is None:
            continue

        in_dev = col in dev.columns
        in_val = col in val.columns

        if in_dev != in_val:
            raise ValueError(
                f"{col} is present in only one DataFrame."
            )

        if in_dev:

            for name, df in [
                ("Developer", dev),
                ("Validator", val)
            ]:

                df[col] = pd.to_numeric(
                    df[col],
                    errors="raise"
                )

                if np.isinf(df[col]).any():
                    raise ValueError(
                        f"{name}: infinite values in {col}."
                    )

            optional_metrics[metric] = (
                col,
                tolerance
            )

    # --------------------------------------------------
    # 4. Match records by unique key
    # --------------------------------------------------

    compare_cols = [
        target_col,
        probability_col
    ] + [
        col for col, _ in optional_metrics.values()
    ]

    selected_cols = list(
        dict.fromkeys(key_columns + compare_cols)
    )

    merged = dev[selected_cols].merge(
        val[selected_cols],
        on=key_columns,
        how="outer",
        suffixes=("_DEV", "_VAL"),
        indicator=True,
        validate="one_to_one"
    )

    missing_records = merged.loc[
        merged["_merge"] != "both",
        key_columns + ["_merge"]
    ].copy()

    missing_records["Issue"] = (
        missing_records["_merge"]
        .astype(str)
        .replace({
            "left_only": "Missing in Validator",
            "right_only": "Extra in Validator"
        })
    )

    missing_records = missing_records[
        key_columns + ["Issue"]
    ].reset_index(drop=True)

    matched = merged.loc[
        merged["_merge"] == "both"
    ].copy()

    # --------------------------------------------------
    # 5. Record-level comparisons
    # --------------------------------------------------

    record_metrics = {
        "Target": (target_col, 0.0),
        "Predicted_Probability": (
            probability_col,
            probability_tolerance
        )
    }

    record_metrics.update(optional_metrics)

    record_rows = []
    mismatch_frames = []

    for metric, (col, tolerance) in record_metrics.items():

        dev_values = matched[f"{col}_DEV"]
        val_values = matched[f"{col}_VAL"]

        absolute_diff = (
            dev_values - val_values
        ).abs()

        passed = (
            dev_values.notna()
            & val_values.notna()
            & (absolute_diff <= tolerance)
        )

        failed = ~passed
        mismatch_count = int(failed.sum())

        record_rows.append({
            "Dataset": dataset_name,
            "Metric": metric,
            "Records_Compared": len(matched),
            "Mismatch_Count": mismatch_count,
            "Mismatch_Percentage": (
                100 * mismatch_count / len(matched)
                if len(matched) else np.nan
            ),
            "Max_Absolute_Difference": absolute_diff.max(),
            "Tolerance": tolerance,
            "Status": (
                "PASS"
                if len(matched) > 0 and mismatch_count == 0
                else "FAIL"
            )
        })

        if mismatch_count:

            detail = matched.loc[
                failed,
                key_columns
            ].copy()

            detail["Dataset"] = dataset_name
            detail["Metric"] = metric
            detail["Developer_Value"] = (
                dev_values.loc[failed]
            )
            detail["Validator_Value"] = (
                val_values.loc[failed]
            )
            detail["Absolute_Difference"] = (
                absolute_diff.loc[failed]
            )
            detail["Tolerance"] = tolerance

            mismatch_frames.append(detail)

    record_summary = pd.DataFrame(record_rows)

    mismatch_columns = key_columns + [
        "Dataset",
        "Metric",
        "Developer_Value",
        "Validator_Value",
        "Absolute_Difference",
        "Tolerance"
    ]

    if mismatch_frames:
        mismatches = pd.concat(
            mismatch_frames,
            ignore_index=True
        )
    else:
        mismatches = pd.DataFrame(
            columns=mismatch_columns
        )

    # --------------------------------------------------
    # 6. Dataset-level performance metrics
    # --------------------------------------------------

    def calculate_performance(df):

        y = df[target_col].to_numpy(dtype=int)
        p = df[probability_col].to_numpy(dtype=float)

        if len(np.unique(y)) != 2:
            raise ValueError(
                f"{dataset_name}: AUC requires both "
                "target classes (0 and 1)."
            )

        return {
            "AUC": roc_auc_score(y, p),
            "Brier_Score": brier_score_loss(y, p),
            "Log_Loss": log_loss(
                y,
                p,
                labels=[0, 1]
            )
        }

    dev_performance = calculate_performance(dev)
    val_performance = calculate_performance(val)

    performance_rows = []

    for metric in dev_performance:

        dev_value = dev_performance[metric]
        val_value = val_performance[metric]

        absolute_diff = abs(dev_value - val_value)

        performance_rows.append({
            "Dataset": dataset_name,
            "Metric": metric,
            "Developer_Value": dev_value,
            "Validator_Value": val_value,
            "Absolute_Difference": absolute_diff,
            "Tolerance": metric_tolerance,
            "Status": (
                "PASS"
                if absolute_diff <= metric_tolerance
                else "FAIL"
            )
        })

    performance_summary = pd.DataFrame(
        performance_rows
    )

    # --------------------------------------------------
    # 7. Aggregate total Loss
    # --------------------------------------------------

    aggregate_rows = []

    if "Loss" in optional_metrics:

        dev_total = dev[loss_col].sum(min_count=1)
        val_total = val[loss_col].sum(min_count=1)

        absolute_diff = abs(dev_total - val_total)

        complete_loss = (
            dev[loss_col].notna().all()
            and val[loss_col].notna().all()
        )

        passed = (
            complete_loss
            and np.isfinite(dev_total)
            and np.isfinite(val_total)
            and absolute_diff <= total_loss_tolerance
        )

        aggregate_rows.append({
            "Dataset": dataset_name,
            "Metric": "Total_Loss",
            "Developer_Value": dev_total,
            "Validator_Value": val_total,
            "Absolute_Difference": absolute_diff,
            "Tolerance": total_loss_tolerance,
            "Status": (
                "PASS" if passed else "FAIL"
            )
        })

    aggregate_summary = pd.DataFrame(
        aggregate_rows,
        columns=[
            "Dataset",
            "Metric",
            "Developer_Value",
            "Validator_Value",
            "Absolute_Difference",
            "Tolerance",
            "Status"
        ]
    )

    # --------------------------------------------------
    # 8. Overall PASS / FAIL
    # --------------------------------------------------

    overall_pass = (
        len(matched) > 0
        and missing_records.empty
        and record_summary["Status"].eq("PASS").all()
        and performance_summary["Status"].eq("PASS").all()
        and (
            aggregate_summary.empty
            or aggregate_summary["Status"].eq("PASS").all()
        )
    )

    overall_status = (
        "PASS" if overall_pass else "FAIL"
    )

    overall_summary = pd.DataFrame([{
        "Dataset": dataset_name,
        "Developer_Rows": len(dev),
        "Validator_Rows": len(val),
        "Matched_Rows": len(matched),
        "Missing_Extra_Rows": len(missing_records),
        "Record_Mismatches": len(mismatches),
        "Overall_Status": overall_status
    }])

    # --------------------------------------------------
    # 9. Optional report export
    # --------------------------------------------------

    report_path = None

    if output_folder is not None:

        output_dir = Path(output_folder)
        output_dir.mkdir(
            parents=True,
            exist_ok=True
        )

        safe_name = "".join(
            c if c.isalnum() or c in "-_" else "_"
            for c in dataset_name
        )

        if not safe_name:
            raise ValueError("Invalid dataset name.")

        report_path = (
            output_dir
            / f"{safe_name}_Reproducibility.xlsx"
        )

        mismatch_path = (
            output_dir
            / f"{safe_name}_Mismatches.csv"
        )

        missing_path = (
            output_dir
            / f"{safe_name}_Missing_Records.csv"
        )

        mismatches.to_csv(
            mismatch_path,
            index=False
        )

        missing_records.to_csv(
            missing_path,
            index=False
        )

        with pd.ExcelWriter(
            report_path,
            engine="openpyxl"
        ) as writer:

            overall_summary.to_excel(
                writer,
                sheet_name="Executive Summary",
                index=False
            )

            record_summary.to_excel(
                writer,
                sheet_name="Record Checks",
                index=False
            )

            performance_summary.to_excel(
                writer,
                sheet_name="Performance Metrics",
                index=False
            )

            aggregate_summary.to_excel(
                writer,
                sheet_name="Aggregate Loss",
                index=False
            )

            mismatches.head(1_048_575).to_excel(
                writer,
                sheet_name="Mismatches",
                index=False
            )

            missing_records.head(1_048_575).to_excel(
                writer,
                sheet_name="Missing Records",
                index=False
            )

    # --------------------------------------------------
    # 10. Display and return
    # --------------------------------------------------

    print(f"\n{'=' * 60}")
    print(f"REPRODUCIBILITY: {dataset_name}")
    print(f"{'=' * 60}")

    print("\nRecord-Level Checks:")
    print(record_summary.to_string(index=False))

    print("\nDataset-Level Metrics:")
    print(performance_summary.to_string(index=False))

    if not aggregate_summary.empty:
        print("\nAggregate Total Loss:")
        print(aggregate_summary.to_string(index=False))

    print(f"\nMissing/Extra Records: {len(missing_records)}")
    print(f"Overall Status: {overall_status}")

    if report_path is not None:
        print(f"Report: {report_path}")

    return {
        "dataset": dataset_name,
        "overall_status": overall_status,
        "overall_summary": overall_summary,
        "record_summary": record_summary,
        "performance_summary": performance_summary,
        "aggregate_summary": aggregate_summary,
        "mismatches": mismatches,
        "missing_records": missing_records,
        "report_path": (
            str(report_path)
            if report_path is not None else None
        )
    }
