def combo_name(row):
    names = []
    if row["COLLECTION_FLAG"] == 1:
        names.append("Collection")
    if row["BANKRUPTCY_FLAG"] == 1:
        names.append("Bankruptcy/Foreclosure")
    if row["JUDGEMENT_FLAG"] == 1:
        names.append("Judgement")
    if row["DEROG_FLAG"] == 1:
        names.append("Public Derog")
    return " + ".join(names)

overlap["Combination"] = overlap.apply(combo_name, axis=1)
