def get_best_model(trainer):
    if hasattr(trainer, "best_model_"):
        return trainer.best_model_
    if hasattr(trainer, "best_model"):
        return trainer.best_model
    raise AttributeError("No best model found.")


def score_validation_set(trainer, experiment_name):
    model = get_best_model(trainer)

    X_valid = trainer._cache["X_test"]
    valid_df = trainer._cache["d_test"].copy()

    valid_df[PROB_COL] = model.predict_proba(X_valid)[:, 1]

    lr_calc = LRCalculator()
    valid_df = lr_calc.transform(
        valid_df,
        {LR_COL: PROB_COL}
    )

    valid_df["experiment"] = experiment_name
    return valid_df


def lr_summary(scored_df):
    actual_loss = scored_df["ACTUAL_LOSS"].sum()
    predicted_loss = scored_df[LOSS_COL].sum()

    return pd.Series({
        "experiment": scored_df["experiment"].iloc[0],
        "n": len(scored_df),
        "mean_lr": scored_df[LR_COL].mean(),
        "median_lr": scored_df[LR_COL].median(),
        "sum_predicted_loss": predicted_loss,
        "sum_actual_loss": actual_loss,
        "diff_predicted_minus_actual": predicted_loss - actual_loss,
        "pct_diff_predicted_vs_actual": (
            (predicted_loss - actual_loss) / actual_loss * 100
            if actual_loss != 0 else np.nan
        ),
    })


def lr_by_bookyear(scored_df):
    out = (
        scored_df
        .groupby(["experiment", YEAR_COL])
        .agg(
            n=(LR_COL, "size"),
            mean_lr=(LR_COL, "mean"),
            median_lr=(LR_COL, "median"),
            sum_predicted_loss=(LOSS_COL, "sum"),
            sum_actual_loss=("ACTUAL_LOSS", "sum"),
        )
        .reset_index()
    )

    out["diff_predicted_minus_actual"] = (
        out["sum_predicted_loss"] - out["sum_actual_loss"]
    )

    out["pct_diff_predicted_vs_actual"] = np.where(
        out["sum_actual_loss"] != 0,
        out["diff_predicted_minus_actual"] / out["sum_actual_loss"] * 100,
        np.nan,
    )

    return out

from sklearn.metrics import roc_auc_score

def get_train_test_auc(trainer):
    model = get_best_model(trainer)

    X_train = trainer._cache["X_train"]
    y_train = trainer._cache["y_train"]

    X_test = trainer._cache["X_test"]
    y_test = trainer._cache["y_test"]

    train_auc = roc_auc_score(
        y_train,
        model.predict_proba(X_train)[:, 1]
    )

    test_auc = roc_auc_score(
        y_test,
        model.predict_proba(X_test)[:, 1]
    )

    return pd.Series({
        "train_auc": train_auc,
        "test_auc": test_auc,
        "auc_gap": train_auc - test_auc
    })
