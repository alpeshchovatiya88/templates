def prepare_ventiles(df):
    """Extract numeric ventiles, including labels such as 'Ventile 01'."""
    result = df.copy()
    result["_ventile_num"] = pd.to_numeric(
        result[VENTILE_COL]
        .astype(str)
        .str.extract(r"(\d+)")[0],
        errors="coerce"
    )
    return result

# ============================================================
# 8A. VALIDATION ONLY:
#     CREATE ADDITIONAL COLUMNS FOR CSV
# ============================================================

# This section does not change PD_ORIGINAL,
# PD_CALIBRATED, the fitted isotonic model,
# performance tables or calibration curves.
#
# No display or print output is generated here.

odds_factor = 34.05


# ------------------------------------------------------------
# 8A.1 ADJUST ORIGINAL AND CALIBRATED PD
# ------------------------------------------------------------

def adjust_pd(p):

    p = pd.to_numeric(
        p,
        errors="coerce"
    )

    return p / (
        p + odds_factor * (1 - p)
    )


df_val["PD_ORIGINAL_adjusted"] = adjust_pd(
    df_val["PD_ORIGINAL"]
)

df_val["PD_CALIBRATED_adjusted"] = adjust_pd(
    df_val["PD_CALIBRATED"]
)


# ------------------------------------------------------------
# 8A.2 CALCULATE CALIBRATED LOSS
# ------------------------------------------------------------

# SMI3_LOSS_calibrated =
# lgd * PD_CALIBRATED_adjusted * 0.62

lgd = pd.to_numeric(
    df_val["lgd"],
    errors="coerce"
)

premium = pd.to_numeric(
    df_val["premium"],
    errors="coerce"
).replace(0, np.nan)


df_val["SMI3_LOSS_calibrated"] = (
    lgd
    * df_val["PD_CALIBRATED_adjusted"]
    * 0.62
)


# ------------------------------------------------------------
# 8A.3 CALCULATE CALIBRATED LOSS RATIO
# ------------------------------------------------------------

# SMI3_L-R_calibrated =
# (SMI3_LOSS_calibrated / premium) * 100

df_val["SMI3_L-R_calibrated"] = (
    df_val["SMI3_LOSS_calibrated"]
    / premium
) * 100


# ------------------------------------------------------------
# 8A.4 DEFINE D1-D20 RANKING FUNCTION
# ------------------------------------------------------------

# D1 = highest loss ratio
# D20 = lowest loss ratio
#
# Approximately equal-sized groups.
# Ties are broken by row order.

def create_20_deciles(scores):

    scores = pd.to_numeric(
        scores,
        errors="coerce"
    )

    result = pd.Series(
        pd.NA,
        index=scores.index,
        dtype="string"
    )

    valid = scores.notna()

    if valid.any():

        ranks = scores.loc[valid].rank(
            method="first",
            ascending=False
        )

        groups = pd.qcut(
            ranks,
            q=min(20, int(valid.sum())),
            labels=False
        ) + 1

        result.loc[valid] = (
            "D" + groups.astype(str)
        )

    return result


# ------------------------------------------------------------
# 8A.5 CALCULATE ORIGINAL ADJUSTED LOSS RATIO
#      FOR ORIGINAL DECILE RANKING
# ------------------------------------------------------------

# Does NOT overwrite existing SMI3_LR.

original_lr_adjusted = (
    lgd
    * df_val["PD_ORIGINAL_adjusted"]
    * 0.62
    / premium
) * 100


# ------------------------------------------------------------
# 8A.6 CREATE ORIGINAL AND CALIBRATED DECILES
# ------------------------------------------------------------

df_val["decile_original"] = (
    create_20_deciles(
        original_lr_adjusted
    )
)

df_val["decile_calibrated"] = (
    create_20_deciles(
        df_val["SMI3_L-R_calibrated"]
    )
)

from sklearn.metrics import (
    roc_auc_score,
    brier_score_loss,
    log_loss
)


def evaluate(df, name):

    # Exclude missing target rows from evaluation
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty:
        print(f"{name}: No observed targets.")
        return None

    y = data[target_col].astype(int)

    original = data["PD_ORIGINAL"]
    calibrated = data["PD_CALIBRATED"]

    results = {
        "Dataset": name,
        "N_Observed": len(data),
        "Actual_Bad_Rate": y.mean(),
        "Original_Mean_PD": original.mean(),
        "Calibrated_Mean_PD": calibrated.mean(),
        "Original_Brier": brier_score_loss(y, original),
        "Calibrated_Brier": brier_score_loss(y, calibrated),
        "Original_LogLoss": log_loss(y, original),
        "Calibrated_LogLoss": log_loss(y, calibrated)
    }

    # AUC requires both classes
    if y.nunique() == 2:

        results["Original_AUC"] = roc_auc_score(
            y, original
        )

        results["Calibrated_AUC"] = roc_auc_score(
            y, calibrated
        )

    else:

        results["Original_AUC"] = np.nan
        results["Calibrated_AUC"] = np.nan

    return results

from sklearn.calibration import calibration_curve
import matplotlib.pyplot as plt


def plot_calibration(df, name):

    # Use observed targets only
    data = df[
        df[target_col].isin([0, 1])
    ].copy()

    if data.empty or data[target_col].nunique() < 2:

        print(
            f"{name}: Insufficient target data."
        )
        return

    y = data[target_col].astype(int)

    plt.figure(figsize=(8, 6))

    for col, label in [
        ("PD_ORIGINAL", "Original XGBoost"),
        ("PD_CALIBRATED", "Calibrated XGBoost")
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            data[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
    )

    plt.xlabel("Mean Predicted Probability")
    plt.ylabel("Observed Bad Rate")
    plt.title(name + " Calibration Curve")
    plt.legend()
    plt.grid(True)
    plt.show()


plot_calibration(df_train, "TRAIN")
plot_calibration(df_test, "TEST")
plot_calibration(df_val, "VALIDATION")
