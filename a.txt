
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill
from openpyxl.worksheet.datavalidation import DataValidation
from openpyxl.utils import get_column_letter

# Build movement counts by region, including Overall
temp["Economic Region"] = temp[region_col].astype(str)

overall = temp.copy()
overall["Economic Region"] = "Overall"

all_data = pd.concat([overall, temp], ignore_index=True)

all_data["Direction"] = "No Movement"
all_data.loc[all_data["movement"] < 0, "Direction"] = "Lower"
all_data.loc[all_data["movement"] > 0, "Direction"] = "Higher"

counts = (
    all_data.groupby(
        ["Economic Region", "deciles_moved", "Direction"]
    ).size().reset_index(name="Count")
)

totals = (
    all_data.groupby("Economic Region")
    .size().to_dict()
)

wb = Workbook()
ws = wb.active
ws.title = "Movement Summary"
data_ws = wb.create_sheet("Data")

# Store all summary counts in a hidden sheet
data_ws.append(["Region", "Movement", "Direction", "Count"])

for row in counts.itertuples(index=False, name=None):
    data_ws.append(list(row))

data_ws["F1"] = "Region"
data_ws["G1"] = "Total"

for i, (region, total) in enumerate(totals.items(), start=2):
    data_ws.cell(i, 6, region)
    data_ws.cell(i, 7, total)

last_row = data_ws.max_row
total_last = len(totals) + 1

# Visible summary
ws["A1"] = "Economic Region"
ws["B1"] = "Overall"

headers = [
    "Deciles Moved", "Lower", "Higher",
    "No Movement", "Total Files", "Percentage"
]

for col, header in enumerate(headers, start=1):
    ws.cell(3, col, header)

for r in range(4, 24):
    ws.cell(r, 1, r - 4)

    for col, direction in [(2, "Lower"), (3, "Higher"), (4, "No Movement")]:
        ws.cell(
            r, col,
            f'=SUMIFS(Data!$D$2:$D${last_row},'
            f'Data!$A$2:$A${last_row},$B$1,'
            f'Data!$B$2:$B${last_row},$A{r},'
            f'Data!$C$2:$C${last_row},"{direction}")'
        )

    ws.cell(r, 5, f"=SUM(B{r}:D{r})")
    ws.cell(
        r, 6,
        f'=IFERROR(E{r}/SUMIF('
        f'Data!$F$2:$F${total_last},$B$1,'
        f'Data!$G$2:$G${total_last}),0)'
    )
    ws.cell(r, 6).number_format = "0.00%"

# Dropdown for region selection
dropdown = DataValidation(
    type="list",
    formula1=f"'Data'!$F$2:$F${total_last}"
)
ws.add_data_validation(dropdown)
dropdown.add(ws["B1"])

# Formatting
for cell in ws[3]:
    cell.fill = PatternFill("solid", fgColor="17365D")
    cell.font = Font(color="FFFFFF", bold=True)

ws["A1"].font = Font(bold=True)
ws["B1"].fill = PatternFill("solid", fgColor="FFF2CC")

for c in range(1, 7):
    ws.column_dimensions[get_column_letter(c)].width = 19

data_ws.sheet_state = "hidden"

filename = f"Ventile_Movement_{name1}_vs_{name2}.xlsx"
wb.save(filename)

display(ws.title)
print("Saved:", filename)
