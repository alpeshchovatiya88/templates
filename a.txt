import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Convert continuous time to completed-year bucket
df_final["DELQ_YEAR"] = np.floor(df_final["TIME_TO_DELQ_YR"])
df_final["CLAIM_YEAR"] = np.floor(df_final["TIME_TO_CLAIM_YR"])

# Keep 0-17 years
years = np.arange(0, 18)

# Count events in each year
delq_count = (
    df_final.loc[df_final["DELQ_YEAR"].between(0, 17), "DELQ_YEAR"]
    .value_counts()
    .reindex(years, fill_value=0)
    .sort_index()
)

claim_count = (
    df_final.loc[df_final["CLAIM_YEAR"].between(0, 17), "CLAIM_YEAR"]
    .value_counts()
    .reindex(years, fill_value=0)
    .sort_index()
)

# Plot
plt.figure(figsize=(12, 6))

plt.plot(
    years,
    delq_count.values,
    marker="o",
    label="90+ Delinquency"
)

plt.plot(
    years,
    claim_count.values,
    marker="o",
    label="Claim"
)

plt.xticks(years)
plt.xlabel("Years from Effective Date")
plt.ylabel("Number of Applications")
plt.title("Time to Delinquency and Claim")
plt.legend()
plt.grid(alpha=0.3)

plt.tight_layout()
plt.show()
