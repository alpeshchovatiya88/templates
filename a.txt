results = []

features = [
    "CRA_BANKRUPTCY_FORECLOSURE_IND",
    "CRA_JUDGEMENT",
    "CRD_PUBLIC_DEROG_TU"
]

for score_bin in sorted(bad["CRD_SCORE_bin"].dropna().unique()):

    bad_bin = bad[bad["CRD_SCORE_bin"] == score_bin]

    for col in features:

        if col == "CRD_PUBLIC_DEROG_TU":
            feature_flag = bad_bin[col] > 0
        else:
            feature_flag = bad_bin[col] == "Y"

        collection_flag = bad_bin["CRA_COLLECTION"] == "Y"

        total_feature = feature_flag.sum()

        overlap = (collection_flag & feature_flag).sum()

        overlap_pct = round(overlap / total_feature * 100, 1) if total_feature > 0 else 0

        results.append({
            "CRD_SCORE_bin": score_bin,
            "Feature": col,
            "Total_Bad_Count": total_feature,
            "Overlap_With_CRA_COLLECTION": overlap,
            "Overlap_%": overlap_pct
        })

    # Combined row
    combined_flag = (
        (bad_bin["CRA_BANKRUPTCY_FORECLOSURE_IND"] == "Y")
        | (bad_bin["CRA_JUDGEMENT"] == "Y")
        | (bad_bin["CRD_PUBLIC_DEROG_TU"] > 0)
    )

    combined_total = combined_flag.sum()

    combined_overlap = (
        (bad_bin["CRA_COLLECTION"] == "Y")
        & combined_flag
    ).sum()

    combined_pct = round(
        combined_overlap / combined_total * 100,
        1
    ) if combined_total > 0 else 0

    results.append({
        "CRD_SCORE_bin": score_bin,
        "Feature": "ALL_3_COMBINED",
        "Total_Bad_Count": combined_total,
        "Overlap_With_CRA_COLLECTION": combined_overlap,
        "Overlap_%": combined_pct
    })

results = pd.DataFrame(results)
