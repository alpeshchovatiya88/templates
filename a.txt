
import pandas as pd
import numpy as np

from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side

# =====================================================
# 1. EXPERIMENTS - CHANGE ONLY THIS DICTIONARY
# =====================================================

dfs = {
    "HPI_+4": df_hpi_4,
    "HPI_+3": df_hpi_3,
    "HPI_+5": df_hpi_5,
}

DECILE_COL = "rr_model_decile"
LOSS_COL = "SMI3_LOSS"
LR_COL = "SMI3_LR"

# Set to 100 if LR values are stored as decimals
# Set to 1 if LR values are already percentages
LR_SCALE = 1

OUTPUT_FILE = "seed_experiments.xlsx"

# Optional descriptions
descriptions = {
    # "HPI_+4": "Experiment description",
}

# =====================================================
# 2. CALCULATE TOTAL LOSS AND AVERAGE LR
#    FOR EACH EXPERIMENT
# =====================================================

results = []

for experiment, df in dfs.items():

    temp = df.copy()

    required = [DECILE_COL, LOSS_COL, LR_COL]
    missing = [c for c in required if c not in temp.columns]

    if missing:
        raise ValueError(
            f"{experiment}: Missing columns {missing}"
        )

    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL]
        .astype(str)
        .str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    temp[LOSS_COL] = pd.to_numeric(
        temp[LOSS_COL], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan)

    temp[LR_COL] = pd.to_numeric(
        temp[LR_COL], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan)

    decile1 = temp[temp["_decile"] == 1]

    results.append({
        "Experiment": experiment,

        "Overall Loss": temp[LOSS_COL].sum(min_count=1),
        "Overall LR": temp[LR_COL].mean() * LR_SCALE,

        "Decile 1 Loss": decile1[LOSS_COL].sum(min_count=1),
        "Decile 1 LR": decile1[LR_COL].mean() * LR_SCALE,
    })

summary = pd.DataFrame(results)

if summary.empty:
    raise ValueError("The dfs dictionary is empty.")

# =====================================================
# 3. MIN AND MAX ACROSS ALL EXPERIMENTS
# =====================================================

ranges = {
    "Overall": [
        summary["Overall Loss"].min(),
        summary["Overall Loss"].max(),
        summary["Overall LR"].min(),
        summary["Overall LR"].max()
    ],

    "Decile 1": [
        summary["Decile 1 Loss"].min(),
        summary["Decile 1 Loss"].max(),
        summary["Decile 1 LR"].min(),
        summary["Decile 1 LR"].max()
    ]
}

# Show calculated results in notebook
print("\nIndividual experiment results:")
print(summary.to_string(index=False))

print("\nMin/max across experiments:")
for label, values in ranges.items():
    print(
        f"{label}: "
        f"Loss ${values[0]:,.0f} - ${values[1]:,.0f}; "
        f"LR {values[2]:.2f} - {values[3]:.2f}"
    )

# =====================================================
# 4. CREATE EXCEL - SINGLE SHEET
# =====================================================

wb = Workbook()
ws = wb.active
ws.title = "Seed Experiments"

yellow = PatternFill("solid", fgColor="FFFF00")
blue = PatternFill("solid", fgColor="DDEBF7")

thin = Side(style="thin", color="B7B7B7")
border = Border(
    left=thin,
    right=thin,
    top=thin,
    bottom=thin
)

# No row-height or column-width changes

# =====================================================
# 5. START AT ROW 4
#    A1, A2, A3 REMAIN EMPTY
# =====================================================

row = 4

for experiment in dfs:

    # -----------------------------------------
    # Experiment name - Yellow
    # -----------------------------------------

    cell = ws.cell(row, 1, str(experiment))
    cell.fill = yellow
    cell.font = Font(bold=True, size=12)

    # -----------------------------------------
    # Group headers
    # -----------------------------------------

    ws.merge_cells(
        start_row=row,
        start_column=2,
        end_row=row,
        end_column=3
    )

    ws.merge_cells(
        start_row=row,
        start_column=4,
        end_row=row,
        end_column=5
    )

    for col, text in [
        (2, "Loss range"),
        (4, "Loss ratio range")
    ]:
        cell = ws.cell(row, col, text)
        cell.font = Font(bold=True, size=12)
        cell.alignment = Alignment(
            horizontal="center",
            vertical="center"
        )

    # -----------------------------------------
    # Description - Blue
    # -----------------------------------------

    desc = ws.cell(
        row + 1,
        1,
        descriptions.get(experiment, "")
    )

    desc.fill = blue
    desc.alignment = Alignment(
        vertical="center",
        wrap_text=False
    )

    # -----------------------------------------
    # Min/max headings
    # -----------------------------------------

    for col, heading in zip(
        [2, 3, 4, 5],
        ["min", "max", "min", "max"]
    ):
        cell = ws.cell(row + 1, col, heading)
        cell.font = Font(bold=True, size=12)
        cell.alignment = Alignment(
            horizontal="center",
            vertical="center"
        )

    # -----------------------------------------
    # Overall and Decile 1
    # -----------------------------------------

    for offset, label in enumerate(
        ["Overall", "Decile 1"],
        start=2
    ):

        r = row + offset

        ws.cell(r, 1, label)

        for col, value in enumerate(
            ranges[label],
            start=2
        ):

            if pd.isna(value):
                value = None
            elif isinstance(value, (np.integer, np.floating)):
                value = value.item()

            cell = ws.cell(r, col, value)

            if col in [2, 3]:
                cell.number_format = '$#,##0'
            else:
                cell.number_format = '0.00'

            cell.alignment = Alignment(
                horizontal="right",
                vertical="center"
            )

    # -----------------------------------------
    # Borders
    # -----------------------------------------

    for r in range(row, row + 4):
        for col in range(1, 6):
            ws.cell(r, col).border = border

    # Two blank rows between experiments
    row += 6

# =====================================================
# 6. SAVE EXCEL
# =====================================================

wb.save(OUTPUT_FILE)

print(f"\nExcel saved: {OUTPUT_FILE}")
