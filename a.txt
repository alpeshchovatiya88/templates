def round_half_up(x):
    return np.floor(x + 0.5)

def agg_one(
    series,
    prefix,
    round_stats=True
):

    s = pd.to_numeric(
        series,
        errors="coerce"
    )

    g = pd.DataFrame(
        {"val": s},
        index=b.index
    ).groupby(
        grp_idx,
        sort=False
    )

    agg_map = {
        f"{prefix}_min":
            ("val", "min"),

        f"{prefix}_max":
            ("val", "max"),

        f"{prefix}_avg":
            ("val", "mean"),

        f"{prefix}_median":
            ("val", "median"),
    }

    agg_map = {
        col: spec
        for col, spec in agg_map.items()
        if col.rsplit("_", 1)[-1]
        in stats
    }

    result = g.agg(
        **agg_map
    )

    if round_stats:

        for stat in [
            "avg",
            "median"
        ]:

            col = f"{prefix}_{stat}"

            if col in result:

                result[col] = (
                    round_half_up(
                        result[col]
                    )
                    .where(
                        result[col].notna()
                    )
                )

    if "wavg" in stats:

        w_valid = w.where(
            s.notna(),
            0.0
        )

        numerator = (
            (s * w_valid)
            .groupby(
                grp_idx,
                sort=False
            )
            .sum(min_count=1)
        )

        denominator = (
            w_valid
            .groupby(
                grp_idx,
                sort=False
            )
            .sum(min_count=1)
            .replace(
                0,
                np.nan
            )
        )

        wavg = (
            numerator
            / denominator
        ).replace(
            0,
            np.nan
        )

        result[
            f"{prefix}_wavg"
        ] = (
            round_half_up(wavg)
            .where(wavg.notna())

            if round_stats

            else wavg
        )

    return result
