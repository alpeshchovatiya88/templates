import pandas as pd
import numpy as np

from pathlib import Path
from datetime import datetime

from sklearn.metrics import (
    roc_auc_score,
    accuracy_score,
    precision_score,
    recall_score,
    brier_score_loss,
    log_loss
)

def check_reproducibility(
    developer_file,
    validator_file,
    dataset_name,
    key_columns,
    target_col="Target",
    probability_col="Predicted_Probability",
    label_col="Prediction_Label",
    probability_tolerance=1e-8,
    metric_tolerance=1e-8,
    loss_tolerance=1e-6,
    loss_ratio_tolerance=1e-8,
    total_loss_tolerance=1e-4,
    loss_col="Loss",
    loss_ratio_col="Loss_Ratio",
    output_folder="reproducibility_reports"
):
    """
    Reproducibility testing between developer and validator CSVs.

    Checks:
      1. Record alignment, missing/extra IDs
      2. Exact target and predicted-label comparison
      3. Predicted probability comparison
      4. Dataset-level AUC, accuracy, precision, recall,
         Brier score and log loss
      5. Optional Loss and Loss Ratio comparisons
      6. Optional aggregate Total Loss
      7. Excel and CSV exception reports

    Loss and Loss Ratio are optional:
      - Both files contain the column: compare it.
      - Neither file contains the column: skip it.
      - Only one file contains it: raise an error.

    AUC and other performance metrics are calculated
    independently from target, label and probability.
    """

    # ========================================================
    # 1. Validate configuration
    # ========================================================

    key_columns = list(key_columns)

    if not key_columns:
        raise ValueError("At least one key column is required.")

    required_columns = (
        key_columns
        + [target_col, probability_col, label_col]
    )

    if len(set(required_columns)) != len(required_columns):
        raise ValueError(
            "Key, target, probability and label columns "
            "must have distinct names."
        )

    tolerances = {
        "Probability": probability_tolerance,
        "Performance": metric_tolerance,
        "Loss": loss_tolerance,
        "Loss_Ratio": loss_ratio_tolerance,
        "Total_Loss": total_loss_tolerance
    }

    for name, tolerance in tolerances.items():
        if not np.isfinite(tolerance) or tolerance < 0:
            raise ValueError(
                f"{name} tolerance must be finite and >= 0."
            )

    # ========================================================
    # 2. Read files
    # ========================================================

    key_dtypes = {
        col: "string" for col in key_columns
    }

    dev = pd.read_csv(
        developer_file,
        dtype=key_dtypes,
        low_memory=False
    )

    val = pd.read_csv(
        validator_file,
        dtype=key_dtypes,
        low_memory=False
    )

    print(f"\nDataset: {dataset_name}")
    print(f"Developer records: {len(dev):,}")
    print(f"Validator records: {len(val):,}")

    # ========================================================
    # 3. Validate required columns and IDs
    # ========================================================

    for name, df in [
        ("Developer", dev),
        ("Validator", val)
    ]:

        if not df.columns.is_unique:
            raise ValueError(
                f"{name}: duplicate column names."
            )

        missing_cols = [
            col for col in required_columns
            if col not in df.columns
        ]

        if missing_cols:
            raise ValueError(
                f"{name}: missing columns {missing_cols}"
            )

        if df[key_columns].isna().any().any():
            raise ValueError(
                f"{name}: missing record IDs."
            )

        if df.duplicated(subset=key_columns).any():
            raise ValueError(
                f"{name}: duplicate record IDs."
            )

        for col in [
            target_col,
            probability_col,
            label_col
        ]:

            original = df[col]

            converted = pd.to_numeric(
                original,
                errors="coerce"
            )

            invalid = (
                original.notna()
                & converted.isna()
            )

            if invalid.any():
                raise ValueError(
                    f"{name}: nonnumeric values in {col}."
                )

            if converted.isna().any():
                raise ValueError(
                    f"{name}: missing values in {col}."
                )

            if not np.isfinite(converted).all():
                raise ValueError(
                    f"{name}: non-finite values in {col}."
                )

            df[col] = converted

        # Binary classification validation.
        if not df[target_col].isin([0, 1]).all():
            raise ValueError(
                f"{name}: target must be 0 or 1."
            )

        if not df[label_col].isin([0, 1]).all():
            raise ValueError(
                f"{name}: predicted label must be 0 or 1."
            )

        if not df[probability_col].between(0, 1).all():
            raise ValueError(
                f"{name}: probability must be in [0, 1]."
            )

    # ========================================================
    # 4. Identify optional columns
    # ========================================================

    optional_metrics = {}

    for metric_name, column, tolerance in [
        ("Loss", loss_col, loss_tolerance),
        ("Loss_Ratio", loss_ratio_col, loss_ratio_tolerance)
    ]:

        if column is None:
            continue

        in_dev = column in dev.columns
        in_val = column in val.columns

        if in_dev != in_val:
            raise ValueError(
                f"{column} exists in only one CSV."
            )

        if in_dev and in_val:

            for name, df in [
                ("Developer", dev),
                ("Validator", val)
            ]:

                original = df[column]

                converted = pd.to_numeric(
                    original,
                    errors="coerce"
                )

                invalid = (
                    original.notna()
                    & converted.isna()
                )

                if invalid.any():
                    raise ValueError(
                        f"{name}: nonnumeric {column}."
                    )

                if np.isinf(converted).any():
                    raise ValueError(
                        f"{name}: infinite {column}."
                    )

                df[column] = converted

            optional_metrics[metric_name] = (
                column,
                tolerance
            )

    # ========================================================
    # 5. Align records using unique identifiers
    # ========================================================

    compare_columns = (
        [target_col, probability_col, label_col]
        + [
            column
            for column, _ in optional_metrics.values()
        ]
    )

    selected_columns = (
        key_columns + list(dict.fromkeys(compare_columns))
    )

    merged = dev[selected_columns].merge(
        val[selected_columns],
        on=key_columns,
        how="outer",
        suffixes=("_DEV", "_VAL"),
        indicator=True,
        validate="one_to_one"
    )

    missing_records = merged.loc[
        merged["_merge"] != "both",
        key_columns + ["_merge"]
    ].copy()

    missing_records["Issue"] = (
        missing_records["_merge"]
        .astype(str)
        .replace({
            "left_only": "Missing in Validator",
            "right_only": "Extra in Validator"
        })
    )

    missing_records = missing_records[
        key_columns + ["Issue"]
    ].reset_index(drop=True)

    matched = merged.loc[
        merged["_merge"] == "both"
    ].copy()

    # ========================================================
    # 6. Record-level comparisons
    # ========================================================

    record_config = {
        "Target": (target_col, 0.0),
        "Prediction_Label": (label_col, 0.0),
        "Predicted_Probability": (
            probability_col,
            probability_tolerance
        )
    }

    record_config.update(optional_metrics)

    record_summary_rows = []
    mismatch_frames = []

    for metric_name, (column, tolerance) in (
        record_config.items()
    ):

        dev_values = matched[f"{column}_DEV"]
        val_values = matched[f"{column}_VAL"]

        difference = (dev_values - val_values).abs()

        valid = (
            dev_values.notna()
            & val_values.notna()
        )

        passed = valid & (difference <= tolerance)

        failed = ~passed
        mismatch_count = int(failed.sum())

        record_summary_rows.append({
            "Dataset": dataset_name,
            "Metric": metric_name,
            "Level": "Record",
            "Records_Compared": len(matched),
            "Mismatch_Count": mismatch_count,
            "Mismatch_Percentage": (
                100 * mismatch_count / len(matched)
                if len(matched) else np.nan
            ),
            "Max_Absolute_Difference": difference.max(),
            "Tolerance": tolerance,
            "Status": (
                "PASS"
                if len(matched) > 0 and mismatch_count == 0
                else "FAIL"
            )
        })

        if mismatch_count > 0:

            detail = matched.loc[
                failed,
                key_columns
            ].copy()

            detail["Dataset"] = dataset_name
            detail["Metric"] = metric_name
            detail["Developer_Value"] = (
                dev_values.loc[failed]
            )
            detail["Validator_Value"] = (
                val_values.loc[failed]
            )
            detail["Absolute_Difference"] = (
                difference.loc[failed]
            )
            detail["Tolerance"] = tolerance
            detail["Status"] = "FAIL"

            mismatch_frames.append(detail)

    record_summary = pd.DataFrame(record_summary_rows)

    if mismatch_frames:
        mismatches = pd.concat(
            mismatch_frames,
            ignore_index=True
        )
    else:
        mismatches = pd.DataFrame(
            columns=key_columns + [
                "Dataset",
                "Metric",
                "Developer_Value",
                "Validator_Value",
                "Absolute_Difference",
                "Tolerance",
                "Status"
            ]
        )

    # ========================================================
    # 7. Dataset-level performance metrics
    # ========================================================

    def calculate_performance(df):

        y_true = df[target_col].to_numpy(dtype=int)
        y_pred = df[label_col].to_numpy(dtype=int)
        y_prob = df[probability_col].to_numpy(dtype=float)

        if len(y_true) == 0:
            raise ValueError(
                "Cannot calculate metrics on empty data."
            )

        if len(np.unique(y_true)) != 2:
            raise ValueError(
                "AUC requires both target classes "
                f"in {dataset_name}."
            )

        return {
            "AUC": roc_auc_score(y_true, y_prob),
            "Accuracy": accuracy_score(y_true, y_pred),
            "Precision": precision_score(
                y_true,
                y_pred,
                zero_division=0
            ),
            "Recall": recall_score(
                y_true,
                y_pred,
                zero_division=0
            ),
            "Brier_Score": brier_score_loss(
                y_true,
                y_prob
            ),
            "Log_Loss": log_loss(
                y_true,
                y_prob,
                labels=[0, 1]
            )
        }

    dev_metrics = calculate_performance(dev)
    val_metrics = calculate_performance(val)

    performance_rows = []

    for metric_name in dev_metrics:

        dev_value = dev_metrics[metric_name]
        val_value = val_metrics[metric_name]

        difference = abs(dev_value - val_value)

        performance_rows.append({
            "Dataset": dataset_name,
            "Metric": metric_name,
            "Level": "Dataset",
            "Developer_Value": dev_value,
            "Validator_Value": val_value,
            "Absolute_Difference": difference,
            "Tolerance": metric_tolerance,
            "Status": (
                "PASS"
                if difference <= metric_tolerance
                else "FAIL"
            )
        })

    performance_summary = pd.DataFrame(
        performance_rows
    )

    # ========================================================
    # 8. Aggregate total loss comparison
    # ========================================================

    aggregate_rows = []

    if "Loss" in optional_metrics:

        dev_total = dev[loss_col].sum(min_count=1)
        val_total = val[loss_col].sum(min_count=1)

        difference = abs(dev_total - val_total)

        complete_loss = (
            dev[loss_col].notna().all()
            and val[loss_col].notna().all()
        )

        passed = (
            complete_loss
            and np.isfinite(dev_total)
            and np.isfinite(val_total)
            and difference <= total_loss_tolerance
        )

        aggregate_rows.append({
            "Dataset": dataset_name,
            "Metric": "Total_Loss",
            "Developer_Value": dev_total,
            "Validator_Value": val_total,
            "Absolute_Difference": difference,
            "Tolerance": total_loss_tolerance,
            "Status": "PASS" if passed else "FAIL"
        })

    aggregate_summary = pd.DataFrame(
        aggregate_rows,
        columns=[
            "Dataset",
            "Metric",
            "Developer_Value",
            "Validator_Value",
            "Absolute_Difference",
            "Tolerance",
            "Status"
        ]
    )

    # ========================================================
    # 9. Overall reproducibility assessment
    # ========================================================

    records_pass = (
        missing_records.empty
        and len(matched) > 0
    )

    record_metrics_pass = (
        record_summary["Status"].eq("PASS").all()
    )

    performance_pass = (
        performance_summary["Status"].eq("PASS").all()
    )

    aggregate_pass = (
        aggregate_summary.empty
        or aggregate_summary["Status"].eq("PASS").all()
    )

    overall_pass = all([
        records_pass,
        record_metrics_pass,
        performance_pass,
        aggregate_pass
    ])

    overall_status = (
        "PASS" if overall_pass else "FAIL"
    )

    overall_summary = pd.DataFrame([{
        "Dataset": dataset_name,
        "Developer_File": str(developer_file),
        "Validator_File": str(validator_file),
        "Developer_Rows": len(dev),
        "Validator_Rows": len(val),
        "Matched_Rows": len(matched),
        "Missing_Extra_Rows": len(missing_records),
        "Record_Mismatches": len(mismatches),
        "Execution_Time": datetime.now().isoformat(
            timespec="seconds"
        ),
        "Overall_Status": overall_status
    }])

    # ========================================================
    # 10. Export reproducibility report
    # ========================================================

    output_dir = Path(output_folder)
    output_dir.mkdir(
        parents=True,
        exist_ok=True
    )

    safe_name = "".join(
        c if c.isalnum() or c in "-_" else "_"
        for c in dataset_name
    )

    if not safe_name:
        raise ValueError("Invalid dataset name.")

    report_path = (
        output_dir
        / f"{safe_name}_Reproducibility.xlsx"
    )

    mismatch_csv = (
        output_dir
        / f"{safe_name}_Mismatches.csv"
    )

    missing_csv = (
        output_dir
        / f"{safe_name}_Missing_Records.csv"
    )

    mismatches.to_csv(
        mismatch_csv,
        index=False
    )

    missing_records.to_csv(
        missing_csv,
        index=False
    )

    excel_limit = 1_048_575

    with pd.ExcelWriter(
        report_path,
        engine="openpyxl"
    ) as writer:

        overall_summary.to_excel(
            writer,
            sheet_name="Executive Summary",
            index=False
        )

        record_summary.to_excel(
            writer,
            sheet_name="Record Checks",
            index=False
        )

        performance_summary.to_excel(
            writer,
            sheet_name="Performance Metrics",
            index=False
        )

        aggregate_summary.to_excel(
            writer,
            sheet_name="Aggregate Loss",
            index=False
        )

        mismatches.head(excel_limit).to_excel(
            writer,
            sheet_name="Mismatches",
            index=False
        )

        missing_records.head(excel_limit).to_excel(
            writer,
            sheet_name="Missing Records",
            index=False
        )

    # ========================================================
    # 11. Display results
    # ========================================================

    print("\n" + "=" * 65)
    print(f"REPRODUCIBILITY RESULTS: {dataset_name}")
    print("=" * 65)

    print("\nRECORD-LEVEL CHECKS")
    print(record_summary.to_string(index=False))

    print("\nDATASET-LEVEL PERFORMANCE METRICS")
    print(performance_summary.to_string(index=False))

    if not aggregate_summary.empty:
        print("\nAGGREGATE LOSS")
        print(aggregate_summary.to_string(index=False))

    print(f"\nMissing / Extra Records: {len(missing_records)}")
    print(f"Total Record Mismatches: {len(mismatches)}")

    print(f"\nOVERALL STATUS: {overall_status}")
    print(f"Report: {report_path}")

    return {
        "dataset": dataset_name,
        "overall_status": overall_status,
        "overall_summary": overall_summary,
        "record_summary": record_summary,
        "performance_summary": performance_summary,
        "aggregate_summary": aggregate_summary,
        "mismatches": mismatches,
        "missing_records": missing_records,
        "report_path": str(report_path)
    }
