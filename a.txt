### ===============================================
### SHAP comparison between all vs funded with hpi
### ===============================================
# =========================
# PLOT SETTINGS
# =========================

plt.rcParams["figure.dpi"] = 150

n_features = len(df_plot)
fig, ax = plt.subplots(figsize=(22, 10))

y = np.arange(n_features)
bar_height = 0.35

color_all = "#A9CCE3"      # light blue
color_funded = "#F9E79F"   # light yellow

# =========================
# BARS
# =========================

ax.barh(
    y - bar_height / 2,
    all_shap_with_hpi,
    height=bar_height,
    label="All Model SHAP With HPI",
    color=color_all,
    edgecolor="black",
    linewidth=0.8
)

ax.barh(
    y + bar_height / 2,
    funded_shap_with_hpi,
    height=bar_height,
    label="Funded Model SHAP With HPI",
    color=color_funded,
    edgecolor="black",
    linewidth=0.8
)

# =========================
# AXES
# =========================

ax.set_yticks(y)
ax.set_yticklabels(features, fontsize=18, fontweight="bold")
ax.tick_params(axis="x", labelsize=16)

ax.set_xlabel("Absolute SHAP Value", fontsize=20, fontweight="bold")

ax.set_title(
    "SHAP Feature Importance Comparison\n(All Model vs Funded Model: With HPI)",
    fontsize=24,
    fontweight="bold",
    pad=25
)

ax.legend(fontsize=15, loc="upper right")
ax.invert_yaxis()

# =========================
# LABELS
# =========================

max_value = max(all_shap_with_hpi.max(), funded_shap_with_hpi.max())
inside_offset = max_value * 0.03
outside_offset = max_value * 0.015

for i in range(n_features):

    # All model label: rank change from Funded rank to All rank
    rank_change_all = int(funded_rank_with_hpi.iloc[i]) - int(all_rank_with_hpi.iloc[i])

    if rank_change_all > 0:
        rank_move_all = f"↑{rank_change_all}"
    elif rank_change_all < 0:
        rank_move_all = f"↓{abs(rank_change_all)}"
    else:
        rank_move_all = "–"

    label_all = (
        f"{all_shap_with_hpi.iloc[i]:.3f} "
        f"(R{int(all_rank_with_hpi.iloc[i])}, {rank_move_all})"
    )

    value_all = all_shap_with_hpi.iloc[i]

    if value_all > max_value * 0.18:
        x_all = value_all - inside_offset
        ha_all = "right"
    else:
        x_all = value_all + outside_offset
        ha_all = "left"

    ax.text(
        x_all,
        y[i] - bar_height / 2,
        label_all,
        va="center",
        ha=ha_all,
        fontsize=14,
        fontweight="bold",
        color="black"
    )

    # Funded model label: rank change from All rank to Funded rank
    rank_change_funded = int(all_rank_with_hpi.iloc[i]) - int(funded_rank_with_hpi.iloc[i])

    if rank_change_funded > 0:
        rank_move_funded = f"↑{rank_change_funded}"
    elif rank_change_funded < 0:
        rank_move_funded = f"↓{abs(rank_change_funded)}"
    else:
        rank_move_funded = "–"

    label_funded = (
        f"{funded_shap_with_hpi.iloc[i]:.3f} "
        f"(R{int(funded_rank_with_hpi.iloc[i])}, {rank_move_funded})"
    )

    value_funded = funded_shap_with_hpi.iloc[i]

    if value_funded > max_value * 0.18:
        x_funded = value_funded - inside_offset
        ha_funded = "right"
    else:
        x_funded = value_funded + outside_offset
        ha_funded = "left"

    ax.text(
        x_funded,
        y[i] + bar_height / 2,
        label_funded,
        va="center",
        ha=ha_funded,
        fontsize=14,
        fontweight="bold",
        color="black"
    )

# =========================
# GRID + SAVE
# =========================

ax.grid(axis="x", linestyle="--", alpha=0.35)
ax.set_xlim(0, max_value * 1.25)

plt.tight_layout()

plt.savefig(
    "SHAP_Comparison_All_vs_Funded_With_HPI.png",
    dpi=400,
    bbox_inches="tight"
)
