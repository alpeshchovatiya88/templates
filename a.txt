import plotly.express as px

bookyear_bad = (
    df.groupby("BOOKYEAR")
      .agg(
          total=("target", "size"),
          bads=("target", "sum"),
          bad_rate=("target", "mean")
      )
      .reset_index()
)

bookyear_bad["bad_rate_pct"] = bookyear_bad["bad_rate"] * 100

fig = px.scatter(
    bookyear_bad,
    x="BOOKYEAR",
    y="bad_rate_pct",
    size="total",
    text="bad_rate_pct",
    custom_data=["total", "bads"],
    size_max=35,
    title="Bad Rate by BOOKYEAR"
)

fig.update_traces(
    texttemplate="%{text:.2f}%",
    textposition="top center",
    hovertemplate=
        "<b>BOOKYEAR: %{x}</b><br>" +
        "Total: %{customdata[0]:,.0f}<br>" +
        "Bads: %{customdata[1]:,.0f}<br>" +
        "Bad Rate: %{y:.2f}%<extra></extra>"
)

fig.update_layout(
    xaxis_title="BOOKYEAR",
    yaxis_title="Bad Rate (%)",
    yaxis_ticksuffix="%",
    xaxis=dict(type="category")
)

fig.show()
