
import pandas as pd
import plotly.graph_objects as go


def plot_by_decile(
    dfs,
    decile_col,
    value_col,
    agg="sum",
    title=None,
    decimals=2,
    label_font_size=15,
    height=750
):

    # --------------------------------------------------
    # 1. Aggregate metric by ventile for each seed
    # --------------------------------------------------

    summary = pd.DataFrame({
        name: (
            df.assign(
                _decile=pd.to_numeric(
                    df[decile_col]
                    .astype(str)
                    .str.extract(r"(\d+)", expand=False),
                    errors="coerce"
                )
            )
            .groupby("_decile")[value_col]
            .agg(agg)
        )
        for name, df in dfs.items()
    }).sort_index()

    if summary.empty:
        raise ValueError(
            "No valid ventile data found."
        )

    # --------------------------------------------------
    # 2. Create grouped bar chart
    # --------------------------------------------------

    fig = go.Figure()

    colors = {
        "RS42": "#17365D",
        "RS07": "#E66B23",
        "RS101": "#09988A",
        "RS151": "#7952B3"
    }

    for name in summary.columns:

        values = summary[name]

        fig.add_trace(
            go.Bar(
                x=summary.index,
                y=values,
                name=name,

                marker_color=colors.get(
                    name, "#64748B"
                ),

                # Format labels with commas
                text=[
                    f"{v:,.{decimals}f}"
                    if pd.notna(v) else ""
                    for v in values
                ],

                # Show values above bars
                textposition="outside",

                # Rotate labels vertically
                textangle=-90,

                # Bold black text
                textfont=dict(
                    family="Arial Black",
                    size=label_font_size,
                    color="black"
                ),

                # Prevent labels from being clipped
                cliponaxis=False,

                hovertemplate=(
                    "Ventile: %{x}<br>"
                    f"Seed: {name}<br>"
                    "Value: %{y:,.2f}"
                    "<extra></extra>"
                )
            )
        )

    # --------------------------------------------------
    # 3. Ventile dropdown
    # --------------------------------------------------

    ventiles = summary.index.tolist()

    buttons = [
        dict(
            label="All Ventiles",
            method="relayout",
            args=[
                {
                    "xaxis.range": [
                        min(ventiles) - 0.5,
                        max(ventiles) + 0.5
                    ]
                }
            ]
        )
    ]

    for d in ventiles:

        buttons.append(
            dict(
                label=f"Ventile {int(d)}",
                method="relayout",
                args=[
                    {
                        "xaxis.range": [
                            float(d) - 0.5,
                            float(d) + 0.5
                        ]
                    }
                ]
            )
        )

    # --------------------------------------------------
    # 4. Add extra headroom for vertical labels
    # --------------------------------------------------

    max_value = summary.max().max()
    min_value = summary.min().min()

    if pd.isna(max_value):
        max_value = 0

    if pd.isna(min_value):
        min_value = 0

    max_value = float(max_value)
    min_value = float(min_value)

    span = max(
        max_value - min_value,
        abs(max_value),
        abs(min_value),
        1
    )

    upper = max(0, max_value) + span * 0.35
    lower = min(0, min_value) - span * 0.10

    # --------------------------------------------------
    # 5. Chart layout
    # --------------------------------------------------

    fig.update_layout(

        title=dict(
            text=(
                title
                or f"{value_col} by Ventile Across Random Seeds"
            ),
            x=0.5,
            xanchor="center",
            font=dict(
                size=22,
                family="Arial Black"
            )
        ),

        xaxis_title="Ventile",
        yaxis_title=value_col,

        barmode="group",

        height=height,

        margin=dict(
            t=180,
            b=90,
            l=90,
            r=60
        ),

        font=dict(
            family="Arial",
            size=13
        ),

        legend=dict(
            orientation="h",
            x=0.5,
            xanchor="center",
            y=1.02,
            yanchor="bottom"
        ),

        uniformtext=dict(
            minsize=10,
            mode="show"
        ),

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,

                x=1,
                xanchor="right",

                y=1.24,
                yanchor="top",

                bgcolor="white",
                bordercolor="#CBD5E1",
                borderwidth=1,

                font=dict(
                    size=13
                )
            )
        ],

        plot_bgcolor="white",
        paper_bgcolor="white",

        bargap=0.20,
        bargroupgap=0.05
    )

    # --------------------------------------------------
    # 6. X-axis
    # --------------------------------------------------

    fig.update_xaxes(
        tickmode="array",
        tickvals=ventiles,
        type="linear",

        range=[
            min(ventiles) - 0.5,
            max(ventiles) + 0.5
        ],

        showgrid=False,

        tickfont=dict(
            size=12
        )
    )

    # --------------------------------------------------
    # 7. Y-axis
    # --------------------------------------------------

    fig.update_yaxes(
        range=[lower, upper],
        automargin=True,
        gridcolor="#E5E7EB",
        zeroline=True,
        zerolinecolor="#94A3B8"
    )

    # --------------------------------------------------
    # 8. Display chart
    # --------------------------------------------------

    fig.show()

    return summary
