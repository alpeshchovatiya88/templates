training_reference = pd.read_csv(

    FILES["train"],

    usecols=FEATURES,

)

def training_scale(column):

    values = pd.to_numeric(

        training_reference[column],

        errors="coerce",

    ).replace([np.inf, -np.inf], np.nan).dropna()

    if len(values) < 2:

        raise ValueError(

            f"{column}: insufficient numeric training observations"

        )

    scale = float(values.std(ddof=1))

    if not np.isfinite(scale) or scale <= 0:

        raise ValueError(

            f"{column}: no usable training variability. "

            "Specify a justified sensitivity magnitude manually."

        )

    return scale
