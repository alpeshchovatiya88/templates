import re

import plotly.graph_objects as go

# Your seven experiment DataFrames
dfs = {
    "RS42": df_rs42,
    "RS532": df_rs532,
    "RS3458": df_rs3458,
    "RS499": df_rs499,
    "RS345": df_rs345,
    "RS1999": df_rs1999,
    "RS2324": df_rs2324,
}

# Experiment colors
colors = {
    "RS42":   "#17365D",  # Navy blue - Baseline
    "RS532":  "#007F73",  # Dark teal
    "RS3458": "#6A1B9A",  # Deep purple
    "RS499":  "#B45309",  # Burnt orange
    "RS345":  "#9F1239",  # Dark rose
    "RS1999": "#166534",  # Forest green
    "RS2324": "#1D4ED8",  # Royal blue
}

VENTILE_COL = "rr_model_decile"
VENTILES = list(range(1, 21))

def prepare_ventiles(df):
    """Extract numeric ventiles, including labels such as 'Ventile 01'."""
    result = df.copy()
    result["_ventile_num"] = pd.to_numeric(
        result[VENTILE_COL]
        .astype(str)
        .str.extract(r"(\d+)")[0],
        errors="coerce"
    )
    return result

===========


# ==========================================================
# SMI3_LOSS - HORIZONTAL GROUPED BAR CHART
# ==========================================================

fig_loss = go.Figure()

for seed, df in dfs.items():

    temp = prepare_ventiles(df)

    temp["SMI3_LOSS"] = pd.to_numeric(
        temp["SMI3_LOSS"], errors="coerce"
    )

    ventile_loss = (
        temp.groupby("_ventile_num")["SMI3_LOSS"]
        .sum(min_count=1)
        .reindex(VENTILES)
    )

    fig_loss.add_trace(
        go.Bar(
            name=seed,
            y=[str(v) for v in VENTILES],
            x=ventile_loss.tolist(),
            orientation="h",
            marker_color=colors[seed],
            text=[
                f"{v:,.2f}" if pd.notna(v) else ""
                for v in ventile_loss
            ],
            textposition="outside",
            textfont=dict(size=12),
            cliponaxis=False,
            hovertemplate=(
                f"<b>{seed}</b><br>"
                "Ventile: %{y}<br>"
                "Total SMI3_LOSS: %{x:,.2f}"
                "<extra></extra>"
            ),
        )
    )

# Dynamic height for more experiments
chart_height = max(
    900,
    len(VENTILES) * len(dfs) * 25
)

fig_loss.update_layout(
    title=dict(
        text="Random Seed Sensitivity - SMI3_LOSS by Ventile",
        x=0.5,
        font=dict(size=22),
    ),

    barmode="group",
    height=chart_height,
    width=1500,

    bargap=0.20,
    bargroupgap=0.05,

    xaxis=dict(
        title="Total SMI3_LOSS",
        tickformat=",.0f",
        rangemode="tozero",
    ),

    yaxis=dict(
        title="Ventile",
        type="category",
        autorange="reversed",
    ),

    legend=dict(
        orientation="h",
        x=0.5,
        xanchor="center",
        y=1.02,
        yanchor="bottom",
    ),

    margin=dict(
        l=110,
        r=170,
        t=130,
        b=80,
    ),

    plot_bgcolor="white",
    paper_bgcolor="white",

    font=dict(size=13),

    uniformtext=dict(
        minsize=11,
        mode="show",
    ),
)

fig_loss.update_xaxes(showgrid=True, gridcolor="#E5E7EB")
fig_loss.update_yaxes(showgrid=False)

fig_loss.show()
