import pandas as pd
import numpy as np

years = np.arange(1, 18)

def create_timing_summary(df):

    results = []

    regions = ["ALL"] + sorted(
        df["ECON_REGION"].dropna().unique()
    )

    for region in regions:

        if region == "ALL":
            temp = df.copy()
        else:
            temp = df[
                df["ECON_REGION"] == region
            ].copy()

        # -------------------------
        # Delinquency
        # Claim takes priority
        # -------------------------
        delq = temp[
            (temp["DELQ90_IND"] == 1.0) &
            (temp["CLAIM_IND"] != 1.0) &
            (temp["DELQ_YEAR"].between(1, 17))
        ].copy()

        # -------------------------
        # Claim
        # -------------------------
        claim = temp[
            (temp["CLAIM_IND"] == 1.0) &
            (temp["CLAIM_YEAR"].between(1, 17))
        ].copy()

        # Counts
        delq_count = len(delq)
        claim_count = len(claim)

        # Peak years
        delq_peak = (
            delq["DELQ_YEAR"].mode().iloc[0]
            if delq_count > 0 else np.nan
        )

        claim_peak = (
            claim["CLAIM_YEAR"].mode().iloc[0]
            if claim_count > 0 else np.nan
        )

        # Median years
        delq_median = (
            delq["DELQ_YEAR"].median()
            if delq_count > 0 else np.nan
        )

        claim_median = (
            claim["CLAIM_YEAR"].median()
            if claim_count > 0 else np.nan
        )

        # % of events occurring Years 3-5
        delq_pct_3_5 = (
            delq["DELQ_YEAR"].between(3, 5).mean() * 100
            if delq_count > 0 else np.nan
        )

        claim_pct_3_5 = (
            claim["CLAIM_YEAR"].between(3, 5).mean() * 100
            if claim_count > 0 else np.nan
        )

        results.append({
            "ECON_REGION": region,
            "DELQ_COUNT": delq_count,
            "DELQ_PEAK_YEAR": delq_peak,
            "DELQ_MEDIAN_YEAR": delq_median,
            "CLAIM_COUNT": claim_count,
            "CLAIM_PEAK_YEAR": claim_peak,
            "CLAIM_MEDIAN_YEAR": claim_median,
            "DELQ_%_YR3_5": delq_pct_3_5,
            "CLAIM_%_YR3_5": claim_pct_3_5
        })

    return pd.DataFrame(results)
