    df_psi[date_col]
    .astype(str)
    .str.replace(".0", "", regex=False)
    .str.strip()
)

# Extract year and month
df_psi["year"] = pd.to_numeric(
    df_psi[date_col].str[:4],
    errors="coerce"
)

df_psi["month"] = pd.to_numeric(
    df_psi[date_col].str[4:6],
    errors="coerce"
)

# Keep baseline year and later only
df_psi = df_psi[
    (df_psi["year"] >= baseline_year)
    & (df_psi["month"].between(1, 12))
].copy()

# Create quarter
df_psi["quarter"] = (
    ((df_psi["month"] - 1) // 3) + 1
).astype(int)

# 2021 is full-year baseline.
# Later years use their actual year and quarter.
df_psi["period"] = np.where(
    df_psi["year"] == baseline_year,
    str(baseline_year),
    (
        df_psi["year"].astype(int).astype(str)
        + "_Q"
        + df_psi["quarter"].astype(str)
    )
)
