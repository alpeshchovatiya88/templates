from sklearn.model_selection import train_test_split

def apply_custom_time_split(trainer, train_mask, valid_mask):
    d_train = trainer.df.loc[train_mask].copy()
    d_test = trainer.df.loc[valid_mask].copy()

    target_col = trainer.cfg.target_col
    drop_cols = trainer.cfg.drop_cols

    X_train = d_train.drop(columns=drop_cols + [target_col], errors="ignore")
    y_train = d_train[target_col]

    X_test = d_test.drop(columns=drop_cols + [target_col], errors="ignore")
    y_test = d_test[target_col]

    trainer._cache["X_train"] = X_train
    trainer._cache["X_test"] = X_test
    trainer._cache["y_train"] = y_train
    trainer._cache["y_test"] = y_test
    trainer._cache["d_train"] = d_train
    trainer._cache["d_test"] = d_test

    trainer._cache["X_full"] = pd.concat([X_train, X_test], axis=0)
    trainer._cache["y_full"] = pd.concat([y_train, y_test], axis=0)
    trainer._cache["d_full"] = pd.concat([d_train, d_test], axis=0)

    return trainer
