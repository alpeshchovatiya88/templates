# Always start from the original dataframe
df_psi = df.copy()

# Clean YYYYMM, including values such as 202201.0
df_psi["YYYYMM"] = (
    df_psi["YYYYMM"]
    .astype(str)
    .str.replace(".0", "", regex=False)
    .str.strip()
)

df_psi["year"] = pd.to_numeric(
    df_psi["YYYYMM"].str[:4],
    errors="coerce"
)

df_psi["month"] = pd.to_numeric(
    df_psi["YYYYMM"].str[4:6],
    errors="coerce"
)

# Keep baseline year and later only
baseline_year = 2021

df_psi = df_psi[
    (df_psi["year"] >= baseline_year)
    & (df_psi["month"].between(1, 12))
].copy()

df_psi["quarter"] = (
    ((df_psi["month"] - 1) // 3) + 1
).astype(int)

# Full-year baseline; quarterly periods afterward
df_psi["period"] = np.where(
    df_psi["year"] == baseline_year,
    str(baseline_year),
    (
        df_psi["year"].astype(int).astype(str)
        + "_Q"
        + df_psi["quarter"].astype(str)
    )
)

# Verify that records are assigned correctly
print(
    df_psi.groupby("period")
    .size()
    .sort_index()
)
