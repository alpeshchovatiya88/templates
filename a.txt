import pandas as pd
import numpy as np

# Make sure year buckets exist
df_new["DELQ_YEAR"] = np.ceil(df_new["TIME_TO_DELQ_YR"])
df_new["CLAIM_YEAR"] = np.ceil(df_new["TIME_TO_CLAIM_YR"])


def regional_event_summary(df):

    rows = []

    # Include ALL + each region
    region_list = ["ALL"] + sorted(df["ECON_REGION"].dropna().unique())

    for region in region_list:

        if region == "ALL":
            temp = df.copy()
        else:
            temp = df[df["ECON_REGION"] == region].copy()

        # -------------------------
        # Delinquency population
        # Claim takes priority
        # -------------------------
        delq = temp.loc[
            (temp["DELQ90_IND"] == 1.0) &
            (temp["CLAIM_IND"] != 1.0) &
            (temp["DELQ_YEAR"].between(1, 17))
        ].copy()

        # -------------------------
        # Claim population
        # -------------------------
        claim = temp.loc[
            (temp["CLAIM_IND"] == 1.0) &
            (temp["CLAIM_YEAR"].between(1, 17))
        ].copy()

        # Peak year
        delq_peak = (
            delq["DELQ_YEAR"].value_counts().idxmax()
            if len(delq) > 0 else np.nan
        )

        claim_peak = (
            claim["CLAIM_YEAR"].value_counts().idxmax()
            if len(claim) > 0 else np.nan
        )

        # Median event year
        delq_median = (
            delq["DELQ_YEAR"].median()
            if len(delq) > 0 else np.nan
        )

        claim_median = (
            claim["CLAIM_YEAR"].median()
            if len(claim) > 0 else np.nan
        )

        # Average deterministic HPI +4
        hpi_plus4_avg = temp["HPI_+4"].mean()

        # Counts
        delq_count = len(delq)
        claim_count = len(claim)

        rows.append({
            "ECON_REGION": region,
            "DELQ_COUNT": delq_count,
            "CLAIM_COUNT": claim_count,
            "DELQ_PEAK_YEAR": delq_peak,
            "CLAIM_PEAK_YEAR": claim_peak,
            "DELQ_MEDIAN_YEAR": delq_median,
            "CLAIM_MEDIAN_YEAR": claim_median,
            "AVG_HPI_+4": hpi_plus4_avg
        })

    return pd.DataFrame(rows)
