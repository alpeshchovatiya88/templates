import pandas as pd
import plotly.graph_objects as go

def plot_by_decile(dfs, decile_col, value_col, agg="sum", title=None):

    summary = pd.DataFrame({
        name: df.assign(
            _decile=pd.to_numeric(
                df[decile_col].astype(str).str.extract(r"(\d+)", expand=False),
                errors="coerce"
            )
        ).groupby("_decile")[value_col].agg(agg)
        for name, df in dfs.items()
    }).sort_index()

    fig = go.Figure()

    for name in summary.columns:
        fig.add_trace(go.Bar(
            x=summary.index,
            y=summary[name],
            name=name,
            text=summary[name].round(2),
            textposition="outside",
            hovertemplate=(
                "Decile: %{x}<br>"
                f"Seed: {name}<br>"
                "Value: %{y:,.2f}<extra></extra>"
            )
        ))

    buttons = [
        dict(
            label="All Deciles",
            method="relayout",
            args=[{"xaxis.range": [0.5, 20.5]}]
        )
    ]

    for d in summary.index:
        buttons.append(dict(
            label=f"Decile {int(d)}",
            method="relayout",
            args=[{"xaxis.range": [d - 0.5, d + 0.5]}]
        ))

    fig.update_layout(
        title=title or f"{value_col} by Decile Across Random Seeds",
        xaxis_title="Decile",
        yaxis_title=value_col,
        barmode="group",
        height=600,
        margin=dict(t=110),
        updatemenus=[dict(
            buttons=buttons,
            direction="down",
            x=1,
            xanchor="right",
            y=1.18,
            yanchor="top"
        )]
    )

    fig.update_xaxes(
        tickmode="array",
        tickvals=summary.index,
        type="linear"
    )

    fig.update_yaxes(rangemode="tozero")
    fig.show()

    return summary
