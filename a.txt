
# ============================================================
# 1. IMPORT LIBRARIES
# ============================================================

import numpy as np
import pandas as pd
import joblib
import matplotlib.pyplot as plt

from sklearn.calibration import (
    CalibratedClassifierCV,
    calibration_curve
)
from sklearn.frozen import FrozenEstimator
from sklearn.metrics import (
    roc_auc_score,
    brier_score_loss,
    log_loss
)

# Requires scikit-learn >= 1.6


# ============================================================
# 2. LOAD DATASETS
# ============================================================

df_train_file = "df_train_scored.csv"
df_test_file = "df_test_scored.csv"
df_val_file = "df_val_scored.csv"

df_train = pd.read_csv(df_train_file, low_memory=False)
df_test = pd.read_csv(df_test_file, low_memory=False)
df_val = pd.read_csv(df_val_file, low_memory=False)

print("Train:", df_train.shape)
print("Test:", df_test.shape)
print("Validation:", df_val.shape)


# ============================================================
# 3. LOAD TRAINED XGBOOST MODEL
# ============================================================

xgb_model = joblib.load("xgb.joblib")

print("Loaded model:", type(xgb_model))


# ============================================================
# 4. DEFINE TARGET AND FEATURES
# ============================================================

# Replace with your actual binary target column
target_col = "BAD_FLAG"

# IMPORTANT:
# Features must match the features used to train XGBoost,
# including their names, order and preprocessing.

# Prefer feature names stored in the fitted model.
if hasattr(xgb_model, "feature_names_in_"):
    feature_cols = list(xgb_model.feature_names_in_)
else:
    booster = xgb_model.get_booster()
    feature_cols = booster.feature_names

    if feature_cols is None:
        raise ValueError(
            "Feature names unavailable. Supply the exact "
            "original training feature list in its original order."
        )

print("Number of features:", len(feature_cols))

for name, df in [
    ("Train", df_train),
    ("Test", df_test),
    ("Validation", df_val)
]:
    missing = [
        c for c in feature_cols + [target_col]
        if c not in df.columns
    ]

    if missing:
        raise ValueError(
            f"{name} is missing columns: {missing}"
        )

X_train = df_train[feature_cols]
y_train = df_train[target_col]

X_test = df_test[feature_cols]
y_test = df_test[target_col]

X_val = df_val[feature_cols]
y_val = df_val[target_col]

# Validate binary labels and observed outcomes
for name, X, y in [
    ("Train", X_train, y_train),
    ("Test", X_test, y_test),
    ("Validation", X_val, y_val)
]:
    if y.isna().any() or not y.isin([0, 1]).all():
        raise ValueError(
            f"{name} target must contain observed 0/1 labels."
        )

    if X.isna().all(axis=1).any():
        print(
            f"Warning: {name} has rows with all features missing."
        )

if y_test.nunique() != 2:
    raise ValueError(
        "Calibration data must contain both target classes."
    )


# ============================================================
# 5. GET ORIGINAL XGBOOST PROBABILITIES
# ============================================================

df_train["PD_ORIGINAL"] = (
    xgb_model.predict_proba(X_train)[:, 1]
)

df_test["PD_ORIGINAL"] = (
    xgb_model.predict_proba(X_test)[:, 1]
)

df_val["PD_ORIGINAL"] = (
    xgb_model.predict_proba(X_val)[:, 1]
)


# ============================================================
# 6. FIT ISOTONIC CALIBRATION
# ============================================================

# FrozenEstimator prevents XGBoost from being retrained.
# Calibration is fitted using 2021-2023 data only.

calibrated_clf = CalibratedClassifierCV(
    estimator=FrozenEstimator(xgb_model),
    method="isotonic"
)

calibrated_clf.fit(X_test, y_test)

print("Calibration fitted successfully.")


# ============================================================
# 7. GENERATE CALIBRATED PROBABILITIES
# ============================================================

df_train["PD_CALIBRATED"] = (
    calibrated_clf.predict_proba(X_train)[:, 1]
)

df_test["PD_CALIBRATED"] = (
    calibrated_clf.predict_proba(X_test)[:, 1]
)

df_val["PD_CALIBRATED"] = (
    calibrated_clf.predict_proba(X_val)[:, 1]
)


# ============================================================
# 8. COMPARE PERFORMANCE
# ============================================================

def evaluate(df, name):

    y = df[target_col]
    original = df["PD_ORIGINAL"]
    calibrated = df["PD_CALIBRATED"]

    results = {
        "Dataset": name,
        "Actual_Bad_Rate": y.mean(),
        "Original_Mean_PD": original.mean(),
        "Calibrated_Mean_PD": calibrated.mean(),
        "Original_AUC": roc_auc_score(y, original),
        "Calibrated_AUC": roc_auc_score(y, calibrated),
        "Original_Brier": brier_score_loss(y, original),
        "Calibrated_Brier": brier_score_loss(y, calibrated),
        "Original_LogLoss": log_loss(y, original),
        "Calibrated_LogLoss": log_loss(y, calibrated)
    }

    return results


results = pd.DataFrame([
    evaluate(df_train, "TRAIN_2009_2015"),
    evaluate(df_test, "TEST_2021_2023"),
    evaluate(df_val, "VAL_202510_202605")
])

print("\nCALIBRATION RESULTS")
print(results.to_string(index=False))


# ============================================================
# 9. PLOT CALIBRATION CURVES
# ============================================================

def plot_calibration(df, name):

    y = df[target_col]

    plt.figure(figsize=(8, 6))

    for col, label in [
        ("PD_ORIGINAL", "Original XGBoost"),
        ("PD_CALIBRATED", "Calibrated XGBoost")
    ]:

        prob_true, prob_pred = calibration_curve(
            y,
            df[col],
            n_bins=10,
            strategy="quantile"
        )

        plt.plot(
            prob_pred,
            prob_true,
            marker="o",
            label=label
        )

    plt.plot(
        [0, 1],
        [0, 1],
        "k--",
        label="Perfect Calibration"
    )

    plt.xlabel("Mean Predicted Probability")
    plt.ylabel("Observed Bad Rate")
    plt.title(name + " Calibration Curve")
    plt.legend()
    plt.grid(True)
    plt.show()


plot_calibration(df_train, "TRAIN")
plot_calibration(df_test, "TEST")
plot_calibration(df_val, "VALIDATION")


# ============================================================
# 10. SAVE CALIBRATED DATASETS
# ============================================================

df_train.to_csv(
    "df_train_calibrated.csv",
    index=False
)

df_test.to_csv(
    "df_test_calibrated.csv",
    index=False
)

df_val.to_csv(
    "df_val_calibrated.csv",
    index=False
)


# ============================================================
# 11. SAVE CALIBRATED MODEL
# ============================================================

joblib.dump(
    calibrated_clf,
    "xgb_isotonic_calibrated.joblib"
)

results.to_csv(
    "calibration_performance.csv",
    index=False
)

print("Calibration completed successfully.")
