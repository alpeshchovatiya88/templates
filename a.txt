bar_idx = 0

for container in ax.containers:

    labels = []

    for bar in container:

        height = bar.get_height()

        if bar_idx < len(agg_df):

            row = agg_df.iloc[bar_idx]

            total = int(row["total_count"])

            count = int(row["bad_count"])

            labels.append(
                f"T: {total}\nB: {count}\n({height:.1%})"
            )

        else:

            labels.append("")

        bar_idx += 1

    ax.bar_label(
        container,
        labels=labels,
        padding=4,
        fontsize=7,
        fontweight="bold",
        rotation=0
    )
