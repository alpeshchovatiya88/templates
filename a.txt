
import os
import sys
import argparse
from types import SimpleNamespace

import joblib
import numpy as np
import pandas as pd

# Your ORIGINAL shap_plots.py
import shap_plots


def run_shap(
    model_path,
    data_path,
    filter_col=None,
    out_dir="shap_results",
    model_name="xgb",
    shap_sample=10000,
    shap_top_k=5,
    shap_dep_enabled=True,
    shap_interaction_index=None,
    shap_min_rows=200,
    shap_max_levels=12,
    random_state=42,
    reproduce_train_order=False,
):
    """
    Standalone SHAP analysis using the original make_shap().

    Does NOT train or refit the model.

    Reuses the original:
        - SHAP explainer logic
        - SHAP sampling
        - One-hot grouping
        - SHAP importance calculations
        - Row-level SHAP exports
        - Top 5 predictors
        - Summary plot
        - Dependence plots

    Parameters
    ----------
    model_path:
        Saved fitted XGBoost pipeline (.joblib).

    data_path:
        CSV containing the same feature-engineered model
        inputs used in the original training run.

    filter_col:
        Exactly one optional subgroup column.

    reproduce_train_order:
        If True, sort by APPLICATION_NUMBER and shuffle
        using the original train.py sequence.

        Use only if your input CSV represents the correct
        stage of your original training data processing.
    """

    os.makedirs(out_dir, exist_ok=True)

    # =====================================================
    # 1. LOAD THE EXISTING TRAINED MODEL
    # =====================================================

    print("=" * 60)
    print("LOADING TRAINED XGBOOST PIPELINE")
    print("=" * 60)

    if not os.path.exists(model_path):
        raise FileNotFoundError(
            f"Model not found: {model_path}"
        )

    pipeline = joblib.load(model_path)

    if not hasattr(pipeline, "named_steps"):
        raise TypeError(
            "Saved object must be a fitted sklearn Pipeline."
        )

    if "pre" not in pipeline.named_steps:
        raise KeyError(
            "Pipeline is missing preprocessing step 'pre'."
        )

    if "model" not in pipeline.named_steps:
        raise KeyError(
            "Pipeline is missing model step 'model'."
        )

    pre = pipeline.named_steps["pre"]
    model = pipeline.named_steps["model"]

    print("Model:", type(model).__name__)

    # =====================================================
    # 2. LOAD INPUT DATA
    # =====================================================

    print("\nLOADING INPUT DATA")

    if not os.path.exists(data_path):
        raise FileNotFoundError(
            f"Input data not found: {data_path}"
        )

    df = pd.read_csv(data_path)

    if df.empty:
        raise ValueError("Input dataset is empty.")

    if "APPLICATION_NUMBER" not in df.columns:
        raise KeyError(
            "APPLICATION_NUMBER is required."
        )

    if filter_col is not None:

        if filter_col not in df.columns:
            raise KeyError(
                f"Filter column '{filter_col}' not found."
            )

    print("Input rows:", len(df))
    print("Input columns:", len(df.columns))

    # =====================================================
    # 3. OPTIONAL ORIGINAL TRAINING ROW ORDER
    # =====================================================

    if reproduce_train_order:

        print("\nREPRODUCING TRAINING ROW ORDER")

        df = (
            df.sort_values(
                "APPLICATION_NUMBER",
                kind="mergesort"
            )
            .reset_index(drop=True)
        )

        df = (
            df.sample(
                frac=1,
                random_state=random_state
            )
            .reset_index(drop=True)
        )

    # =====================================================
    # 4. RESTORE ORIGINAL MODEL FEATURES
    # =====================================================

    # The saved fitted preprocessor knows the raw
    # predictor columns used during model training.

    model_features = list(pre.feature_names_in_)

    missing = [
        col
        for col in model_features
        if col not in df.columns
    ]

    if missing:
        raise ValueError(
            "Input data is missing trained model features:\n"
            f"{missing}"
        )

    print("\nRaw model predictors:", len(model_features))
    print("Predictors:", model_features)

    # Only original trained features are treated
    # as model predictors.
    X_shap = df[model_features].copy()

    # =====================================================
    # 5. PRESERVE APPLICATION NUMBER AND ONE FILTER
    # =====================================================

    # The original make_shap() requires
    # APPLICATION_NUMBER inside its X_train argument.

    # Extra identifier/filter columns are safe only
    # when the fitted ColumnTransformer selects its
    # trained feature columns and ignores extras.

    if "APPLICATION_NUMBER" not in X_shap.columns:

        X_shap.insert(
            0,
            "APPLICATION_NUMBER",
            df["APPLICATION_NUMBER"].to_numpy()
        )

    if filter_col is not None:

        if filter_col not in X_shap.columns:

            X_shap[filter_col] = (
                df[filter_col].to_numpy()
            )

    # Check that extra metadata columns do not change
    # the fitted preprocessor's transformed feature count.
    check_n = min(5, len(X_shap))

    transformed_check = pre.transform(
        X_shap.iloc[:check_n]
    )

    expected_names = list(
        pre.get_feature_names_out()
    )

    if transformed_check.shape[1] != len(expected_names):
        raise ValueError(
            "Transformed features do not match "
            "saved preprocessor feature names."
        )

    # =====================================================
    # 6. USE EXACT FEATURE-NAME LOGIC FROM train.py
    # =====================================================

    feature_names = []

    for name in expected_names:

        clean_name = name.split("__", 1)[-1]

        feature_names.append(clean_name)

    print(
        "Encoded model features:",
        len(feature_names)
    )

    # =====================================================
    # 7. ORIGINAL SHAP CONFIGURATION
    # =====================================================

    cfg = SimpleNamespace()

    cfg.shap_sample = int(shap_sample)

    cfg.shap_top_k = int(shap_top_k)

    cfg.shap_dep_enabled = bool(
        shap_dep_enabled
    )

    cfg.shap_interaction_index = (
        shap_interaction_index
    )

    cfg.random_state = int(random_state)

    cfg.shap_min_rows = int(shap_min_rows)

    cfg.shap_max_levels = int(shap_max_levels)

    # =====================================================
    # 8. ONE FILTER ONLY
    # =====================================================

    if filter_col is None:

        subgroup_cols = []

    else:

        subgroup_cols = [filter_col]

    print("\nSHAP Configuration:")
    print("Sample:", cfg.shap_sample)
    print("Random state:", cfg.random_state)
    print("Filter:", filter_col)
    print("Min subgroup rows:", cfg.shap_min_rows)
    print("Max subgroup levels:", cfg.shap_max_levels)

    # =====================================================
    # 9. USE ORIGINAL SHAP FUNCTION
    # =====================================================

    print("\n" + "=" * 60)
    print("RUNNING ORIGINAL SHAP LOGIC")
    print("=" * 60)

    # Original shap_plots.py calls display().
    # Make display available in a standalone script.
    try:
        from IPython.display import display
        shap_plots.display = display
    except ImportError:
        shap_plots.display = print

    # This calls your ORIGINAL implementation without
    # replacing its explainer, grouping, or sampling.
    shap_plots.make_shap(
        pipeline=pipeline,
        X_train=X_shap,
        out_dir=out_dir,
        cfg=cfg,
        model_name=model_name,
        feat_names=feature_names,
        subgroup_cols=subgroup_cols,
        one_hot_group=True,
    )

    # =====================================================
    # 10. LOAD ORIGINAL SHAP FEATURE IMPORTANCE OUTPUT
    # =====================================================

    importance_path = os.path.join(
        out_dir,
        f"shap_feature_importance_{model_name}.csv"
    )

    importance = pd.read_csv(importance_path)

    print("\n" + "=" * 60)
    print("SHAP COMPLETED")
    print("=" * 60)

    print("\nFeature importance file:")
    print(importance_path)

    print("\nSubgroups:")
    print(
        importance["Sub-Group"]
        .drop_duplicates()
        .tolist()
    )

    print("\nOverall Top Predictors:")

    overall = importance.loc[
        importance["Sub-Group"] == "All"
    ]

    print(overall.head(12).to_string(index=False))

    return importance


# ============================================================
# 11. RUN STANDALONE
# ============================================================

if __name__ == "__main__":

    result = run_shap(

        # EXISTING TRAINED MODEL
        model_path="best_xgb.joblib",

        # PREPARED INPUT DATA
        data_path="train.csv",

        # ONE FILTER
        filter_col="ECON_REGION",

        # OUTPUT LOCATION
        out_dir="shap_results",

        model_name="xgb",

        # MUST MATCH ORIGINAL TRAINING CONFIG
        shap_sample=10000,
        shap_top_k=5,
        shap_dep_enabled=True,
        shap_interaction_index=None,
        shap_min_rows=200,
        shap_max_levels=12,
        random_state=42,

        # Set True only if you need to recreate the
        # original initial sort/shuffle from train.py.
        reproduce_train_order=False,
    )
