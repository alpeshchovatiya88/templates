def psi_by_period(
    df,
    period_col,
    periods,
    numeric_cols,
    categorical_cols
):
    results = []

    for i in range(1, len(periods)):
        previous_period = periods[i - 1]
        current_period = periods[i]

        expected = df[df[period_col] == previous_period]
        actual = df[df[period_col] == current_period]

        if expected.empty or actual.empty:
            print(f"Missing data: {previous_period} vs {current_period}")
            continue

        temp = run_psi_for_model(
            expected=expected,
            actual=actual,
            model_name="funded_model",
            numeric_cols=numeric_cols,
            categorical_cols=categorical_cols
        )

        temp["period"] = current_period
        results.append(temp)

    result = pd.concat(results, ignore_index=True)

    # PSI values with periods as columns
    psi_wide = result.pivot(
        index="feature",
        columns="period",
        values="psi"
    ).reset_index()

    # Percentage change in PSI value
    period_columns = [
        col for col in periods
        if col in psi_wide.columns
    ]

    psi_pct_diff = psi_wide.copy()

    psi_pct_diff[period_columns] = (
        psi_pct_diff[period_columns]
        .pct_change(axis=1)
        .mul(100)
    )

    return psi_wide, psi_pct_diff
