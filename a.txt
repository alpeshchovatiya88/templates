def population_stability_index(expected, actual, buckets=10, eps=1e-4):

    # Numeric
    if pd.api.types.is_numeric_dtype(expected):
        data_type = "numeric"

        e = pd.to_numeric(expected, errors="coerce")
        a = pd.to_numeric(actual, errors="coerce")

        clean = e.dropna()

        if clean.nunique() < 2:
            return np.nan, data_type, "Not calculated"

        _, edges = pd.qcut(
            clean,
            q=buckets,
            retbins=True,
            duplicates="drop"
        )

        edges[0], edges[-1] = -np.inf, np.inf

        e = pd.cut(e, edges, include_lowest=True).astype("string").fillna("MISSING")
        a = pd.cut(a, edges, include_lowest=True).astype("string").fillna("MISSING")

    # Categorical
    else:
        data_type = "categorical"

        e = expected.astype("string").fillna("MISSING")
        a = actual.astype("string").fillna("MISSING")

    # Distribution
    levels = e.unique().tolist()

    for x in a.unique():
        if x not in levels:
            levels.append(x)

    e_pct = e.value_counts(normalize=True).reindex(levels, fill_value=0)
    a_pct = a.value_counts(normalize=True).reindex(levels, fill_value=0)

    e_pct = e_pct.replace(0, eps)
    a_pct = a_pct.replace(0, eps)

    # PSI
    psi = float(
        ((a_pct - e_pct) * np.log(a_pct / e_pct)).sum()
    )

    # Interpretation
    if psi < 0.10:
        interpretation = "Stable"
    elif psi < 0.25:
        interpretation = "Moderate shift"
    else:
        interpretation = "Significant shift"

    return psi, data_type, interpretation
