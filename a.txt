import plotly.express as px

bookyear_bad = (
    df.groupby("BOOKYEAR")
      .agg(
          total=("target", "size"),
          bads=("target", "sum"),
          bad_rate=("target", "mean")
      )
      .reset_index()
)

bookyear_bad["bad_rate_pct"] = (
    bookyear_bad["bad_rate"] * 100
).round(2)

bookyear_bad["Label"] = (
    "Total: " + bookyear_bad["total"].map("{:,.0f}".format) +
    "<br>Bads: " + bookyear_bad["bads"].map("{:,.0f}".format) +
    "<br>Bad Rate: " + bookyear_bad["bad_rate_pct"].map("{:.2f}%".format)
)

fig = px.line(
    bookyear_bad,
    x="BOOKYEAR",
    y="bad_rate_pct",
    markers=True,
    text="Label",
    hover_data=["total", "bads", "bad_rate_pct"],
    title="Bad Rate by BOOKYEAR"
)

fig.update_traces(
    textposition="top center",
    marker=dict(size=8)
)

fig.update_layout(
    xaxis_title="BOOKYEAR",
    yaxis_title="Bad Rate (%)",
    yaxis_ticksuffix="%",
    hovermode="x unified"
)

fig.show()
