def export_interactive_seed_report(
    dfs,
    filename="Random_Seed_Sensitivity_Report.html",
    decile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    metrics=None,
    baseline="RS42",
    claims_share=0.62,
    colors=None,
    decimals=2,
    inside_font_size=13,
    chart_height=950
):
    """
    Export an offline interactive random seed sensitivity report.

    Inside each bar:
      - Min, Max, Avg, P10, P50, P90
      - Difference versus baseline
      - Predicted Delq + Claims
      - Predicted Claims
      - Predicted Delqs

    Filters:
      - Metric
      - Decile
      - ECON_REGION
      - POD_MARKET_CODE
      - PRK_TENURE

    The function does not modify the input DataFrames.
    """

    if baseline not in dfs:
        raise ValueError(f"Baseline {baseline} not found in dfs")

    if not 0 <= claims_share <= 1:
        raise ValueError("claims_share must be between 0 and 1")

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            },
            "Predicted_n_Delq_and_Claims": {
                "agg": "sum",
                "is_ratio": False
            },
            "Predicted_n_Claims": {
                "agg": "sum",
                "is_ratio": False
            },
            "Predicted_n_Delqs": {
                "agg": "sum",
                "is_ratio": False
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    predicted_metric_names = [
        "Predicted_n_Delq_and_Claims",
        "Predicted_n_Claims",
        "Predicted_n_Delqs"
    ]

    for metric, config in metrics.items():
        if config["agg"] not in ("sum", "mean"):
            raise ValueError(
                f"{metric}: agg must be 'sum' or 'mean'"
            )

    records = []
    decile_values = set()
    region_values = set()
    market_values = set()
    tenure_values = set()

    # --------------------------------------------------------
    # Prepare each seed
    # --------------------------------------------------------

    for seed, source_df in dfs.items():

        required = [
            decile_col,
            region_col,
            market_col,
            tenure_col,
            probability_col
        ]

        required += [
            m for m in metrics
            if m not in predicted_metric_names
        ]

        missing = [
            col for col in required
            if col not in source_df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed} is missing columns: {missing}"
            )

        df = source_df.copy()

        probability = pd.to_numeric(
            df[probability_col],
            errors="coerce"
        )

        # Predicted expected counts per record
        df["Predicted_n_Delq_and_Claims"] = probability
        df["Predicted_n_Claims"] = (
            probability * claims_share
        )
        df["Predicted_n_Delqs"] = (
            probability * (1 - claims_share)
        )

        deciles = pd.to_numeric(
            df[decile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        regions = (
            df[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        markets = (
            df[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        tenures = (
            df[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        decile_values.update(
            deciles.dropna().astype(int).tolist()
        )
        region_values.update(regions.tolist())
        market_values.update(markets.tolist())
        tenure_values.update(tenures.tolist())

        # One record per observation per metric.
        # Format:
        # [seed, decile, region, market, tenure, metric, value]

        for metric in metrics:

            if metric not in df.columns:
                raise ValueError(
                    f"{seed}: metric {metric} not available"
                )

            values = (
                pd.to_numeric(df[metric], errors="coerce")
                .replace([np.inf, -np.inf], np.nan)
            )

            valid = values.notna()

            for idx in np.flatnonzero(valid.to_numpy()):

                d = deciles.iloc[idx]

                records.append([
                    str(seed),
                    int(d) if pd.notna(d) else None,
                    str(regions.iloc[idx]),
                    str(markets.iloc[idx]),
                    str(tenures.iloc[idx]),
                    metric,
                    float(values.iloc[idx])
                ])

    report_data = {
        "seeds": list(dfs.keys()),
        "baseline": baseline,
        "metrics": metrics,
        "colors": colors,
        "deciles": sorted(decile_values),
        "regions": sorted(region_values),
        "markets": sorted(market_values),
        "tenures": sorted(tenure_values),
        "records": records,
        "decimals": decimals,
        "fontSize": inside_font_size,
        "chartHeight": chart_height
    }

    report_json = json.dumps(
        report_data,
        ensure_ascii=False,
        allow_nan=False,
        separators=(",", ":")
    ).replace("<", "\\u003c")

    html = r"""

function number(v, digits=report.decimals) {
    if (v === null || !Number.isFinite(v)) return "N/A";
    return v.toLocaleString("en-US", {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
    });
}

function metricNumber(v, metric) {
    if (v === null || !Number.isFinite(v)) return "N/A";

    const cfg = report.metrics[metric];

    if (cfg.is_ratio) {
        return number(
            v / (cfg.ratio_scale || 1) * 100
        ) + "%";
    }

    return number(v);
}

function deltaNumber(v, metric) {
    if (v === null || !Number.isFinite(v)) return "N/A";

    const cfg = report.metrics[metric];
    const sign = v > 0 ? "+" : "";

    if (cfg.is_ratio) {
        return sign + number(
            v / (cfg.ratio_scale || 1) * 100
        ) + " pp";
    }

    return sign + number(v);
}

function pctNumber(v) {
    if (v === null || !Number.isFinite(v)) return "N/A";
    return (v > 0 ? "+" : "") + number(v) + "%";
}

function quantile(sorted, p) {
    if (!sorted.length) return null;
    if (sorted.length === 1) return sorted[0];

    const position = (sorted.length - 1) * p;
    const low = Math.floor(position);
    const high = Math.ceil(position);
    const weight = position - low;

    return sorted[low] * (1 - weight) +
           sorted[high] * weight;
}

function stats(values, agg) {

    if (!values.length) {
        return {
            count: 0,
            total: null,
            min: null,
            max: null,
            avg: null,
            p10: null,
            p50: null,
            p90: null
        };
    }

    const sorted = [...values].sort((a, b) => a - b);
    const sum = values.reduce((a, b) => a + b, 0);
    const avg = sum / values.length;

    return {
        count: values.length,
        total: agg === "sum" ? sum : avg,
        min: sorted[0],
        max: sorted[sorted.length - 1],
        avg: avg,
        p10: quantile(sorted, 0.10),
        p50: quantile(sorted, 0.50),
        p90: quantile(sorted, 0.90)
    };
}

function update() {

    const metric = $("metric").value;
    const cfg = report.metrics[metric];

    const decile = $("decile").value === ""
        ? null

    // Filter once, retaining all metrics.

        (decile === null || r[1] === decile) &&
        (region === "" || r[2] === region) &&
        (market === "" || r[3] === market) &&

    );

    const grouped = {};

        grouped[seed] = {};

            grouped[seed][m] = [];

        grouped[r[0]][r[5]].push(r[6]);

    const result = {};

    report.seeds.forEach(seed => {

            grouped[seed][metric],

        const delta = (
            s.total !== null &&
            baselineValue !== null

            delta !== null &&
            baselineValue !== 0

            : null;

        return {
            seed,
            ...s,
            delta,
            deltaPct
        };
    });

    // Chart

            p["Predicted_n_Claims"];

            p["Predicted_n_Delqs"];

        const deltaText = r.seed === report.baseline

                number(predictedClaims) + "</b>",
            "<b>Pred Delqs: " +
                number(predictedDelqs) + "</b>",
            "────────────",
            "<b>Δ: " + deltaText + "</b>"
        ];

            lines.push("<b>Δ%: " + pctText + "</b>");

        traces.push({
            type: "bar",
