
import pandas as pd
import matplotlib.pyplot as plt

def plot_hpi_training(df):

    data = df.copy()

    data["BOOKYEAR"] = pd.to_numeric(
        data["BOOKYEAR"], errors="coerce"
    )

    data["HPI_+4"] = pd.to_numeric(
        data["HPI_+4"], errors="coerce"
    )

    # Filter training period
    train = data[
        data["BOOKYEAR"].between(2009, 2015)
    ].copy()

    fig, ax = plt.subplots(figsize=(12, 6))

    # Individual regions - dotted lines
    for region, region_df in train.groupby("ECON_REGION"):

        region_hpi = (
            region_df.groupby("BOOKYEAR")["HPI_+4"]
            .mean()
            .sort_index()
        )

        ax.plot(
            region_hpi.index,
            region_hpi.values,
            linestyle=":",
            linewidth=1.5,
            alpha=0.75,
            label=str(region)
        )

    # Overall training average - solid blue
    overall_hpi = (
        train.groupby("BOOKYEAR")["HPI_+4"]
        .mean()
        .sort_index()
    )

    ax.plot(
        overall_hpi.index,
        overall_hpi.values,
        color="blue",
        linestyle="-",
        linewidth=3,
        marker="o",
        label="Overall Training HPI",
        zorder=10
    )

    ax.set_title(
        "HPI +4 - Training (2009-2015)"
    )

    ax.set_xlabel("BOOKYEAR")
    ax.set_ylabel("HPI_+4")

    ax.set_xticks(range(2009, 2016))
    ax.grid(alpha=0.3)

    ax.legend(
        title="Economic Region",
        bbox_to_anchor=(1.02, 1),
        loc="upper left"
    )

    plt.tight_layout()
    plt.show()
