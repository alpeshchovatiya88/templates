
import pandas as pd
import numpy as np
import plotly.graph_objects as go
import ipywidgets as widgets
from IPython.display import display

def plot_seed_sensitivity(
    dfs,
    decile_col,
    value_col,
    region_col="ECON_REGION",
    agg="sum",
    baseline="RS42",
    decimals=2,
    title=None,
    colors=None,
    inside_font_size=16,
    total_font_size=18
):
    seeds = list(dfs)

    # Customize colors here, or pass a dictionary to the function
    if colors is None:
        colors = {
            "RS42":  "#264E86",   # Navy blue
            "RS07":  "#D97736",   # Burnt orange
            "RS101": "#21867A",   # Teal green
            "RS151": "#7856A6"    # Muted purple
        }

    fallback_colors = [
        "#264E86", "#D97736", "#21867A", "#7856A6"
    ]

    def fmt(v):
        if pd.isna(v):
            return "N/A"
        if abs(v) >= 1e6:
            return f"{v / 1e6:,.2f}M"
        if abs(v) >= 1e3:
            return f"{v / 1e3:,.2f}K"
        return f"{v:,.{decimals}f}"

    def signed(v):
        if pd.isna(v):
            return "N/A"
        return ("+" if v > 0 else "") + fmt(v)

    # Prepare data
    frames = []

    for seed, df in dfs.items():
        temp = df[[decile_col, value_col, region_col]].copy()

        temp["Seed"] = seed
        temp["Decile"] = pd.to_numeric(
            temp[decile_col].astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )
        temp[value_col] = pd.to_numeric(
            temp[value_col], errors="coerce"
        )

        frames.append(temp)

    data = pd.concat(frames, ignore_index=True)
    data = data.dropna(subset=["Decile"])
    data["Decile"] = data["Decile"].astype(int)

    deciles = sorted(data["Decile"].unique())
    regions = sorted(
        data[region_col].dropna().unique(), key=str
    )

    decile_filter = widgets.Dropdown(
        options=[("All Deciles", None)] +
                [(f"Decile {d}", d) for d in deciles],
        value=None,
        description="Decile:"
    )

    region_filter = widgets.Dropdown(
        options=[("All Regions", None)] +
                [(str(r), r) for r in regions],
        value=None,
        description="ECON_REGION:",
        style={"description_width": "initial"}
    )

    output = widgets.Output()

    def update_chart(change=None):
        with output:
            output.clear_output(wait=True)

            subset = data

            if region_filter.value is not None:
                subset = subset[
                    subset[region_col] == region_filter.value
                ]

            if decile_filter.value is not None:
                subset = subset[
                    subset["Decile"] == decile_filter.value
                ]

            stats = (
                subset.groupby(["Seed", "Decile"])[value_col]
                .agg(
                    total=agg,
                    min="min",
                    max="max",
                    avg="mean",
                    p10=lambda x: x.quantile(0.10),
                    p20=lambda x: x.quantile(0.20),
                    p50="median"
                )
                .reset_index()
            )

            base = (
                stats[stats["Seed"] == baseline]
                [["Decile", "total"]]
                .rename(columns={"total": "baseline_total"})
            )

            stats = stats.merge(
                base, on="Decile", how="left"
            )

            stats["difference"] = (
                stats["total"] - stats["baseline_total"]
            )

            stats["pct_difference"] = np.where(
                stats["baseline_total"].notna() &
                (stats["baseline_total"] != 0),
                stats["difference"] /
                stats["baseline_total"] * 100,
                np.nan
            )

            stats.loc[
                stats["Seed"] == baseline,
                ["difference", "pct_difference"]
            ] = 0

            selected_deciles = (
                [decile_filter.value]
                if decile_filter.value is not None
                else deciles
            )

            single_decile = decile_filter.value is not None
            fig = go.Figure()

            for i, seed in enumerate(seeds):
                s = (
                    stats[stats["Seed"] == seed]
                    .set_index("Decile")
                    .reindex(selected_deciles)
                )

                inside_text = []
                hover_text = []
                total_text = []

                for d, r in s.iterrows():
                    if pd.isna(r["total"]):
                        inside_text.append("")
                        hover_text.append("No data")
                        total_text.append("")
                        continue

                    diff = (
                        "Baseline" if seed == baseline
                        else signed(r["difference"])
                    )

                    pct = (
                        "Baseline" if seed == baseline
                        else (
                            f"{r['pct_difference']:+.2f}%"
                            if pd.notna(r["pct_difference"])
                            else "N/A"
                        )
                    )

                    details = (
                        f"<b>Min: {fmt(r['min'])}</b><br>"
                        f"<b>Max: {fmt(r['max'])}</b><br>"
                        f"<b>Avg: {fmt(r['avg'])}</b><br>"
                        f"<b>P10: {fmt(r['p10'])}</b><br>"
                        f"<b>P20: {fmt(r['p20'])}</b><br>"
                        f"<b>P50: {fmt(r['p50'])}</b><br>"
                        f"────────────<br>"
                        f"<b>Δ: {diff}</b><br>"
                        f"<b>Δ%: {pct}</b>"
                    )

                    inside_text.append(details)

                    hover_text.append(
                        f"<b>{seed} | Decile {d}</b><br>"
                        f"<b>Total SUM: "
                        f"{r['total']:,.{decimals}f}</b><br>"
                        + details
                    )

                    total_text.append(
                        f"<b>SUM: {fmt(r['total'])}</b>"
                    )

                # Main bar with statistics inside
                fig.add_trace(go.Bar(
                    x=[str(d) for d in selected_deciles],
                    y=s["total"],
                    name=seed,
                    marker=dict(
                        color=colors.get(
                            seed,
                            fallback_colors[i % len(fallback_colors)]
                        ),
                        line=dict(color="white", width=1.5)
                    ),
                    text=inside_text if single_decile else total_text,
                    textposition="inside" if single_decile else "outside",
                    insidetextanchor="middle",
                    textfont=dict(
                        size=inside_font_size if single_decile
                             else total_font_size,
                        color="white" if single_decile
                              else "#172033",
                        family="Arial"
                    ),
                    customdata=hover_text,
                    hovertemplate="%{customdata}<extra></extra>",
                    cliponaxis=False
                ))

                # Separate bold SUM label above each bar
                # in the individual-decile view
                if single_decile:
                    fig.add_trace(go.Scatter(
                        x=[str(d) for d in selected_deciles],
                        y=s["total"],
                        mode="text",
                        text=total_text,
                        textposition="top center",
                        textfont=dict(
                            size=total_font_size,
                            color="#172033",
                            family="Arial Black"
                        ),
                        showlegend=False,
                        hoverinfo="skip"
                    ))

            fig.update_layout(
                title=dict(
                    text=title or
                    f"{value_col} by Decile Across Random Seeds",
                    x=0.5,
                    font=dict(size=22, color="#172033")
                ),
                template="plotly_white",
                barmode="group",
                height=800,
                bargap=0.30,
                bargroupgap=0.08,
                plot_bgcolor="white",
                paper_bgcolor="white",
                font=dict(
                    family="Arial",
                    size=15,
                    color="#172033"
                ),
                legend=dict(
                    orientation="h",
                    y=1.10,
                    x=0.5,
                    xanchor="center",
                    font=dict(size=15)
                ),
                margin=dict(
                    t=130, b=80, l=100, r=60
                ),
                uniformtext=dict(
                    minsize=12,
                    mode="show"
                )
            )

            fig.update_xaxes(
                title=dict(text="Decile", font=dict(size=17)),
                type="category",
                categoryorder="array",
                categoryarray=[str(d) for d in deciles],
                tickfont=dict(size=14),
                showgrid=False
            )

            fig.update_yaxes(
                title=dict(text=value_col, font=dict(size=17)),
                tickformat="~s",
                tickfont=dict(size=14),
                gridcolor="#E7ECF2",
                zeroline=True
            )

            fig.show()

            # Numerical comparison table
            table = stats[
                ["Decile", "Seed", "total",
                 "difference", "pct_difference",
                 "min", "max", "avg",
                 "p10", "p20", "p50"]
            ].copy()

            display(
                table.style.format({
                    "total": f"{{:,.{decimals}f}}",
                    "difference": f"{{:+,.{decimals}f}}",
                    "pct_difference": "{:+.2f}%",
                    "min": f"{{:,.{decimals}f}}",
                    "max": f"{{:,.{decimals}f}}",
                    "avg": f"{{:,.{decimals}f}}",
                    "p10": f"{{:,.{decimals}f}}",
                    "p20": f"{{:,.{decimals}f}}",
                    "p50": f"{{:,.{decimals}f}}"
                }, na_rep="N/A")
            )

    decile_filter.observe(update_chart, names="value")
    region_filter.observe(update_chart, names="value")

    display(
        widgets.HBox([decile_filter, region_filter]),
        output
    )
    update_chart()
