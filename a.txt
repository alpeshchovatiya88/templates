# SECTION 4 — PERFORMANCE, ROC, PR AND THRESHOLDS
STATUS_COLUMN = "STATUS"   # Change to "STAUS" if needed

# Full scored population: used for total n.
test_all = frames["test"]
declined_all = test_all[STATUS_COLUMN].astype("string").str.contains(
    "declined", case=False, regex=False, na=False
)

# Labeled population: used for outcome metrics.
test = evaluation["test"]
declined = test[STATUS_COLUMN].astype("string").str.contains(
    "declined", case=False, regex=False, na=False
)

populations = {
    "train": frames["train"],
    "test with declined": test_all,
    "test without declined": test_all.loc[~declined_all],
    "val": frames["val"],
}
scored = {
    "test with declined": test,
    "test without declined": test.loc[~declined],
}
for name in ["train", "val"]:
    if name in evaluation:
        scored[name] = evaluation[name]

# 4A. Performance table
rows = []
for label, full in populations.items():
    d = scored.get(label)
    n_evaluated = len(d) if d is not None else 0

    row = {
        "population": label,
        "n": len(full),
        "n_evaluated": n_evaluated,
        "n_not_evaluated": len(full) - n_evaluated,
        "mean_pd_all": full["_p"].mean(),
    }

    if d is not None and not d.empty:
        row.update(performance(d))
        row["n"] = len(full)  # Preserve total population count.

    rows.append(row)

summary = pd.DataFrame(rows).set_index("population")
show_table(summary, "02_performance")

# 4B. ROC and precision–recall curves
fig, axes = plt.subplots(1, 2, figsize=(12, 4))

for label in populations:
    d = scored.get(label)
    if d is None or d["_y"].nunique() < 2:
        continue

    fpr, tpr, _ = roc_curve(d["_y"], d["_p"])
    precision, recall, _ = precision_recall_curve(d["_y"], d["_p"])

    axes[0].plot(
        fpr, tpr,
        label=f"{label} (AUC={roc_auc_score(d['_y'], d['_p']):.3f})"
    )
    axes[1].plot(recall, precision, label=label)

axes[0].plot([0, 1], [0, 1], "k--")
axes[0].set(title="ROC", xlabel="False positive rate", ylabel="Recall")
axes[1].set(
    title="Precision–recall", xlabel="Recall", ylabel="Precision"
)
for ax in axes:
    ax.legend(fontsize=8)

show_plot("02_discrimination_with_without_declined")

# 4C. Threshold sensitivity
threshold_rows = []
for label in populations:
    d = scored.get(label)
    if d is None or d.empty:
        continue

    for threshold in [.01, .025, .05, .10, .20, .50]:
        metrics = performance(d, threshold)
        metrics["n"] = len(populations[label])
        threshold_rows.append({
            "population": label,
            "threshold": threshold,
            "n_evaluated": len(d),
            **metrics,
        })

thresholds = pd.DataFrame(threshold_rows)
show_table(thresholds, "02_threshold_sensitivity")
