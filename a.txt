
import pandas as pd
import numpy as np
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side

# =====================================================
# 1. EXPERIMENTS - CHANGE THIS DICTIONARY
# =====================================================

dfs = {
    "HPI_+4": df_hpi_4,
    "HPI_+3": df_hpi_3,
    "HPI_+5": df_hpi_5,
}

DECILE_COL = "rr_model_decile"
LOSS_COL = "SMI3_LOSS"
LR_COL = "SMI3_LR"

LR_SCALE = 1  # Change to 100 if LR is stored as decimals

OUTPUT_FILE = "experiment_comparison.xlsx"

# =====================================================
# 2. CALCULATE METRICS FOR EACH EXPERIMENT
# =====================================================

results = []

for experiment, df in dfs.items():

    temp = df.copy()

    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL].astype(str).str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    temp[LOSS_COL] = pd.to_numeric(
        temp[LOSS_COL], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan)

    temp[LR_COL] = pd.to_numeric(
        temp[LR_COL], errors="coerce"
    ).replace([np.inf, -np.inf], np.nan)

    decile1 = temp[temp["_decile"] == 1]

    results.append({
        "Experiment": experiment,
        "Overall Loss": temp[LOSS_COL].sum(min_count=1),
        "Overall LR": temp[LR_COL].mean() * LR_SCALE,
        "Decile 1 Loss": decile1[LOSS_COL].sum(min_count=1),
        "Decile 1 LR": decile1[LR_COL].mean() * LR_SCALE
    })

summary = pd.DataFrame(results)

if summary.empty:
    raise ValueError("No experiments found in dfs.")

# =====================================================
# 3. MIN / MAX ACROSS ALL EXPERIMENTS
# =====================================================

overall = [
    summary["Overall Loss"].min(),
    summary["Overall Loss"].max(),
    summary["Overall LR"].min(),
    summary["Overall LR"].max()
]

decile1 = [
    summary["Decile 1 Loss"].min(),
    summary["Decile 1 Loss"].max(),
    summary["Decile 1 LR"].min(),
    summary["Decile 1 LR"].max()
]

# =====================================================
# 4. CREATE ONE EXCEL TABLE
# =====================================================

wb = Workbook()
ws = wb.active
ws.title = "Experiment Comparison"

yellow = PatternFill("solid", fgColor="FFFF00")
blue = PatternFill("solid", fgColor="DDEBF7")

thin = Side(style="thin", color="B7B7B7")
border = Border(
    left=thin,
    right=thin,
    top=thin,
    bottom=thin
)

# Start at row 4
START_ROW = 4

# Experiment comparison label
experiment_names = " vs ".join(dfs.keys())

# Optional heading in row 2, as in screenshot
ws.merge_cells("B2:E2")
ws["B2"] = f"{experiment_names} Experiments"
ws["B2"].font = Font(size=16, bold=True)
ws["B2"].alignment = Alignment(horizontal="center")

# Row 4: Yellow label and grouped headers
ws.cell(START_ROW, 1, "Experiments")
ws.cell(START_ROW, 1).fill = yellow
ws.cell(START_ROW, 1).font = Font(size=12, bold=True)

ws.merge_cells(
    start_row=START_ROW,
    start_column=2,
    end_row=START_ROW,
    end_column=3
)

ws.merge_cells(
    start_row=START_ROW,
    start_column=4,
    end_row=START_ROW,
    end_column=5
)

ws.cell(START_ROW, 2, "Loss range")
ws.cell(START_ROW, 4, "Loss ratio range")

for col in [2, 4]:
    cell = ws.cell(START_ROW, col)
    cell.font = Font(size=12, bold=True)
    cell.alignment = Alignment(horizontal="center")

# Row 5: Min / Max headers
ws.cell(START_ROW + 1, 1).fill = blue

for col, text in zip(
    [2, 3, 4, 5],
    ["min", "max", "min", "max"]
):
    cell = ws.cell(START_ROW + 1, col, text)
    cell.font = Font(size=12, bold=True)
    cell.alignment = Alignment(horizontal="center")

# Rows 6 and 7: Overall and Decile 1
for r, label, values in [
    (START_ROW + 2, "Overall", overall),
    (START_ROW + 3, "Decile 1", decile1)
]:

    ws.cell(r, 1, label)

    for col, value in enumerate(values, start=2):

        if pd.isna(value):
            value = None
        elif isinstance(value, (np.integer, np.floating)):
            value = value.item()

        cell = ws.cell(r, col, value)

        cell.number_format = (
            '$#,##0' if col in [2, 3] else '0.00'
        )

        cell.alignment = Alignment(horizontal="right")

# Borders around the table
for r in range(START_ROW, START_ROW + 4):
    for col in range(1, 6):
        ws.cell(r, col).border = border

# =====================================================
# 5. SAVE EXCEL
# =====================================================

wb.save(OUTPUT_FILE)

print(f"Saved: {OUTPUT_FILE}")

print("\nExperiment results:")
print(summary.to_string(index=False))

print("\nOverall min/max:", overall)
print("Decile 1 min/max:", decile1)
