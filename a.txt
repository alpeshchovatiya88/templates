def fill_pol_missing_values(
    df_pol,
    divide_actltv_by_100=True
):

    df_pol = df_pol.copy()

    # EFFECT_DAY_SID:
    # use RECEIVED_DAY_SID when -1 or missing
    effect_day = (
        df_pol["EFFECT_DAY_SID"]
        .replace(
            -1,
            np.nan
        )
    )

    df_pol["EFFECT_DAY_SID"] = (
        effect_day.fillna(
            df_pol["RECEIVED_DAY_SID"]
        )
    )

    df_pol["EFFECT_SID"] = (
        df_pol["EFFECT_DAY_SID"]
        // 100
    ).astype("Int64")


    # PRD_CONTRACT_RATE:
    # fill with PRD_QUAL_INT_RATE
    missing_rate = (
        df_pol[
            "PRD_CONTRACT_RATE"
        ].isna()
    )

    df_pol.loc[
        missing_rate,
        "filled_PRD_CONTRACT_RATE_by"
    ] = "PRD_QUAL_INT_RATE"

    df_pol["PRD_CONTRACT_RATE"] = (
        df_pol[
            "PRD_CONTRACT_RATE"
        ]
        .fillna(
            df_pol[
                "PRD_QUAL_INT_RATE"
            ]
        )
        .astype(float)
    )


    # POD_INSURANCE_IN_FORCE
    df_pol["ACTLTV_lgd"] = (
        df_pol["ACTLTV"] / 100
        if divide_actltv_by_100
        else df_pol["ACTLTV"]
    )

    missing_iif = (
        df_pol[
            "POD_INSURANCE_IN_FORCE"
        ].isna()
    )

    iif_from_purchase_price = (
        df_pol[
            "PRK_PURCHASE_PRICE"
        ].notna()
        & df_pol[
            "ACTLTV_lgd"
        ].notna()
    )

    df_pol.loc[
        missing_iif
        & iif_from_purchase_price,
        "filled_POD_INSURANCE_IN_FORCE_by"
    ] = (
        "PRK_PURCHASE_PRICE * ACTLTV_lgd"
    )

    df_pol.loc[
        missing_iif
        & ~iif_from_purchase_price,
        "filled_POD_INSURANCE_IN_FORCE_by"
    ] = "PRD_BASE_LOAN_AMT"

    df_pol[
        "POD_INSURANCE_IN_FORCE"
    ] = (
        df_pol[
            "POD_INSURANCE_IN_FORCE"
        ]
        .fillna(
            df_pol[
                "PRK_PURCHASE_PRICE"
            ]
            * df_pol[
                "ACTLTV_lgd"
            ]
        )
        .fillna(
            df_pol[
                "PRD_BASE_LOAN_AMT"
            ]
        )
        .astype(float)
    )


    # PRK_PROPERTY_VALUE
    property_value = (
        df_pol[
            "PRK_PROPERTY_VALUE"
        ]
        .where(
            df_pol[
                "PRK_PROPERTY_VALUE"
            ] > 1,
            np.nan
        )
    )

    missing_property_value = (
        property_value.isna()
    )

    df_pol.loc[
        missing_property_value
        & df_pol[
            "PRK_PURCHASE_PRICE"
        ].notna(),
        "filled_PRK_PROPERTY_VALUE_by"
    ] = "PRK_PURCHASE_PRICE"

    df_pol.loc[
        missing_property_value
        & df_pol[
            "PRK_PURCHASE_PRICE"
        ].isna()
        & df_pol[
            "PRK_DECLARED_VALUE"
        ].notna(),
        "filled_PRK_PROPERTY_VALUE_by"
    ] = "PRK_DECLARED_VALUE"

    df_pol[
        "PRK_PROPERTY_VALUE"
    ] = (
        property_value
        .fillna(
            df_pol[
                "PRK_PURCHASE_PRICE"
            ]
        )
        .fillna(
            df_pol[
                "PRK_DECLARED_VALUE"
            ]
        )
        .astype(float)
    )


    # PRK_PROPERTY_AGE
    missing_property_age = (
        df_pol[
            "PRK_PROPERTY_AGE"
        ].isna()
    )

    df_pol.loc[
        missing_property_age
        & df_pol[
            "AVM_YEAR_BUILT"
        ].notna(),
        "PRK_PROPERTY_AGE"
    ] = (
        df_pol["BOOKYEAR"]
        - df_pol[
            "AVM_YEAR_BUILT"
        ]
    )

    df_pol.loc[
        df_pol[
            "PRK_CONS_TYPE"
        ].eq("N"),
        "PRK_PROPERTY_AGE"
    ] = 1

    return df_pol
