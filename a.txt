import pandas as pd
import numpy as np

# ==================================================
# 1. READ CSV FILES
# ==================================================

dev = pd.read_csv("dev.csv", keep_default_na=False)
val = pd.read_csv("val.csv", keep_default_na=False)

# ==================================================
# 2. CLEAN COLUMN NAMES
# ==================================================

dev.columns = dev.columns.str.strip()
val.columns = val.columns.str.strip()

# ==================================================
# 3. DEFINE METRICS
# ==================================================

metrics = [
    "n",
    "ACTUAL_LOSS",
    "SMI3_LOSS",
    "ACTUAL_LR",
    "SMI3_LR",
    "Actual_n_Delq_and_Claims",
    "Predicted_n_Delq_and_Claims"
]

# ==================================================
# 4. FILTER MODEL VERSION (OPTIONAL)
# ==================================================

# Uncomment if comparing one specific model version

# dev = dev[dev["Model_Ver"] == "V3.0.52"].copy()
# val = val[val["Model_Ver"] == "V3.0.52"].copy()

# ==================================================
# 5. CLEAN BOOKYEAR
# ==================================================

for df in [dev, val]:

    df["BOOKYEAR"] = pd.to_numeric(
        df["BOOKYEAR"].astype(str).str.strip(),
        errors="coerce"
    )

# ==================================================
# 6. KEEP ONLY BOOKYEAR ROWS
# ==================================================

# Removes "-" and other non-Bookyear rows

dev = dev[
    dev["BOOKYEAR"].between(2009, 2015)
].copy()

val = val[
    val["BOOKYEAR"].between(2009, 2015)
].copy()

# ==================================================
# 7. SORT BY BOOKYEAR
# ==================================================

dev = dev.sort_values(
    "BOOKYEAR"
).reset_index(drop=True)

val = val.sort_values(
    "BOOKYEAR"
).reset_index(drop=True)

# Convert BOOKYEAR to integer
dev["BOOKYEAR"] = dev["BOOKYEAR"].astype(int)
val["BOOKYEAR"] = val["BOOKYEAR"].astype(int)

print("DEV BOOKYEARS:")
print(dev["BOOKYEAR"].tolist())

print("\nVAL BOOKYEARS:")
print(val["BOOKYEAR"].tolist())

# ==================================================
# 8. CHECK DUPLICATE BOOKYEARS
# ==================================================

if dev["BOOKYEAR"].duplicated().any():
    raise ValueError(
        "Duplicate BOOKYEAR values found in DEV"
    )

if val["BOOKYEAR"].duplicated().any():
    raise ValueError(
        "Duplicate BOOKYEAR values found in VAL"
    )

# ==================================================
# 9. KEEP REQUIRED COLUMNS
# ==================================================

columns = ["BOOKYEAR"] + metrics

dev = dev[columns].copy()
val = val[columns].copy()

# ==================================================
# 10. MERGE DEV AND VAL
# ==================================================

merged = pd.merge(
    dev,
    val,
    on="BOOKYEAR",
    how="outer",
    suffixes=("_DEV", "_VAL"),
    indicator=True,
    validate="one_to_one"
)

merged = merged.sort_values(
    "BOOKYEAR"
).reset_index(drop=True)

# ==================================================
# 11. CALCULATE METRIC DIFFERENCES
# ==================================================

tolerance = 1e-8

for metric in metrics:

    dev_col = f"{metric}_DEV"
    val_col = f"{metric}_VAL"

    # Convert metric values to numeric
    for col in [dev_col, val_col]:

        merged[col] = pd.to_numeric(
            merged[col]
            .astype("string")
            .str.replace(",", "", regex=False)
            .str.strip(),
            errors="coerce"
        )

    # Calculate absolute difference
    merged[f"{metric}_Difference"] = (
        merged[dev_col] - merged[val_col]
    ).abs()

    # Assign PASS / FAIL
    merged[f"{metric}_Status"] = np.where(
        (
            merged["_merge"].eq("both")
            &
            merged[f"{metric}_Difference"].notna()
            &
            (
                merged[f"{metric}_Difference"]
                <= tolerance
            )
        ),
        "PASS",
        "FAIL"
    )

# ==================================================
# 12. CREATE WIDE-FORMAT REPORT
# ==================================================

report_columns = ["BOOKYEAR"]

for metric in metrics:

    report_columns.extend([
        f"{metric}_DEV",
        f"{metric}_VAL",
        f"{metric}_Difference",
        f"{metric}_Status"
    ])

wide_report = merged[report_columns].copy()

# ==================================================
# 13. OVERALL STATUS BY BOOKYEAR
# ==================================================

status_cols = [
    f"{metric}_Status"
    for metric in metrics
]

wide_report["Overall_Status"] = np.where(
    wide_report[status_cols].eq("PASS").all(axis=1),
    "PASS",
    "FAIL"
)

# ==================================================
# 14. CREATE SIMPLE PASS/FAIL SUMMARY
# ==================================================

status_report = wide_report[
    ["BOOKYEAR"] + status_cols + ["Overall_Status"]
].copy()

status_report.columns = (
    ["BOOKYEAR"] + metrics + ["Overall_Status"]
)

# ==================================================
# 15. OVERALL REPRODUCIBILITY
# ==================================================

overall_result = (
    "PASS"
    if len(wide_report) > 0
    and wide_report["Overall_Status"].eq("PASS").all()
    else "FAIL"
)

# ==================================================
# 16. PRINT RESULTS
# ==================================================

print("\nPERFORMANCE REPRODUCIBILITY REPORT\n")

print(wide_report.to_string(index=False))

print("\nPASS/FAIL SUMMARY BY BOOKYEAR\n")

print(status_report.to_string(index=False))

print("\nOVERALL REPRODUCIBILITY:", overall_result)

# ==================================================
# 17. EXPORT TO EXCEL
# ==================================================

output_file = "Performance_Reproducibility_Report.xlsx"

with pd.ExcelWriter(
    output_file,
    engine="openpyxl"
) as writer:

    wide_report.to_excel(
        writer,
        sheet_name="Detailed Comparison",
        index=False
    )

    status_report.to_excel(
        writer,
        sheet_name="PASS FAIL Summary",
        index=False
    )

    merged.to_excel(
        writer,
        sheet_name="Merged Data",
        index=False
    )

print("\nReport saved successfully:", output_file)
