import plotly.graph_objects as go

def create_swapset_html(df, filename="swapset_by_region.html"):

    regions = sorted(df["ECON_REGION"].dropna().unique())

    fig = go.Figure()

    # -------------------------
    # ALL REGIONS
    # -------------------------
    swap = pd.crosstab(
        df["decile_hpi4_num"],
        df["decile_hpi3_num"],
        margins=True
    )

    fig.add_trace(
        go.Table(
            header=dict(
                values=["HPI +4"] + list(swap.columns)
            ),
            cells=dict(
                values=[
                    swap.index
                ] + [
                    swap[col] for col in swap.columns
                ]
            ),
            visible=True
        )
    )

    # -------------------------
    # EACH ECON REGION
    # -------------------------
    for region in regions:

        temp = df[df["ECON_REGION"] == region]

        swap = pd.crosstab(
            temp["decile_hpi4_num"],
            temp["decile_hpi3_num"],
            margins=True
        )

        fig.add_trace(
            go.Table(
                header=dict(
                    values=["HPI +4"] + list(swap.columns)
                ),
                cells=dict(
                    values=[
                        swap.index
                    ] + [
                        swap[col] for col in swap.columns
                    ]
                ),
                visible=False
            )
        )

    # -------------------------
    # DROPDOWN
    # -------------------------
    buttons = []

    buttons.append(
        dict(
            label="ALL",
            method="update",
            args=[
                {"visible": [True] + [False] * len(regions)},
                {"title": "Swapset: HPI +4 vs HPI +3 — ALL"}
            ]
        )
    )

    for i, region in enumerate(regions):

        visible = [False] * (len(regions) + 1)
        visible[i + 1] = True

        buttons.append(
            dict(
                label=str(region),
                method="update",
                args=[
                    {"visible": visible},
                    {
                        "title":
                        f"Swapset: HPI +4 vs HPI +3 — {region}"
                    }
                ]
            )
        )

    fig.update_layout(
        title="Swapset: HPI +4 vs HPI +3 — ALL",

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,
                x=0,
                y=1.15
            )
        ]
    )

    # Save interactive HTML
    fig.write_html(filename)

    fig.show()
