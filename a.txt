import pandas as pd
import numpy as np

# ==================================================
# 1. READ CSV FILES
# ==================================================

dev = pd.read_csv("dev.csv")
val = pd.read_csv("val.csv")

# ==================================================
# 2. SELECT REQUIRED ROWS
# ==================================================

# Row numbers refer to data rows, excluding header
dev = dev.iloc[9:16].copy()     # DEV rows 10 to 16
val = val.iloc[14:21].copy()    # VAL rows 15 to 21

# ==================================================
# 3. CLEAN COLUMN NAMES
# ==================================================

dev.columns = dev.columns.str.strip()
val.columns = val.columns.str.strip()

# ==================================================
# 4. DEFINE METRICS
# ==================================================

# Based on your screenshot
metrics = [
    "n",
    "ACTUAL_LOSS",
    "SMI3_LOSS",
    "ACTUAL_LR",
    "SMI3_LR",
    "Actual_n_Delq_and_Claims",
    "Predicted_n_Delq_and_Claims"
]

columns = ["BOOKYEAR"] + metrics

# Keep only required columns
dev = dev[columns].copy()
val = val[columns].copy()

# ==================================================
# 5. CLEAN BOOKYEAR
# ==================================================

dev["BOOKYEAR"] = pd.to_numeric(
    dev["BOOKYEAR"],
    errors="coerce"
)

val["BOOKYEAR"] = pd.to_numeric(
    val["BOOKYEAR"],
    errors="coerce"
)

# Do not silently discard invalid Bookyears
if dev["BOOKYEAR"].isna().any():
    raise ValueError("DEV contains missing or invalid BOOKYEAR values")

if val["BOOKYEAR"].isna().any():
    raise ValueError("VAL contains missing or invalid BOOKYEAR values")

# ==================================================
# 6. SORT BY BOOKYEAR
# ==================================================

dev = dev.sort_values(
    by="BOOKYEAR"
).reset_index(drop=True)

val = val.sort_values(
    by="BOOKYEAR"
).reset_index(drop=True)

print("DEV BOOKYEARS:")
print(dev["BOOKYEAR"].tolist())

print("\nVAL BOOKYEARS:")
print(val["BOOKYEAR"].tolist())

# ==================================================
# 7. CHECK FOR DUPLICATE BOOKYEARS
# ==================================================

if dev["BOOKYEAR"].duplicated().any():
    raise ValueError("Duplicate BOOKYEAR found in DEV")

if val["BOOKYEAR"].duplicated().any():
    raise ValueError("Duplicate BOOKYEAR found in VAL")

# ==================================================
# 8. MERGE DEV AND VAL BY BOOKYEAR
# ==================================================

merged = pd.merge(
    dev,
    val,
    on="BOOKYEAR",
    how="outer",
    suffixes=("_DEV", "_VAL"),
    indicator=True,
    validate="one_to_one"
)

merged = merged.sort_values(
    "BOOKYEAR"
).reset_index(drop=True)

# ==================================================
# 9. CALCULATE DIFFERENCES AND STATUS
# ==================================================

tolerance = 1e-8

# Convert only metric columns to numeric
for metric in metrics:

    for suffix in ["DEV", "VAL"]:

        col = f"{metric}_{suffix}"

        merged[col] = pd.to_numeric(
            merged[col]
            .astype("string")
            .str.replace(",", "", regex=False)
            .str.strip(),
            errors="coerce"
        )

    # Absolute difference
    merged[f"{metric}_Difference"] = abs(
        merged[f"{metric}_DEV"] -
        merged[f"{metric}_VAL"]
    )

    # PASS / FAIL
    merged[f"{metric}_Status"] = np.where(
        (
            merged["_merge"].eq("both")
            &
            merged[f"{metric}_Difference"].notna()
            &
            (
                merged[f"{metric}_Difference"]
                <= tolerance
            )
        ),
        "PASS",
        "FAIL"
    )

# ==================================================
# 10. CREATE WIDE FORMAT REPORT
# ==================================================

report_columns = ["BOOKYEAR"]

for metric in metrics:

    report_columns.extend([
        f"{metric}_DEV",
        f"{metric}_VAL",
        f"{metric}_Difference",
        f"{metric}_Status"
    ])

wide_report = merged[report_columns].copy()

# ==================================================
# 11. OVERALL STATUS BY BOOKYEAR
# ==================================================

status_cols = [
    f"{metric}_Status"
    for metric in metrics
]

wide_report["Overall_Status"] = np.where(
    wide_report[status_cols].eq("PASS").all(axis=1),
    "PASS",
    "FAIL"
)

# ==================================================
# 12. CREATE SIMPLE PASS/FAIL MATRIX
# ==================================================

status_report = wide_report[
    ["BOOKYEAR"] + status_cols + ["Overall_Status"]
].copy()

# Rename columns to metric names
status_report.columns = (
    ["BOOKYEAR"] + metrics + ["Overall_Status"]
)

# ==================================================
# 13. OVERALL REPRODUCIBILITY
# ==================================================

overall_result = (
    "PASS"
    if wide_report["Overall_Status"].eq("PASS").all()
    else "FAIL"
)

# ==================================================
# 14. PRINT RESULTS
# ==================================================

print("\nPERFORMANCE REPRODUCIBILITY REPORT")
print(wide_report.to_string(index=False))

print("\nPASS/FAIL SUMMARY BY BOOKYEAR")
print(status_report.to_string(index=False))

print("\nOVERALL REPRODUCIBILITY:", overall_result)

# ==================================================
# 15. EXPORT TO EXCEL
# ==================================================

output_file = "Performance_Reproducibility_Report.xlsx"

with pd.ExcelWriter(
    output_file,
    engine="openpyxl"
) as writer:

    wide_report.to_excel(
        writer,
        sheet_name="Detailed Comparison",
        index=False
    )

    status_report.to_excel(
        writer,
        sheet_name="PASS FAIL Summary",
        index=False
    )

    merged.to_excel(
        writer,
        sheet_name="Merged Data",
        index=False
    )

print("\nReport saved:", output_file)
