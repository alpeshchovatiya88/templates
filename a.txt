import pandas as pd
import numpy as np

# ----------------------------------
# 1. Read CSV files
# ----------------------------------

dev = pd.read_csv("dev.csv")
val = pd.read_csv("val.csv")

# ----------------------------------
# 2. Select required rows
# ----------------------------------

dev = dev.iloc[9:16].copy()
val = val.iloc[14:21].copy()

# Clean column names
dev.columns = dev.columns.str.strip()
val.columns = val.columns.str.strip()

# ----------------------------------
# 3. Sort by BOOKYEAR
# ----------------------------------

dev["BOOKYEAR"] = pd.to_numeric(
    dev["BOOKYEAR"], errors="raise"
)

val["BOOKYEAR"] = pd.to_numeric(
    val["BOOKYEAR"], errors="raise"
)

dev = dev.sort_values(
    by="BOOKYEAR",
    ascending=True
).reset_index(drop=True)

val = val.sort_values(
    by="BOOKYEAR",
    ascending=True
).reset_index(drop=True)

# ----------------------------------
# 4. Define metrics
# ----------------------------------

metrics = [
    "Total Application",
    "Actual Loss",
    "Predicted Loss",
    "Actual LR",
    "Predicted LR",
    "Actual N of Delq",
    "Claims"
]

# ----------------------------------
# 5. Match by BOOKYEAR
# ----------------------------------

merged = pd.merge(
    dev[["BOOKYEAR"] + metrics],
    val[["BOOKYEAR"] + metrics],
    on="BOOKYEAR",
    how="outer",
    suffixes=("_DEV", "_VAL"),
    indicator=True,
    validate="one_to_one"
)

merged = merged.sort_values(
    "BOOKYEAR"
).reset_index(drop=True)

# ----------------------------------
# 6. Calculate metric differences
# ----------------------------------

tolerance = 1e-8
results = []

for _, row in merged.iterrows():

    bookyear = row["BOOKYEAR"]

    for metric in metrics:

        dev_value = pd.to_numeric(
            str(row[f"{metric}_DEV"]).replace(",", ""),
            errors="coerce"
        )

        val_value = pd.to_numeric(
            str(row[f"{metric}_VAL"]).replace(",", ""),
            errors="coerce"
        )

        if pd.notna(dev_value) and pd.notna(val_value):
            abs_diff = abs(dev_value - val_value)
        else:
            abs_diff = np.nan

        status = (
            "PASS"
            if row["_merge"] == "both"
            and pd.notna(abs_diff)
            and abs_diff <= tolerance
            else "FAIL"
        )

        results.append({
            "BOOKYEAR": bookyear,
            "Metric": metric,
            "Developer_Value": dev_value,
            "Validator_Value": val_value,
            "Absolute_Difference": abs_diff,
            "Tolerance": tolerance,
            "Status": status
        })

# ----------------------------------
# 7. Final reproducibility report
# ----------------------------------

report = pd.DataFrame(results)

report = report.sort_values(
    by=["BOOKYEAR", "Metric"]
).reset_index(drop=True)

print("\nPERFORMANCE REPRODUCIBILITY REPORT\n")
print(report.to_string(index=False))

# ----------------------------------
# 8. Overall status by BOOKYEAR
# ----------------------------------

summary = (
    report.groupby("BOOKYEAR")["Status"]
    .apply(
        lambda x: "PASS"
        if (x == "PASS").all()
        else "FAIL"
    )
    .reset_index(name="Overall_Status")
)

print("\nBOOKYEAR SUMMARY\n")
print(summary.to_string(index=False))

# ----------------------------------
# 9. Export report
# ----------------------------------

with pd.ExcelWriter(
    "Performance_Reproducibility.xlsx",
    engine="openpyxl"
) as writer:

    report.to_excel(
        writer,
        sheet_name="Metric Comparison",
        index=False
    )

    summary.to_excel(
        writer,
        sheet_name="Bookyear Summary",
        index=False
    )
