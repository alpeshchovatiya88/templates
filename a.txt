
import pandas as pd
import plotly.graph_objects as go

def plot_hpi_training(df):

    data = df.copy()

    data["BOOKYEAR"] = pd.to_numeric(
        data["BOOKYEAR"], errors="coerce"
    )
    data["HPI_+4"] = pd.to_numeric(
        data["HPI_+4"], errors="coerce"
    )

    # Filter training period
    train = data[
        data["BOOKYEAR"].between(2009, 2015)
    ].copy()

    # Overall average
    overall = (
        train.groupby("BOOKYEAR")["HPI_+4"]
        .mean()
        .sort_index()
    )

    regions = sorted(
        train["ECON_REGION"].dropna().unique()
    )

    # Distinct colors
    colors = [
        "#E41A1C",  # Red
        "#FF7F00",  # Orange
        "#228B22",  # Green
        "#984EA3",  # Purple
        "#00A6A6",  # Teal
        "#A65628",  # Brown
        "#D81B60",  # Pink
        "#808000",  # Olive
        "#000000",  # Black
        "#17BECF",  # Cyan
        "#8B008B",  # Dark magenta
        "#B8860B",  # Gold
    ]

    # Different marker shapes
    markers = [
        "square",
        "diamond",
        "triangle-up",
        "cross",
        "x",
        "star",
        "triangle-down",
        "pentagon",
        "hexagon",
        "triangle-left",
        "triangle-right",
        "hourglass"
    ]

    # Different line patterns
    dash_styles = [
        "dot",
        "dash",
        "dashdot",
        "longdash",
        "longdashdot"
    ]

    fig = go.Figure()

    # Plot each region
    for i, region in enumerate(regions):

        region_df = train[
            train["ECON_REGION"] == region
        ]

        yearly = (
            region_df.groupby("BOOKYEAR")["HPI_+4"]
            .mean()
            .sort_index()
        )

        fig.add_trace(
            go.Scatter(
                x=yearly.index,
                y=yearly.values,
                mode="lines+markers",
                name=str(region),

                line=dict(
                    color=colors[i % len(colors)],
                    width=3,
                    dash=dash_styles[
                        i % len(dash_styles)
                    ]
                ),

                marker=dict(
                    symbol=markers[i % len(markers)],
                    size=10,
                    color=colors[i % len(colors)],
                    line=dict(
                        color="white",
                        width=1
                    )
                ),

                hovertemplate=(
                    f"Region: {region}<br>"
                    "Year: %{x}<br>"
                    "HPI: %{y:.4f}"
                    "<extra></extra>"
                )
            )
        )

    # Overall HPI: bold blue
    fig.add_trace(
        go.Scatter(
            x=overall.index,
            y=overall.values,
            mode="lines+markers",
            name="Overall Training HPI",
            line=dict(
                color="#0047FF",
                width=5,
                dash="solid"
            ),
            marker=dict(
                symbol="circle",
                size=12,
                color="#0047FF",
                line=dict(
                    color="white",
                    width=2
                )
            ),
            hovertemplate=(
                "Overall HPI<br>"
                "Year: %{x}<br>"
                "HPI: %{y:.4f}"
                "<extra></extra>"
            )
        )
    )

    fig.update_layout(
        title="HPI +4 - Training (2009-2015)",
        xaxis_title="BOOKYEAR",
        yaxis_title="HPI_+4",
        template="plotly_white",
        height=700,
        width=1300,

        hovermode="closest",

        legend=dict(
            title="Economic Region",
            x=1.02,
            y=1,
            xanchor="left"
        ),

        margin=dict(
            l=80,
            r=240,
            t=80,
            b=70
        )
    )

    fig.update_xaxes(
        tickmode="linear",
        dtick=1
    )

    fig.show()

    return fig
