grouped = econ.groupby(
    "ECON_REGION",
    group_keys=False
)

for year in range(1, 5):
    months = 12 * year

    econ[f"UE_-{year}"] = grouped["UE"].shift(months)

    econ[f"UE_+{year}"] = grouped["UE"].shift(-months)

    econ[f"HPI_+{year}"] = (
        grouped["HPI"].shift(-months)
        / econ["HPI"]
    )

    econ[f"HPI_-{year}"] = (
        econ["HPI"]
        / grouped["HPI"].shift(months)
    )
