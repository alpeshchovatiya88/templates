cutoff = 10

df["decision_8020"] = np.where(
    df["decile_8020_num"] <= cutoff,
    "A",
    "B"
)

df["decision_ts"] = np.where(
    df["decile_ts_num"] <= cutoff,
    "A",
    "B"
)

====================
df["swap_set"] = np.select(
    [
        (df["decision_8020"] == "A") &
        (df["decision_ts"] == "A"),

        (df["decision_8020"] == "A") &
        (df["decision_ts"] == "B"),

        (df["decision_8020"] == "B") &
        (df["decision_ts"] == "A"),

        (df["decision_8020"] == "B") &
        (df["decision_ts"] == "B")
    ],
    [
        "A -> A",
        "A -> B",
        "B -> A",
        "B -> B"
    ]
)

swap_analysis = (

      .agg(
          applications=("APPLICATION_NUMBER", "count"),
          bads=("BAD_FLAG", "sum"),
          bad_rate=("BAD_FLAG", "mean"),
          avg_8020_score=("SMI3_LR_8020", "mean"),
          avg_ts_score=("SMI3_LR_ts", "mean")
