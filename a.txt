import pandas as pd
import numpy as np

# Convert effective date: 20150908 -> 2015-09-08
df_new["EFFECT_DATE"] = pd.to_datetime(
    df_new["EFFECT_DAY_SID"].astype("Int64").astype(str),
    format="%Y%m%d",
    errors="coerce"
)

# Convert event dates
df_new["DELQ90_IND_DATE"] = pd.to_datetime(
    df_new["DELQ90_IND_DATE"],
    errors="coerce"
)

df_new["CLAIM_IND_DATE"] = pd.to_datetime(
    df_new["CLAIM_IND_DATE"],
    errors="coerce"
)

# Time to delinquency:
# only DELQ=1 and CLAIM != 1
df_new["TIME_TO_DELQ_YR"] = np.where(
    (df_new["DELQ90_IND"] == 1.0) &
    (df_new["CLAIM_IND"] != 1.0),
    (df_new["DELQ90_IND_DATE"] - df_new["EFFECT_DATE"]).dt.days / 365.25,
    np.nan
)

# Time to claim:
# CLAIM takes priority if both indicators = 1
df_new["TIME_TO_CLAIM_YR"] = np.where(
    df_new["CLAIM_IND"] == 1.0,
    (df_new["CLAIM_IND_DATE"] - df_new["EFFECT_DATE"]).dt.days / 365.25,
    np.nan
)
