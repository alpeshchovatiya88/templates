feature = "PRD_LOAN_PURPOSE"

dev_pct = (
    all_model_dev[feature]
    .astype("string").fillna("MISSING")
    .value_counts(normalize=True)
)

oot_pct = (
    all_model_oot[feature]
    .astype("string").fillna("MISSING")
    .value_counts(normalize=True)
)

psi_detail = pd.concat([dev_pct, oot_pct], axis=1).fillna(0)
psi_detail.columns = ["DEV_pct", "OOT_pct"]

eps = 0.0001

e = psi_detail["DEV_pct"].replace(0, eps)
a = psi_detail["OOT_pct"].replace(0, eps)

psi_detail["PSI_contribution"] = (
    (a - e) * np.log(a / e)
)

# Add type
psi_detail["type"] = "categorical"

# Add interpretation
psi_detail["interpretation"] = np.select(
    [
        psi_detail["PSI_contribution"] < 0.10,
        psi_detail["PSI_contribution"] < 0.25
    ],
    [
        "Stable",
        "Moderate shift"
    ],
    default="Significant shift"
)

# Convert to %
psi_detail["DEV_pct"] *= 100
psi_detail["OOT_pct"] *= 100

psi_detail = (
    psi_detail
    .sort_values("PSI_contribution", ascending=False)
)

display(psi_detail)
