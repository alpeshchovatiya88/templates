import plotly.express as px

econ_dist = df["ECON_REGION"].value_counts().reset_index()
econ_dist.columns = ["ECON_REGION", "Count"]

econ_dist["Percentage"] = (
    econ_dist["Count"] / econ_dist["Count"].sum() * 100
)

econ_dist["Label"] = (
    econ_dist["Count"].map("{:,}".format)
    + " (" +
    econ_dist["Percentage"].map("{:.1f}%".format)
    + ")"
)

fig = px.bar(
    econ_dist,
    x="ECON_REGION",
    y="Count",
    text="Label",
    title="ECON_REGION Data Distribution"
)

fig.update_traces(textposition="outside")

fig.update_layout(
    xaxis_title="ECON_REGION",
    yaxis_title="Number of Records"
)

fig.show()
