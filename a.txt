
import pandas as pd
import plotly.graph_objects as go


def plot_lr_by_ventile(
    dfs,
    ventile_col="rr_model_decile",
    lr_col="SMI3_LR",
    lr_scale=100,
    title="Random Seed Sensitivity - Loss Ratio by Ventile",
    decimals=2,
    label_font_size=13,
    height=800
):
    """
    Plot mean SMI3_LR by ventile for each random seed.

    lr_scale=100: LR stored as 0.1847 -> 18.47%
    lr_scale=1:   LR stored as 18.47 -> 18.47%
    """

    # 1. Calculate mean LR by ventile
    summary = pd.DataFrame({
        seed: (
            df.assign(
                _ventile=pd.to_numeric(
                    df[ventile_col]
                    .astype(str)
                    .str.extract(r"(\d+)", expand=False),
                    errors="coerce"
                )
            )
            .assign(
                _lr=pd.to_numeric(
                    df[lr_col],
                    errors="coerce"
                )
            )
            .groupby("_ventile")["_lr"]
            .mean()
        )
        for seed, df in dfs.items()
    }).sort_index()

    if summary.empty:
        raise ValueError("No valid ventile data found.")

    # Convert to percentage display units
    summary = summary * lr_scale

    colors = {
        "RS42": "#17365D",
        "RS07": "#E66B23",
        "RS101": "#09988A",
        "RS151": "#7952B3"
    }

    fig = go.Figure()

    # 2. Create grouped bars
    for seed in summary.columns:

        values = summary[seed]

        fig.add_trace(
            go.Bar(
                x=summary.index,
                y=values,
                name=seed,
                marker_color=colors.get(seed, "#64748B"),

                # Vertical percentage labels
                text=[
                    f"{v:,.{decimals}f}%"
                    if pd.notna(v) else ""
                    for v in values
                ],

                textposition="outside",
                textangle=-90,

                textfont=dict(
                    family="Arial Black",
                    size=label_font_size,
                    color="black"
                ),

                cliponaxis=False,

                hovertemplate=(
                    "Ventile: %{x}<br>"
                    f"Seed: {seed}<br>"
                    "Mean LR: %{y:,.2f}%"
                    "<extra></extra>"
                )
            )
        )

    # 3. Dropdown for ventiles
    ventiles = summary.index.tolist()

    buttons = [
        dict(
            label="All Ventiles",
            method="relayout",
            args=[{
                "xaxis.range": [
                    min(ventiles) - 0.5,
                    max(ventiles) + 0.5
                ]
            }]
        )
    ]

    for v in ventiles:
        buttons.append(
            dict(
                label=f"Ventile {int(v)}",
                method="relayout",
                args=[{
                    "xaxis.range": [
                        float(v) - 0.5,
                        float(v) + 0.5
                    ]
                }]
            )
        )

    # 4. Extra space for vertical labels
    max_lr = summary.max().max()
    min_lr = summary.min().min()

    if pd.isna(max_lr):
        max_lr = 0

    if pd.isna(min_lr):
        min_lr = 0

    max_lr = float(max_lr)
    min_lr = float(min_lr)

    span = max(
        max_lr - min_lr,
        abs(max_lr),
        abs(min_lr),
        1
    )

    upper = max(0, max_lr) + span * 0.35
    lower = min(0, min_lr) - span * 0.10

    # 5. Layout
    fig.update_layout(
        title=dict(
            text=title,
            x=0.5,
            xanchor="center",
            font=dict(
                size=22,
                family="Arial Black"
            )
        ),

        xaxis_title="Ventile",
        yaxis_title="Average SMI3_LR (%)",

        barmode="group",
        height=height,

        margin=dict(
            t=180,
            b=90,
            l=90,
            r=60
        ),

        font=dict(
            family="Arial",
            size=13
        ),

        legend=dict(
            orientation="h",
            x=0.5,
            xanchor="center",
            y=1.02,
            yanchor="bottom"
        ),

        uniformtext=dict(
            minsize=10,
            mode="show"
        ),

        updatemenus=[
            dict(
                buttons=buttons,
                direction="down",
                showactive=True,
                x=1,
                xanchor="right",
                y=1.24,
                yanchor="top",
                bgcolor="white",
                bordercolor="#CBD5E1",
                borderwidth=1
            )
        ],

        plot_bgcolor="white",
        paper_bgcolor="white",

        bargap=0.20,
        bargroupgap=0.05
    )

    fig.update_xaxes(
        tickmode="array",
        tickvals=ventiles,
        type="linear",
        range=[
            min(ventiles) - 0.5,
            max(ventiles) + 0.5
        ],
        showgrid=False
    )

    fig.update_yaxes(
        range=[lower, upper],
        ticksuffix="%",
        gridcolor="#E5E7EB",
        zeroline=True,
        zerolinecolor="#94A3B8"
    )

    fig.show()

    return summary
