import math
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path

# ==========================================
# 1. DATAFRAMES
# ==========================================

dfs = {
    "RS42": df_rs42,
    "RS532": df_rs532,
    "RS3458": df_rs3458,
    "RS499": df_rs499,
    "RS345": df_rs345,
    "RS1999": df_rs1999,
    "RS2324": df_rs2324,
}

colors = {
    "RS42": "#17365D",
    "RS532": "#007F73",
    "RS3458": "#6A1B9A",
    "RS499": "#B45309",
    "RS345": "#9F1239",
    "RS1999": "#166534",
    "RS2324": "#1D4ED8",
}

DECILE_COL = "rr_model_decile"
LR_SCALE = 100  # Change to 1 if SMI3_LR is already in %

output_dir = Path("seed_sensitivity_plots")
output_dir.mkdir(exist_ok=True)

seeds = list(dfs.keys())

# ==========================================
# 2. PREPARE DATA
# ==========================================

prepared = {}

for seed, df in dfs.items():
    temp = df.copy()

    temp["_decile"] = pd.to_numeric(
        temp[DECILE_COL].astype(str).str.extract(r"(\d+)")[0],
        errors="coerce"
    )

    temp["SMI3_LOSS"] = pd.to_numeric(
        temp["SMI3_LOSS"], errors="coerce"
    )

    temp["SMI3_LR"] = pd.to_numeric(
        temp["SMI3_LR"], errors="coerce"
    )

    prepared[seed] = temp.dropna(subset=["_decile"])

deciles = sorted(set().union(
    *(set(df["_decile"].unique()) for df in prepared.values())
))

# ==========================================
# 3. PLOTTING FUNCTION
# ==========================================

def plot_metric(metric, aggregation, scale=1):

    ncols = 4
    nrows = math.ceil(len(deciles) / ncols)

    fig, axes = plt.subplots(
        nrows,
        ncols,
        figsize=(24, nrows * 4.5)
    )

    axes = np.array(axes).flatten()

    aggregated = {}

    for seed, df in prepared.items():

        grouped = df.groupby("_decile")[metric]

        if aggregation == "sum":
            values = grouped.sum(min_count=1)
        else:
            values = grouped.mean()

        aggregated[seed] = values * scale

    for i, decile in enumerate(deciles):

        ax = axes[i]

        values = [
            aggregated[seed].get(decile, np.nan)
            for seed in seeds
        ]

        bars = ax.bar(
            seeds,
            values,
            color=[colors[s] for s in seeds],
            width=0.7
        )

        # Value labels above bars
        for bar, value in zip(bars, values):

            if pd.isna(value):
                continue

            if metric == "SMI3_LOSS":
                label = f"{value / 1_000_000:.2f}M"
            else:
                label = f"{value:.2f}%"

            ax.annotate(
                label,
                xy=(
                    bar.get_x() + bar.get_width() / 2,
                    bar.get_height()
                ),
                xytext=(0, 4),
                textcoords="offset points",
                ha="center",
                va="bottom",
                fontsize=9
            )

        ax.set_title(
            f"VENTILE: {int(decile)}",
            fontsize=14,
            fontweight="bold"
        )

        ax.tick_params(axis="x", rotation=45, labelsize=9)
        ax.tick_params(axis="y", labelsize=9)

        ax.grid(
            axis="y",
            linestyle="--",
            alpha=0.3
        )

        ax.set_axisbelow(True)

        valid = [v for v in values if pd.notna(v)]

        if valid and min(valid) >= 0:
            ymax = max(valid)
            ax.set_ylim(0, ymax * 1.25 if ymax else 1)

    # Hide unused subplots
    for j in range(len(deciles), len(axes)):
        axes[j].axis("off")

    fig.suptitle(
        f"Random Seed Sensitivity - {metric} by Ventile",
        fontsize=22,
        fontweight="bold",
        y=1.01
    )

    plt.tight_layout()

    # Save directly as PNG
    filename = output_dir / f"{metric}_by_decile.png"

    fig.savefig(
        filename,
        dpi=300,
        bbox_inches="tight",
        facecolor="white"
    )

    print(f"Saved: {filename}")

    plt.show()

    return fig


# ==========================================
# 4. CREATE AND SAVE BOTH PNG FIGURES
# ==========================================

fig_loss = plot_metric(
    metric="SMI3_LOSS",
    aggregation="sum"
)

fig_lr = plot_metric(
    metric="SMI3_LR",
    aggregation="mean",
    scale=LR_SCALE
)
