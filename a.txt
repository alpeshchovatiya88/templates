summary = (
    df.groupby(["BOOKYEAR", "which_score", "CRD_SCORE_bin"])
      .agg(
          total_number=("target", "count"),
          total_bad_count=("target", "sum"),
          bad_rate_pct=("target", lambda x: round(x.mean() * 100, 2))
      )
)

out = summary.unstack(["which_score", "CRD_SCORE_bin"])

formatted = (
    out["total_number"].astype("Int64").astype(str)
    + " (bad="
    + out["total_bad_count"].astype("Int64").astype(str)
    + ", "
    + out["bad_rate_pct"].astype(str)
    + "%)"
)

formatted
