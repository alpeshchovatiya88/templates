
import pandas as pd
import numpy as np
import plotly.graph_objects as go

def export_seed_sensitivity_html(
    dfs,
    decile_col,
    value_col,
    filename="seed_sensitivity.html",
    region_col="ECON_REGION",
    agg="sum",
    baseline="RS42",
    is_ratio=False,
    ratio_scale=1,
    colors=None
):
    seeds = list(dfs)

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#C75B39",
            "RS101": "#147D74",
            "RS151": "#7952B3"
        }

    frames = []

    for seed, df in dfs.items():
        t = df[[decile_col, value_col, region_col]].copy()
        t["Seed"] = seed
        t["Decile"] = pd.to_numeric(
            t[decile_col].astype(str).str.extract(
                r"(\d+)", expand=False
            ),
            errors="coerce"
        )
        t[value_col] = pd.to_numeric(
            t[value_col], errors="coerce"
        )
        frames.append(t)

    data = pd.concat(frames, ignore_index=True)
    data = data.dropna(subset=["Decile"])

    deciles = sorted(data["Decile"].astype(int).unique())
    regions = sorted(data[region_col].dropna().unique(), key=str)

    def fmt(v):
        if pd.isna(v):
            return "N/A"
        if is_ratio:
            return f"{v / ratio_scale * 100:.2f}%"
        if abs(v) >= 1e6:
            return f"{v / 1e6:.2f}M"
        if abs(v) >= 1e3:
            return f"{v / 1e3:.2f}K"
        return f"{v:.2f}"

    def make_chart(selected_decile=None, selected_region=None):
        sub = data

        if selected_decile is not None:
            sub = sub[sub["Decile"] == selected_decile]

        if selected_region is not None:
            sub = sub[sub[region_col] == selected_region]

        stats = sub.groupby("Seed")[value_col].agg(
            total=agg,
            min="min",
            max="max",
            avg="mean",
            p10=lambda x: x.quantile(.10),
            p20=lambda x: x.quantile(.20),
            p50="median"
        ).reindex(seeds)

        base = stats.loc[baseline, "total"]

        fig = go.Figure()

        for i, seed in enumerate(seeds):
            r = stats.loc[seed]

            if pd.isna(r["total"]):
                continue

            delta = r["total"] - base if pd.notna(base) else np.nan
            delta_pct = (
                delta / base * 100
                if pd.notna(base) and base != 0
                else np.nan
            )

            delta_text = (
                "Baseline" if seed == baseline else fmt(delta)
            )

            pct_text = (
                "Baseline" if seed == baseline
                else f"{delta_pct:+.2f}%"
                if pd.notna(delta_pct) else "N/A"
            )

            details = (
                f"<b>Min: {fmt(r['min'])}</b><br>"
                f"<b>Max: {fmt(r['max'])}</b><br>"
                f"<b>Avg: {fmt(r['avg'])}</b><br>"
                f"<b>P10: {fmt(r['p10'])}</b><br>"
                f"<b>P20: {fmt(r['p20'])}</b><br>"
                f"<b>P50: {fmt(r['p50'])}</b><br>"
                f"<b>Δ: {delta_text}</b><br>"
                f"<b>Δ%: {pct_text}</b>"
            )

            fig.add_trace(go.Bar(
                x=[seed],
                y=[r["total"]],
                marker_color=colors.get(seed, "#64748B"),
                text=[details],
                textposition="inside",
                insidetextanchor="middle",
                textfont=dict(size=15, color="white"),
                hovertemplate=details + "<extra></extra>",
                showlegend=False
            ))

            fig.add_annotation(
                x=seed,
                y=float(r["total"]),
                text=f"<b>{agg.upper()}: {fmt(r['total'])}</b>",
                showarrow=False,
                yshift=18,
                font=dict(size=17, color="#172033")
            )

        label = (
            "All Deciles"
            if selected_decile is None
            else f"Decile {selected_decile}"
        )

        region_label = (
            "All Regions"
            if selected_region is None
            else str(selected_region)
        )

        fig.update_layout(
            title=f"{value_col} — {label} | {region_label}",
            template="plotly_white",
            height=750,
            width=1200,
            margin=dict(t=120, b=80, l=100, r=50),
            xaxis_title="Random Seed",
            yaxis_title=value_col,
            yaxis_tickformat=(
                ".1%" if is_ratio and ratio_scale == 1
                else "~s"
            )
        )

        return fig

    # Create one standalone HTML file with all filter combinations.
    # Each chart is embedded as an independent Plotly visualization.
    from plotly.io import to_html

    html_parts = [
        "<html><head><meta charset='utf-8'></head><body>",
        "<h2>Random Seed Sensitivity Report</h2>"
    ]

    combinations = (
        [(None, None)] +
        [(d, None) for d in deciles] +
        [(None, r) for r in regions] +
        [(d, r) for d in deciles for r in regions]
    )

    for idx, (d, r) in enumerate(combinations):
        fig = make_chart(d, r)

        html_parts.append(
            to_html(
                fig,
                full_html=False,
                include_plotlyjs=True if idx == 0 else False
            )
        )

    html_parts.append("</body></html>")

    with open(filename, "w", encoding="utf-8") as f:
        f.write("\n".join(html_parts))

    print(f"HTML report saved: {filename}")
