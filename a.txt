def apply_dsr_logic(self, df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()

    mask_extract = df["EFFECT_SID"].between(200901, 201610)
    mask_comp = df["EFFECT_SID"].between(201611, 202512)

    # 200901 to 201610
    df_extract = self.extract_dsr(df.loc[mask_extract])

    # 201611 to 202512
    df_comp = self.use_comp_dsr(df.loc[mask_comp])

    # Anything outside the above ranges
    df_other = df.loc[~(mask_extract | mask_comp)]

    return pd.concat(
        [df_extract, df_comp, df_other]
    ).sort_index()
