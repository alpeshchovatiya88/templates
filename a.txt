def export_interactive_seed_report(

    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    metrics=None,
    baseline="RS42",
    claims_share=0.62,
    colors=None,
    decimals=2,
    inside_font_size=12,
    chart_height=900,

    """
    Standalone offline HTML random seed sensitivity report.

    Features:
      - Metric dropdown: SMI3_LOSS / SMI3_LR
      - Multi-select Ventile checkboxes
      - Select All / Clear All Ventiles
      - ECON_REGION, POD_MARKET_CODE, PRK_TENURE filters
      - Predicted Delq + Claims, Claims, Delqs inside every bar
      - Min, Max, Avg, P10, P50, P90 inside every bar
      - SMI3_LOSS: absolute delta and relative delta %
      - SMI3_LR: delta in percentage points only
      - Detailed table
      - Offline self-contained HTML

    Note:
      No ventile values selected = All Ventiles.

    The input DataFrames are not modified.
    """

    if not dfs:
        raise ValueError("dfs cannot be empty")

    if baseline not in dfs:
        raise ValueError(
            f"Baseline '{baseline}' is not in dfs"
        )

    if not 0 <= claims_share <= 1:
        raise ValueError(
            "claims_share must be between 0 and 1"
        )

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    for metric, config in metrics.items():
        if config.get("agg") not in ("sum", "mean"):
            raise ValueError(
                f"{metric}: aggregation must be sum or mean"
            )

    # --------------------------------------------------------
    # Prepare data
    #
    # Each source observation appears ONCE.
    # This prevents predicted counts being duplicated
    # when the selected metric changes.
    #
    # Record structure:
    # [
    #   seed,
    #   ventile,
    #   region,
    #   market,
    #   tenure,
    #   metric_1_value,
    #   metric_2_value,
    #   ...,
    #   prediction_probability
    # ]
    # --------------------------------------------------------

    records = []

    all_ventiles = set()
    all_regions = set()
    all_markets = set()
    all_tenures = set()

    metric_names = list(metrics.keys())

    required_cols = list(dict.fromkeys([
        ventile_col,
        region_col,
        market_col,
        tenure_col,
        probability_col,
        *metric_names
    ]))

    for seed, source_df in dfs.items():

        missing = [
            col for col in required_cols
            if col not in source_df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed}: missing columns {missing}"
            )

        df = source_df[required_cols].copy()

        ventiles = pd.to_numeric(
            df[ventile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        regions = (
            df[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        markets = (
            df[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        tenures = (
            df[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        probability = (
            pd.to_numeric(
                df[probability_col],
                errors="coerce"
            )
            .replace([np.inf, -np.inf], np.nan)
        )

        metric_series = {}

        for metric in metric_names:
            metric_series[metric] = (
                pd.to_numeric(
                    df[metric],
                    errors="coerce"
                )
                .replace([np.inf, -np.inf], np.nan)
            )

        all_ventiles.update(
            ventiles.dropna().astype(int).tolist()
        )
        all_regions.update(regions.tolist())
        all_markets.update(markets.tolist())
        all_tenures.update(tenures.tolist())

        # Convert to compact Python lists before iteration.

        v_list = ventiles.tolist()
        r_list = regions.tolist()
        m_list = markets.tolist()
        t_list = tenures.tolist()
        p_list = probability.tolist()

        metric_lists = [
            metric_series[metric].tolist()
            for metric in metric_names
        ]

        for i in range(len(df)):

            ventile = v_list[i]

            record = [
                str(seed),
                int(ventile)
                    if pd.notna(ventile)
                    else None,
                str(r_list[i]),
                str(m_list[i]),
                str(t_list[i])
            ]

            for values in metric_lists:
                value = values[i]

                record.append(
                    float(value)
                    if pd.notna(value)
                    else None
                )

            p = p_list[i]

            record.append(
                float(p)
                if pd.notna(p)
                else None
            )

            records.append(record)

    report_data = {
        "seeds": list(dfs.keys()),
        "baseline": baseline,
        "metrics": metrics,
        "metricNames": metric_names,
        "colors": colors,
        "ventiles": sorted(all_ventiles),
        "regions": sorted(all_regions),
        "markets": sorted(all_markets),
        "tenures": sorted(all_tenures),
        "records": records,
        "claimsShare": claims_share,
        "decimals": decimals,
        "insideFontSize": inside_font_size,
        "chartHeight": chart_height
    }

    report_json = json.dumps(
        report_data,
        ensure_ascii=False,
        allow_nan=False,
        separators=(",", ":")
    ).replace("<", "\\u003c")

(function () {

"use strict";

const report = JSON.parse(
    document.getElementById("reportData").textContent
);

const $ = id => document.getElementById(id);

const selectedVentiles = new Set();

const metricIndex = {};
report.metricNames.forEach((metric, i) => {
    metricIndex[metric] = 5 + i;
});

const probabilityIndex = 5 + report.metricNames.length;

function addOption(select, value, label) {
    const option = document.createElement("option");
    option.value = value;
    option.textContent = label;
    select.appendChild(option);
}

// ------------------------------------------------------------
// Standard dropdowns

report.metricNames.forEach(metric => {
    addOption($("metric"), metric, metric);
});

addOption($("region"), "", "All Regions");
report.regions.forEach(value => {
    addOption($("region"), value, value);
});

addOption($("market"), "", "All Markets");
report.markets.forEach(value => {
    addOption($("market"), value, value);
});

addOption($("tenure"), "", "All Tenures");
report.tenures.forEach(value => {
    addOption($("tenure"), value, value);
});

function updateVentileLabel() {

    const values = [...selectedVentiles]
        .sort((a, b) => a - b);

    if (
        values.length === 0 ||
        values.length === report.ventiles.length
    ) {
        $("ventileLabel").textContent = "All Ventiles";
    } else if (values.length <= 3) {
        $("ventileLabel").textContent =
            "Ventile " + values.join(", ");
    } else {
        $("ventileLabel").textContent =
            values.length + " Ventiles Selected";
    }
}

function syncVentileCheckboxes() {
    document
        .querySelectorAll(".ventile-checkbox")
        .forEach(checkbox => {
            checkbox.checked = selectedVentiles.has(
                Number(checkbox.value)
            );
        });
}

report.ventiles.forEach(ventile => {

    const label = document.createElement("label");
    label.className = "ventile-option";

    const checkbox = document.createElement("input");
    checkbox.type = "checkbox";
    checkbox.className = "ventile-checkbox";
    checkbox.value = String(ventile);

    const text = document.createElement("span");
    text.textContent = "Ventile " + ventile;

    checkbox.addEventListener("change", () => {

        if (checkbox.checked) {
            selectedVentiles.add(ventile);
        } else {
            selectedVentiles.delete(ventile);
        }

        updateVentileLabel();
        update();
    });

    label.appendChild(checkbox);
    label.appendChild(text);

    $("ventileOptions").appendChild(label);
});

// ------------------------------------------------------------
// Formatting helpers
// ------------------------------------------------------------

function number(value, digits=report.decimals) {

    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    return value.toLocaleString("en-US", {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
    });
}

function metricNumber(value, metric) {

    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    const cfg = report.metrics[metric];

    if (cfg.is_ratio) {
        return number(
            value / (cfg.ratio_scale || 1) * 100
        ) + "%";
    }

    return number(value);
}

function deltaNumber(value, metric) {

    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    const cfg = report.metrics[metric];
    const sign = value > 0 ? "+" : "";

    if (cfg.is_ratio) {
        return sign + number(
            value / (cfg.ratio_scale || 1) * 100
        ) + " pp";
    }

    return sign + number(value);
}

function percentNumber(value) {

    if (value === null || !Number.isFinite(value)) {
        return "N/A";
    }

    return (
        (value > 0 ? "+" : "") +
        number(value) +
        "%"
    );
}

function quantile(sorted, p) {

    if (!sorted.length) return null;

    if (sorted.length === 1) return sorted[0];

    const position = (sorted.length - 1) * p;

    const low = Math.floor(position);
    const high = Math.ceil(position);
    const weight = position - low;

    return (
        sorted[low] * (1 - weight) +
        sorted[high] * weight
    );
}

function calculateStats(values, agg) {

    if (!values.length) {
        return {
            count: 0,
            total: null,
            min: null,
            max: null,
            avg: null,
            p10: null,
            p50: null,
            p90: null
        };
    }

    const sorted = [...values].sort((a, b) => a - b);

    const sum = values.reduce(
        (a, b) => a + b,
        0
    );

    const avg = sum / values.length;

    return {
        count: values.length,
        total: agg === "sum" ? sum : avg,
        min: sorted[0],
        max: sorted[sorted.length - 1],
        avg: avg,
        p10: quantile(sorted, 0.10),
        p50: quantile(sorted, 0.50),
        p90: quantile(sorted, 0.90)
    };
}

// ------------------------------------------------------------
// Update chart and table
// ------------------------------------------------------------

function update() {

    const metric = $("metric").value;
    const cfg = report.metrics[metric];

    const region = $("region").value;
    const market = $("market").value;
    const tenure = $("tenure").value;

    const allVentilesSelected =
        selectedVentiles.size === 0 ||
        selectedVentiles.size === report.ventiles.length;

    const grouped = {};

    report.seeds.forEach(seed => {
        grouped[seed] = {
            values: [],
            probabilitySum: 0,
            probabilityCount: 0
        };
    });

    const selectedMetricIndex = metricIndex[metric];

    // Each source observation is processed once.
    // Predicted counts are therefore NOT double counted.

    report.records.forEach(r => {

        const seed = r[0];
        const ventile = r[1];
        const rowRegion = r[2];
        const rowMarket = r[3];
        const rowTenure = r[4];

        if (
            !allVentilesSelected &&
            !selectedVentiles.has(ventile)
        ) return;

        if (region && rowRegion !== region) return;
        if (market && rowMarket !== market) return;
        if (tenure && rowTenure !== tenure) return;

        const group = grouped[seed];

        const value = r[selectedMetricIndex];
        const probability = r[probabilityIndex];

        if (value !== null) {
            group.values.push(value);
        }

        if (probability !== null) {
            group.probabilitySum += probability;
            group.probabilityCount += 1;
        }
    });

    const result = {};

    report.seeds.forEach(seed => {

        const group = grouped[seed];

        const s = calculateStats(
            group.values,
            cfg.agg
        );

        const predictedTotal =
            group.probabilityCount > 0
            ? group.probabilitySum
            : null;

        s.predictedTotal = predictedTotal;

        s.predictedClaims =
            predictedTotal === null
            ? null
            : predictedTotal * report.claimsShare;

        s.predictedDelqs =
            predictedTotal === null
            ? null
            : predictedTotal * (1 - report.claimsShare);

        result[seed] = s;
    });

    const baselineValue =
        result[report.baseline].total;

    const rows = report.seeds.map(seed => {

        const s = result[seed];

        const delta = (
            s.total !== null &&
            baselineValue !== null
        )
            ? s.total - baselineValue
            : null;

        const deltaPct = (
            delta !== null &&
            baselineValue !== 0
        )
            ? delta / baselineValue * 100
            : null;

        return {
            seed,
            ...s,
            delta,
            deltaPct
        };
    });

    // --------------------------------------------------------
    // Chart
    // --------------------------------------------------------

    const traces = [];
    const annotations = [];

    rows.forEach((r, i) => {

        if (r.total === null) return;

        const deltaText =
            r.seed === report.baseline
            ? "Baseline"
            : deltaNumber(r.delta, metric);

        const deltaPctText =
            r.seed === report.baseline
            ? "Baseline"
            : percentNumber(r.deltaPct);

        const lines = [
            "<b>Min: " + metricNumber(r.min, metric) + "</b>",
            "<b>Max: " + metricNumber(r.max, metric) + "</b>",
            "<b>Avg: " + metricNumber(r.avg, metric) + "</b>",
            "<b>P10: " + metricNumber(r.p10, metric) + "</b>",
            "<b>P50: " + metricNumber(r.p50, metric) + "</b>",
            "<b>P90: " + metricNumber(r.p90, metric) + "</b>",
            "────────────",
            "<b>Pred Delq + Claims: " +
                number(r.predictedTotal) + "</b>",
            "<b>Pred Claims: " +
                number(r.predictedClaims) + "</b>",
            "<b>Pred Delqs: " +
                number(r.predictedDelqs) + "</b>",
            "────────────",
            "<b>Δ: " + deltaText + "</b>"
        ];

        if (!cfg.is_ratio) {
            lines.push(
                "<b>Δ%: " + deltaPctText + "</b>"
            );
        }

        const insideText = lines.join("<br>");

        traces.push({
            type: "bar",
            x: [i],
            y: [r.total],
            width: 0.78,

            marker: {
                color: report.colors[r.seed] || "#64748b"
            },

            text: [insideText],
            textposition: "inside",
            insidetextanchor: "middle",
            constraintext: "none",
            cliponaxis: false,

            textfont: {
                size: report.insideFontSize,
                color: "white",
                family: "Arial"
            },

            hovertemplate:
                "<b>" + r.seed + "</b><br>" +
                "Aggregate: " +
                metricNumber(r.total, metric) +
                "<br>" + insideText +
                "<extra></extra>",

            showlegend: false
        });

        annotations.push({

    });

// ------------------------------------------------------------

# ============================================================
# NEW PDF FUNCTION
#
# PDF CONTAINS CHARTS ONLY — NO TABLE
#
# Pages:
#   1. Overall / All Ventiles
#   2. Ventile 1
#   3. Ventile 2
#   ...
#   Every ECON_REGION
#   Every POD_MARKET_CODE
#   Every PRK_TENURE
#
# The same sequence is created for each metric.
# ============================================================

def export_seed_pdf_report(
    dfs,
    filename="Random_Seed_Sensitivity_Loss_Comparison.pdf",
    ventile_col="rr_model_decile",
    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",
    probability_col="prediction_probability",
    metrics=None,
    baseline="RS42",
    claims_share=0.62,
    colors=None,
    decimals=2,
    inside_font_size=17,
    page_width=1600,
    page_height=1000,
):

    if not dfs:
        raise ValueError("dfs cannot be empty")

    if baseline not in dfs:
        raise ValueError(
            f"Baseline '{baseline}' is not in dfs"
        )

    if not 0 <= claims_share <= 1:
        raise ValueError(
            "claims_share must be between 0 and 1"
        )

    if metrics is None:
        metrics = {
            "SMI3_LOSS": {
                "agg": "sum",
                "is_ratio": False
            },
            "SMI3_LR": {
                "agg": "mean",
                "is_ratio": True,
                "ratio_scale": 100
            }
        }

    if colors is None:
        colors = {
            "RS42": "#17365D",
            "RS07": "#E66B23",
            "RS101": "#09988A",
            "RS151": "#7952B3"
        }

    metric_names = list(metrics.keys())

    required_cols = list(dict.fromkeys([
        ventile_col,
        region_col,
        market_col,
        tenure_col,
        probability_col,
        *metric_names
    ]))

    # --------------------------------------------------------
    # Prepare each seed once
    # --------------------------------------------------------

    prepared = {}

    all_ventiles = set()
    all_regions = set()
    all_markets = set()
    all_tenures = set()

    for seed, source_df in dfs.items():

        missing = [
            col for col in required_cols
            if col not in source_df.columns
        ]

        if missing:
            raise ValueError(
                f"{seed}: missing columns {missing}"
            )

        df = source_df[required_cols].copy()

        df["_Ventile"] = pd.to_numeric(
            df[ventile_col]
            .astype("string")
            .str.extract(r"(\d+)", expand=False),
            errors="coerce"
        )

        df["_Region"] = (
            df[region_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Market"] = (
            df[market_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Tenure"] = (
            df[tenure_col]
            .astype("string")
            .fillna("(Missing)")
        )

        df["_Probability"] = (
            pd.to_numeric(
                df[probability_col],
                errors="coerce"
            )
            .replace([np.inf, -np.inf], np.nan)
        )

        for metric in metric_names:
            df[metric] = (
                pd.to_numeric(
                    df[metric],
                    errors="coerce"
                )
                .replace([np.inf, -np.inf], np.nan)
            )

        all_ventiles.update(
            df["_Ventile"]
            .dropna()
            .astype(int)
            .tolist()
        )

        all_regions.update(
            df["_Region"].tolist()
        )

        all_markets.update(
            df["_Market"].tolist()
        )

        all_tenures.update(
            df["_Tenure"].tolist()
        )

        prepared[str(seed)] = df

    all_ventiles = sorted(all_ventiles)
    all_regions = sorted(all_regions)
    all_markets = sorted(all_markets)
    all_tenures = sorted(all_tenures)

    seeds = list(dfs.keys())

    # --------------------------------------------------------
    # Same quantile logic as HTML
    # --------------------------------------------------------

    def quantile(values, p):

        if len(values) == 0:
            return np.nan

        if len(values) == 1:
            return values[0]

        position = (len(values) - 1) * p

        low = int(np.floor(position))
        high = int(np.ceil(position))
        weight = position - low

        return (
            values[low] * (1 - weight)
            + values[high] * weight
        )

    # --------------------------------------------------------
    # Formatting — same idea as HTML
    # --------------------------------------------------------

    def number(value, digits=decimals):

        if value is None or not np.isfinite(value):
            return "N/A"

        return f"{value:,.{digits}f}"

    def metric_number(value, metric):

        if value is None or not np.isfinite(value):
            return "N/A"

        cfg = metrics[metric]

        if cfg.get("is_ratio", False):

            ratio_scale = cfg.get(
                "ratio_scale",
                1
            )

            display_value = (
                value / ratio_scale * 100
            )

            return (
                f"{display_value:,.{decimals}f}%"
            )

        return f"{value:,.{decimals}f}"

    def delta_number(value, metric):

        if value is None or not np.isfinite(value):
            return "N/A"

        sign = "+" if value > 0 else ""

        cfg = metrics[metric]

        if cfg.get("is_ratio", False):

            ratio_scale = cfg.get(
                "ratio_scale",
                1
            )

            display_value = (
                value / ratio_scale * 100
            )

            return (
                f"{sign}"
                f"{display_value:,.{decimals}f} pp"
            )

        return (
            f"{sign}{value:,.{decimals}f}"
        )

    def percent_number(value):

        if value is None or not np.isfinite(value):
            return "N/A"

        sign = "+" if value > 0 else ""

        return (
            f"{sign}{value:,.{decimals}f}%"
        )

    # --------------------------------------------------------
    # Calculate one chart page
    #
    # filter_type:
    #     None
    #     ventile
    #     region
    #     market
    #     tenure
    # --------------------------------------------------------

    def calculate_page(
        metric,
        filter_type=None,
        filter_value=None
    ):

        cfg = metrics[metric]

        results = {}

        for seed in seeds:

            df = prepared[str(seed)]

            if filter_type == "ventile":

                subset = df[
                    df["_Ventile"] == filter_value
                ]

            elif filter_type == "region":

                subset = df[
                    df["_Region"] == str(filter_value)
                ]

            elif filter_type == "market":

                subset = df[
                    df["_Market"] == str(filter_value)
                ]

            elif filter_type == "tenure":

                subset = df[
                    df["_Tenure"] == str(filter_value)
                ]

            else:
                subset = df

            values = (
                subset[metric]
                .dropna()
                .to_numpy(dtype=float)
            )

            values.sort()

            if len(values) == 0:

                total = np.nan
                minimum = np.nan
                maximum = np.nan
                average = np.nan
                p10 = np.nan
                p50 = np.nan
                p90 = np.nan

            else:

                total_sum = float(
                    np.sum(values)
                )

                average = float(
                    np.mean(values)
                )

                total = (
                    total_sum
                    if cfg["agg"] == "sum"
                    else average
                )

                minimum = float(values[0])
                maximum = float(values[-1])

                p10 = float(
                    quantile(values, 0.10)
                )

                p50 = float(
                    quantile(values, 0.50)
                )

                p90 = float(
                    quantile(values, 0.90)
                )

            probability = (
                subset["_Probability"]
                .dropna()
            )

            if len(probability) > 0:

                predicted_total = float(
                    probability.sum()
                )

                predicted_claims = (
                    predicted_total
                    * claims_share
                )

                predicted_delqs = (
                    predicted_total
                    * (1 - claims_share)
                )

            else:

                predicted_total = np.nan
                predicted_claims = np.nan
                predicted_delqs = np.nan

            results[str(seed)] = {
                "count": len(values),
                "total": total,
                "min": minimum,
                "max": maximum,
                "avg": average,
                "p10": p10,
                "p50": p50,
                "p90": p90,
                "predictedTotal": predicted_total,
                "predictedClaims": predicted_claims,
                "predictedDelqs": predicted_delqs
            }

        baseline_value = (
            results[str(baseline)]["total"]
        )

        for seed in seeds:

            seed = str(seed)

            total = results[seed]["total"]

            if (
                np.isfinite(total)
                and np.isfinite(baseline_value)
            ):

                delta = (
                    total - baseline_value
                )

            else:

                delta = np.nan

            if (
                np.isfinite(delta)
                and np.isfinite(baseline_value)
                and baseline_value != 0
            ):

                delta_pct = (
                    delta
                    / baseline_value
                    * 100
                )

            else:

                delta_pct = np.nan

            results[seed]["delta"] = delta
            results[seed]["deltaPct"] = delta_pct

        return results

    # --------------------------------------------------------
    # Create one Plotly page
    # --------------------------------------------------------

    def create_figure(
        metric,
        page_subtitle,
        filter_type=None,
        filter_value=None
    ):

        cfg = metrics[metric]

        results = calculate_page(
            metric=metric,
            filter_type=filter_type,
            filter_value=filter_value
        )

        fig = go.Figure()

        totals = []

        for seed in seeds:

            seed = str(seed)

            r = results[seed]

            if np.isfinite(r["total"]):
                totals.append(r["total"])

        if totals:

            min_value = min(0, min(totals))
            max_value = max(0, max(totals))

            span = max(
                max_value - min_value,
                1e-9
            )

        else:

            min_value = 0
            max_value = 1
            span = 1

        for i, seed in enumerate(seeds):

            seed = str(seed)

            r = results[seed]

            if not np.isfinite(r["total"]):
                continue

            if seed == str(baseline):

                delta_text = "Baseline"
                delta_pct_text = "Baseline"

            else:

                delta_text = delta_number(
                    r["delta"],
                    metric
                )

                delta_pct_text = percent_number(
                    r["deltaPct"]
                )

            lines = [
                "<b>Min: "
                + metric_number(r["min"], metric)
                + "</b>",

                "<b>Max: "
                + metric_number(r["max"], metric)
                + "</b>",

                "<b>Avg: "
                + metric_number(r["avg"], metric)
                + "</b>",

                "<b>P10: "
                + metric_number(r["p10"], metric)
                + "</b>",

                "<b>P50: "
                + metric_number(r["p50"], metric)
                + "</b>",

                "<b>P90: "
                + metric_number(r["p90"], metric)
                + "</b>",

                "────────────",

                "<b>Pred Delq + Claims: "
                + number(r["predictedTotal"])
                + "</b>",

                "<b>Pred Claims: "
                + number(r["predictedClaims"])
                + "</b>",

                "<b>Pred Delqs: "
                + number(r["predictedDelqs"])
                + "</b>",

                "────────────",

                "<b>Δ: "
                + delta_text
                + "</b>"
            ]

            if not cfg.get(
                "is_ratio",
                False
            ):

                lines.append(
                    "<b>Δ%: "
                    + delta_pct_text
                    + "</b>"
                )

            inside_text = "<br>".join(lines)

            fig.add_trace(
                go.Bar(
                    x=[i],
                    y=[r["total"]],
                    width=0.78,

                    marker=dict(
                        color=colors.get(
                            seed,
                            "#64748b"
                        )
                    ),

                    text=[inside_text],

                    textposition="inside",

                    insidetextanchor="middle",

                    constraintext="none",

                    cliponaxis=False,

                    textfont=dict(
                        size=inside_font_size,
                        color="white",
                        family="Arial"
                    ),

                    hoverinfo="skip",

                    showlegend=False
                )
            )

            fig.add_annotation(
                x=i,
                y=r["total"],

                text=(
                    "<b>"
                    + metric_number(
                        r["total"],
                        metric
                    )
                    + "</b>"
                ),

                showarrow=False,

                yshift=25
                    if r["total"] >= 0
                    else -25,

                font=dict(
                    size=22,
                    color="#172033",
                    family="Arial"
                )
            )

        # ----------------------------------------------------
        # Title/subtitle
        # ----------------------------------------------------

        title_text = (
            "<b>"
            + metric
            + " — Random Seed Sensitivity"
            + "</b>"
            + "<br>"
            + "<sup>"
            + str(page_subtitle)
            + "</sup>"
        )

        fig.update_layout(

            title=dict(
                text=title_text,
                x=0.5,
                xanchor="center",

                font=dict(
                    size=28,
                    family="Arial",
                    color="#172033"
                )
            ),

            width=page_width,
            height=page_height,

            template="plotly_white",

            margin=dict(
                t=145,
                b=105,
                l=115,
                r=80
            ),

            xaxis=dict(

                tickmode="array",

                tickvals=list(
                    range(len(seeds))
                ),

                ticktext=[
                    str(seed)
                    for seed in seeds
                ],

                range=[
                    -0.6,
                    len(seeds) - 0.4
                ],

                showgrid=False,

                title=dict(
                    text="Experiment",
                    font=dict(size=17)
                ),

                tickfont=dict(
                    size=16
                )
            ),

            yaxis=dict(

                title=dict(
                    text=metric,
                    font=dict(size=17)
                ),

                range=[
                    min_value
                    - span * 0.08,

                    max_value
                    + span * 0.25
                ],

                ticksuffix=(
                    "%"
                    if (
                        cfg.get(
                            "is_ratio",
                            False
                        )
                        and cfg.get(
                            "ratio_scale",
                            1
                        ) == 100
                    )
                    else ""
                ),

                tickfont=dict(
                    size=15
                ),

                gridcolor="#e5e7eb"
            ),

            uniformtext=dict(
                minsize=10,
                mode="show"
            ),

            paper_bgcolor="white",
            plot_bgcolor="white",

            font=dict(
                family="Arial"
            ),

            bargap=0.12
        )

        return fig

    # --------------------------------------------------------
    # Build page definitions
    # --------------------------------------------------------

    pages = []

    for metric in metric_names:

        # Overall page
        pages.append({
            "metric": metric,
            "subtitle": "All Ventiles",
            "filter_type": None,
            "filter_value": None
        })

        # Each Ventile
        for ventile in all_ventiles:

            pages.append({
                "metric": metric,
                "subtitle": (
                    f"Ventile {ventile}"
                ),
                "filter_type": "ventile",
                "filter_value": ventile
            })

        # Each ECON_REGION
        for region in all_regions:

            pages.append({
                "metric": metric,
                "subtitle": (
                    f"ECON_REGION: {region}"
                ),
                "filter_type": "region",
                "filter_value": region
            })

        # Each POD_MARKET_CODE
        for market in all_markets:

            pages.append({
                "metric": metric,
                "subtitle": (
                    f"POD_MARKET_CODE: {market}"
                ),
                "filter_type": "market",
                "filter_value": market
            })

        # Each PRK_TENURE
        for tenure in all_tenures:

            pages.append({
                "metric": metric,
                "subtitle": (
                    f"PRK_TENURE: {tenure}"
                ),
                "filter_type": "tenure",
                "filter_value": tenure
            })

    print(
        f"Creating {len(pages)} PDF pages..."
    )

    # --------------------------------------------------------
    # Convert every Plotly chart to an image
    # --------------------------------------------------------

    pdf_images = []

    for page_number, page in enumerate(
        pages,
        start=1
    ):

        print(
            f"Creating page "
            f"{page_number}/{len(pages)}: "
            f"{page['metric']} - "
            f"{page['subtitle']}"
        )

        fig = create_figure(
            metric=page["metric"],
            page_subtitle=page["subtitle"],
            filter_type=page["filter_type"],
            filter_value=page["filter_value"]
        )

        try:

            png_bytes = fig.to_image(
                format="png",
                width=page_width,
                height=page_height,
                scale=1
            )

        except Exception as e:

            raise RuntimeError(
                "\nPDF image export failed.\n"
                "Install Kaleido in your notebook:\n\n"
                "%pip install kaleido\n\n"
                "Then restart the kernel and rerun.\n"
            ) from e

        image = PILImage.open(
            BytesIO(png_bytes)
        ).convert("RGB")

        pdf_images.append(image)

    # --------------------------------------------------------
    # Save all chart images into one multipage PDF
    # --------------------------------------------------------

    output = Path(filename).resolve()

    if not pdf_images:
        raise ValueError(
            "No PDF pages were generated"
        )

    first_image = pdf_images[0]

    remaining_images = pdf_images[1:]

    first_image.save(
        output,
        "PDF",
        resolution=120.0,
        save_all=True,
        append_images=remaining_images
    )

    # Close image objects
    for image in pdf_images:
        image.close()

    print()
    print(f"PDF report created: {output}")

    print(
        f"File size: "
        f"{output.stat().st_size / (1024 ** 2):.2f} MB"
    )

    print(
        f"Total pages: {len(pages)}"
    )

    return output

    region_col="ECON_REGION",
    market_col="POD_MARKET_CODE",
    tenure_col="PRK_TENURE",

    probability_col="prediction_probability",

    metrics=None,

    baseline="RS42",

    claims_share=0.62,

    colors=None,

    decimals=2,

    inside_font_size=12,

    chart_height=900
):

    # --------------------------------------------------------
    # HTML — your existing report

        ventile_col=ventile_col,

        region_col=region_col,

        market_col=market_col,

        tenure_col=tenure_col,
