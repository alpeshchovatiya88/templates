"""Run independent MoDeVa validation on the fitted SMI3 model."""

from __future__ import annotations

import argparse
import hashlib
import json
import logging
import os
import tarfile
import traceback
from pathlib import Path
from tempfile import TemporaryDirectory

import joblib
import pandas as pd
import yaml
from sklearn.metrics import (
    brier_score_loss,
    log_loss,
    roc_auc_score,
)

LOG = logging.getLogger("modeva_validation")


def load_model(path: str):
    source = Path(path)

    if source.name.endswith(".tar.gz"):
        with TemporaryDirectory() as temp_dir:
            with tarfile.open(source, "r:gz") as archive:
                archive.extractall(temp_dir, filter="data")

            preferred = (
                Path(temp_dir)
                / "deployed_model"
                / "best_xgb.joblib"
            )
            matches = (
                [preferred]
                if preferred.is_file()
                else list(
                    Path(temp_dir).rglob("best_xgb.joblib")
                )
            )

            if len(matches) != 1:
                raise ValueError(
                    "Expected exactly one best_xgb.joblib "
                    f"in {path}; found {len(matches)}"
                )

            # Only load artifacts from your trusted training job.
            model = joblib.load(matches[0])
    else:
        model = joblib.load(source)

    if (
        not hasattr(model, "named_steps")
        or not hasattr(model, "predict_proba")
    ):
        raise TypeError(
            "Expected a fitted sklearn Pipeline "
            "with predict_proba"
        )

    model_hash = hashlib.sha256(
        source.read_bytes()
    ).hexdigest()
    return model, model_hash


def model_frame(
    frame: pd.DataFrame,
    model,
) -> pd.DataFrame:
    features = [
        str(column)
        for column in (
            model.named_steps["pre"].feature_names_in_
        )
    ]
    missing = sorted(
        set(features) - set(frame.columns)
    )

    if missing:
        raise ValueError(
            f"Missing model features: {missing}"
        )

    return frame.loc[:, features].copy()


def save_result(
    name: str,
    result,
    output_dir: Path,
) -> None:
    folder = output_dir / name
    folder.mkdir(parents=True, exist_ok=True)

    if isinstance(result.table, pd.DataFrame):
        result.table.to_csv(
            folder / "table.csv",
            index=False,
        )
    elif isinstance(result.table, dict):
        for key, table in result.table.items():
            if isinstance(table, pd.DataFrame):
                safe_name = "".join(
                    character
                    if character.isalnum()
                    or character in "_-"
                    else "_"
                    for character in str(key)
                )
                table.to_csv(
                    folder / f"{safe_name}.csv",
                    index=False,
                )

    metadata = {
        "key": result.key,
        "inputs": result.inputs,
    }
    (folder / "metadata.json").write_text(
        json.dumps(
            metadata,
            indent=2,
            default=str,
        ),
        encoding="utf-8",
    )

    try:
        result.plot_save(
            file_name=str(folder / "figure"),
            format="png",
        )
    except Exception as exc:
        LOG.warning(
            "Could not save figures for %s: %s",
            name,
            exc,
        )


def run_test(
    report: dict,
    output_dir: Path,
    name: str,
    test_function,
) -> None:
    try:
        result = test_function()
        save_result(
            name,
            result,
            output_dir,
        )
        report["tests"][name] = {
            "status": "completed"
        }
    except Exception as exc:
        LOG.exception(
            "MoDeVa test failed: %s",
            name,
        )
        report["tests"][name] = {
            "status": "failed",
            "error": str(exc),
            "traceback": traceback.format_exc(
                limit=5
            ),
        }


def log_validation_to_mlflow(
    report: dict,
    output_dir: Path,
) -> str | None:
    tracking_arn = os.environ.get(
        "MLFLOW_TRACKING_SERVER_ARN"
    )
    if not tracking_arn:
        return None

    import mlflow

    mlflow.set_tracking_uri(tracking_arn)

    experiment_name = os.environ.get(
        "MLFLOW_EXPERIMENT_NAME"
    )
    if experiment_name:
        mlflow.set_experiment(
            experiment_name
        )

    # Otherwise MLflow may resume the parent run.
    os.environ.pop(
        "MLFLOW_RUN_ID",
        None,
    )

    parent_id = os.environ.get(
        "SMI3_PARENT_RUN_ID"
    )
    tags = {
        "validation_tool": "MoDeVa",
        "validation_status": report["status"],
    }
    if parent_id:
        tags["mlflow.parentRunId"] = (
            parent_id
        )

    with mlflow.start_run(
        run_name="SMI3-MoDeVa-Validation",
        tags=tags,
    ) as run:
        if report.get("model_sha256"):
            mlflow.log_param(
                "model_sha256",
                report["model_sha256"],
            )

        for name, count in report.get(
            "counts", {}
        ).items():
            mlflow.log_metric(
                f"validation_{name}",
                float(count),
            )

        for name, value in report.get(
            "metrics", {}
        ).items():
            mlflow.log_metric(
                name,
                float(value),
            )

        for name, test in report[
            "tests"
        ].items():
            mlflow.set_tag(
                f"modeva.{name}",
                test["status"],
            )

        mlflow.log_artifacts(
            str(output_dir),
            artifact_path=(
                "modeva_validation"
            ),
        )
        return run.info.run_id


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--model-path",
        required=True,
    )
    parser.add_argument(
        "--train",
        required=True,
    )
    parser.add_argument(
        "--test",
        required=True,
    )
    parser.add_argument(
        "--recent",
        required=True,
    )
    parser.add_argument(
        "--config",
        required=True,
    )
    parser.add_argument(
        "--output",
        default=(
            "/opt/ml/processing/validation"
        ),
    )
    args = parser.parse_args()

    output_dir = Path(args.output)
    output_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    report = {
        "tests": {},
        "findings": [],
        "counts": {},
    }

    try:
        from modeva import (
            DataSet,
            TestSuite,
        )
        from modeva.models import (
            MoScoredClassifier,
            MoSKLearnClassifier,
        )

        config = yaml.safe_load(
            Path(args.config).read_text(
                encoding="utf-8"
            )
        ) or {}

        label = config.get(
            "label",
            "target",
        )
        eligible_status = config.get(
            "eligible_status",
            "Funded",
        )

        model, model_hash = load_model(
            args.model_path
        )
        report["model_sha256"] = (
            model_hash
        )

        raw = {
            "train": pd.read_csv(
                args.train
            ),
            "test": pd.read_csv(
                args.test
            ),
            "recent": pd.read_csv(
                args.recent
            ),
        }

        # Outcome tests use the population
        # with observable loan outcomes.
        eligible = {}

        for name, frame in raw.items():
            report["counts"][
                f"{name}_input"
            ] = len(frame)

            if "STATUS" in frame.columns:
                frame = frame.loc[
                    frame["STATUS"].eq(
                        eligible_status
                    )
                ].copy()

            eligible[name] = frame

            report["counts"][
                f"{name}_eligible"
            ] = len(frame)
            report["counts"][
                f"{name}_excluded"
            ] = (
                len(raw[name])
                - len(frame)
            )

        if "category" in raw[
            "test"
        ].columns:
            declined = raw["test"].loc[
                raw["test"][
                    "category"
                ].eq("Declined")
            ]

            if (
                len(declined)
                and label in declined.columns
                and declined[
                    label
                ].notna().any()
            ):
                report["findings"].append(
                    "REVIEW: declined applications "
                    "have non-null test targets. "
                    "They were excluded from "
                    "outcome validation. Confirm "
                    "the target definition."
                )

        labeled = {}

        for name in (
            "train",
            "test",
        ):
            if label not in eligible[
                name
            ].columns:
                raise ValueError(
                    f"{name} has no "
                    f"{label!r} column"
                )

            frame = eligible[
                name
            ].dropna(
                subset=[label]
            ).copy()

            if (
                frame.empty
                or set(
                    frame[
                        label
                    ].unique()
                ) != {0, 1}
            ):
                raise ValueError(
                    f"{name} needs "
                    "eligible labeled "
                    "observations of "
                    "both classes"
                )

            labeled[name] = frame
            report["counts"][
                f"{name}_labeled"
            ] = len(frame)

        train = model_frame(
            labeled["train"],
            model,
        ).assign(
            **{
                label: (
                    labeled["train"][
                        label
                    ]
                    .astype(int)
                    .to_numpy()
                )
            }
        )

        test = model_frame(
            labeled["test"],
            model,
        ).assign(
            **{
                label: (
                    labeled["test"][
                        label
                    ]
                    .astype(int)
                    .to_numpy()
                )
            }
        )

        y_test = (
            labeled["test"][
                label
            ]
            .astype(int)
            .to_numpy()
        )
        probability = (
            model.predict_proba(
                model_frame(
                    labeled["test"],
                    model,
                )
            )[:, 1]
        )

        report["metrics"] = {
            "validation_test_auc": (
                roc_auc_score(
                    y_test,
                    probability,
                )
            ),
            "validation_test_log_loss": (
                log_loss(
                    y_test,
                    probability,
                )
            ),
            "validation_test_brier": (
                brier_score_loss(
                    y_test,
                    probability,
                )
            ),
        }

        ds = DataSet(
            name="smi3_validator"
        )
        ds.load_dataframe_train_test(
            train=train,
            test=test,
        )
        ds.set_target(label)
        ds.set_task_type(
            "Classification"
        )

        wrapped_model = (
            MoSKLearnClassifier(
                estimator=model,
                name="smi3_xgb",
            )
        )
        suite = TestSuite(
            dataset=ds,
            model=wrapped_model,
        )

        standard_tests = {
            "train_summary": (
                lambda: ds.summary(
                    dataset="train"
                )
            ),
            "test_summary": (
                lambda: ds.summary(
                    dataset="test"
                )
            ),
            "train_test_drift": (
                lambda:
                    ds.data_drift_test(
                        dataset1="train",
                        dataset2="test",
                        distance_metric="PSI",
                    )
            ),
            "accuracy": (
                lambda:
                    suite.diagnose_accuracy_table(
                        metric=(
                            "AUC",
                            "ACC",
                            "F1",
                            "LogLoss",
                            "Brier",
                        )
                    )
            ),
            "reliability": (
                lambda:
                    suite.diagnose_reliability(
                        train_dataset="test",
                        test_dataset="test",
                        test_size=0.5,
                        alpha=0.1,
                        random_state=42,
                    )
            ),
            "resilience": (
                lambda:
                    suite.diagnose_resilience(
                        dataset="test",
                        metric="AUC",
                        random_state=42,
                    )
            ),
            "robustness": (
                lambda:
                    suite.diagnose_robustness(
                        dataset="test",
                        metric="AUC",
                        n_repeats=5,
                        perturb_features=(
                            tuple(
                                config.get(
                                    "perturb_features",
                                    [],
                                )
                            )
                            or None
                        ),
                        noise_levels=0.05,
                        random_state=42,
                    )
            ),
            "permutation_importance": (
                lambda:
                    suite.explain_pfi(
                        dataset="test",
                        sample_size=1000,
                        n_repeats=5,
                        random_state=42,
                    )
            ),
        }

        for name, test_function in (
            standard_tests.items()
        ):
            run_test(
                report,
                output_dir,
                name,
                test_function,
            )

        for feature in config.get(
            "slice_features",
            [],
        ):
            if feature not in train.columns:
                report["tests"][
                    f"slice_{feature}"
                ] = {
                    "status": "skipped",
                    "reason": (
                        "Not a model input"
                    ),
                }
                continue

            run_test(
                report,
                output_dir,
                f"slice_{feature}",
                lambda feature=feature:
                    suite.diagnose_slicing_accuracy(
                        features=feature,
                        dataset="test",
                        metric="AUC",
                        method="quantile",
                        bins=5,
                    ),
            )

            run_test(
                report,
                output_dir,
                f"overfit_{feature}",
                lambda feature=feature:
                    suite.diagnose_slicing_overfit(
                        features=feature,
                        train_dataset="train",
                        test_dataset="test",
                        metric="AUC",
                        method="quantile",
                        bins=5,
                    ),
            )

            run_test(
                report,
                output_dir,
                f"ale_{feature}",
                lambda feature=feature:
                    suite.explain_ale(
                        features=feature,
                        dataset="test",
                        sample_size=1000,
                        random_state=42,
                    ),
            )

        if config.get(
            "run_expensive_tests",
            False,
        ):
            run_test(
                report,
                output_dir,
                "weakness_regions",
                lambda:
                    suite.diagnose_weakness_region(
                        train_dataset="train",
                        test_dataset="test",
                        metric="AUC",
                        random_state=42,
                    ),
            )

            run_test(
                report,
                output_dir,
                "local_shap",
                lambda: suite.explain_shap(
                    dataset="test",
                    sample_index=0,
                    baseline_dataset="train",
                    baseline_sample_size=100,
                    random_state=42,
                ),
            )

            for feature in config.get(
                "slice_features",
                [],
            ):
                if feature in test.columns:
                    run_test(
                        report,
                        output_dir,
                        f"pdp_{feature}",
                        lambda feature=feature:
                            suite.explain_pdp(
                                features=feature,
                                dataset="test",
                                sample_size=1000,
                                random_state=42,
                            ),
                    )

        if len(
            eligible["recent"]
        ):
            recent_x = model_frame(
                eligible["recent"],
                model,
            )
            drift_ds = DataSet(
                name=(
                    "smi3_recent_shift"
                )
            )
            drift_ds.load_dataframe_train_test(
                train=model_frame(
                    labeled["train"],
                    model,
                ),
                test=recent_x,
            )

            run_test(
                report,
                output_dir,
                "train_recent_drift",
                lambda:
                    drift_ds.data_drift_test(
                        dataset1="train",
                        dataset2="test",
                        distance_metric="PSI",
                    ),
            )

            report["tests"][
                "recent_performance"
            ] = {
                "status": "skipped",
                "reason": (
                    "Recent outcome "
                    "maturity has not "
                    "been established"
                ),
            }

        groups = config.get(
            "fairness_groups",
            {},
        )

        if groups:
            scored = model_frame(
                labeled["test"],
                model,
            ).reset_index(
                drop=True
            )
            scored[label] = (
                labeled["test"][
                    label
                ]
                .astype(int)
                .to_numpy()
            )
            scored[
                "_validation_probability"
            ] = (
                model.predict_proba(
                    model_frame(
                        labeled["test"],
                        model,
                    )
                )[:, 1]
            )

            group_columns = sorted(
                {
                    definition[
                        "feature"
                    ]
                    for definition
                    in groups.values()
                }
            )
            for column in group_columns:
                scored[column] = (
                    labeled["test"][
                        column
                    ].to_numpy()
                )

            fair_ds = DataSet(
                name="smi3_fairness"
            )
            fair_ds.load_dataframe(
                scored
            )
            fair_ds.set_target(
                label
            )
            fair_ds.set_task_type(
                "Classification"
            )
            fair_ds.set_prediction_proba(
                "_validation_probability"
            )
            fair_ds.set_inactive_features(
                group_columns
            )

            fair_model = (
                MoScoredClassifier(
                    dataset=fair_ds,
                    name="smi3_scored",
                )
            )
            fair_suite = TestSuite(
                dataset=fair_ds,
                model=fair_model,
            )

            run_test(
                report,
                output_dir,
                "fairness",
                lambda:
                    fair_suite.diagnose_fairness(
                        group_config=groups,
                        favorable_label=int(
                            config[
                                "favorable_label"
                            ]
                        ),
                        dataset="main",
                    ),
            )
        else:
            report["tests"][
                "fairness"
            ] = {
                "status": "skipped",
                "reason": (
                    "No approved group "
                    "definitions configured"
                ),
            }

        incumbent_path = config.get(
            "incumbent_model_path"
        )

        if incumbent_path:
            incumbent, incumbent_hash = (
                load_model(
                    incumbent_path
                )
            )
            report[
                "incumbent_sha256"
            ] = incumbent_hash

            incumbent_features = [
                str(column)
                for column in (
                    incumbent
                    .named_steps["pre"]
                    .feature_names_in_
                )
            ]
            challenger_features = [
                str(column)
                for column in (
                    model
                    .named_steps["pre"]
                    .feature_names_in_
                )
            ]

            if (
                incumbent_features
                != challenger_features
            ):
                raise ValueError(
                    "Incumbent and "
                    "challenger require "
                    "different input "
                    "features"
                )

            comparison = TestSuite(
                dataset=ds,
                models=[
                    wrapped_model,
                    MoSKLearnClassifier(
                        estimator=incumbent,
                        name="incumbent",
                    ),
                ],
            )

            run_test(
                report,
                output_dir,
                "incumbent_comparison",
                lambda:
                    comparison.compare_accuracy_table(
                        metric=(
                            "AUC",
                            "LogLoss",
                            "Brier",
                        )
                    ),
            )

    except Exception as exc:
        LOG.exception(
            "Validation setup failed"
        )
        report["setup_error"] = (
            str(exc)
        )
        report[
            "setup_traceback"
        ] = traceback.format_exc(
            limit=8
        )

    failed = (
        "setup_error" in report
        or any(
            test["status"] == "failed"
            for test
            in report["tests"].values()
        )
    )
    report["status"] = (
        "FAILED"
        if failed
        else (
            "REVIEW_REQUIRED"
            if report["findings"]
            else "COMPLETED"
        )
    )

    report_path = (
        output_dir
        / "validation_report.json"
    )
    report_path.write_text(
        json.dumps(
            report,
            indent=2,
            default=str,
        ),
        encoding="utf-8",
    )

    try:
        run_id = (
            log_validation_to_mlflow(
                report,
                output_dir,
            )
        )
        if run_id:
            report[
                "mlflow_run_id"
            ] = run_id
            report_path.write_text(
                json.dumps(
                    report,
                    indent=2,
                    default=str,
                ),
                encoding="utf-8",
            )
    except Exception as exc:
        LOG.exception(
            "MLflow logging failed"
        )
        report[
            "mlflow_error"
        ] = str(exc)
        report["status"] = (
            "FAILED"
        )
        report_path.write_text(
            json.dumps(
                report,
                indent=2,
                default=str,
            ),
            encoding="utf-8",
        )
        failed = True

    LOG.info(
        "Validation status: %s",
        report["status"],
    )
    return 2 if failed else 0


if __name__ == "__main__":
    logging.basicConfig(
        level=logging.INFO
    )
    raise SystemExit(main())
