import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==========================
# Read Excel File
# ==========================
file_path = "feature_importance.xlsx"   # <-- Update filename

df = pd.read_excel(file_path)

# ==========================
# Sort by All Model Rank
# ==========================
df = df.sort_values("SHAP Rank_All Model")

# ==========================
# Extract Columns
# ==========================
features = df["Feature"]

all_val = df["Abs Shap Value_All Model"]
funded_val = df["Abs Shap Value_Funded Model"]

all_rank = df["SHAP Rank_All Model"]
funded_rank = df["SHAP Rank_Funded Model"]

# ==========================
# Plot Settings
# ==========================
n_features = len(df)

fig, ax = plt.subplots(
    figsize=(20, max(12, n_features * 0.55))
)

y = np.arange(n_features)
bar_height = 0.38

# ==========================
# Bars
# ==========================
bars1 = ax.barh(
    y - bar_height/2,
    all_val,
    height=bar_height,
    label="All Model"
)

bars2 = ax.barh(
    y + bar_height/2,
    funded_val,
    height=bar_height,
    label="Funded Model"
)

# ==========================
# Labels
# ==========================
ax.set_yticks(y)
ax.set_yticklabels(features, fontsize=11)

ax.set_xlabel(
    "Mean Absolute SHAP Value",
    fontsize=14
)

ax.set_title(
    "SHAP Feature Importance Comparison\n(All Model vs Funded Model)",
    fontsize=18,
    fontweight="bold"
)

ax.legend(fontsize=12)

# Highest rank at top
ax.invert_yaxis()

# ==========================
# Add Value + Rank Labels
# ==========================
x_offset = max(
    all_val.max(),
    funded_val.max()
) * 0.01

for i in range(n_features):

    ax.text(
        all_val.iloc[i] + x_offset,
        y[i] - bar_height/2,
        f"{all_val.iloc[i]:.3f} (R{int(all_rank.iloc[i])})",
        va="center",
        fontsize=9
    )

    ax.text(
        funded_val.iloc[i] + x_offset,
        y[i] + bar_height/2,
        f"{funded_val.iloc[i]:.3f} (R{int(funded_rank.iloc[i])})",
        va="center",
        fontsize=9
    )

# ==========================
# Grid
# ==========================
ax.grid(
    axis="x",
    linestyle="--",
    alpha=0.4
)

# Leave room for labels
plt.tight_layout()

# ==========================
# Save High Resolution Image
# ==========================
plt.savefig(
    "SHAP_Comparison_All_Features.png",
    dpi=300,
    bbox_inches="tight"
)

plt.show()


ax.text(
    funded_val.iloc[i] + x_offset,
    y[i] + bar_height/2,
    f"{funded_val.iloc[i]:.3f} (R{int(funded_rank.iloc[i])}, Δ{int(funded_rank.iloc[i]-all_rank.iloc[i])})",
    va="center",
    fontsize=9
)
