import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ============================================================
# READ EXCEL
# ============================================================
df = pd.read_excel("your_file.xlsx")   # change file name

# ============================================================
# CLEAN COLUMN NAMES
# ============================================================
df.columns = df.columns.str.strip()

print(df.columns.tolist())

# ============================================================
# NORMALIZE FUNCTION
# ============================================================
def normalize_col(series):
    series = pd.to_numeric(series, errors="coerce")
    return series / series.max()

# ============================================================
# NORMALIZE SHAP AND IV
# ============================================================
df["SHAP_All_Norm"] = normalize_col(df["SHAP_All"])
df["SHAP_Funded_Norm"] = normalize_col(df["SHAP_Funded"])

df["IV_All_Norm"] = normalize_col(df["IV_All"])
df["IV_Funded_Norm"] = normalize_col(df["IV_Funded"])

# ============================================================
# SAFE RANK FORMATTER
# ============================================================
def fmt_rank(x):
    if pd.isna(x):
        return "NA"
    return str(int(x))

# ============================================================
# SORT DATA
# ============================================================
df = df.sort_values("SHAP_Rank_All").reset_index(drop=True)

# ============================================================
# VARIABLES
# ============================================================
features = df["Feature"]

shap_all = df["SHAP_All_Norm"]
iv_all = df["IV_All_Norm"]

shap_funded = df["SHAP_Funded_Norm"]
iv_funded = df["IV_Funded_Norm"]

shap_rank_all = df["SHAP_Rank_All"]
iv_rank_all = df["IV_Rank_All"]

shap_rank_funded = df["SHAP_Rank_Funded"]
iv_rank_funded = df["IV_Rank_Funded"]

n_features = len(df)
y = np.arange(n_features)

# ============================================================
# PLOT SETTINGS
# ============================================================
plt.rcParams["figure.dpi"] = 150

fig, ax = plt.subplots(figsize=(32, 60))

bar_height = 0.16

# ============================================================
# FOUR BARS PER FEATURE
# ============================================================
ax.barh(
    y - 1.5 * bar_height,
    shap_all,
    height=bar_height,
    label="SHAP All"
)

ax.barh(
    y - 0.5 * bar_height,
    iv_all,
    height=bar_height,
    label="IV All"
)

ax.barh(
    y + 0.5 * bar_height,
    shap_funded,
    height=bar_height,
    label="SHAP Funded"
)

ax.barh(
    y + 1.5 * bar_height,
    iv_funded,
    height=bar_height,
    label="IV Funded"
)

# ============================================================
# AXIS FORMATTING
# ============================================================
ax.set_yticks(y)
ax.set_yticklabels(features, fontsize=16)

ax.set_xlabel(
    "Normalized Value",
    fontsize=22,
    fontweight="bold"
)

ax.set_title(
    "Normalized SHAP vs IV Comparison\nAll Model and Funded Model",
    fontsize=26,
    fontweight="bold",
    pad=30
)

ax.tick_params(axis="x", labelsize=16)

ax.legend(
    fontsize=18,
    loc="upper right"
)

ax.invert_yaxis()

# ============================================================
# LABELS: VALUE + RANK
# ============================================================
max_value = max(
    shap_all.max(),
    iv_all.max(),
    shap_funded.max(),
    iv_funded.max()
)

offset = max_value * 0.01

for i in range(n_features):

    ax.text(
        shap_all.iloc[i] + offset,
        y[i] - 1.5 * bar_height,
        f"{shap_all.iloc[i]:.3f} (R{fmt_rank(shap_rank_all.iloc[i])})",
        va="center",
        fontsize=10
    )

    ax.text(
        iv_all.iloc[i] + offset,
        y[i] - 0.5 * bar_height,
        f"{iv_all.iloc[i]:.3f} (R{fmt_rank(iv_rank_all.iloc[i])})",
        va="center",
        fontsize=10
    )

    ax.text(

        f"{shap_funded.iloc[i]:.3f} (R{fmt_rank(shap_rank_funded.iloc[i])})",
        va="center",
        fontsize=10
    )

    ax.text(
        iv_funded.iloc[i] + offset,
        y[i] + 1.5 * bar_height,
        f"{iv_funded.iloc[i]:.3f} (R{fmt_rank(iv_rank_funded.iloc[i])})",
        va="center",
        fontsize=10
    )
