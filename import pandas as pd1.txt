import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ============================================================
# READ EXCEL
# ============================================================
df = pd.read_excel("feature_importance.xlsx")

# ============================================================
# SORT BY ALL MODEL RANK
# ============================================================
df = df.sort_values("Rank_All")

# ============================================================
# COLUMNS
# ============================================================
features = df["Feature"]

all_val = df["All Model"]
funded_val = df["Funded Model"]

all_rank = df["Rank_All"]
funded_rank = df["Rank_Funded"]

# ============================================================
# PLOT SETTINGS
# ============================================================
plt.rcParams["figure.dpi"] = 150

n_features = len(df)

fig, ax = plt.subplots(
    figsize=(26, 40)      # <-- BIG HEIGHT
)

# ============================================================
# BAR SETTINGS
# ============================================================
y = np.arange(n_features)

bar_height = 0.30

ax.barh(
    y - bar_height/2,
    all_val,
    height=bar_height,
    label="All Model"
)

ax.barh(
    y + bar_height/2,
    funded_val,
    height=bar_height,
    label="Funded Model"
)

# ============================================================
# AXES
# ============================================================
ax.set_yticks(y)

ax.set_yticklabels(
    features,
    fontsize=15
)

ax.tick_params(
    axis='x',
    labelsize=14
)

ax.set_xlabel(
    "Mean Absolute SHAP Value",
    fontsize=18,
    fontweight="bold"
)

ax.set_title(
    "SHAP Feature Importance Comparison\n(All Model vs Funded Model)",
    fontsize=22,
    fontweight="bold",
    pad=25
)

ax.legend(
    fontsize=15,
    loc="upper right"
)

ax.invert_yaxis()

# ============================================================
# LABELS (VALUE + RANK)
# ============================================================
max_value = max(
    all_val.max(),
    funded_val.max()
)

offset = max_value * 0.01

for i in range(n_features):

    ax.text(
        all_val.iloc[i] + offset,
        y[i] - bar_height/2,
        f"{all_val.iloc[i]:.3f} (R{int(all_rank.iloc[i])})",
        va="center",
        fontsize=12
    )

    ax.text(
        funded_val.iloc[i] + offset,
        y[i] + bar_height/2,
        f"{funded_val.iloc[i]:.3f} (R{int(funded_rank.iloc[i])})",
        va="center",
        fontsize=12
    )

# ============================================================
# GRID
# ============================================================
ax.grid(
    axis="x",
    linestyle="--",
    alpha=0.4
)

# ============================================================
# EXTRA SPACE FOR LABELS
# ============================================================
ax.set_xlim(
    0,
    max_value * 1.40
)

# ============================================================
# LAYOUT
# ============================================================
plt.tight_layout()

# ============================================================
# SAVE
# ============================================================
plt.savefig(
    "SHAP_Comparison.png",
    dpi=400,
    bbox_inches="tight"
)

plt.show()
