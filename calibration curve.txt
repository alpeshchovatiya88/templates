from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.calibration import CalibrationDisplay
from sklearn.metrics import (
    roc_auc_score,
    average_precision_score,
    log_loss,
    roc_curve,
    precision_recall_curve,
)

FILE = "df_scored_test_comparison.csv"
TARGET = "target"
ID = "APPLICATION_NUMBER"

SCORES = {
    "XGBoost": "pred_proba",
    "CatBoost": "catboost_pred_proba",
}

OUTCOMES_MATURE = True  # Confirm before running outcome-based comparisons.
GROUP_COLUMN = None    # Optional borrower/group identifier.
N_BINS = 10
BOOTSTRAPS = 500
SEED = 42

OUTPUT = Path("xgb_catboost_comparison")
OUTPUT.mkdir(exist_ok=True)

COLORS = {
    "XGBoost": "tab:blue",
    "CatBoost": "tab:orange",
}

df = pd.read_csv(FILE)

# Apply your approved population filter here, if required.
# df = df.query('STATUS == "Funded"').copy()

if not OUTCOMES_MATURE:
    raise ValueError(
        "Calibration and performance comparison require mature outcomes."
    )

if df.empty:
    raise ValueError("Empty comparison population")

if df[ID].isna().any() or df[ID].duplicated().any():
    raise ValueError("Missing or duplicate application IDs")

df[TARGET] = pd.to_numeric(df[TARGET], errors="coerce")

if not df[TARGET].isin([0, 1]).all():
    raise ValueError("Missing/nonbinary outcomes")

df[TARGET] = df[TARGET].astype(int)

if df[TARGET].nunique() != 2:
    raise ValueError("Both outcome classes required for this comparison")

for model_name, column in SCORES.items():
    df[column] = pd.to_numeric(df[column], errors="coerce")

    if (
        not np.isfinite(df[column]).all()
        or not df[column].between(0, 1).all()
    ):
        raise ValueError(f"{model_name}: invalid/missing probabilities")

if GROUP_COLUMN:
    if (
        GROUP_COLUMN not in df
        or df[GROUP_COLUMN].isna().any()
    ):
        raise ValueError("Missing borrower/group identifiers")

print(
    f"Applications: {len(df):,}\n"
    f"Bad events: {df[TARGET].sum():,}\n"
    f"Bad-event rate: {df[TARGET].mean():.2%}"
)


def save_plot(name):
    plt.tight_layout()
    plt.savefig(
        OUTPUT / f"{name}.png",
        dpi=160,
        bbox_inches="tight",
    )
    plt.show()
    plt.close()

def model_metrics(y, p):
    expected = p.sum()

    return {
        "AUC": roc_auc_score(y, p),
        "Average_precision": average_precision_score(y, p),
        "Brier": np.mean((y - p) ** 2),
        "Log_loss": log_loss(y, p, labels=[0, 1]),
        "Mean_PD": p.mean(),
        "Observed_bad_rate": y.mean(),
        "Observed_expected": (
            y.sum() / expected if expected else np.nan
        ),
    }


y = df[TARGET].to_numpy()

comparison = pd.DataFrame({
    model_name: model_metrics(
        y, df[column].to_numpy()
    )
    for model_name, column in SCORES.items()
}).T

display(comparison)

comparison.to_csv(
    OUTPUT / "metric_comparison.csv",
    index=True,
)

fig, axes = plt.subplots(1, 2, figsize=(13, 5))

for model_name, column in SCORES.items():
    p = df[column].to_numpy()

    fpr, tpr, _ = roc_curve(y, p)

    axes[0].plot(
        fpr,

Cell 4 — paired bootstrap: are the differences distinguishable?
==============================================================================
rng = np.random.default_rng(SEED)

METRICS = [
    "AUC",
    "Average_precision",
    "Brier",
    "Log_loss",
]

units = None

if GROUP_COLUMN:
    units = list(
        df.groupby(GROUP_COLUMN, sort=False).indices.values()
    )

    if len(units) < 2:
        raise ValueError("At least two independent groups required")

draws = []

for _ in range(BOOTSTRAPS):
    if units is None:
        indices = rng.integers(0, len(df), len(df))
    else:
        selected = rng.integers(0, len(units), len(units))
        indices = np.concatenate([units[i] for i in selected])

    sample = df.iloc[indices]
    sample_y = sample[TARGET].to_numpy()

    if np.unique(sample_y).size != 2:
        continue

    xgb = model_metrics(
        sample_y,
        sample[SCORES["XGBoost"]].to_numpy(),
    )

    cat = model_metrics(
        sample_y,
        sample[SCORES["CatBoost"]].to_numpy(),
    )

    draws.append({
        metric: cat[metric] - xgb[metric]
        for metric in METRICS
    })

if not draws:
    raise ValueError("No valid bootstrap comparisons")

draws = pd.DataFrame(draws)

difference = pd.DataFrame({
    "CatBoost_minus_XGBoost": (
        comparison.loc["CatBoost", METRICS]
        - comparison.loc["XGBoost", METRICS]
    ),
    "CI_lower": draws.quantile(0.025),
    "CI_upper": draws.quantile(0.975),
    "Valid_bootstraps": len(draws),
})

display(difference)

difference.to_csv(
    OUTPUT / "paired_metric_differences.csv",
    index=True,
)

fig, ax = plt.subplots(figsize=(9, 5))
positions = np.arange(len(difference))

ax.plot(
    difference["CatBoost_minus_XGBoost"],
    positions,
    "o",
)

ax.hlines(
    positions,
    difference["CI_lower"],
    difference["CI_upper"],
)

ax.axvline(0, color="black", linestyle="--")

ax.set(
    yticks=positions,
    yticklabels=difference.index,
    xlabel="CatBoost − XGBoost",
    title="Paired bootstrap 95% intervals",
)

save_plot("paired_metric_differences")
