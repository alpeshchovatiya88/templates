def jsd_interpretation(jsd):
    if pd.isna(jsd):
        return "Not calculated"
    elif jsd < 0.05:
        return "Stable"
    elif jsd < 0.10:
        return "Moderate shift"
    else:
        return "Significant shift"

# Numeric jsd

from scipy.spatial.distance import jensenshannon

def calculate_jsd_numeric(
    expected,
    actual,
    feature,
    bins=10,
    eps=1e-6
):
    expected_data = (
        expected[feature]
        .replace([np.inf, -np.inf], np.nan)
        .dropna()
    )

    actual_data = (
        actual[feature]
        .replace([np.inf, -np.inf], np.nan)
        .dropna()
    )

    if expected_data.nunique() < 2 or actual_data.nunique() < 2:
        return np.nan

    try:
        _, bin_edges = pd.qcut(
            expected_data,
            q=bins,
            retbins=True,
            duplicates="drop"
        )

        expected_bins = pd.cut(
            expected_data,
            bins=bin_edges,
            include_lowest=True
        )

        actual_bins = pd.cut(
            actual_data,
            bins=bin_edges,
            include_lowest=True
        )

        expected_pct = (
            expected_bins
            .value_counts(normalize=True, sort=False)
            .fillna(0)
        )

        actual_pct = (
            actual_bins
            .value_counts(normalize=True, sort=False)
            .fillna(0)
        )

        p = expected_pct.values + eps
        q = actual_pct.values + eps

        p = p / p.sum()
        q = q / q.sum()

        jsd = jensenshannon(p, q, base=2) ** 2

        return jsd

    except Exception:
        return np.nan

# categorical jsd

from scipy.spatial.distance import jensenshannon

def calculate_jsd_categorical(
    expected,
    actual,
    feature,
    eps=1e-6
):
    expected_data = (
        expected[feature]
        .astype("object")
        .where(expected[feature].notna(), "MISSING")
        .astype(str)
    )

    actual_data = (
        actual[feature]
        .astype("object")
        .where(actual[feature].notna(), "MISSING")
        .astype(str)
    )

    levels = sorted(
        set(expected_data.unique())
        .union(set(actual_data.unique()))
    )

    expected_pct = (
        expected_data
        .value_counts(normalize=True)
        .reindex(levels, fill_value=0)
    )

    actual_pct = (
        actual_data
        .value_counts(normalize=True)
        .reindex(levels, fill_value=0)
    )

    p = expected_pct.values + eps
    q = actual_pct.values + eps

    p = p / p.sum()
    q = q / q.sum()

    jsd = jensenshannon(p, q, base=2) ** 2

    return jsd

# Run JSD For One Model

def run_jsd_for_model(
    expected,
    actual,
    model_name,
    numeric_cols,
    categorical_cols
):
    results = []

    for col in numeric_cols:

        jsd_value = calculate_jsd_numeric(
            expected=expected,
            actual=actual,
            feature=col,
            bins=10
        )

        results.append({
            "model": model_name,
            "feature": col,
            "type": "numeric",
            "jsd": jsd_value,
            "interpretation": jsd_interpretation(jsd_value)
        })

    for col in categorical_cols:

        jsd_value = calculate_jsd_categorical(
            expected=expected,
            actual=actual,
            feature=col
        )

        results.append({
            "model": model_name,
            "feature": col,
            "type": "categorical",
            "jsd": jsd_value,
            "interpretation": jsd_interpretation(jsd_value)
        })

    return pd.DataFrame(results)
