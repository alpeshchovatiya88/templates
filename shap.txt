def run_shap(
    pipeline,
    X_train,
    out_dir,
    model_name="xgb",
    sample_size=5000,
    random_state=42,
    feat_names=None
):
    import os
    import shap
    import numpy as np
    import pandas as pd
    import matplotlib.pyplot as plt
    import scipy.sparse as sp

    os.makedirs(out_dir, exist_ok=True)

    Xs_raw = X_train.sample(
        min(len(X_train), sample_size),
        random_state=random_state
    )

    Xs = pipeline.named_steps["pre"].transform(Xs_raw)

    if sp.issparse(Xs):
        Xs = Xs.toarray()

    model = pipeline.named_steps["model"]

    if feat_names is None:
        try:
            feat_names = pipeline.named_steps["pre"].get_feature_names_out().tolist()
        except Exception:
            feat_names = [f"f{i}" for i in range(Xs.shape[1])]

    explainer = shap.TreeExplainer(model)
    shap_values = explainer.shap_values(Xs)

    np.save(
        os.path.join(out_dir, f"shap_values_{model_name}.npy"),
        shap_values
    )

    shap.summary_plot(
        shap_values,
        Xs,
        feature_names=feat_names,
        show=False
    )
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"shap_summary_{model_name}.png"))
    plt.show()

    mean_abs = np.nanmean(np.abs(shap_values), axis=0)

    shap_importance = pd.DataFrame({
        "feature": feat_names,
        "mean_abs_shap": mean_abs
    })

    shap_importance["feature"] = (
        shap_importance["feature"]
        .str.replace("cat__", "", regex=False)
        .str.replace("num__", "", regex=False)
        .str.replace("remainder__", "", regex=False)
    )

    for col in sorted(X_train.columns, key=len, reverse=True):
        shap_importance.loc[
            shap_importance["feature"].str.startswith(col + "_"),
            "feature"
        ] = col

    shap_importance = (
        shap_importance
        .groupby("feature", as_index=False)["mean_abs_shap"]
        .sum()
        .sort_values("mean_abs_shap", ascending=False)
    )

    shap_importance.to_csv(
        os.path.join(out_dir, f"shap_feature_importance_{model_name}.csv"),
        index=False
    )

    display(shap_importance.head(20))

    return {
        "shap_values": shap_values,
        "shap_importance": shap_importance
    }
