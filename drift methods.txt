def psi_interpretation(psi):
    if pd.isna(psi):
        return "Not calculated"
    elif psi < 0.10:
        return "Stable"
    elif psi < 0.25:
        return "Moderate shift"
    else:
        return "Significant shift"

# 3. PSI functions
# ============================================================

def calculate_psi_numeric(expected, actual, feature, bins=10, eps=1e-6):
    expected_data = expected[feature].replace([np.inf, -np.inf], np.nan).dropna()
    actual_data = actual[feature].replace([np.inf, -np.inf], np.nan).dropna()

    if expected_data.nunique() < 2 or actual_data.nunique() < 2:
        return np.nan

    try:
        _, bin_edges = pd.qcut(
            expected_data,
            q=bins,
            retbins=True,
            duplicates="drop"
        )

        expected_bins = pd.cut(expected_data, bins=bin_edges, include_lowest=True)
        actual_bins = pd.cut(actual_data, bins=bin_edges, include_lowest=True)

        expected_pct = expected_bins.value_counts(normalize=True, sort=False).fillna(0)
        actual_pct = actual_bins.value_counts(normalize=True, sort=False).fillna(0)

        p = expected_pct.values + eps
        q = actual_pct.values + eps

        return np.sum((q - p) * np.log(q / p))

    except Exception:
        return np.nan


def calculate_psi_categorical(expected, actual, feature, eps=1e-6):
    expected_data = (
        expected[feature]
        .astype("object")
        .where(expected[feature].notna(), "MISSING")
        .astype(str)
    )

    actual_data = (
        actual[feature]
        .astype("object")
        .where(actual[feature].notna(), "MISSING")
        .astype(str)
    )

    levels = sorted(set(expected_data.unique()).union(set(actual_data.unique())))

    expected_pct = expected_data.value_counts(normalize=True).reindex(levels, fill_value=0)
    actual_pct = actual_data.value_counts(normalize=True).reindex(levels, fill_value=0)

    p = expected_pct.values + eps
    q = actual_pct.values + eps

    return np.sum((q - p) * np.log(q / p))

def calculate_jsd_numeric(expected, actual, feature, bins=10, eps=1e-6):
    expected_data = expected[feature].replace([np.inf, -np.inf], np.nan).dropna()
    actual_data = actual[feature].replace([np.inf, -np.inf], np.nan).dropna()

    if expected_data.nunique() < 2 or actual_data.nunique() < 2:
        return np.nan

    try:
        _, bin_edges = pd.qcut(
            expected_data,
            q=bins,
            retbins=True,
            duplicates="drop"
        )

        expected_bins = pd.cut(expected_data, bins=bin_edges, include_lowest=True)
        actual_bins = pd.cut(actual_data, bins=bin_edges, include_lowest=True)

        expected_pct = expected_bins.value_counts(normalize=True, sort=False).fillna(0)
        actual_pct = actual_bins.value_counts(normalize=True, sort=False).fillna(0)

        p = expected_pct.values + eps
        q = actual_pct.values + eps

        p = p / p.sum()
        q = q / q.sum()

        return jensenshannon(p, q, base=2) ** 2

    except Exception:
        return np.nan

# ============================================================
# 5. Wasserstein for numeric variables
# ============================================================

def calculate_wasserstein_numeric(expected, actual, feature):
    expected_data = expected[feature].replace([np.inf, -np.inf], np.nan).dropna()
    actual_data = actual[feature].replace([np.inf, -np.inf], np.nan).dropna()

    if len(expected_data) == 0 or len(actual_data) == 0:
        return np.nan

    try:
        return wasserstein_distance(expected_data, actual_data)
    except Exception:
        return np.nan

# 6. Hellinger for categorical variables
# ============================================================

def calculate_hellinger_categorical(expected, actual, feature, eps=1e-6):
    expected_data = (
        expected[feature]
        .astype("object")
        .where(expected[feature].notna(), "MISSING")
        .astype(str)
    )

    actual_data = (
        actual[feature]
        .astype("object")
        .where(actual[feature].notna(), "MISSING")
        .astype(str)
    )

    levels = sorted(set(expected_data.unique()).union(set(actual_data.unique())))

    expected_pct = expected_data.value_counts(normalize=True).reindex(levels, fill_value=0)
    actual_pct = actual_data.value_counts(normalize=True).reindex(levels, fill_value=0)

    p = expected_pct.values + eps
    q = actual_pct.values + eps

    p = p / p.sum()
    q = q / q.sum()

    return np.sqrt(np.sum((np.sqrt(p) - np.sqrt(q)) ** 2)) / np.sqrt(2)

# 7. Final comparison function
# ============================================================

def compare_drift_metrics_for_model(
    expected,
    actual,
    model_name,
    numeric_cols,
    categorical_cols,
    bins=10
):
    results = []

    for col in numeric_cols:
        psi_value = calculate_psi_numeric(expected, actual, col, bins=bins)
        jsd_value = calculate_jsd_numeric(expected, actual, col, bins=bins)
        wd_value = calculate_wasserstein_numeric(expected, actual, col)

        results.append({
            "model": model_name,
            "feature": col,
            "feature_type": "numeric",
            "psi": psi_value,
            "psi_interpretation": psi_interpretation(psi_value),
            "jsd": jsd_value,
            "jsd_interpretation": jsd_interpretation(jsd_value),
            "wasserstein": wd_value,
            "wasserstein_interpretation": wasserstein_interpretation(wd_value),
            "hellinger": np.nan,
            "hellinger_interpretation": np.nan
        })

    for col in categorical_cols:
        psi_value = calculate_psi_categorical(expected, actual, col)
        jsd_value = calculate_jsd_categorical(expected, actual, col)
        hd_value = calculate_hellinger_categorical(expected, actual, col)

        results.append({
            "model": model_name,
            "feature": col,
            "feature_type": "categorical",
            "psi": psi_value,
            "psi_interpretation": psi_interpretation(psi_value),
            "jsd": jsd_value,
            "jsd_interpretation": jsd_interpretation(jsd_value),
            "wasserstein": np.nan,
            "wasserstein_interpretation": np.nan,
            "hellinger": hd_value,
            "hellinger_interpretation": hellinger_interpretation(hd_value)
        })

    return pd.DataFrame(results)
