from sklearn.calibration import calibration_curve
import matplotlib.pyplot as plt
import os

def plot_calibration_curve(
    model,
    X_test,
    y_test,
    out_dir="model_outputs",
    label="oot_2021_2023",
    n_bins=10
):
    os.makedirs(out_dir, exist_ok=True)

    proba = model.predict_proba(X_test)[:, 1]

    prob_true, prob_pred = calibration_curve(
        y_true=y_test,
        y_prob=proba,
        n_bins=n_bins,
        strategy="quantile"
    )

    plt.figure(figsize=(6, 6))

    plt.plot(
        prob_pred,
        prob_true,
        marker="o",
        linewidth=2,
        label="XGBoost"
    )

    plt.plot(
        [0, 1],
        [0, 1],
        linestyle="--",
        label="Perfect calibration"
    )

    plt.xlabel("Mean predicted probability")
    plt.ylabel("Observed default rate")
    plt.title(f"Calibration Curve - {label}")
    plt.legend()
    plt.grid(True)
    plt.tight_layout()

    file_path = os.path.join(out_dir, f"{label}_calibration_curve.png")
    plt.savefig(file_path, dpi=300, bbox_inches="tight")
    plt.show()

    print("Saved calibration curve to:", file_path)

    return file_path

from sklearn.calibration import CalibratedClassifierCV, calibration_curve
import matplotlib.pyplot as plt
import os

def calibrate_model(
    model,
    X_train,
    y_train,
    method="isotonic",
    cv=5
):
    calibrated_model = CalibratedClassifierCV(
        estimator=model,
        method=method,   # "isotonic" or "sigmoid"
        cv=cv
    )

    calibrated_model.fit(X_train, y_train)

    return calibrated_model

    plt.ylabel("Observed default rate")

    plt.title("Calibration Curve (5-fold OOF)")

    plt.legend()

    plt.grid(True)

    plt.show()

from sklearn.calibration import CalibrationDisplay
import matplotlib.pyplot as plt

def plot_oof_calibration_display(y_true, oof_proba, n_bins=5):

    fig, ax = plt.subplots(figsize=(7,6))

    CalibrationDisplay.from_predictions(
        y_true=y_true,
        y_prob=oof_proba,
        n_bins=n_bins,
        strategy="quantile",
        name="XGBoost (OOF)",
        ax=ax
    )

    ax.set_title("Calibration Curve (5-fold OOF)")
    ax.grid(True)

    plt.show()

from sklearn.model_selection import train_test_split
from sklearn.calibration import CalibratedClassifierCV, CalibrationDisplay
import matplotlib.pyplot as plt

# 1. Split training data
X_fit, X_cal, y_fit, y_cal = train_test_split(
    X_train,
    y_train,
    test_size=0.2,
    random_state=42,
    stratify=y_train
)

# 2. Build full pipeline: preprocessing + XGBoost
base_model = build_pipeline(
    X_fit,
    credit_use="bin",
    random_state=42
)

# 3. Fit base model
base_model.fit(X_fit, y_fit)

# 4. Calibrate fitted model
calibrated_model = CalibratedClassifierCV(
    estimator=base_model,
    method="sigmoid",   # better for your rare default rate
    cv="prefit"
)

calibrated_model.fit(X_cal, y_cal)

# 5. Evaluate calibrated model on OOT
calibrated_probs = calibrated_model.predict_proba(X_test)[:, 1]

fig, ax = plt.subplots(figsize=(7, 6))

CalibrationDisplay.from_predictions(
    y_true=y_test,
    y_prob=calibrated_probs,
    n_bins=5,
    strategy="quantile",
    name="Calibrated XGBoost",
    ax=ax
)

ax.set_title("Calibration Curve - Calibrated Model on OOT")
ax.grid(True)
plt.show()
