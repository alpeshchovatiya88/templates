Change 2: local file read/write

Find write2file() and loadfile(). They use s3fs.S3FileSystem(), so local CSV may fail.

Replace both functions with:

def write2file(df, filename: str, output_path: str) -> None:
    output_path = Path(output_path)
    output_path.mkdir(parents=True, exist_ok=True)
    df.to_csv(output_path / filename, index=False)


def loadfile(path: str, filename: str | None = None):
    full_path = Path(path) if filename is None else Path(path) / filename

    if str(full_path).lower().endswith((".parquet", ".pq")):
        return pd.read_parquet(full_path)

    if str(full_path).endswith("FSA_T26_mapping.csv"):
        df = pd.read_csv(full_path, usecols=["T26", "fsa"])
        return set(df.loc[df["T26"] == "Toronto", "fsa"].astype(str))

    return pd.read_csv(full_path)

That changes I/O only, not model logic.
