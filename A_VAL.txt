def divide(a, b):
    return a / b if b else np.nan


def performance(df):
    y = df["_y"].to_numpy()
    p = df["_p"].to_numpy()

    both_classes = np.unique(y).size == 2

    tn, fp, fn, tp = confusion_matrix(
        y, p >= THRESHOLD, labels=[0, 1]
    ).ravel()

    auc = roc_auc_score(y, p) if both_classes else np.nan
    ks = np.nan

    if both_classes:
        fpr, tpr, _ = roc_curve(
            y, p, drop_intermediate=False
        )
        ks = float(np.max(abs(tpr - fpr)))

    return {
        "n": len(y),
        "bad_events": int(y.sum()),
        "bad_rate": y.mean(),
        "mean_pd": p.mean(),
        "auc": auc,
        "gini": 2 * auc - 1,
        "ks": ks,
        "average_precision": (
            average_precision_score(y, p)
            if both_classes else np.nan
        ),
        "brier": np.mean((y - p) ** 2),
        "log_loss": log_loss(y, p, labels=[0, 1]),
        "precision": divide(tp, tp + fp),
        "recall": divide(tp, tp + fn),
        "fpr": divide(fp, fp + tn),
        "f1": divide(2 * tp, 2 * tp + fp + fn),
        "flag_rate": (p >= THRESHOLD).mean(),
        "observed_expected": divide(y.sum(), p.sum()),
    }


def bootstrap_indices(df, rng):
    if not GROUP_COLUMN:
        return rng.integers(0, len(df), len(df))

    units = list(
        df.groupby(
            GROUP_COLUMN, sort=False
        ).indices.values()
    )

    if len(units) < 2:
        raise ValueError("At least two independent groups required")

    selected = rng.integers(
        0, len(units), len(units)
    )

    return np.concatenate([units[i] for i in selected])


def performance_intervals(df):
    rng = np.random.default_rng(SEED)

    draws = pd.DataFrame([
        performance(
            df.iloc[bootstrap_indices(df, rng)]
        )
        for _ in range(BOOTSTRAPS)
    ])

    result = pd.DataFrame({
        "estimate": pd.Series(performance(df)),
        "lower": draws.quantile(0.025),
        "upper": draws.quantile(0.975),
        "valid_bootstraps": draws.notna().sum(),
    })

    result.loc[
        ["n", "bad_events"], ["lower", "upper"]
    ] = np.nan

    return result


def split_data(df, stratify=True):
    if GROUP_COLUMN:
        splitter = GroupShuffleSplit(
            n_splits=1,
            test_size=0.5,
            random_state=SEED,
        )

        fitting, assessment = next(splitter.split(
            df, groups=df[GROUP_COLUMN]
        ))

        return (
            df.iloc[fitting].copy(),
            df.iloc[assessment].copy(),
        )

    return train_test_split(
        df,
        test_size=0.5,
        random_state=SEED,
        stratify=df["_y"] if stratify else None,
    )


def calibration_table(df, n_bins=10, strategy="uniform"):
    p = df["_p"]

    edges = (
        np.linspace(0, 1, n_bins + 1)
        if strategy == "uniform"
        else np.quantile(p, np.linspace(0, 1, n_bins + 1))
    )

    work = df.assign(
        bin=np.searchsorted(edges[1:-1], p)
    )

    result = work.groupby("bin").agg(
        n=("_y", "size"),
        bad_events=("_y", "sum"),
        mean_pd=("_p", "mean"),
        observed_rate=("_y", "mean"),
    )

    n = result["n"]
    rate = result["observed_rate"]
    denominator = 1 + 1.96**2 / n

    center = (
        rate + 1.96**2 / (2 * n)
    ) / denominator

    half_width = (
        1.96 * np.sqrt(
            rate * (1 - rate) / n
            + 1.96**2 / (4 * n**2)
        ) / denominator
    )

    result["lower"] = center - half_width
    result["upper"] = center + half_width
    result["gap"] = (
        result["observed_rate"] - result["mean_pd"]
    )
    result["small_bin"] = n < MIN_N

    return result


def segment_performance(df, columns):
    rows = []

    for column in columns:
        if column not in df:
            continue

        for value, group in df.groupby(
            column, dropna=False
        ):
            row = performance(group)

            row.update({
                "feature": column,
                "value": str(value),
                "sufficient_sample": (
                    len(group) >= MIN_N
                    and group["_y"].sum() >= MIN_EVENTS
                    and (1 - group["_y"]).sum() >= MIN_EVENTS
                ),
            })

            rows.append(row)

    return pd.DataFrame(rows)

def psi_table(reference, current):
    interior = np.unique(np.quantile(
        reference, np.linspace(0, 1, 11)[1:-1]
    ))

    if reference.min() == reference.max():
        value = float(reference.iloc[0])
        interior = np.array([
            np.nextafter(value, -np.inf),
            np.nextafter(value, np.inf),
        ])

    edges = np.r_[-np.inf, interior, np.inf]
    a = np.histogram(reference, edges)[0]
    b = np.histogram(current, edges)[0]

    aa = (a + 0.5) / (a.sum() + 0.5 * len(a))
    bb = (b + 0.5) / (b.sum() + 0.5 * len(b))

    return pd.DataFrame({
        "left": edges[:-1],
        "right": edges[1:],
        "reference_fraction": a / a.sum(),
        "current_fraction": b / b.sum(),
        "psi_contribution": (bb - aa) * np.log(bb / aa),
    })

    nonconformity = np.where(

def model_probability(X, estimator=None):
    estimator = model if estimator is None else estimator

    p = estimator.predict_proba(X)[
        :, list(estimator.classes_).index(1)
    ]

    if (
        not np.isfinite(p).all()
        or ((p < 0) | (p > 1)).any()
    ):
        raise ValueError("Invalid model probabilities")

    return p
