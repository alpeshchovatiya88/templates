def divide(a, b):
    return a / b if b else np.nan


def wilson_interval(events, n):
    """Independent-observation 95% interval for a proportion."""
    if n == 0:
        return np.nan, np.nan

    rate = events / n
    z = 1.96
    denominator = 1 + z**2 / n

    center = (rate + z**2 / (2 * n)) / denominator

    half = (
        z * np.sqrt(
            rate * (1 - rate) / n
            + z**2 / (4 * n**2)
        ) / denominator
    )

    return center - half, center + half

def performance(df, threshold=THRESHOLD):
    y = df["_y"].to_numpy()
    p = df["_p"].to_numpy()

    tn, fp, fn, tp = confusion_matrix(
        y,
        p >= threshold,
        labels=[0, 1],
    ).ravel()

    both_classes = np.unique(y).size == 2

    auc = (
        roc_auc_score(y, p)
        if both_classes else np.nan
    )

    ks = np.nan

    if both_classes:
        fpr, tpr, _ = roc_curve(
            y,
            p,
            drop_intermediate=False,
        )
        ks = float(np.max(np.abs(tpr - fpr)))

    prevalence = y.mean()
    brier = np.mean((y - p) ** 2)

    return {
        "n": len(y),
        "bad_events": int(y.sum()),
        "bad_rate": prevalence,
        "mean_pd": p.mean(),
        "auc": auc,
        "gini": 2 * auc - 1,
        "ks": ks,
        "average_precision": (
            average_precision_score(y, p)
            if both_classes else np.nan
        ),
        "brier": brier,
        "within_cohort_brier_skill": (
            1 - divide(
                brier,
                prevalence * (1 - prevalence),
            )
        ),
        "log_loss": log_loss(y, p, labels=[0, 1]),
        "accuracy": divide(tp + tn, len(y)),
        "precision": divide(tp, tp + fp),
        "recall": divide(tp, tp + fn),
        "specificity": divide(tn, tn + fp),
        "false_positive_rate": divide(fp, fp + tn),
        "f1": divide(2 * tp, 2 * tp + fp + fn),
        "flag_rate": (p >= threshold).mean(),
        "observed_expected": divide(y.sum(), p.sum()),
    }


def resampling_units(df):
    if GROUP_COLUMN:
        units = list(
            df.groupby(
                GROUP_COLUMN, sort=False
            ).indices.values()
        )
        if len(units) < 2:
            raise ValueError("At least two independent groups required")
        return units

    return None


def sample_indices(df, rng, units=None):
    if units is None:
        return rng.integers(0, len(df), len(df))

    selected = rng.integers(0, len(units), len(units))
    return np.concatenate([units[i] for i in selected])


def bootstrap_performance(df):
    rng = np.random.default_rng(SEED)
    units = resampling_units(df)

    draws = pd.DataFrame([
        performance(
            df.iloc[sample_indices(df, rng, units)]
        )
        for _ in range(BOOTSTRAPS)
    ])

    result = pd.DataFrame({
        "estimate": pd.Series(performance(df)),
        "lower": draws.quantile(0.025),
        "upper": draws.quantile(0.975),
        "valid_bootstraps": draws.notna().sum(),
    })

    result.loc[
        ["n", "bad_events"],
        ["lower", "upper"],
    ] = np.nan

    return result


def split_data(df, fraction=0.5, stratify=True):
    if GROUP_COLUMN:
        splitter = GroupShuffleSplit(
            n_splits=1,
            test_size=fraction,
            random_state=SEED,
        )
        a, b = next(splitter.split(
            df, groups=df[GROUP_COLUMN]
        ))
        return df.iloc[a].copy(), df.iloc[b].copy()

    return train_test_split(
        df,
        test_size=fraction,
        random_state=SEED,
        stratify=df["_y"] if stratify else None,
    )


def calibration_table(df):
    work = df.assign(
        band=pd.cut(
            df["_p"],
            np.linspace(0, 1, 11),
            include_lowest=True,
        )
    )

    result = work.groupby(
        "band", observed=True
    ).agg(
        n=("_y", "size"),
        bad_events=("_y", "sum"),
        mean_pd=("_p", "mean"),
        observed_rate=("_y", "mean"),
    )

    intervals = [
        wilson_interval(events, n)
        for events, n in zip(
            result["bad_events"], result["n"]
        )
    ]

    result["lower"] = [x[0] for x in intervals]
    result["upper"] = [x[1] for x in intervals]

    result["gap"] = (
        result["observed_rate"] - result["mean_pd"]
    )

    return result


def segment_performance(df, columns):
    rows = []

    for column in columns:
        if column not in df:
            continue

        for value, group in df.groupby(
            column, dropna=False
        ):
            row = performance(group)

            row.update({
                "feature": column,
                "value": str(value),
                "sufficient_sample": (
                    len(group) >= MIN_N
                    and group["_y"].sum() >= MIN_EVENTS
                    and (1 - group["_y"]).sum() >= MIN_EVENTS
                ),
            })

            rows.append(row)

    return pd.DataFrame(rows)

def model_probability(X, estimator=None):
    estimator = model if estimator is None else estimator

    p = estimator.predict_proba(X)[
        :, list(estimator.classes_).index(1)
    ]

    if (
        not np.isfinite(p).all()
        or ((p < 0) | (p > 1)).any()
    ):
        raise ValueError("Invalid model probabilities")

    return p

def psi_table(reference, current):
    interior = np.unique(np.quantile(
        reference, np.linspace(0, 1, 11)[1:-1]
    ))

    if reference.min() == reference.max():
        value = float(reference.iloc[0])
        interior = np.array([
            np.nextafter(value, -np.inf),
            np.nextafter(value, np.inf),
        ])

    edges = np.r_[-np.inf, interior, np.inf]
    a = np.histogram(reference, edges)[0]
    b = np.histogram(current, edges)[0]

    # Explicit pseudocount smoothing and normalization.
    aa = (a + 0.5) / (a.sum() + 0.5 * len(a))
    bb = (b + 0.5) / (b.sum() + 0.5 * len(b))

    return pd.DataFrame({
        "left": edges[:-1],
        "right": edges[1:],
        "reference_fraction": a / a.sum(),
        "current_fraction": b / b.sum(),
        "psi_contribution": (bb - aa) * np.log(bb / aa),
    })

    data = evaluation["train"]
    X = data[FEATURES]
    y = data["_y"]

    groups = (
        data[GROUP_COLUMN]
        if GROUP_COLUMN else None
    )

    def cv_splits(X, y, groups, folds, seed):
        if groups is not None:
            cv = StratifiedGroupKFold(
                n_splits=folds,
                shuffle=True,
                random_state=seed,
            )
        else:
            cv = StratifiedKFold(
                n_splits=folds,
                shuffle=True,
                random_state=seed,
            )

        splits = list(cv.split(X, y, groups))

        for fitting, assessment in splits:
            if (
                y.iloc[fitting].nunique() != 2
                or y.iloc[assessment].nunique() != 2
            ):
                raise ValueError(
                    "A fold lacks an outcome class. Revise CV design."
                )

        return splits

    outer = cv_splits(X, y, groups, 5, SEED)
    rows = []

    for fold, (fitting, assessment) in enumerate(outer):
        inner_groups = (
            groups.iloc[fitting]
            if groups is not None else None
        )

        inner = cv_splits(
            X.iloc[fitting],
            y.iloc[fitting],
            inner_groups,
            3,
            SEED + 1,
        )

        search = GridSearchCV(
            clone(model),
            PARAM_GRID,
            cv=inner,
            scoring="roc_auc",
            n_jobs=1,
            error_score="raise",
        )

        search.fit(
            X.iloc[fitting],
            y.iloc[fitting],
        )

        assessed = data.iloc[assessment].copy()

        assessed["_p"] = model_probability(
            X.iloc[assessment],
            search.best_estimator_,
        )

        rows.append({
            "outer_fold": fold,
            **performance(assessed),
        })

    nested = pd.DataFrame(rows)
    show_table(nested, "13_nested_tuning")

    incumbent_metrics = performance(incumbent)

    selected_metrics = [
        "auc", "average_precision", "brier", "log_loss"
    ]

    rng = np.random.default_rng(SEED)
    units = resampling_units(df)
    draws = []

    for _ in range(BOOTSTRAPS):
        indices = sample_indices(df, rng, units)

        current = performance(df.iloc[indices])
        previous = performance(incumbent.iloc[indices])

        draws.append({
            metric: current[metric] - previous[metric]
            for metric in selected_metrics
        })

    draws = pd.DataFrame(draws)

    result = pd.DataFrame({
        "current": {
            metric: current_metrics[metric]
            for metric in selected_metrics
        },
        "incumbent": {
            metric: incumbent_metrics[metric]
            for metric in selected_metrics
        },
        "difference": {
            metric: (
                current_metrics[metric] - incumbent_metrics[metric]
            )
            for metric in selected_metrics
        },
        "difference_lower": draws.quantile(0.025),
        "difference_upper": draws.quantile(0.975),
        "valid_bootstraps": draws.notna().sum(),
    })

    show_table(result, f"15_{name}_paired_comparison")

    fig, ax = plt.subplots(figsize=(8, 4))
    positions = np.arange(len(result))

    ax.plot(result["difference"], positions, "o")
    ax.hlines(
        positions,
        result["difference_lower"],
        result["difference_upper"],
    )
    ax.axvline(0, color="gray", linestyle="--")

    ax.set(
        yticks=positions,
        yticklabels=result.index,
        xlabel="Current − incumbent",
        title=f"{name}: paired metric differences",
    )

    show_plot(f"15_{name}_paired_comparison")

        passed = value <= limit

        "metric": metric,
        "value": value,
        "approved_limit": limit,
        "status": (
            "NOT EVALUABLE"
            if not np.isfinite(value)
            else "PASS"
            if passed
            else "REVIEW"
        ),
    })

        "16_approved_validation_criteria",
    )
