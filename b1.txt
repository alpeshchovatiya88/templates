def psi(reference, actual, bins=10):
    reference = np.asarray(reference, float)
    actual = np.asarray(actual, float)
    reference = reference[np.isfinite(reference)]
    actual = actual[np.isfinite(actual)]
    edges = np.unique(np.quantile(reference, np.linspace(0, 1, bins + 1)))
    if len(edges) < 2:
        return np.nan
    edges[0], edges[-1] = -np.inf, np.inf
    a = np.histogram(reference, edges)[0] + 0.5
    b = np.histogram(actual, edges)[0] + 0.5
    a, b = a / a.sum(), b / b.sum()
    return float(np.sum((b - a) * np.log(b / a)))


def performance(d, threshold=THRESHOLD):
    y = d["_y"].to_numpy(int)
    p = d["_p"].to_numpy(float)
    both = len(np.unique(y)) == 2
    tn, fp, fn, tp = confusion_matrix(y, p >= threshold, labels=[0, 1]).ravel()
    ratio = lambda a, b: a / b if b else np.nan
    auc = roc_auc_score(y, p) if both else np.nan
    fpr, tpr, _ = roc_curve(y, p) if both else ([], [], [])
    return dict(
        n=len(d), bad_events=int(y.sum()), bad_rate=float(y.mean()),
        mean_pd=float(p.mean()), auc=auc,
        gini=(2 * auc - 1) if both else np.nan,
        ks=float(np.max(np.abs(np.array(tpr) - fpr))) if both else np.nan,
        average_precision=average_precision_score(y, p) if both else np.nan,
        brier=float(np.mean((y - p) ** 2)),
        log_loss=log_loss(y, np.clip(p, 1e-6, 1 - 1e-6), labels=[0, 1]),
        recall=ratio(tp, tp + fn), precision=ratio(tp, tp + fp),
        fpr=ratio(fp, fp + tn), flag_rate=float(np.mean(p >= threshold)),
        observed_expected=ratio(y.sum(), p.sum()),
    )


def calibration_table(d, bins=10, strategy="uniform"):
    edges = (np.linspace(0, 1, bins + 1) if strategy == "uniform"
             else np.quantile(d["_p"], np.linspace(0, 1, bins + 1)))
    edges = np.unique(edges)
    if len(edges) < 3:
        edges = np.linspace(0, 1, bins + 1)
    t = d.assign(bin=np.searchsorted(edges[1:-1], d["_p"])).groupby("bin").agg(
        n=("_y", "size"), events=("_y", "sum"), mean_pd=("_p", "mean")
    )
    r, n, z = t["events"] / t["n"], t["n"], 1.96
    center = (r + z * z / (2 * n)) / (1 + z * z / n)
    half = z * np.sqrt(r * (1 - r) / n + z * z / (4 * n * n)) / (1 + z * z / n)
    return t.assign(
        observed_rate=r, lower=center - half, upper=center + half,
        gap=r - t["mean_pd"], small_bin=n < MIN_N,
    )


def ece(d, bins=10, strategy="quantile"):
    t = calibration_table(d, bins, strategy)
    return float(np.average(np.abs(t["gap"]), weights=t["n"]))


def split_data(d, stratify=True):
    if GROUP_COLUMN:
        a, b = next(GroupShuffleSplit(
            n_splits=1, test_size=0.5, random_state=SEED
        ).split(d, groups=d[GROUP_COLUMN]))
        return d.iloc[a].copy(), d.iloc[b].copy()
    return train_test_split(
        d, test_size=0.5, random_state=SEED,
        stratify=d["_y"] if stratify and d["_y"].nunique() == 2 else None,
    )


def bootstrap_indices(d, rng):
    if not GROUP_COLUMN:
        return rng.integers(len(d), size=len(d))
    groups = list(d.groupby(GROUP_COLUMN, sort=False).indices.values())
    if len(groups) < 2:
        raise ValueError("Bootstrap requires at least two groups.")
    return np.concatenate([
        groups[i] for i in rng.integers(len(groups), size=len(groups))
    ])


def labeled_frame(d, y):
    out = d.loc[y.isin([0, 1])].copy()
    out["_y"] = y.loc[out.index].astype(int)
    return out

                ("statistical_stress", direction * 0.5 * scale),
            ]:
                altered = X.copy()
                altered[c] = pd.to_numeric(altered[c], errors="coerce") + delta
                p = model_probability(altered)
                changes.append(dict(
                    feature=c, scenario=label,
                    mean_pd_change=float(np.mean(p - base)),
                    mean_absolute_change=float(np.mean(np.abs(p - base))),
                    p95_absolute_change=float(np.quantile(np.abs(p - base), 0.95)),
                    threshold_flip_rate=float(np.mean((p >= THRESHOLD) != (base >= THRESHOLD))),
                ))
        show_table(pd.DataFrame(changes), f"09_{name}_sensitivity")

        bin_changes = []
        for c in ["CRD_SCORE_bin", "ACTLTV_bin", "DSR_DIFF_bin"]:
            if c not in X:
                continue
            levels = np.sort(pd.to_numeric(frames["train"][c], errors="coerce").dropna().unique())
            v = pd.to_numeric(X[c], errors="coerce")
            known = v.dropna()
            if not known.isin(levels).all():
                record_status(f"{name}_{c}", "REVIEW", "unseen bin levels; bin step skipped")
                continue
            for step in [-1, 1]:
                altered = X.copy()
                pos = np.searchsorted(levels, v.fillna(levels[0]))
                altered[c] = np.where(v.isna(), np.nan, levels[np.clip(pos + step, 0, len(levels) - 1)])
                p = model_probability(altered)
                bin_changes.append(dict(
                    feature=c, step=step,
                    mean_pd_change=float(np.nanmean(p - base)),
                    mean_absolute_change=float(np.nanmean(np.abs(p - base))),
                ))
        show_table(pd.DataFrame(bin_changes), f"09_{name}_bin_sensitivity")

    for name, d in {**evaluation, **partial}.items():
        sample = d.sample(min(1500, len(d)), random_state=SEED)
        if sample["_y"].nunique() < 2:
            continue
        result = permutation_importance(
            model, sample[FEATURES], sample["_y"], scoring="roc_auc",
            n_repeats=3, random_state=SEED,
        )
        t = pd.DataFrame(dict(

        tag = name if name in evaluation else f"{name}_partial"
        show_table(t, f"09_{tag}_permutation_importance")
        t["auc_drop"].sort_values().plot.barh()
        show_plot(f"09_{tag}_permutation_importance")

    X = frames["test"][FEATURES].sample(min(1000, len(frames["test"])), random_state=SEED)

    def cv_splits(X, y, groups, folds):

                raise ValueError("A CV fold contains only one outcome class")
        return splits
