# -----------------------------
# Train-test split by time
# -----------------------------

def prepare_train_test(
    df,
    features,
    target="target",
    train_years=(2009, 2017),
    test_years=(2021, 2023),
    year_col="BOOKYEAR"
):
    df = df.copy()

    df_train = df[df[year_col].between(*train_years)].copy()
    df_test = df[df[year_col].between(*test_years)].copy()

    features = [
        c for c in features
        if c not in ["APPLICATION_NUMBER", "BOOKYEAR", target]
    ]

    X_train = df_train[features].copy()
    y_train = df_train[target].astype(int).to_numpy().ravel()

    X_test = df_test[features].copy()
    y_test = df_test[target].astype(int).to_numpy().ravel()

    return df_train, df_test, X_train, X_test, y_train, y_test, features

# -----------------------------
# Preprocessor
# -----------------------------

def build_preprocessor(X, credit_use="bin"):
    ordinal_cols = ["PRK_NUM_OF_UNITS", "NUM_OF_BOR"]

    if credit_use == "bin" and "CRD_SCORE_bin" in X.columns:
        ordinal_cols = ["CRD_SCORE_bin"] + ordinal_cols

    log_cols = ["HPI_+4"] if "HPI_+4" in X.columns else []

    num_cols = X.select_dtypes(include=["number", "bool"]).columns.tolist()
    cat_cols = X.select_dtypes(exclude=["number", "bool"]).columns.tolist()

    ordinal_cols = [c for c in ordinal_cols if c in X.columns]
    num_cols = [c for c in num_cols if c not in ordinal_cols + log_cols]
    cat_cols = [c for c in cat_cols if c not in ordinal_cols]

    num_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="median")),
        ("scaler", StandardScaler())
    ])

    log_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="median")),
        ("scaler", StandardScaler())
    ])

    ord_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="most_frequent")),
        ("ord", OrdinalEncoder(
            handle_unknown="use_encoded_value",
            unknown_value=-1
        ))
    ])

    cat_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="most_frequent")),
        ("ohe", OneHotEncoder(
            handle_unknown="ignore",
            sparse_output=False
        ))
    ])

    preprocessor = ColumnTransformer([
        ("lognum", log_tf, log_cols),
        ("num", num_tf, num_cols),
        ("ord", ord_tf, ordinal_cols),
        ("cat", cat_tf, cat_cols)
    ], sparse_threshold=0)

    return preprocessor

# -----------------------------
# Model
# -----------------------------

def build_model(random_state=42):
    return XGBClassifier(
        n_estimators=500,
        max_depth=4,
        learning_rate=0.03,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_weight=50,
        gamma=1,
        eval_metric="auc",
        random_state=random_state,
        n_jobs=-1,
        tree_method="hist"
    )


def build_pipeline(X, credit_use="bin", random_state=42):
    preprocessor = build_preprocessor(X, credit_use=credit_use)
    model = build_model(random_state=random_state)

    return Pipeline([
        ("pre", preprocessor),
        ("model", model)
    ])


def train_model(X_train, y_train, credit_use="bin", random_state=42):
    pipeline = build_pipeline(
        X_train,
        credit_use=credit_use,
        random_state=random_state
    )

    pipeline.fit(X_train, y_train)
    return pipeline

# -----------------------------
# Evaluation
# -----------------------------

def evaluate_model(model, X, y, df, out_dir, label="oot"):
    os.makedirs(out_dir, exist_ok=True)

    proba = model.predict_proba(X)[:, 1]
    auc = roc_auc_score(y, proba)

    scored = df.copy()
    scored["pred_proba"] = proba
    scored["pred_class"] = (proba >= 0.5).astype(int)

    print(f"{label} AUC:", auc)
    print(classification_report(y, scored["pred_class"]))

    scored.to_csv(
        os.path.join(out_dir, f"{label}_scored.csv"),
        index=False
    )

    return {
        "auc": auc,
        "proba": proba,
        "scored": scored
    }

# -----------------------------
# Diagnostics
# -----------------------------

def plot_model_diagnostics(
    y_true,
    proba,
    df,
    out_dir,
    category_col="category",
    loan_purpose_col="PRD_LOAN_PURPOSE",
    bins=40,
    density=True
):
    os.makedirs(out_dir, exist_ok=True)

    y_true = np.asarray(y_true)
    proba = np.asarray(proba)

    # ROC curve
    fpr, tpr, _ = roc_curve(y_true, proba)
    auc = roc_auc_score(y_true, proba)

    plt.figure(figsize=(7, 6))
    plt.plot(fpr, tpr, label=f"AUC = {auc:.4f}")
    plt.plot([0, 1], [0, 1], linestyle="--")
    plt.xlabel("False Positive Rate")
    plt.ylabel("True Positive Rate")
    plt.title("ROC Curve")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "roc_curve.png"), dpi=150)
    plt.show()

    # Score distribution
    plt.figure(figsize=(8, 5))
    sns.histplot(proba, bins=bins, kde=True, stat="density" if density else "count")
    plt.xlabel("Predicted Probability")
    plt.title("Prediction Probability Distribution")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "score_distribution.png"), dpi=150)
    plt.show()

    # Confusion matrix
    pred = (proba >= 0.5).astype(int)
    cm = confusion_matrix(y_true, pred)

    plt.figure(figsize=(5, 4))
    sns.heatmap(cm, annot=True, fmt="d", cmap="Blues")
    plt.xlabel("Predicted")
    plt.ylabel("Actual")
    plt.title("Confusion Matrix")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "confusion_matrix.png"), dpi=150)
    plt.show()

    return {
        "auc": auc,
        "confusion_matrix": cm
    }


def run_diagnostics(y_true, proba, scored_df, out_dir):
    return plot_model_diagnostics(
        y_true=np.asarray(y_true),
        proba=np.asarray(proba),
        df=scored_df,
        out_dir=out_dir,
        category_col="category",
        loan_purpose_col="PRD_LOAN_PURPOSE",
        bins=40,
        density=True
    )

# -----------------------------
# Cross-validation OOF
# -----------------------------

def run_cv_oof(
    X,
    y,
    n_splits=5,
    credit_use="bin",
    random_state=42
):
    y = np.asarray(y).reshape(-1)

    print("X:", X.shape)
    print("y:", y.shape)

    assert len(X) == len(y), f"X rows {len(X)} != y rows {len(y)}"

    skf = StratifiedKFold(
        n_splits=n_splits,
        shuffle=True,
        random_state=random_state
    )

    oof_proba = np.zeros(len(X))

    for fold, (tr_idx, va_idx) in enumerate(skf.split(X, y), 1):
        X_tr = X.iloc[tr_idx]
        y_tr = y[tr_idx]

        X_va = X.iloc[va_idx]
        y_va = y[va_idx]

        fold_model = build_pipeline(
            X_tr,
            credit_use=credit_use,
            random_state=random_state
        )

        fold_model.fit(X_tr, y_tr)

        oof_proba[va_idx] = fold_model.predict_proba(X_va)[:, 1]

        fold_auc = roc_auc_score(y_va, oof_proba[va_idx])
        print(f"Fold {fold} AUC:", fold_auc)

    oof_auc = roc_auc_score(y, oof_proba)
    print("OOF AUC:", oof_auc)

    return oof_proba, oof_auc

# -----------------------------
# SHAP
# -----------------------------

def run_shap(
    model,
    df_train,
    features,
    out_dir,
    model_name="xgb",
    sample_size=5000,
    random_state=42
):
    import shap

    os.makedirs(out_dir, exist_ok=True)

    shap_cols = list(features)

    X_sample = df_train[shap_cols].sample(
        min(sample_size, len(df_train)),
        random_state=random_state
    ).copy()

    pre = model.named_steps["pre"]
    xgb = model.named_steps["model"]

    X_trans = pre.transform(X_sample)

    try:
        feature_names = pre.get_feature_names_out()
    except:
        feature_names = [f"feature_{i}" for i in range(X_trans.shape[1])]

    explainer = shap.TreeExplainer(xgb)
    shap_values = explainer.shap_values(X_trans)

    shap_df = pd.DataFrame(
        shap_values,
        columns=feature_names
    )

    shap_df.to_csv(
        os.path.join(out_dir, f"{model_name}_shap_values.csv"),
        index=False
    )

    shap_importance = pd.DataFrame({
        "feature": feature_names,
        "mean_abs_shap": np.abs(shap_values).mean(axis=0)
    }).sort_values("mean_abs_shap", ascending=False)

    shap_importance.to_csv(
        os.path.join(out_dir, f"{model_name}_shap_importance.csv"),
        index=False
    )

    print("Top SHAP features:")
    display(shap_importance.head(20))

    shap.summary_plot(
        shap_values,
        X_trans,
        feature_names=feature_names,
        plot_type="bar",
        show=False
    )
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{model_name}_shap_bar.png"), dpi=150)
    plt.show()

    shap.summary_plot(
        shap_values,
        X_trans,
        feature_names=feature_names,
        show=False
    )
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{model_name}_shap_summary.png"), dpi=150)
    plt.show()

    return {
        "shap_values": shap_values,
        "shap_importance": shap_importance
    }

# -----------------------------
# Risk ranking
# -----------------------------

def run_risk_ranking(
    scored_df,
    out_dir,
    prob_col="pred_proba",
    category_col="category"
):
    os.makedirs(out_dir, exist_ok=True)

    risk_df = scored_df.copy()

    risk_df = risk_df.sort_values(prob_col, ascending=False).copy()
    risk_df["risk_rank"] = range(1, len(risk_df) + 1)

    risk_df["risk_decile"] = pd.qcut(
        risk_df[prob_col].rank(method="first"),
        q=10,
        labels=False
    ) + 1

    risk_df["risk_decile"] = 11 - risk_df["risk_decile"]

    risk_df["risk_group"] = pd.qcut(
        risk_df[prob_col].rank(method="first"),
        q=20,
        labels=False
    ) + 1

    risk_df["risk_group"] = 21 - risk_df["risk_group"]

    risk_summary = (
        risk_df
        .groupby("risk_decile")
        .agg(
            records=(prob_col, "size"),
            avg_score=(prob_col, "mean"),
            min_score=(prob_col, "min"),
            max_score=(prob_col, "max"),
            bad_rate=("target", "mean")
        )
        .reset_index()
        .sort_values("risk_decile")
    )

    risk_df.to_csv(
        os.path.join(out_dir, "scored_with_deciles.csv"),
        index=False
    )

    risk_summary.to_csv(
        os.path.join(out_dir, "risk_decile_summary.csv"),
        index=False
    )

    print("Risk decile summary:")
    display(risk_summary)

    print("Top 20 highest-risk records:")
    display(risk_df.head(20))

    return risk_summary

# -----------------------------
# Full workflow
# -----------------------------

def full_model_workflow(
    df,
    features,
    target="target",
    out_dir="model_outputs",
    credit_use="bin",
    run_cv=True,
    run_shap_flag=True,
    run_risk_flag=True,
    random_state=42
):
    os.makedirs(out_dir, exist_ok=True)

    df_train, df_test, X_train, X_test, y_train, y_test, features = prepare_train_test(
        df=df,
        features=features,
        target=target
    )

    if run_cv:
        oof_proba, oof_auc = run_cv_oof(
            X_train,
            y_train,
            n_splits=5,
            credit_use=credit_use,
            random_state=random_state
        )

        df_train_oof = df_train.copy()
        df_train_oof["oof_pred_proba"] = oof_proba

        df_train_oof.to_csv(
            os.path.join(out_dir, "train_oof_predictions.csv"),
            index=False
        )

    model = train_model(
        X_train,
        y_train,
        credit_use=credit_use,
        random_state=random_state
    )

    joblib.dump(
        model,
        os.path.join(out_dir, "best_xgb_model.joblib")
    )

    eval_result = evaluate_model(
        model=model,
        X=X_test,
        y=y_test,
        df=df_test,
        out_dir=out_dir,
        label="oot_2021_2023"
    )

    run_diagnostics(
        y_true=y_test,
        proba=eval_result["proba"],
        scored_df=eval_result["scored"],
        out_dir=out_dir
    )

    if run_shap_flag:
        shap_result = run_shap(
            model=model,
            df_train=df_train,
            features=features,
            out_dir=out_dir,
            model_name="xgb"
        )
    else:
        shap_result = None

    if run_risk_flag:
        risk_table = run_risk_ranking(
            scored_df=eval_result["scored"],
            out_dir=out_dir,
            prob_col="pred_proba"
        )
    else:
        risk_table = None

    return {
        "model": model,
        "test_auc": eval_result["auc"],
        "test_scored": eval_result["scored"],
        "risk_table": risk_table,
        "shap_result": shap_result
    }

from sklearn.model_selection import RandomizedSearchCV, StratifiedKFold
from sklearn.pipeline import Pipeline
from xgboost import XGBClassifier

def build_model(random_state=42):
    return XGBClassifier(
        objective="binary:logistic",
        eval_metric="auc",
        tree_method="hist",
        random_state=random_state,
        n_jobs=-1
    )


def build_pipeline(X, credit_use="bin", random_state=42):
    preprocessor = build_preprocessor(X, credit_use=credit_use)
    model = build_model(random_state=random_state)

    return Pipeline([
        ("pre", preprocessor),
        ("model", model)
    ])


def train_model(
    X_train,
    y_train,
    credit_use="bin",
    random_state=42,
    tune=True
):
    pipeline = build_pipeline(
        X_train,
        credit_use=credit_use,
        random_state=random_state
    )

    if not tune:
        pipeline.fit(X_train, y_train)
        return pipeline

    param_dist = {
        "model__n_estimators": [300, 500, 700],
        "model__max_depth": [3, 4, 5, 6],
        "model__learning_rate": [0.01, 0.03, 0.05, 0.06],
        "model__subsample": [0.7, 0.8, 0.9],
        "model__colsample_bytree": [0.7, 0.8, 0.9],
        "model__min_child_weight": [25, 50, 75, 100],
        "model__gamma": [0, 0.5, 1, 2],
        "model__reg_lambda": [1, 2, 5, 10],
        "model__reg_alpha": [0, 0.1, 0.5, 1]
    }

    cv = StratifiedKFold(
        n_splits=5,
        shuffle=True,
        random_state=random_state
    )

    search = RandomizedSearchCV(
        estimator=pipeline,
        param_distributions=param_dist,
        n_iter=25,
        scoring="roc_auc",
        cv=cv,
        verbose=2,
        n_jobs=-1,
        random_state=random_state,
        refit=True
    )

    search.fit(X_train, y_train)

    print("Best CV AUC:", search.best_score_)
    print("Best parameters:")
    print(search.best_params_)

    return search.best_estimator_
