# Shared helpers

def show_table(df, name):

def show_plot(name):

def record_status(test, state, detail=""):
    execution_status.append(dict(test=test, state=state, detail=detail))

        OUTPUT / "execution_status.csv", index=False

def performance(d, threshold=THRESHOLD):
    y, p = d["_y"].to_numpy(int), d["_p"].to_numpy(float)
    both = len(np.unique(y)) == 2
    tn, fp, fn, tp = confusion_matrix(
        y, p >= threshold, labels=[0, 1]
    ).ravel()
    ratio = lambda a, b: a / b if b else np.nan
    auc = roc_auc_score(y, p) if both else np.nan
    fpr, tpr, _ = roc_curve(y, p) if both else ([], [], [])
    return dict(
        n=len(d), bad_events=int(y.sum()), bad_rate=y.mean(),
        mean_pd=p.mean(), auc=auc, gini=2 * auc - 1,
        ks=max(np.abs(np.array(tpr) - fpr)) if both else np.nan,
        average_precision=average_precision_score(y, p) if both else np.nan,
        brier=np.mean((y - p)**2), log_loss=log_loss(y, p, labels=[0, 1]),
        recall=ratio(tp, tp + fn), precision=ratio(tp, tp + fp),
        fpr=ratio(fp, fp + tn), flag_rate=np.mean(p >= threshold),
        observed_expected=ratio(y.sum(), p.sum()),
    )

def calibration_table(d, bins=10, strategy="uniform"):
    edges = (np.linspace(0, 1, bins + 1) if strategy == "uniform"
             else np.quantile(d["_p"], np.linspace(0, 1, bins + 1)))
    t = d.assign(bin=np.searchsorted(edges[1:-1], d["_p"])).groupby(
        "bin"
    ).agg(n=("_y", "size"), events=("_y", "sum"), mean_pd=("_p", "mean"))
    r, n, z = t["events"] / t["n"], t["n"], 1.96
    center = (r + z*z/(2*n)) / (1 + z*z/n)
    half = z*np.sqrt(r*(1-r)/n + z*z/(4*n*n)) / (1 + z*z/n)
    return t.assign(observed_rate=r, lower=center-half, upper=center+half,
                    gap=r-t["mean_pd"], small_bin=n < MIN_N)

def split_data(d, stratify=True):
    if GROUP_COLUMN:
        a, b = next(GroupShuffleSplit(
            n_splits=1, test_size=.5, random_state=SEED
        ).split(d, groups=d[GROUP_COLUMN]))
        return d.iloc[a].copy(), d.iloc[b].copy()
    return train_test_split(
        d, test_size=.5, random_state=SEED,
        stratify=d["_y"] if stratify else None
    )

def bootstrap_indices(d, rng):
    if not GROUP_COLUMN:
        return rng.integers(len(d), size=len(d))
    groups = list(d.groupby(GROUP_COLUMN, sort=False).indices.values())
    if len(groups) < 2:
        raise ValueError("Bootstrap requires at least two groups.")
    return np.concatenate([groups[i] for i in rng.integers(
        len(groups), size=len(groups)
    )])

# Load train, test, val

frames, evaluation, audit = {}, {}, []

for name, path in FILES.items():
    d = pd.read_csv(path, dtype={ID_COLUMN: "string"}, low_memory=False)
    raw_n = len(d)
    if POPULATION_FILTERS.get(name):
        d = d.query(POPULATION_FILTERS[name], engine="python").copy()
    d = d.reset_index(drop=True)
    if d.empty:
        raise ValueError(f"{name}: empty population")

    for c in [ID_COLUMN] + ([GROUP_COLUMN] if GROUP_COLUMN else []):
        s = d[c].astype("string")
        if (s.isna() | s.str.strip().eq("")).any():
            raise ValueError(f"{name}: missing/blank {c}")
    if d[ID_COLUMN].duplicated().any():
        raise ValueError(f"{name}: duplicate application IDs")

    d["_p"] = pd.to_numeric(
        d[SCORE_COLUMNS[name]], errors="coerce"
    ).to_numpy(dtype=float, na_value=np.nan)
    if not (np.isfinite(d["_p"]) & d["_p"].between(0, 1)).all():
        raise ValueError(f"{name}: invalid probabilities")

    d = d.drop(columns="_y", errors="ignore")
    labeled_n, missing_n = 0, np.nan
    if TARGET in d:
        s = d[TARGET].astype("string").str.strip()
        missing = s.isna() | s.eq("").fillna(False)
        y = pd.to_numeric(s, errors="coerce")
        if ((~missing) & (~y.isin([0, 1]))).any():
            raise ValueError(f"{name}: unexpected target codes")
        labeled_n, missing_n = int(y.isin([0, 1]).sum()), int(missing.sum())

    if MATURE[name]:
        if TARGET not in d:
            raise ValueError(f"{name}: target column missing")
        if missing_n and not ALLOW_MISSING_TARGET[name]:
            raise ValueError(f"{name}: missing outcomes")
        if not labeled_n:
            raise ValueError(f"{name}: no labeled outcomes")
        d["_y"] = y.to_numpy(dtype=float, na_value=np.nan)
        evaluation[name] = d.loc[y.isin([0, 1])].copy()
        evaluation[name]["_y"] = evaluation[name]["_y"].astype(int)
        if missing_n:
            record_status(name + "_targets", "REVIEW",
                          "Labeled-subset evaluation; missing outcomes unresolved.")

    frames[name] = d
    audit.append(dict(cohort=name, raw_rows=raw_n, n=len(d),
                      n_labeled_available=labeled_n,
                      n_missing_target=missing_n,
                      n_evaluated=len(evaluation.get(name, []))))

show_table(pd.DataFrame({

# Model replay, robustness, stress and explainability

model = None
if RUN_MODEL_TESTS:
    import joblib
    from sklearn.inspection import permutation_importance

    model = joblib.load(MODEL_PATH)       # Trusted model artifact only
    if set(model.classes_) != {0, 1}:
        raise ValueError("Expected model classes 0 and 1")
    expected = list(getattr(model, "feature_names_in_", FEATURES))
    if set(expected) != set(FEATURES):
        raise ValueError("Model feature names do not match FEATURES")
    FEATURES = expected

    def model_probability(X, estimator=None):
        estimator = model if estimator is None else estimator
        p = estimator.predict_proba(X)[:, list(estimator.classes_).index(1)]
        if not (np.isfinite(p) & (p >= 0) & (p <= 1)).all():
            raise ValueError("Invalid model probabilities")
        return p

    for name, d in frames.items():
        X = d[FEATURES].copy()
        base = model_probability(X)

        # Full fitted model should not be compared to training OOF scores.
        if name != "train":
            show_table(pd.DataFrame([dict(
                mean_absolute_gap=np.mean(abs(base-d["_p"])),
                max_absolute_gap=np.max(abs(base-d["_p"]))
            )]), f"09_{name}_replay")

        changes = []
        for c, direction in [("HPI_+4", -1), ("UE_+4", 1)]:
            scale = pd.to_numeric(frames["train"][c], errors="coerce").std()
            if not np.isfinite(scale) or scale == 0:
                continue
            rng = np.random.default_rng(SEED)
            for label, delta in [
                ("measurement_noise", rng.normal(0, .05*scale, len(X))),
                ("statistical_stress", direction*.5*scale)
            ]:
                altered = X.copy()
                altered[c] = pd.to_numeric(altered[c], errors="raise") + delta
                p = model_probability(altered)
                changes.append(dict(feature=c, scenario=label,
                    mean_pd_change=np.mean(p-base),
                    mean_absolute_change=np.mean(abs(p-base)),
                    p95_absolute_change=np.quantile(abs(p-base), .95),
                    threshold_flip_rate=np.mean((p>=THRESHOLD)!=(base>=THRESHOLD))))
        show_table(pd.DataFrame(changes), f"09_{name}_sensitivity")

        # Adjacent observed bin levels; no Gaussian noise on bin codes.
        bin_changes = []
        for c in ["CRD_SCORE_bin", "ACTLTV_bin", "DSR_DIFF_bin"]:
            levels = np.sort(pd.to_numeric(
                frames["train"][c], errors="raise").dropna().unique())
            v = pd.to_numeric(X[c], errors="raise")
            if not v.dropna().isin(levels).all():
                raise ValueError(f"{name}: unseen {c} levels")
            for step in [-1, 1]:
                altered = X.copy()
                pos = np.searchsorted(levels, v.fillna(levels[0]))
                altered[c] = np.where(v.isna(), np.nan,
                    levels[np.clip(pos+step, 0, len(levels)-1)])
                p = model_probability(altered)
                bin_changes.append(dict(feature=c, step=step,
                    mean_pd_change=np.mean(p-base),
                    mean_absolute_change=np.mean(abs(p-base))))
        show_table(pd.DataFrame(bin_changes), f"09_{name}_bin_sensitivity")

    for name, d in evaluation.items():
        sample = d.sample(min(1500, len(d)), random_state=SEED)
        if sample["_y"].nunique() < 2:
            continue
        result = permutation_importance(
            model, sample[FEATURES], sample["_y"], scoring="roc_auc",
            n_repeats=3, random_state=SEED
        )
        t = pd.DataFrame(dict(feature=FEATURES,
            auc_drop=result.importances_mean,
            repeat_sd=result.importances_std)).set_index("feature")
        show_table(t, f"09_{name}_permutation_importance")
        t["auc_drop"].sort_values().plot.barh()
        show_plot(f"09_{name}_permutation_importance")

    # Partial dependence using the full fitted pipeline.
    X = frames["test"][FEATURES].sample(
        min(1000, len(frames["test"])), random_state=SEED
    )
    for c in ["HPI_+4", "UE_+4"]:
        grid = np.unique(np.nanquantile(pd.to_numeric(X[c]), np.linspace(.05,.95,10)))
        means = [model_probability(X.assign(**{c: value})).mean() for value in grid]

    for metric in ["auc", "average_precision", "brier", "log_loss"]:
        comparison.append(dict(cohort=name, metric=metric,

        for metric, v in pd.DataFrame(draws).items():
            comparison_intervals.append(dict(cohort=name, metric=metric,
                lower=v.quantile(.025), upper=v.quantile(.975),
                valid_bootstraps=v.notna().sum()))

    def cv_splits(X, y, groups, folds):
        cv = (StratifiedGroupKFold(folds, shuffle=True, random_state=SEED)
              if groups is not None else
              StratifiedKFold(folds, shuffle=True, random_state=SEED))
        splits = list(cv.split(X, y, groups))
        for a, b in splits:
            if y.iloc[a].nunique() < 2 or y.iloc[b].nunique() < 2:
                raise ValueError("A CV fold contains only one outcome class")
        return splits

    groups = d[GROUP_COLUMN] if GROUP_COLUMN else None
    nested = []
    for fold, (a, b) in enumerate(cv_splits(X, y, groups, 5), 1):
        inner_groups = groups.iloc[a] if groups is not None else None
        search = GridSearchCV(
            clone(model), PARAM_GRID, scoring="roc_auc",
            cv=cv_splits(X.iloc[a], y.iloc[a], inner_groups, 3),
            n_jobs=1, error_score="raise"
        ).fit(X.iloc[a], y.iloc[a])
        assess = d.iloc[b].copy()
        assess["_p"] = model_probability(X.iloc[b], search.best_estimator_)
        nested.append(dict(fold=fold, **performance(assess)))
    show_table(pd.DataFrame(nested), "11_nested_cv")

    search = GridSearchCV(
        clone(model), PARAM_GRID, scoring="roc_auc",
        cv=cv_splits(X, y, groups, 3), n_jobs=1, error_score="raise"
    ).fit(X, y)
    show_table(pd.DataFrame(search.cv_results_), "11_tuning_results")
    print("Best parameters:", search.best_params_)
