def filter_data(df_raw, approach):
    df = df_raw.copy()

    df = df[df["STATUS"] != "Others"].copy()

    if approach == "all_data":

        mask_funded = df["STATUS"] != "Declined"

        mask_declined = (
            (df["STATUS"] == "Declined") &
            (df["CRD_SCORE"] < 700) &
            (df["RAS_OMNI"] == "F") &
            (df["EFFECT_SID"].between(200900, 201712))
        )

        df = df[mask_funded | mask_declined].copy()

    elif approach == "funded_only":

        df = df[df["STATUS"] != "Declined"].copy()

    else:
        raise ValueError("approach must be 'all_data' or 'funded_only'")

    df = df[
        df["BOOKYEAR"].between(*TRAIN_YEARS) |
        df["BOOKYEAR"].between(*TEST_YEARS)
    ].copy()

    return df

def prepare_train_test(df, features, target=TARGET):

    df_train = df[df[YEAR_COL].between(*TRAIN_YEARS)].copy()
    df_test = df[df[YEAR_COL].between(*TEST_YEARS)].copy()

    model_features = [
        c for c in features
        if c not in [ID_COL, YEAR_COL, target, "STATUS"]
    ]

    X_train = df_train[model_features].copy()
    y_train = df_train[target].astype(int).to_numpy().ravel()

    X_test = df_test[model_features].copy()
    y_test = df_test[target].astype(int).to_numpy().ravel()

    return df_train, df_test, X_train, X_test, y_train, y_test, model_features

# ============================================================
# PREPROCESSOR
# ============================================================

def build_preprocessor(X, credit_use="bin"):

    ordinal_cols = ["PRK_NUM_OF_UNITS", "NUM_OF_BOR"]

    if credit_use == "bin" and "CRD_SCORE_bin" in X.columns:
        ordinal_cols = ["CRD_SCORE_bin"] + ordinal_cols

    log_cols = ["HPI_+4"] if "HPI_+4" in X.columns else []

    num_cols = X.select_dtypes(include=["number", "bool"]).columns.tolist()
    cat_cols = X.select_dtypes(exclude=["number", "bool"]).columns.tolist()

    ordinal_cols = [c for c in ordinal_cols if c in X.columns]
    num_cols = [c for c in num_cols if c not in ordinal_cols + log_cols]
    cat_cols = [c for c in cat_cols if c not in ordinal_cols]

    num_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="median")),
        ("scaler", StandardScaler())
    ])

    log_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="median")),
        ("scaler", StandardScaler())
    ])

    ord_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="most_frequent")),
        ("ord", OrdinalEncoder(
            handle_unknown="use_encoded_value",
            unknown_value=-1
        ))
    ])

    cat_tf = Pipeline([
        ("imputer", SimpleImputer(strategy="most_frequent")),
        ("ohe", OneHotEncoder(
            handle_unknown="ignore",
            sparse_output=False
        ))
    ])

    return ColumnTransformer([
        ("lognum", log_tf, log_cols),
        ("num", num_tf, num_cols),
        ("ord", ord_tf, ordinal_cols),
        ("cat", cat_tf, cat_cols)
    ], sparse_threshold=0)

# MODEL
# ============================================================

def build_model(model_type="xgb", random_state=42):

    if model_type == "xgb":
        return XGBClassifier(
            objective="binary:logistic",
            eval_metric="auc",
            tree_method="hist",
            random_state=random_state,
            n_jobs=-1
        )

    elif model_type == "rf":
        return RandomForestClassifier(
            random_state=random_state,
            n_jobs=-1
        )

    else:
        raise ValueError("model_type must be 'xgb' or 'rf'")

def get_param_dist(model_type="xgb"):

    if model_type == "xgb":
        return {
            "model__n_estimators": [300, 500, 700],
            "model__max_depth": [3, 4, 5, 6],
            "model__learning_rate": [0.01, 0.03, 0.05, 0.06],
            "model__subsample": [0.7, 0.8, 0.9],
            "model__colsample_bytree": [0.7, 0.8, 0.9],
            "model__min_child_weight": [25, 50, 75, 100],
            "model__gamma": [0, 0.5, 1, 2],
            "model__reg_lambda": [1, 2, 5, 10],
            "model__reg_alpha": [0, 0.1, 0.5, 1]
        }

    elif model_type == "rf":
        return {
            "model__n_estimators": [300, 500],
            "model__max_depth": [6, 10, 12],
            "model__min_samples_split": [25, 50, 100],
            "model__min_samples_leaf": [10, 25, 50],
            "model__max_features": ["sqrt", "log2", 0.5],
            "model__bootstrap": [True]
        }

def build_pipeline(X, model_type="xgb", credit_use="bin", random_state=42):

    preprocessor = build_preprocessor(X, credit_use=credit_use)
    model = build_model(model_type=model_type, random_state=random_state)

    return Pipeline([
        ("pre", preprocessor),
        ("model", model)
    ])

# ============================================================
# TRAIN MODEL
# ============================================================

def train_model(
    X_train,
    y_train,
    model_type="xgb",
    credit_use="bin",
    tune=True,
    n_iter=10,
    random_state=42
):
    pipeline = build_pipeline(
        X_train,
        model_type=model_type,
        credit_use=credit_use,
        random_state=random_state
    )

    if not tune:
        pipeline.fit(X_train, y_train)
        return pipeline

    cv = StratifiedKFold(
        n_splits=5,
        shuffle=True,
        random_state=random_state
    )

    search = RandomizedSearchCV(
        estimator=pipeline,
        param_distributions=get_param_dist(model_type),
        n_iter=n_iter,
        scoring="roc_auc",
        cv=cv,
        verbose=2,
        n_jobs=1,
        random_state=random_state,
        refit=True
    )

    search.fit(X_train, y_train)

    print("Best CV AUC:", search.best_score_)
    print("Best parameters:")
    print(search.best_params_)

    return search.best_estimator_


# ============================================================
# EVALUATION
# ============================================================

def evaluate_model(model, X, y, df, out_dir, label):

    os.makedirs(out_dir, exist_ok=True)

    proba = model.predict_proba(X)[:, 1]
    auc = roc_auc_score(y, proba)

    scored = df.copy()
    scored["pred_proba"] = proba
    scored["pred_class"] = (proba >= 0.5).astype(int)

    print(f"{label} AUC:", auc)
    print(classification_report(y, scored["pred_class"]))

    scored.to_csv(
        os.path.join(out_dir, f"{label}_scored.csv"),
        index=False
    )

    return {
        "auc": auc,
        "proba": proba,
        "scored": scored
    }

# DIAGNOSTICS
# ============================================================

def plot_model_diagnostics(y_true, proba, out_dir, label="oot", bins=40):

    os.makedirs(out_dir, exist_ok=True)

    fpr, tpr, _ = roc_curve(y_true, proba)
    auc = roc_auc_score(y_true, proba)

    plt.figure(figsize=(7, 6))
    plt.plot(fpr, tpr, label=f"AUC = {auc:.4f}")
    plt.plot([0, 1], [0, 1], linestyle="--")
    plt.xlabel("False Positive Rate")
    plt.ylabel("True Positive Rate")
    plt.title(f"ROC Curve - {label}")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{label}_roc_curve.png"), dpi=150)
    plt.show()

    plt.figure(figsize=(8, 5))
    sns.histplot(proba, bins=bins, kde=True)
    plt.xlabel("Predicted Probability")
    plt.title(f"Prediction Probability Distribution - {label}")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{label}_score_distribution.png"), dpi=150)
    plt.show()

    pred = (proba >= 0.5).astype(int)
    cm = confusion_matrix(y_true, pred)

    plt.figure(figsize=(5, 4))
    sns.heatmap(cm, annot=True, fmt="d")
    plt.xlabel("Predicted")
    plt.ylabel("Actual")
    plt.title(f"Confusion Matrix - {label}")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{label}_confusion_matrix.png"), dpi=150)
    plt.show()

    return {
        "auc": auc,
        "confusion_matrix": cm
    }


# ============================================================
# CV OOF
# ============================================================

def run_cv_oof(
    X,
    y,
    model_type="xgb",
    credit_use="bin",
    n_splits=5,
    random_state=42
):
    y = np.asarray(y).reshape(-1)

    skf = StratifiedKFold(
        n_splits=n_splits,
        shuffle=True,
        random_state=random_state
    )

    oof_proba = np.zeros(len(X))

    for fold, (tr_idx, va_idx) in enumerate(skf.split(X, y), 1):

        X_tr = X.iloc[tr_idx]
        y_tr = y[tr_idx]

        X_va = X.iloc[va_idx]
        y_va = y[va_idx]

        fold_model = build_pipeline(
            X_tr,
            model_type=model_type,
            credit_use=credit_use,
            random_state=random_state
        )

        fold_model.fit(X_tr, y_tr)

        oof_proba[va_idx] = fold_model.predict_proba(X_va)[:, 1]

        fold_auc = roc_auc_score(y_va, oof_proba[va_idx])
        print(f"Fold {fold} AUC:", fold_auc)

    oof_auc = roc_auc_score(y, oof_proba)
    print("OOF AUC:", oof_auc)

    return oof_proba, oof_auc

# ============================================================
# SHAP
# ============================================================

def run_shap(
    model,
    df_train,
    features,
    out_dir,
    model_name="xgb",
    sample_size=5000,
    random_state=42
):
    import shap

    os.makedirs(out_dir, exist_ok=True)

    shap_cols = [
        c for c in features
        if c not in [ID_COL, YEAR_COL, TARGET, "STATUS"]
    ]

    X_sample = df_train[shap_cols].sample(
        min(sample_size, len(df_train)),
        random_state=random_state
    ).copy()

    pre = model.named_steps["pre"]
    tree_model = model.named_steps["model"]

    X_trans = pre.transform(X_sample)

    try:
        feature_names = pre.get_feature_names_out()
    except:
        feature_names = [f"feature_{i}" for i in range(X_trans.shape[1])]

    explainer = shap.TreeExplainer(tree_model)
    shap_values = explainer.shap_values(X_trans)

    shap_importance = pd.DataFrame({
        "feature": feature_names,
        "mean_abs_shap": np.abs(shap_values).mean(axis=0)
    }).sort_values("mean_abs_shap", ascending=False)

    shap_importance.to_csv(
        os.path.join(out_dir, f"{model_name}_shap_importance.csv"),
        index=False
    )

    display(shap_importance.head(20))

    shap.summary_plot(
        shap_values,
        X_trans,
        feature_names=feature_names,
        plot_type="bar",
        show=False
    )
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{model_name}_shap_bar.png"), dpi=150)
    plt.show()

    shap.summary_plot(
        shap_values,
        X_trans,
        feature_names=feature_names,
        show=False
    )
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, f"{model_name}_shap_summary.png"), dpi=150)
    plt.show()

    return {
        "shap_values": shap_values,
        "shap_importance": shap_importance
    }

# FULL WORKFLOW
# ============================================================

def full_model_workflow(
    df,
    dataset_name,
    features,
    model_type="xgb",
    credit_use="bin",
    tune=True,
    n_iter=10,
    run_cv=True,
    run_shap_flag=True,
    random_state=42
):

    out_dir = os.path.join(BASE_OUT_DIR, dataset_name, model_type)
    os.makedirs(out_dir, exist_ok=True)

    print("Dataset:", dataset_name)
    print("Model:", model_type)
    print("Output:", out_dir)

    df_train, df_test, X_train, X_test, y_train, y_test, model_features = prepare_train_test(
        df=df,
        features=features,
        target=TARGET
    )

    print("Train shape:", X_train.shape)
    print("Test shape:", X_test.shape)

    oof_result = None

    if run_cv:
        oof_proba, oof_auc = run_cv_oof(
            X=X_train,
            y=y_train,
            model_type=model_type,
            credit_use=credit_use,
            n_splits=5,
            random_state=random_state
        )

        df_train_oof = df_train.copy()
        df_train_oof["oof_pred_proba"] = oof_proba

        df_train_oof.to_csv(
            os.path.join(out_dir, "train_oof_predictions.csv"),
            index=False
        )

        oof_result = {
            "oof_proba": oof_proba,
            "oof_auc": oof_auc
        }

    model = train_model(
        X_train=X_train,
        y_train=y_train,
        model_type=model_type,
        credit_use=credit_use,
        tune=tune,
        n_iter=n_iter,
        random_state=random_state
    )

    joblib.dump(
        model,
        os.path.join(out_dir, f"best_{model_type}_model.joblib")
    )

    train_result = evaluate_model(
        model=model,
        X=X_train,
        y=y_train,
        df=df_train,
        out_dir=out_dir,
        label="train"
    )

    test_result = evaluate_model(
        model=model,
        X=X_test,
        y=y_test,
        df=df_test,
        out_dir=out_dir,
        label="oot_2021_2023"
    )

    plot_model_diagnostics(
        y_true=y_test,
        proba=test_result["proba"],
        out_dir=out_dir,
        label="oot_2021_2023"
    )

    shap_result = None

    if run_shap_flag:
        shap_result = run_shap(
            model=model,
            df_train=df_train,
            features=features,
            out_dir=out_dir,
            model_name=model_type,
            sample_size=5000,
            random_state=random_state
        )

    return {
        "dataset_name": dataset_name,
        "model_type": model_type,
        "model": model,
        "train_auc": train_result["auc"],
        "test_auc": test_result["auc"],
        "train_scored": train_result["scored"],
        "test_scored": test_result["scored"],
        "oof_result": oof_result,
        "shap_result": shap_result,
        "model_features": model_features,
        "out_dir": out_dir
    }
